    LOCAL PROCEDURE Remember@4(VAR Rec@1000000000 : Record 37);
    VAR
      Mark@1000000001 : Record 110233;
      ContextTable@1000000002 : Record 110232;
    BEGIN
      // Временная запись живёт в памяти сеанса и никакой блокировки не создаёт: отмечать
      // по ней документ значит называть виновником того, кто ещё ничего не записал.
      IF Rec.ISTEMPORARY THEN
        EXIT;
      // Строка настройки решает, участвует ли эта таблица. Выключить переходник, не
      // выкладывая объектов, - единственный способ снять его цену на бою в один щелчок.
      IF NOT ContextTable.GET(TableNo) THEN
        EXIT;
      IF NOT ContextTable.Enabled THEN
        EXIT;
      Mark.Mark(TableNo,DocumentNo(Rec));
    END;

    LOCAL PROCEDURE TableNo@5() : Integer;
    BEGIN
      // Номер таблицы у подписки - КОНСТАНТА времени компиляции, настройкой он не
      // задаётся вовсе (FINDINGS, раздел 10). Отсюда и устройство: на каждую таблицу
      // контекста - свой переходник, а настройка отвечает только за разбор.
      EXIT(DATABASE::"Sales Line");
    END;

    LOCAL PROCEDURE DocumentNo@6(VAR Rec@1000000000 : Record 37) : Code[20];
    BEGIN
      // Поле берётся ПО ИМЕНИ, а не по номеру из настройки: номер поля в настройке нужен
      // разбору и экрану, а здесь имя известно на этапе компиляции - как и номер таблицы.
      EXIT(Rec."Document No.");
    END;

    [External]
    PROCEDURE SubscriptionActive@11() : Boolean;
    VAR
      Subscription@1000000000 : Record 2000000140;
    BEGIN
      // Единственный способ узнать, что подписка ЖИВАЯ. Объект, не выложенный или
      // выложенный без компиляции, ничем не отличается от работающего: колонка документа
      // просто остаётся пустой, а пустая колонка читается как "спорили не за документы".
      //
      // Номера ТАБЛИЦЫ у подписки платформа не показывает - в "Event Subscription" такого
      // поля нет вовсе (измерено 08.09.2026: поля 1-4 и 6-13, пятого нет). Поэтому здесь
      // проверяется, что подписчик зарегистрирован и активен, а что таблица та самая -
      // проверяется вторым способом: записью в неё, ниже.
      Subscription.SETRANGE("Subscriber Codeunit ID",CODEUNIT::"LockWatch Context Adapter");
      Subscription.SETRANGE(Active,TRUE);
      EXIT(NOT Subscription.ISEMPTY);
    END;

    [External]
    PROCEDURE SubscriptionCalls@12() : Integer;
    VAR
      Subscription@1000000000 : Record 2000000140;
      Calls@1000000001 : Integer;
    BEGIN
      // Сколько раз подписка сработала с момента старта службы. Ноль на живой таблице -
      // это не "тихо", а "не туда": объект собран под другую таблицу, и узнать об этом
      // иначе нельзя, потому что номера таблицы платформа в списке подписок не держит.
      Subscription.SETRANGE("Subscriber Codeunit ID",CODEUNIT::"LockWatch Context Adapter");
      IF Subscription.FINDSET THEN
        REPEAT
          Calls := Calls + Subscription."Number of Calls";
        UNTIL Subscription.NEXT = 0;
      EXIT(Calls);
    END;

    [External]
    PROCEDURE SelfTest@7();
    VAR
      SalesLine@1000000000 : Record 37;
      Mark@1000000001 : Record 110233;
      ContextTable@1000000002 : Record 110232;
      DocNo@1000000003 : Code[20];
      Existed@1000000004 : Boolean;
    BEGIN
      // Переходник, который ни разу не сработал, ничем не отличается от отсутствующего.
      // Поэтому он проверяет себя сам: пишет строку в таблицу контекста и смотрит, легла
      // ли отметка. Убирает за собой и то, и другое.
      DocNo := TestDocumentNo;
      IF NOT SubscriptionActive THEN
        ERROR(NotSubscribedErr,TableNo);
      Existed := ContextTable.GET(TableNo);
      IF NOT Existed THEN BEGIN
        ContextTable.INIT;
        ContextTable.VALIDATE("Table No.",TableNo);
        ContextTable."Document Field No." := 3;
        ContextTable.Enabled := TRUE;
        ContextTable.INSERT;
      END ELSE
        IF NOT ContextTable.Enabled THEN BEGIN
          ContextTable.Enabled := TRUE;
          ContextTable.MODIFY;
        END;

      Mark.Release;
      SalesLine.INIT;
      SalesLine."Document Type" := SalesLine."Document Type"::Order;
      SalesLine."Document No." := DocNo;
      SalesLine."Line No." := TestLineNo;
      SalesLine.INSERT;

      IF NOT Mark.GET(SERVICEINSTANCEID,SESSIONID) THEN BEGIN
        SalesLine.DELETE;
        CleanUp(Existed);
        ERROR(NoMarkErr);
      END;
      IF Mark."Document No." <> DocNo THEN BEGIN
        SalesLine.DELETE;
        CleanUp(Existed);
        ERROR(WrongDocErr,Mark."Document No.",DocNo);
      END;

      SalesLine.DELETE;
      Mark.Release;
      CleanUp(Existed);
      MESSAGE(OkMsg,DocNo);
    END;

      Переходник-подписчик: ОБРАЗЕЦ, а не готовый объект установки.

      Подписка стоит на штатной Table 37 "Sales Line" нарочно - репозиторий считаем
      публичным, и номеров доработанных объектов заказчика в коде быть не может. На
      установке объект под целевую таблицу СОБИРАЕТСЯ ПОДСТАНОВКОЙ, а не правкой руками:
      `pwsh scripts/New-ContextAdapter.ps1 -TableNo N -ObjectNo M`. Мест для правки пять, и
      промахнуться в них можно молча - подписка на чужой номер таблицы компилируется и
      просто никогда не срабатывает. Копий столько, сколько таблиц контекста: номер таблицы
      в подписке - константа времени компиляции, и одним объектом на все таблицы обойтись
      нельзя (FINDINGS, раздел 10).

      Почему переходник вообще нужен. Отметка ставится ВНУТРИ транзакции пишущего, и
      этим она ценна: транзакция виновника держит на ней монопольную блокировку, а по
      номеру транзакции её находит съём со стороны SQL. Это не совпадение по времени,
      а факт - и он проверен на живой блокировке (FINDINGS, раздел 17).

      Три оговорки, каждая записана нарочно:

      1. Временную запись пропускаем. Она живёт в памяти сеанса, блокировки не создаёт,
         и отмечать по ней документ значит называть виновником того, кто ничего не писал.
      2. Выключатель живёт в СТРОКЕ НАСТРОЙКИ таблицы контекста. Снять цену переходника
         на бою можно щелчком, не выкладывая объектов.
      3. Читающий, пусть даже с LOCKTABLE, отметки не оставит вовсе. Для него остаётся
         дорога через хэш самой блокировки, и это ещё одна причина строить обе.
    }
    END.
