OBJECT Codeunit 110242 LockWatch Hold Bench
{
  OBJECT-PROPERTIES
  {
    Date=18.10.26;
    Time=12:00:00;
    Modified=Yes;
    Version List=LockWatch;
  }
  PROPERTIES
  {
  }
  CODE
  {
    VAR
      HeldTxt@1000000000 : TextConst 'ENU=LOCKHOLD user %1 session %2 ms %3 ownrows %4 laterows %5;RUS=LOCKHOLD user %1 session %2 ms %3 ownrows %4 laterows %5';
      NoRowErr@1000000001 : TextConst 'ENU=The bench mark row is not there. It is put down by the run itself, and holding a lock on a row that does not exist would measure nothing.;RUS=Строки-мишени нет. Её кладёт сам прогон, и держать блокировку на несуществующей строке значит не мерить ничего.';
      LockRowHeldTxt@1000000002 : TextConst 'ENU=LOCKROWHELD user %1 session %2;RUS=LOCKROWHELD user %1 session %2';
      LockRowTookTxt@1000000003 : TextConst 'ENU=LOCKROWTOOK user %1 session %2;RUS=LOCKROWTOOK user %1 session %2';

    [External]
    PROCEDURE Hold@1();
    VAR
      Mark@1000000000 : Record 110233;
      Waited@1000000001 : Integer;
      Locks@1000000002 : Record 2000000154;
      OwnRows@1000000003 : Integer;
      LateRows@1000000004 : Integer;
    BEGIN
      // Блокировку держит СЕССИЯ NAV, и в этом весь смысл. Штатная таблица блокировок
      // показывает только те транзакции, что платформа завела в свою карту, а туда
      // попадают лишь сессии NAV этого экземпляра. Соединение sqlcmd в ней не появится
      // никогда, и проверять дорогу к имени человека им нельзя вовсе.
      Mark.LOCKTABLE;
      IF NOT Mark.GET(TargetInstanceId,TargetSessionId) THEN
        ERROR(NoRowErr);
      Mark."Document No." := HeldDocumentNo;
      Mark.MODIFY;
      // COMMIT здесь не делается нарочно: транзакция держится открытой, и вместе с ней
      // держится блокировка. Отпустит её платформа сама, когда вызов кончится.
      //
      // Штатная таблица читается ОТСЮДА, изнутри собственной открытой транзакции. Это и
      // есть решающий опыт: если держатель не видит даже СВОЕЙ блокировки, значит карта
      // транзакций пуста вовсе; если видит свою, а сторонний наблюдатель не видит ничего,
      // значит карта у каждой сессии своя, и дорога к чужому имени по ней не проходит.
      OwnRows := Locks.COUNT;
      Waited := HoldMs;
      SLEEP(Waited DIV 2);
      // Второй счёт - когда ждущий уже стоит в очереди. Первый мог оказаться слишком
      // ранним: карта наполняется на старте транзакции, а не на взятии блокировки.
      Locks.RESET;
      LateRows := Locks.COUNT;
      SLEEP(Waited DIV 2);
      MESSAGE(HeldTxt,USERID,SESSIONID,Waited,OwnRows,LateRows);
    END;

    LOCAL PROCEDURE HoldMs@2() : Integer;
    BEGIN
      // Минута. Больше предела ожидания NAV (10 000 мс) - чтобы жертва успела отвалиться
      // по нему и эпизод получил честный исход, - и заведомо больше, чем нужно прогону на
      // проход и нажатие кнопки. Меньше минуты не годится: на холодной службе первый
      // проход платит за сборку business assemblies, и окно съедается целиком.
      EXIT(60000);
    END;

    LOCAL PROCEDURE TargetInstanceId@3() : Integer;
    BEGIN
      // Та же мишень, что и у прогона по живой блокировке: отрицательные номера сеанса и
      // экземпляра настоящими не бывают, и столкнуться с чужой отметкой нельзя.
      EXIT(-1);
    END;

    LOCAL PROCEDURE TargetSessionId@4() : Integer;
    BEGIN
      EXIT(-1);
    END;

    LOCAL PROCEDURE HeldDocumentNo@5() : Code[20];
    BEGIN
      EXIT('HELD-BY-NAV');
    END;

    [External]
    PROCEDURE HoldWithUserRow@6();
    VAR
      Register@1000000000 : Record 45;
      Account@1000000001 : Record 15;
    BEGIN
      // Держатель, в чьей транзакции лежит ИМЯ. Дорога "по строке виновника" читает
      // учётку из деловых полей той строки, которую виновник сам записал, и проверять её
      // можно только так: пока транзакция открыта, в G/L Register лежит строка с USERID,
      // а G/L Account держит X-блокировку ключа. Ни одного нового права дороге не нужно -
      // обе таблицы штатные, а имя стоит в той же транзакции, что и спор.
      Register.INIT;
      Register."No." := HeldRegisterNo;
      Register."User ID" := USERID;
      Register.INSERT;
      // Чтение той же строки - не ради ответа, а ради того, чтобы отложенная вставка
      // ушла в SQL: дорога читает её ДРУГИМ соединением, и буфер сессии ей не виден.
      // Ответ здесь ничего не решает, поэтому и проверять его нечего.
      Register.GET(HeldRegisterNo);
      Account.LOCKTABLE;
      IF Account.FINDFIRST THEN
        Account.MODIFY;
      SLEEP(HeldRowMs);
      // ERROR, а не MESSAGE: им кончается и вызов, и транзакция - обе строки откатываются,
      // блокировки отпускаются, и следующий прогон начинает с чистого места. MESSAGE
      // зафиксировал бы след из чужой таблицы учёта в базе заказчика.
      ERROR(LockRowHeldTxt,USERID,SESSIONID);
    END;

    [External]
    PROCEDURE WaitWithUserRow@7();
    VAR
      Register@1000000000 : Record 45;
      Account@1000000001 : Record 15;
    BEGIN
      // Ждущий. Его собственная строка помечена чужой учёткой нарочно: если дорога
      // когда-нибудь перепутает стороны спора и прочтёт строку ЖДУЩЕГО, ответ будет
      // "LOCKWATCH-VICTIM" - именем, которого в системе нет, и прогон это увидит сразу.
      // Своей учёткой он не подписывается: подпись жертвы не должна совпасть с подписью
      // виновника ни при какой ошибке разбора.
      Register.INIT;
      Register."No." := VictimRegisterNo;
      Register."User ID" := VictimUserID;
      Register.INSERT;
      Register.GET(VictimRegisterNo);
      Account.LOCKTABLE;
      // Здесь ждущий УПИРАЕТСЯ в блокировку держателя и стоит на ней до предела ожидания
      // NAV (10 000 мс). Дошёл до MODIFY - значит блокировки не было вовсе, и это
      // отдельный ответ, а не успех: прогон обязан покраснеть.
      IF Account.FINDFIRST THEN
        Account.MODIFY;
      ERROR(LockRowTookTxt,USERID,SESSIONID);
    END;

    LOCAL PROCEDURE HeldRegisterNo@8() : Integer;
    BEGIN
      // Номер строки-мишени в G/L Register. Он же ключ: номер у таблицы первичный, и
      // строка гарантированно одна. 2000000001 не бывает настоящим номером регистра -
      // регистр нумеруется с единицы, - поэтому столкнуться с чужой строкой неоткуда.
      EXIT(2000000001);
    END;

    LOCAL PROCEDURE VictimRegisterNo@9() : Integer;
    BEGIN
      // Вторая мишень, тем же доводом. Разные номера у сторон спора нужны затем, чтобы
      // дорога, прочитавшая не ту транзакцию, не могла случайно наткнуться на строку
      // держателя: у неё другой ключ и другая учётка.
      EXIT(2000000002);
    END;

    LOCAL PROCEDURE VictimUserID@10() : Code[50];
    BEGIN
      // Учётки с таким именем нет и быть не может: оно нужно как метка, а не как
      // пользователь. Настоящая учётка здесь означала бы, что жертву можно спутать с
      // виновником, - а весь опыт затем и ставится, чтобы этого не случилось.
      EXIT('LOCKWATCH-VICTIM');
    END;

    LOCAL PROCEDURE HeldRowMs@11() : Integer;
    BEGIN
      // Тридцать секунд - втрое больше предела ожидания блокировки у NAV (10 000 мс).
      // Ждущий обязан упереться в СВОЙ предел и отвалиться, пока держатель ещё здесь:
      // отпусти держатель блокировку раньше - и прогон прочитал бы тишину вместо спора.
      EXIT(30000);
    END;

    BEGIN
    {
      Держатель блокировки для проверки дороги к имени ЧЕЛОВЕКА.

      Нужен он ровно затем, что этой дороги нельзя проверить соединением sqlcmd: штатная
      таблица блокировок показывает лишь те транзакции, что платформа завела в свою карту,
      а туда попадают только сессии NAV этого экземпляра. Держатель поэтому - сессия NAV,
      запускаемая через Invoke-NAVCodeunit, а ждущим остаётся всё тот же sqlcmd.

      Ответ он отдаёт машиночитаемой строкой и на обоих языках одинаково: разбирать её
      прогону, а не читать человеку, и зависеть от языка службы разбор не должен.

      Объект мерный, как и Codeunit 110233: в бою он не нужен и вреда не приносит - без
      строки-мишени он просто отказывается работать.

      Две мишени - HoldWithUserRow и WaitWithUserRow - ставят опыт для дороги "по строке
      виновника". Держатель пишет в G/L Register свою учётку и держит X-блокировку ключа
      G/L Account; ждущий пишет подставную и упирается в ту же строку. Дорога обязана
      назвать держателя по его собственной строке, а подставную учётку ждущего - не
      назвать никогда. Оба вызова кончаются ERROR: опыт кончается полным откатом, и
      следов в базе заказчика не остаётся.
    }
    END.
  }
}
