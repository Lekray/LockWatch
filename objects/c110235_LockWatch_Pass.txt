    [External]
    PROCEDURE RunPass@1() : Boolean;
    VAR
      Setup@1000000009 : Record 110230;
      Observed@1000000010 : TEMPORARY Record 110231;
      Waiter@1000000011 : ARRAY [500] OF Integer;
      Blocker@1000000012 : ARRAY [500] OF Integer;
      BlockerIsNav@1000000013 : ARRAY [500] OF Boolean;
      Head@1000000014 : ARRAY [500] OF Integer;
      Depth@1000000015 : ARRAY [500] OF Integer;
      Ring@1000000016 : ARRAY [500] OF Boolean;
      Behind@1000000017 : ARRAY [500] OF Integer;
      RowCount@1000000018 : Integer;
      OtherDatabase@1000000019 : Integer;
      ObservedAt@1000000020 : DateTime;
      StartedAt@1000000021 : DateTime;
      Elapsed@1000000022 : Duration;
      Index@1000000023 : Integer;
      Opened@1000000024 : Integer;
      Updated@1000000025 : Integer;
      Closed@1000000026 : Integer;
      Restarted@1000000027 : Integer;
      Note@1000000028 : Text;
      Truncated@1000000029 : Boolean;
      State@1000000030 : Record 110237;
    BEGIN
      // Проход целиком: съём, разматывание цепочек, ведение эпизодов, чистка по сроку и
      // ОДИН COMMIT в конце. Порядок именно такой, потому что каждый шаг дешевле
      // предыдущего, а самый дорогой - запись - должен получить всё, что о нём известно,
      // с первого раза: обновлять строку журнала дважды за проход дороже вдвое.
      Setup.FetchOnce;
      // Состояние сторожа лежит отдельно от настройки нарочно: строку, которую фон
      // переписывает каждый проход, человек не может править вовсе (FINDINGS, раздел 33).
      State.FetchOnce;
      // Язык, на котором проход пишет слова: подсказку, причину, сообщение сторожа и текст
      // тревоги наружу. В сессии с окном язык не трогается вовсе - разбирается это внутри.
      Parse.UseJournalLanguage(Setup."Journal Language ID");
      StartedAt := CURRENTDATETIME;

      IF NOT Sql.ReadQueue(Setup."Max Rows per Pass",Setup."Collect Statement Values",
                           Observed,Waiter,Blocker,BlockerIsNav,
                           RowCount,ObservedAt,OtherDatabase)
      THEN BEGIN
        // Отказ съёма журнал НЕ трогает. Закрыть эпизоды моментом, которого мы не знаем,
        // значило бы объявить их законченными, ничего об этом не выяснив.
        SaveWatchdog(State,STRSUBSTNO(ReadFailedTxt,Sql.LastErrorText),StartedAt,NoTime,0,FALSE);
        EXIT(FALSE);
      END;

      IF RowCount > 0 THEN
        IF NOT Sql.ResolveNames(Observed) THEN
          // Имена - не эпизоды. Без них строка беднее, но она есть и она правдива;
          // выбросить проход из-за имён значило бы потерять сам факт блокировки.
          Note := ' ' + STRSUBSTNO(NamesFailedTxt,Sql.LastErrorText);

      Parse.Chains(Waiter,Blocker,RowCount,Head,Depth,Ring,Behind);

      Journal.BeginPass(ObservedAt,Setup."Poll Period (ms)",Setup."Alert Threshold (ms)");
      Observed.RESET;
      IF Observed.FINDSET THEN
        REPEAT
          Index := Observed."Entry No.";
          Observed."Head SPID" := Head[Index];
          Observed."Chain Depth" := Depth[Index];
          Observed."Chain Ring" := Ring[Index];
          Observed."Victims Behind Head" := Behind[Index];
          Observed."Head Is Foreign" := HeadIsForeign(Blocker,BlockerIsNav,RowCount,Head[Index]);
          Journal.Observe(Observed);
        UNTIL Observed.NEXT = 0;
      Journal.FinishPass;
      Journal.PassCounts(Opened,Updated,Closed,Restarted);

      // Документ и имена разбираются ПОСЛЕ записи и только для идущих эпизодов: мост к
      // ним - блокировка, которую держат прямо сейчас. На закрытом эпизоде разбирать
      // нечего вовсе - ожидание кончилось ровно потому, что блокировку отпустили. Пока
      // же эпизод идёт, вопрос повторяется каждым проходом: отметка виновника появляется
      // тогда, когда он трогает таблицу документа, а это бывает и позже начала ожидания.
      Note := Note + NameDocuments;

      // Тревога поднимается ПОСЛЕ разбора документа: в неё должны попасть и номер
      // документа, и учётная запись, а до разбора их ещё нет.
      Note := Note + RaiseAlerts(Setup);

      // Взаимоблокировки собираются ПОСЛЕ очереди и не каждым проходом. Они приходят
      // готовыми эпизодами и ведению не подлежат: сервер уже разорвал круг.
      Note := Note + CollectDeadlocks(Setup,State);

      // Накопительный слой снимается каждым проходом: он дёшев (доли миллисекунды на всю
      // базу) и, в отличие от очереди, видит ожидания короче периода опроса. Сторожем ему
      // не быть - счётчик прирастает только в момент конца ожидания, - зато он единственный,
      // кто может сказать, какую долю происходящего журнал вообще видит.
      Note := Note + CollectCoverage(Setup,State);

      // Переезд порционный и внутри того же прохода: отдельное задание уборки - это ещё
      // один объект, который забудут включить, и горячая таблица будет расти молча.
      // Строка не уничтожается: срок в настройке - срок ПЕРЕЕЗДА в историю.
      //
      // Пределов два, и второй нужен ровно там, где первый бессилен: в шторм всё, что
      // лежит в журнале, моложе срока. Сколько уехало - в строке сторожа: ненайденный
      // вчерашний эпизод без объяснения стоит часа поисков.
      Note := Note + MovedNote(Journal.MoveOldEntries(Setup."Retention (Days)",
                                                      Setup."Journal Rows Ceiling",
                                                      TrimRowsPerPass(Setup."Max Rows per Pass")));

      Truncated := (Setup."Max Rows per Pass" > 0) AND (RowCount >= Setup."Max Rows per Pass");
      Elapsed := CURRENTDATETIME - StartedAt;
      Note := STRSUBSTNO(CountsTxt,Opened,Updated,Closed,Restarted) + Note;
      SaveWatchdog(State,PassNote(RowCount,OtherDatabase,Truncated,Setup."Max Rows per Pass",Note),
             ObservedAt,Elapsed,RowCount,Truncated);
      EXIT(TRUE);
    END;

    LOCAL PROCEDURE NameDocuments@7() : Text;
    VAR
      Pending@1000000000 : TEMPORARY Record 110231;
      Context@1000000002 : TEMPORARY Record 110232;
      Picked@1000000001 : Integer;
    BEGIN
      // Разбирать нечего - и запроса не будет вовсе. Это и есть ответ на "dm_tran_locks
      // в цикле не трогать никогда": на тихом сервере цикл её не трогает, а на шумном
      // читает один раз на проход, а не один раз на эпизод.
      Picked := Journal.PickUnresolved(Pending,Parse.MaxDocumentsPerPass);
      IF Picked = 0 THEN
        EXIT('');
      LoadContext(Context);
      IF NOT Sql.ResolveDocuments(Pending,Context) THEN
        EXIT(' ' + STRSUBSTNO(DocFailedTxt,Sql.LastErrorText));

      Pending.RESET;
      IF Pending.FINDSET THEN
        REPEAT
          IF Pending."Document No." <> '' THEN
            Journal.SetDocument(Pending."Entry No.",Pending."Document No.",Pending."User Name",
                                Pending."Document Source",'',Pending."Victim User Name")
          ELSE
            // Пустая колонка с ПРИЧИНОЙ честнее пустой колонки без неё: без причины она
            // читается как "блокировок по документам не было".
            Journal.SetDocument(Pending."Entry No.",'','',Parse.DocSourceNone,NoMarkTxt,
                                Pending."Victim User Name");
        UNTIL Pending.NEXT = 0;
      EXIT('');
    END;

    LOCAL PROCEDURE CollectDeadlocks@9(VAR Setup@1000000000 : Record 110230;VAR State@1000000011 : Record 110237) : Text;
    VAR
      Found@1000000001 : TEMPORARY Record 110231;
      WasReadUntil@1000000002 : DateTime;
      OldestAt@1000000003 : DateTime;
      NewestAt@1000000004 : DateTime;
      Elapsed@1000000005 : Duration;
      Period@1000000006 : Duration;
      Rows@1000000007 : Integer;
      Added@1000000008 : Integer;
      BufferPresent@1000000009 : Boolean;
      Note@1000000010 : Text;
    BEGIN
      IF NOT Setup."Deadlocks Enabled" THEN
        EXIT('');
      // Буфер читается НЕ каждым проходом. Он растёт не от наших блокировок, а от чужих
      // взаимоблокировок, и на тихом сервере не меняется вовсе; разбор всего буфера в XML
      // дороже опроса очереди, ради которого проход и заведён.
      Period := Parse.DeadlockReadPeriodMs;
      IF State."Deadlocks Read At" <> 0DT THEN BEGIN
        Elapsed := CURRENTDATETIME - State."Deadlocks Read At";
        IF Elapsed < Period THEN
          EXIT('');
      END;
      // Отметка "читали в" ставится ДО чтения и остаётся стоять при отказе: сервер,
      // который не отвечает, нельзя спрашивать каждые три секунды.
      State."Deadlocks Read At" := CURRENTDATETIME;
      WasReadUntil := State."Deadlocks Read Until";

      IF NOT Sql.ReadDeadlocks(WasReadUntil,Parse.MaxDeadlocksPerPass,
                               Setup."Collect Statement Values",Found,BufferPresent,OldestAt,Rows)
      THEN
        EXIT(' ' + STRSUBSTNO(DeadlockFailedTxt,Sql.LastErrorText));

      IF NOT BufferPresent THEN
        // Выключенная сессия и отсутствие взаимоблокировок дают одну и ту же пустую
        // колонку. Различает их только эта строка.
        EXIT(' ' + NoBufferTxt);

      Found.RESET;
      IF Found.FINDSET THEN
        REPEAT
          IF Found."Ended At" > NewestAt THEN
            NewestAt := Found."Ended At";
          IF Journal.AddDeadlock(Found) THEN
            Added := Added + 1;
        UNTIL Found.NEXT = 0;
      IF NewestAt <> 0DT THEN
        State."Deadlocks Read Until" := NewestAt;

      // Буфер кольцевой. Если самое старое событие в нём уже новее прежней отметки, между
      // ними что-то вытеснено, и молчать об этом нельзя: пустая колонка читается как
      // "взаимоблокировок не было", а это ровно противоположный вывод.
      IF (WasReadUntil <> 0DT) AND (OldestAt <> 0DT) AND (OldestAt > WasReadUntil) THEN
        Note := ' ' + DeadlockLostTxt;
      IF Added > 0 THEN
        Note := Note + ' ' + STRSUBSTNO(DeadlocksTxt,Added);
      EXIT(Note);
    END;

    LOCAL PROCEDURE RaiseAlerts@12(VAR Setup@1000000000 : Record 110230) : Text;
    VAR
      Pending@1000000001 : TEMPORARY Record 110231;
      Picked@1000000002 : Integer;
      Raised@1000000003 : Integer;
      Latched@1000000004 : Integer;
    BEGIN
      IF Setup."Alert Channel" = Setup."Alert Channel"::Off THEN
        EXIT('');
      Picked := Journal.PickAlerting(Pending,Setup."Alert Threshold (ms)",Parse.MaxAlertsPerPass);
      // Тревоги фоном НЕ чистятся, и это решение, а не забывчивость. Тревога - запись о
      // том, что наружу ушло сообщение; стереть её значит расписаться за чужой разговор с
      // мониторингом. Строк здесь на порядки меньше, чем эпизодов: тревога поднимается на
      // голову цепочки один раз за окно защёлки, а не на каждое ожидание. Убирает их
      // человек кнопкой на странице тревог - там же, где их и читает.
      IF Picked = 0 THEN
        EXIT('');
      AlertMgt.BeginPass;
      Pending.RESET;
      IF Pending.FINDSET THEN
        REPEAT
          AlertMgt.Raise(Pending,Setup."Alert Threshold (ms)",
                         Setup."Alert Channel" = Setup."Alert Channel"::EventLog,
                         Setup."Alert Event Source");
        UNTIL Pending.NEXT = 0;
      AlertMgt.PassCounts(Raised,Latched);
      IF (Raised > 0) OR (Latched > 0) THEN
        EXIT(' ' + STRSUBSTNO(AlertsTxt,Raised,Latched));
      EXIT('');
    END;

    LOCAL PROCEDURE CollectCoverage@11(VAR Setup@1000000000 : Record 110230;VAR State@1000000006 : Record 110237) : Text;
    VAR
      Reading@1000000001 : TEMPORARY Record 110235;
      Rows@1000000002 : Integer;
      Added@1000000003 : Integer;
      Changed@1000000004 : Integer;
      Resets@1000000005 : Integer;
    BEGIN
      IF NOT Setup."Coverage Enabled" THEN
        EXIT('');
      IF NOT Sql.ReadCounters(Parse.MaxCoverageRows,Reading,Rows) THEN
        EXIT(' ' + STRSUBSTNO(CoverageFailedTxt,Sql.LastErrorText));

      // Отметка "охват считается с" ставится в тот же проход, что и первое чтение. Без
      // неё журнальная половина охвата считалась бы за всё время, а счётчики - с этого
      // мига, и доля вышла бы больше сотни процентов.
      IF State."Coverage Since" = 0DT THEN
        State."Coverage Since" := CURRENTDATETIME;

      Coverage.Accumulate(Reading);
      Coverage.PassCounts(Added,Changed,Resets);
      IF Resets > 0 THEN
        EXIT(' ' + STRSUBSTNO(CoverageResetTxt,Resets));
      IF Changed > 0 THEN
        EXIT(' ' + STRSUBSTNO(CoverageTxt,Changed));
      EXIT('');
    END;

    LOCAL PROCEDURE LoadContext@8(VAR Context@1000000000 : TEMPORARY Record 110232);
    VAR
      ContextTable@1000000001 : Record 110232;
    BEGIN
      // Список таблиц контекста читается ОДИН раз на проход и уезжает съёму копией. Он
      // короткий - десятки строк, - а обращаться к нему пришлось бы на каждый разбираемый
      // эпизод, то есть внутри цикла, который и без того самый дорогой.
      Context.RESET;
      Context.DELETEALL;
      ContextTable.SETRANGE(Enabled,TRUE);
      IF ContextTable.FINDSET THEN
        REPEAT
          Context := ContextTable;
          Context.INSERT;
        UNTIL ContextTable.NEXT = 0;
    END;

    LOCAL PROCEDURE SaveWatchdog@2(VAR State@1000000000 : Record 110237;Msg@1000000001 : Text;At@1000000002 : DateTime;Elapsed@1000000003 : Duration;Rows@1000000006 : Integer;Truncated@1000000004 : Boolean);
    VAR
      Fresh@1000000007 : Record 110237;
      Milliseconds@1000000005 : Integer;
    BEGIN
      // Сторож пишет ВСЕГДА, и на удачном проходе тоже. Пустое сообщение читается как
      // "всё хорошо", а на деле означает "не знаю": последний проход мог не случиться
      // вовсе, и отличить одно от другого можно только по отметке времени рядом.
      //
      // Запись у прохода ОДНА и в самом конце - как и была, когда лежала в настройке.
      // Строка перечитывается под замком прямо перед ней: между началом прохода и его
      // концом человек мог нажать "стоп" или "начать счёт заново", а NAV сверяет версию
      // строки при записи. Отметки чтения, добытые проходом, переносятся в свежую строку
      // поимённо - их три, и список этот стоит держать на виду.
      Milliseconds := Elapsed;
      Fresh.FetchToWrite;
      Fresh."Deadlocks Read At" := State."Deadlocks Read At";
      Fresh."Deadlocks Read Until" := State."Deadlocks Read Until";
      Fresh."Coverage Since" := State."Coverage Since";
      Fresh."Last Pass At" := At;
      Fresh."Last Pass (ms)" := Milliseconds;
      Fresh."Last Pass Rows" := Rows;
      Fresh."Last Pass Truncated" := Truncated;
      Fresh."Watchdog Message" := COPYSTR(Msg,1,MAXSTRLEN(Fresh."Watchdog Message"));
      Fresh.MODIFY;
      COMMIT;
    END;

    LOCAL PROCEDURE PassNote@3(RowCount@1000000000 : Integer;OtherDatabase@1000000001 : Integer;Truncated@1000000002 : Boolean;Ceiling@1000000003 : Integer;Note@1000000004 : Text) : Text;
    VAR
      Msg@1000000005 : Text;
    BEGIN
      IF RowCount = 0 THEN
        Msg := QuietTxt
      ELSE
        Msg := STRSUBSTNO(BusyTxt,RowCount);
      IF Truncated THEN
        Msg := Msg + ' ' + STRSUBSTNO(TruncatedTxt,Ceiling);
      IF OtherDatabase > 0 THEN
        Msg := Msg + ' ' + STRSUBSTNO(OtherDbTxt,OtherDatabase);
      IF Note <> '' THEN
        Msg := Msg + ' ' + Note;
      EXIT(Msg);
    END;

    LOCAL PROCEDURE NoTime@6() : Duration;
    BEGIN
      // Отказавший проход длительности не имеет: время до отказа - это не время прохода,
      // и записать его в "последний проход, мс" значило бы показать цену того, чего не было.
      EXIT(0);
    END;

    LOCAL PROCEDURE HeadIsForeign@4(VAR Blocker@1000000000 : ARRAY [500] OF Integer;VAR BlockerIsNav@1000000001 : ARRAY [500] OF Boolean;Count@1000000002 : Integer;HeadSpid@1000000003 : Integer) : Boolean;
    VAR
      Index@1000000004 : Integer;
    BEGIN
      // Голова цепочки сама не ждёт, поэтому среди ждущих её нет - зато она обязательно
      // чей-то непосредственный блокирующий, и оттуда о ней и известно, наша она или нет.
      //
      // Не нашли - отвечаем "наша", то есть НЕ обвиняем. Пометить незнакомую сессию чужой
      // значит подсказать заказчику искать виноватого снаружи, а искать надо у себя.
      IF (Count <= 0) OR (HeadSpid = 0) THEN
        EXIT(FALSE);
      FOR Index := 1 TO Count DO
        IF Blocker[Index] = HeadSpid THEN
          EXIT(NOT BlockerIsNav[Index]);
      EXIT(FALSE);
    END;

    LOCAL PROCEDURE TrimRowsPerPass@5(RowsPerPass@1000000000 : Integer) : Integer;
    BEGIN
      // Потолок уборки за проход. Месяц журнала за один COMMIT - это блокировка
      // собственной таблицы посреди наблюдения, то есть ровно то, что мы ловим.
      //
      // Но и меньше потолка НАБЛЮДЕНИЯ уборка быть не может. Если за проход приезжает
      // строк больше, чем уезжает, потолок журнала не выполняется никогда: журнал растёт
      // по построению, и второй предел оказывается написанным зря. Потолок наблюдения
      // живёт в настройке и его поднимают - значит и уборка меряется им, а не своим числом.
      IF RowsPerPass > TrimRowsFloor THEN
        EXIT(RowsPerPass);
      EXIT(TrimRowsFloor);
    END;

    LOCAL PROCEDURE TrimRowsFloor@13() : Integer;
    BEGIN
      // Пол уборки: двести строк за проход разбирают суточный завал за минуты и никому не
      // мешают. Ниже него уборка не опускается, даже если потолок наблюдения занижен до
      // единиц: тогда завал не разобрался бы вовсе.
      EXIT(200);
    END;

    LOCAL PROCEDURE MovedNote@14(Moved@1000000000 : Integer) : Text;
    BEGIN
      // Ноль не пишется. Строку сторожа читают каждым проходом, и постоянное "уехало 0"
      // заслонило бы в ней то, ради чего её читают.
      IF Moved = 0 THEN
        EXIT('');
      EXIT(' ' + STRSUBSTNO(MovedTxt,Moved));
    END;

    BEGIN
    {
      Проход целиком - от очереди SQL до строки в журнале. Здесь нет ни разбора (он в
      Codeunit 110230), ни SQL (он в 110234), ни записи (она в 110232): здесь только
      порядок, в котором они складываются, и решения о том, что делать при отказе.

      Решений об отказе три, и все три - о том, чтобы не соврать молчанием:

      1. Съём не удался - журнал НЕ трогаем вовсе. Закрыть эпизоды моментом, которого мы
         не знаем, значит объявить их законченными без единого основания.
      2. Имена не получены - эпизоды всё равно записываются. Строка без имени таблицы
         беднее, но правдива; потерять из-за имён сам ФАКТ блокировки нельзя.
      3. Сторож пишет сообщение и на удачном проходе. Пустое сообщение читается как
         "всё хорошо", а означает "не знаю".

      Цепочки считаются ОДНИМ вызовом на проход. Голова каждой строки стоит прохода по
      всем ждущим, и посчитать "жертв за головой" отдельно для каждой строки значило бы
      получить куб от числа ждущих: на пятистах это сотни миллионов сравнений.

      COMMIT здесь ровно один на проход - тот самый, чью цену меряет Codeunit 110233.
    }
    END.
  }
