    PROCEDURE ParseWaitResource@1(ResourceText@1000 : Text;VAR Kind@1001 : Text;VAR HobtId@1002 : BigInteger;VAR ObjectId@1003 : Integer;VAR DatabaseId@1004 : Integer;VAR HeldMode@1005 : Text) : Boolean;
    VAR
      Trimmed@1006 : Text;
      SpacePos@1007 : Integer;
    BEGIN
      // Описание ресурса в очереди ожиданий - это род и пары имя=значение:
      //   keylock hobtid=72057594261667840 dbid=7 id=lock1870ff96880 mode=X associatedObjectId=...
      //   objectlock lockPartition=0 objid=1115203073 subresource=FULL dbid=7 id=... mode=X
      // Хэша ключа здесь НЕТ вовсе - он живёт только в dm_tran_locks. Значит дешёвый
      // опрос очереди отвечает "какая таблица и какой ключ", но не "какая запись".
      // mode здесь - режим ДЕРЖИМОЙ блокировки, а wait_type говорит, чего хочет жертва.
      Kind := '';
      CLEAR(HobtId);
      ObjectId := 0;
      DatabaseId := 0;
      HeldMode := '';

      Trimmed := DELCHR(ResourceText,'<>',' ');
      IF Trimmed = '' THEN
        EXIT(FALSE);

      SpacePos := STRPOS(Trimmed,' ');
      IF SpacePos = 0 THEN
        Kind := LOWERCASE(Trimmed)
      ELSE
        Kind := LOWERCASE(COPYSTR(Trimmed,1,SpacePos - 1));

      IF NOT EVALUATE(HobtId,PairValue(Trimmed,'hobtid')) THEN
        CLEAR(HobtId);
      IF NOT EVALUATE(ObjectId,PairValue(Trimmed,'objid')) THEN
        ObjectId := 0;
      IF NOT EVALUATE(DatabaseId,PairValue(Trimmed,'dbid')) THEN
        DatabaseId := 0;
      HeldMode := UPPERCASE(PairValue(Trimmed,'mode'));

      EXIT((Kind <> '') AND (DatabaseId > 0));
    END;

    [External]
    PROCEDURE TrimLockHash@2(ResourceDescription@1000 : Text) : Text;
    VAR
      Value@1001 : Text;
    BEGIN
      // dm_tran_locks отдаёт хэш ключа добитым пробелами до ширины столбца DMV.
      // Сравнение с параметром это переживает - SQL игнорирует хвостовые пробелы, -
      // а склейка хэша в текст запроса не переживает и молча не находит ничего.
      Value := DELCHR(ResourceDescription,'<>',' ');
      IF STRLEN(Value) < 3 THEN
        EXIT('');
      IF (COPYSTR(Value,1,1) <> '(') OR (COPYSTR(Value,STRLEN(Value),1) <> ')') THEN
        EXIT('');
      EXIT(Value);
    END;

    [External]
    PROCEDURE SplitSqlObjectName@3(SqlName@1000 : Text;VAR CompanyName@1001 : Text;VAR TableName@1002 : Text;VAR SiftKeyNo@1003 : Integer) : Boolean;
    VAR
      Value@1004 : Text;
      Marker@1005 : Text;
      Pos@1006 : Integer;
      Suffix@1007 : Text;
    BEGIN
      // Правил разбора два, а не одно, и одно правило теряет самый частый случай молча.
      // У таблицы номер ключа NAV - суффикс имени ИНДЕКСА (см. KeyNoFromIndexName).
      // У SIFT-представления индекс у всех один и тот же, VSIFTIDX, и номера не несёт:
      // номер там - число после ПОСЛЕДНЕГО $VSIFT$ в имени самого объекта.
      CompanyName := '';
      TableName := '';
      SiftKeyNo := UnknownKeyNo;

      Value := DELCHR(SqlName,'<>',' ');
      IF Value = '' THEN
        EXIT(FALSE);

      Marker := '$VSIFT$';
      Pos := LastPos(UPPERCASE(Value),Marker);
      IF Pos > 0 THEN BEGIN
        Suffix := COPYSTR(Value,Pos + STRLEN(Marker));
        IF NOT IsAllDigits(Suffix) THEN
          EXIT(FALSE);
        EVALUATE(SiftKeyNo,Suffix);
        Value := COPYSTR(Value,1,Pos - 1);
      END;

      // Платформенные $ndo$* пользовательскими таблицами не являются: их 19 на базу,
      // и разбирать в них нечего. Отдать их как таблицу компании "" было бы враньём.
      IF COPYSTR(LOWERCASE(Value),1,5) = '$ndo$' THEN
        EXIT(FALSE);

      Pos := STRPOS(Value,'$');
      IF Pos = 0 THEN
        // У таблиц с DataPerCompany=No префикса компании нет вовсе - это не ошибка.
        TableName := Value
      ELSE BEGIN
        CompanyName := COPYSTR(Value,1,Pos - 1);
        TableName := COPYSTR(Value,Pos + 1);
      END;

      EXIT(TableName <> '');
    END;

    [External]
    PROCEDURE KeyNoFromIndexName@4(IndexName@1000 : Text) : Integer;
    VAR
      Value@1001 : Text;
      Pos@1002 : Integer;
      Suffix@1003 : Text;
      KeyNo@1004 : Integer;
    BEGIN
      // Кластерный индекс зовётся <Компания>$<Таблица>$0, вторичные - просто $1, $5, $13.
      // Номера НЕ сплошные: у выключенного ключа индекса в SQL нет вовсе, и дыры в
      // нумерации - норма, а не признак поломки. index_id и номер ключа NAV - разные вещи.
      Value := DELCHR(IndexName,'<>',' ');
      Pos := LastPos(Value,'$');
      IF Pos = 0 THEN
        EXIT(UnknownKeyNo);
      Suffix := COPYSTR(Value,Pos + 1);
      IF NOT IsAllDigits(Suffix) THEN
        EXIT(UnknownKeyNo);
      EVALUATE(KeyNo,Suffix);
      EXIT(KeyNo);
    END;

    PROCEDURE EpisodeStart@5(ObservedAt@1000 : DateTime;WaitDurationMs@1001 : Integer) : DateTime;
    VAR
      Waited@1002 : Duration;
    BEGIN
      // Длительность из DMV точна, а период опроса плавает: сам проход в шторм стоит
      // миллисекунды. Считать начало по своим часам значит внести погрешность периода
      // в цифру, которую потом покажут заказчику. Считаем назад от момента наблюдения,
      // а сам момент берётся SYSDATETIME() тем же запросом - часами SQL, не NAV.
      IF WaitDurationMs <= 0 THEN
        EXIT(ObservedAt);
      Waited := WaitDurationMs;
      EXIT(ObservedAt - Waited);
    END;

    PROCEDURE IsNewEpisode@6(PrevDurationMs@1000 : Integer;NewDurationMs@1001 : Integer) : Boolean;
    BEGIN
      // Жертва может дождаться и тут же встать снова на тот же ресурс: сессия та же,

    [External]
    PROCEDURE Outcome@7(LastDurationMs@1000 : Integer;PeriodMs@1001 : Integer;SessionIsNav@1002 : Boolean) : Integer;
    BEGIN
      // Жертва исчезла из очереди - и дождавшаяся, и отвалившаяся исчезают одинаково.
      // Твёрдо судить можно только в двух случаях из трёх:
      //   наблюдали на пределе и дальше - отказ,
      //   до предела не хватило целого периода - дождался,
      //   зазор меньше периода - не определён, и врать про это нельзя.
      // Чужой сессии исход не приписываем вовсе: предел 10 000 мс - предел NAV,
      // внешнее подключение и задание обслуживания ждут сколько угодно.
      IF NOT SessionIsNav THEN
        EXIT(OutcomeUnknown);
      IF (LastDurationMs <= 0) OR (PeriodMs < 0) THEN
        EXIT(OutcomeUnknown);
      IF LastDurationMs >= NavLockTimeoutMs THEN
        EXIT(OutcomeTimedOut);
      IF LastDurationMs + PeriodMs < NavLockTimeoutMs THEN
        EXIT(OutcomeGranted);
      EXIT(OutcomeUnknown);
    END;

    [External]
    PROCEDURE ChainHead@8(VAR Waiter@1000 : ARRAY [500] OF Integer;VAR Blocker@1001 : ARRAY [500] OF Integer;Count@1002 : Integer;StartSession@1003 : Integer;VAR Depth@1004 : Integer;VAR Cycle@1005 : Boolean) : Integer;
    VAR
      Current@1006 : Integer;
      Next@1007 : Integer;
      Index@1008 : Integer;
      Steps@1009 : Integer;
    BEGIN
      // Очередь называет НЕПОСРЕДСТВЕННОГО блокирующего. В цепочке A -> B -> C у C
      // непосредственный виновник - B, который сам стоит за A. Написать в журнал B
      // значит назвать виновным пострадавшего, и назвать правдоподобно: он ведь и
      // правда держит то, чего ждёт C. По такому журналу чинят не то.
      // Голова - сессия, которой самой нет среди ждущих. Разматывается по уже
      // прочитанным строкам прохода: ни одного лишнего запроса к серверу.
      Depth := 0;
      Cycle := FALSE;
      Current := StartSession;
      IF (Count <= 0) OR (Count > MaxWaitersPerPass) THEN
        EXIT(Current);

      Steps := 0;
      REPEAT
        Next := 0;
        FOR Index := 1 TO Count DO
          IF (Next = 0) AND (Waiter[Index] = Current) THEN
            Next := Blocker[Index];
        IF Next <> 0 THEN BEGIN
          Current := Next;
          Depth := Depth + 1;
          Steps := Steps + 1;
          // Кольцо (A ждёт B, B ждёт A) - это взаимоблокировка на подходе либо уже
          // разобранная сервером. Обрываем: и по возврату в начало, и по потолку шагов,
          // потому что начало может вести В кольцо, не входя в него само.
          IF (Current = StartSession) OR (Steps > MaxChainSteps) THEN BEGIN
            Cycle := TRUE;
            Next := 0;
          END;
        END;
      UNTIL Next = 0;
      EXIT(Current);
    END;

    [External]
    PROCEDURE Chains@9(VAR Waiter@1000 : ARRAY [500] OF Integer;VAR Blocker@1001 : ARRAY [500] OF Integer;Count@1002 : Integer;VAR Head@1003 : ARRAY [500] OF Integer;VAR Depth@1004 : ARRAY [500] OF Integer;VAR Ring@1005 : ARRAY [500] OF Boolean;VAR Behind@1006 : ARRAY [500] OF Integer);
    VAR
      Index@1007 : Integer;
      Other@1008 : Integer;
      StepDepth@1009 : Integer;
      Cycle@1010 : Boolean;
    BEGIN
      // Одна голова с сорока жертвами - это не сорок бед, а одна. Это число и есть честный
      // ответ на "кто пострадал", и по нему решают, что чинить первым.
      //
      // Считается ОДИН раз на проход, а не на строку, и это не вкусовщина. Голова каждой
      // строки стоит прохода по всем ждущим; посчитать "жертв за головой" отдельно для
      // каждой строки значило бы получить КУБ от числа ждущих - на пятистах это сотни
      // миллионов сравнений, то есть наблюдатель встал бы ровно тогда, когда он нужен.
      CLEAR(Head);
      CLEAR(Depth);
      CLEAR(Ring);
      CLEAR(Behind);
      IF (Count <= 0) OR (Count > MaxWaitersPerPass) THEN
        EXIT;
      FOR Index := 1 TO Count DO BEGIN
        Head[Index] := ChainHead(Waiter,Blocker,Count,Waiter[Index],StepDepth,Cycle);
        Depth[Index] := StepDepth;
        Ring[Index] := Cycle;
      END;
      FOR Index := 1 TO Count DO
        FOR Other := 1 TO Count DO
          IF Head[Other] = Head[Index] THEN
            Behind[Index] := Behind[Index] + 1;
    END;

    [External]
    PROCEDURE EpisodeClass@22(CulpritIsSleeping@1000 : Boolean;CulpritHasRequest@1001 : Boolean;CulpritIdleMs@1002 : Integer;EscalationsInWindow@1003 : Integer) : Integer;
    BEGIN
      // Классы лечатся по-разному, поэтому и подписываются отдельно:
      //   эскалация - размером пачки, а не индексом,
      //   забытая транзакция - звонком, а не правкой кода,
      //   обычный спор за строку - кодом или ключом.
      // Эскалация старше сна: она останавливает всех разом, и она же чаще причина,
      // а не следствие. Взаимоблокировку сюда не выводим - её называет разбор графа.
      IF EscalationsInWindow > 0 THEN
        EXIT(ClassEscalation);
      IF CulpritIsSleeping AND (NOT CulpritHasRequest) AND (CulpritIdleMs >= IdleInTranMs) THEN
        EXIT(ClassSleepingTran);
      EXIT(ClassPlain);
    END;

    [External]
    PROCEDURE Advise@51(EpisodeClassNo@1000 : Integer;OutcomeNo@1001 : Integer;MaxWaitMs@1002 : Integer;AlertMs@1003 : Integer;VictimsBehindHead@1004 : Integer;OnSift@1005 : Boolean;NavKeyNo@1006 : Integer;HeadIsForeign@1007 : Boolean;VAR AdviceText@1008 : Text;VAR Evidence@1009 : Text) : Integer;
    BEGIN
      // Совет - это не пересказ строки, а ответ на вопрос "что с этим делать". Отсюда и
      // порядок: беда, у которой есть СВОЁ лечение, старше беды, про которую известна

      // потом три признака устройства, и только в конце длительность как таковая.
      //
      // Улика рядом с советом обязательна. Совет без числа, по которому он выдан, - это
      // мнение: его нечем оспорить и нечем проверить. По улике видно, на чём именно
      // совет держится, и она же ловит совет, выданный не по тому признаку.
      AdviceText := '';
      Evidence := '';

      IF EpisodeClassNo = ClassDeadlock THEN BEGIN
        AdviceText := AdvDeadlockTxt;
        Evidence := STRSUBSTNO(EvDeadlockTxt,MaxWaitMs);
        EXIT(AdviceDeadlock);
      END;

      IF EpisodeClassNo = ClassEscalation THEN BEGIN
        AdviceText := AdvEscalationTxt;
        Evidence := STRSUBSTNO(EvEscalationTxt,MaxWaitMs);
        EXIT(AdviceEscalation);
      END;

      IF EpisodeClassNo = ClassSleepingTran THEN BEGIN
        AdviceText := AdvSleepingTxt;
        Evidence := STRSUBSTNO(EvSleepingTxt,MaxWaitMs);
        EXIT(AdviceSleeping);
      END;

      // Чужая голова старше и агрегата, и очереди: пока цепочку держит не сессия NAV,
      // любой совет про ключи и пачки адресован не туда.
      IF HeadIsForeign THEN BEGIN
        AdviceText := AdvForeignTxt;
        Evidence := STRSUBSTNO(EvForeignTxt,MaxWaitMs);
        EXIT(AdviceForeignHead);
      END;

      IF OnSift THEN BEGIN
        AdviceText := AdvSiftTxt;
        Evidence := STRSUBSTNO(EvSiftTxt,NavKeyNo);
        EXIT(AdviceSift);
      END;

      IF VictimsBehindHead >= AdviceQueueWaiters THEN BEGIN
        AdviceText := AdvQueueTxt;
        Evidence := STRSUBSTNO(EvQueueTxt,VictimsBehindHead);
        EXIT(AdviceQueue);
      END;

      IF OutcomeNo = OutcomeTimedOut THEN BEGIN
        AdviceText := AdvTimedOutTxt;
        Evidence := STRSUBSTNO(EvTimedOutTxt,MaxWaitMs);
        EXIT(AdviceTimedOut);
      END;

      // Ноль в пороге означает "не тревожить вовсе", а не "тревожить всегда". Разница не
      // теоретическая: без этой оговорки выключенная тревога дала бы совет КАЖДОМУ
      // эпизоду, и советчик стал бы шумом, который перестают читать.
      IF (AlertMs > 0) AND (MaxWaitMs >= AlertMs) THEN BEGIN
        AdviceText := AdvLongWaitTxt;
        Evidence := STRSUBSTNO(EvLongWaitTxt,MaxWaitMs,AlertMs);
        EXIT(AdviceLongWait);
      END;

      EXIT(AdviceNone);
    END;

    [External]
    PROCEDURE TextFingerprint@23(Value@1000 : Text) : Code[20];
    VAR
      Hash@1001 : BigInteger;
      Index@1002 : Integer;
      CharCode@1003 : Integer;
      Length@1004 : Integer;
    BEGIN
      // Один и тот же граф взаимоблокировки виден при КАЖДОМ чтении кольцевого буфера,
      // пока не вытеснится. Дедупликация идёт по времени события плюс этому отпечатку,
      // а не по месту в буфере: место меняется при каждом чтении, граф - нет.
      Length := STRLEN(Value);
      IF Length = 0 THEN
        EXIT('');
      Hash := 0;
      FOR Index := 1 TO Length DO BEGIN
        CharCode := Value[Index];
        Hash := (Hash * FingerprintFactor + CharCode) MOD FingerprintModulus;
      END;
      EXIT(FORMAT(Hash));
    END;

    [External]
    PROCEDURE MsPerDay@40() : Decimal;
    BEGIN
      // Суток в миллисекундах. Тип здесь важнее числа: он ОБЯЗАН быть дробным. Тридцать
      // суток - это 2 592 000 000, что больше предельного Integer, и та же величина,
      // объявленная целой, дала бы переполнение на самом обычном значении настройки.
      EXIT(86400000);
    END;

    [External]
    PROCEDURE OutcomeDeadlocked@37() : Integer;
    BEGIN
      // Исход жертвы взаимоблокировки известен точно: сервер выбрал её сам и откатил.
      // Оставить здесь "не определён" значило бы сказать "не знаю" про то, что знаем.
      EXIT(3);
    END;

    [External]
    PROCEDURE MaxDeadlocksPerPass@38() : Integer;
    BEGIN
      // Потолок на одно чтение буфера. Двадцать графов - это уже сломанная система, а
      // остаток никуда не денется: отметка "прочитано до" двигается на последний взятый,
      // и следующее чтение продолжит с него.
      EXIT(20);
    END;

    [External]
    PROCEDURE DeadlockReadPeriodMs@39() : Integer;
    BEGIN
      // Минута. Буфер растёт не от наших блокировок, а от чужих взаимоблокировок, и на
      // тихом сервере не меняется вовсе - читать его каждым проходом значит платить за
      // разбор многомегабайтного XML двадцать раз в минуту впустую. Верхнюю границу
      // ставит оборот буфера: события старше вытесняются, и слишком редкое чтение теряет
      // их молча. Потеря при этом ЗАМЕТНА - проход сверяет старейшее событие с отметкой.
      EXIT(60000);
    END;

    [External]
    PROCEDURE AlertLatchMs@54() : Integer;
    BEGIN
      // Пять минут. Окно защёлки: пока та же голова держит очередь, тревога одна. Меньше -
      // и одна забытая транзакция даст десяток тревог за время своей жизни; больше - и
      // вторая беда на той же голове, начавшаяся после того, как первую разобрали,
      // потеряется молча.
      EXIT(300000);
    END;

    [External]
    PROCEDURE MaxAlertsPerPass@55() : Integer;
    BEGIN
      // Потолок разбираемых на тревогу эпизодов за проход. В шторм их сотни, но тревог из
      // них выйдет единицы - защёлка сведёт их к головам, - а перебирать все сотни внутри
      // прохода незачем: голов в шторме меньше двадцати не бывает разве что в теории.
      EXIT(20);
    END;

    [External]
    PROCEDURE CounterDelta@52(Previous@1000 : BigInteger;Current@1001 : BigInteger;VAR WasReset@1002 : Boolean) : BigInteger;
    BEGIN
      // Накопительные счётчики сервера живут не вечно: они обнуляются, когда объект уходит
      // из кэша метаданных - при перезапуске, при переводе базы в offline, под давлением
      // памяти. Признак у обнуления ровно один: значение стало МЕНЬШЕ прежнего. Разность
      // в этом случае вышла бы отрицательной, и итог поехал бы назад - молча и правдоподобно.
      //
      // После обнуления за прирост принимается ВСЁ текущее значение: это всё, что сервер
      // насчитал с момента обнуления. Случившееся между прошлым чтением и самим обнулением
      // потеряно безвозвратно, и потому обнуления считаются отдельно - чтобы потеря была
      // видна, а не растворилась в сумме.
      WasReset := Current < Previous;
      IF WasReset THEN
        EXIT(Current);
      EXIT(Current - Previous);
    END;

    [External]
    PROCEDURE MaxCoverageRows@53() : Integer;
    BEGIN
      // Потолок строк накопительного слоя за проход. Берутся САМЫЕ ЖДУЩИЕ - запрос
      // упорядочен по счётчику миллисекунд, - и усечение хвоста счёту не вредит: строка,
      // выпавшая из потолка и вернувшаяся, даст прирост за всё время отсутствия разом.
      EXIT(500);
    END;

    [External]
    PROCEDURE SqlNamePattern@34(Name@1000 : Text) : Text;
    VAR
      Pattern@1001 : Text;
      Index@1002 : Integer;
      Ch@1003 : Char;
    BEGIN
      // Имя поля NAV в имя столбца SQL переводит не наше правило, а платформа, и правило
      // это нигде не объявлено: измерением видно, что точка и косая черта становятся
      // подчёркиванием, а скобки, запятая, двоеточие, минус и амперсанд остаются как есть.
      // Повторять чужое правило по памяти - значит однажды промахнуться молча.
      //
      // Поэтому имя переводится не в имя, а в ОБРАЗЕЦ для LIKE: всё, что не буква, не
      // цифра и не пробел, заменяется подчёркиванием, а подчёркивание в LIKE подходит к
      // любому одному знаку. Образец подходит и к замене, и к её отсутствию, и правило
      // платформы знать не требуется вовсе.
      //
      // Буква определяется тем, что у неё есть разный верхний и нижний регистр. Это
      // работает для кириллицы так же, как для латиницы, а список диапазонов - нет:
      // в именах столбцов заказчика кириллица встречается.
      FOR Index := 1 TO STRLEN(Name) DO BEGIN
        Ch := Name[Index];
        IF IsNameChar(Ch) THEN
          Pattern := Pattern + FORMAT(Ch)
        ELSE
          Pattern := Pattern + '_';
      END;
      EXIT(Pattern);
    END;

    [External]
    PROCEDURE SameSqlName@35(SqlName@1000 : Text;NavName@1001 : Text) : Boolean;
    VAR
      Index@1002 : Integer;
    BEGIN
      // Сравнение имени, снятого с SQL, с именем объекта NAV - по тому же правилу, что и
      // образец: расходиться они вправе только там, где в имени NAV стоит знак, который
      // платформа могла заменить. Длина при замене не меняется, поэтому разная длина -
      // сразу нет.
      IF STRLEN(SqlName) <> STRLEN(NavName) THEN
        EXIT(FALSE);
      IF SqlName = '' THEN
        EXIT(FALSE);
      FOR Index := 1 TO STRLEN(NavName) DO
        IF UPPERCASE(FORMAT(SqlName[Index])) <> UPPERCASE(FORMAT(NavName[Index])) THEN
          IF IsNameChar(NavName[Index]) THEN
            EXIT(FALSE);
      EXIT(TRUE);
    END;

    LOCAL PROCEDURE IsNameChar@36(Ch@1000 : Char) : Boolean;
    VAR
      Text@1001 : Text[1];
    BEGIN
      IF Ch = ' ' THEN
        EXIT(TRUE);
      IF (Ch >= '0') AND (Ch <= '9') THEN
        EXIT(TRUE);
      // Буква - это знак, у которого верхний и нижний регистр различаются. Проверка по
      // диапазонам латиницы потеряла бы кириллицу, а она в именах столбцов есть.
      Text := FORMAT(Ch);
      EXIT(UPPERCASE(Text) <> LOWERCASE(Text));
    END;

    LOCAL PROCEDURE FingerprintFactor@24() : Integer;
    BEGIN
      // Нечётный множитель полиномиального свёртывания. 131 - простое, взаимно простое
      // с модулем; конкретное значение неважно, важно, что оно НЕ МЕНЯЕТСЯ никогда:
      // отпечатки лежат в базе, и смена множителя развалит дедупликацию задним числом.
      EXIT(131);
    END;

    LOCAL PROCEDURE FingerprintModulus@25() : Integer;
    BEGIN
      // Простое число Мерсенна 2^31-1: наибольшее, что помещается в Integer C/AL,
      // поэтому промежуточное произведение считается в BigInteger, а результат - нет.
      EXIT(2147483647);
    END;

    LOCAL PROCEDURE PairValue@26(Source@1000 : Text;Name@1001 : Text) : Text;
    VAR
      Padded@1002 : Text;
      Needle@1003 : Text;
      Pos@1004 : Integer;
      Tail@1005 : Text;
      SpacePos@1006 : Integer;
    BEGIN
      // Имя ищется только на ГРАНИЦЕ слова. В описании очереди рядом живут id=lock...
      // и associatedObjectId=..., и поиск простой подстрокой возьмёт не то: "id=" нашлось
      // бы внутри "hobtid=". Отсюда ведущий пробел в игле и в стоге.
      Padded := ' ' + LOWERCASE(Source);
      Needle := ' ' + LOWERCASE(Name) + '=';
      Pos := STRPOS(Padded,Needle);
      IF Pos = 0 THEN
        EXIT('');
      Tail := COPYSTR(Source,Pos + STRLEN(Needle) - 1);
      SpacePos := STRPOS(Tail,' ');
      IF SpacePos > 0 THEN
        Tail := COPYSTR(Tail,1,SpacePos - 1);
      EXIT(Tail);
    END;

    LOCAL PROCEDURE LastPos@27(Source@1000 : Text;Sub@1001 : Text) : Integer;
    VAR
      Found@1002 : Integer;
      Cursor@1003 : Integer;
      Step@1004 : Integer;
    BEGIN
      // В C/AL нет поиска с конца: идём вперёд и запоминаем последнее совпадение.
      // Именно последнее: имя SIFT-представления может нести $VSIFT$ не один раз.
      IF (Source = '') OR (Sub = '') THEN
        EXIT(0);
      Found := 0;
      Cursor := 1;
      REPEAT
        Step := STRPOS(COPYSTR(Source,Cursor),Sub);
        IF Step > 0 THEN BEGIN
          Found := Cursor + Step - 1;
          Cursor := Found + 1;
        END;
      UNTIL (Step = 0) OR (Cursor > STRLEN(Source));
      EXIT(Found);
    END;

    LOCAL PROCEDURE IsAllDigits@28(Value@1000 : Text) : Boolean;
    BEGIN
      IF Value = '' THEN
        EXIT(FALSE);
      EXIT(DELCHR(Value,'=','0123456789') = '');
    END;

    BEGIN
    {
      Разбор того, что отдают DMV, и счёт эпизода. Ни одного обращения к базе: всё
      здесь - чистые функции над данными, и потому проверяются прогоном без стенда
      (Codeunit 110231). Обращения к SQL живут отдельно.

      Два места, где легко ошибиться молча, и оба стоили опыта на стенде:

      1. Описаний ресурса ДВА. В dm_os_waiting_tasks это "keylock hobtid=... dbid=...
         mode=X ...", и хэша ключа там нет вовсе. В dm_tran_locks - только хэш "(...)",
         а hobt лежит отдельным столбцом. Путать их значит искать запись там, где её
         не называют.

      2. Хэш считается ПО КОНКРЕТНОМУ ИНДЕКСУ. Одна и та же строка даёт разный хэш на
         кластерном и на вторичном индексе, а поиск по чужому хэшу возвращает пустой
         набор и никакой ошибки. Ветвление по index_id из hobt_id обязательно.

      Числа с именами собраны наверху отдельными функциями: у каждого рядом довод,
      откуда оно взялось. В настройке им не место - от среды они не зависят.

      Советчик живёт здесь же и по той же причине: совет - это чистая функция от того,
      что уже известно про эпизод, и проверяется он без базы. Порядок правил в нём
      важнее самих правил, и потому он проверяется отдельно: случай, где сходятся сразу
      несколько признаков, обязан дать тот совет, у которого лечение точнее.
    }
    END.
