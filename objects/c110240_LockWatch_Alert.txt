    [External]
    PROCEDURE BeginPass@1();
    BEGIN
      RaisedCount := 0;
      LatchedCount := 0;
      // Очередь писем обнуляется КАЖДЫМ проходом, даже когда канал выключен: иначе проход
      // унаследовал бы очередь прошлого и отправил бы вчерашние письма сегодня.
      MailPendingCount := 0;
    END;

    [External]
    PROCEDURE Raise@2(VAR Episode@1000000000 : Record 110231;ThresholdMs@1000000001 : Integer;ToEventLog@1000000002 : Boolean;SourceName@1000000003 : Text;ToMail@1000000006 : Boolean;NotifGroupID@1000000007 : Integer) : Boolean;
    VAR
      Now@1000000004 : DateTime;
      Latch@1000000005 : Duration;
    BEGIN
      // Защёлка ищется по ГОЛОВЕ цепочки, а не по эпизоду: одна забытая транзакция в шторм
      // держит сотни жертв, и уведомление на каждую - это мусор, который перестают читать.
      // Окно защёлки живёт в разборе функцией с доводом рядом.
      IF ThresholdMs <= 0 THEN
        EXIT(FALSE);
      IF Episode."Max Wait (ms)" < ThresholdMs THEN
        EXIT(FALSE);
      // Эпизод ПОКАЗА не уведомляет никого. Показ держит строку дольше порога уведомления
      // нарочно, и уведомление уходило бы наружу по нажатию кнопки "Демо" - то есть ровно
      // то, после которого перестают читать настоящие.
      IF OnDeliberateShow(Episode) THEN
        EXIT(FALSE);
      Now := CURRENTDATETIME;
      Latch := Parse.AlertLatchMs;

      Alert.RESET;
      Alert.SETCURRENTKEY("Head SPID","Last Seen At");
      Alert.SETRANGE("Head SPID",Episode."Head SPID");
      Alert.SETFILTER("Last Seen At",'>=%1',Now - Latch);
      IF Alert.FINDLAST THEN BEGIN
        // Та же голова в том же окне - уведомление уже создано. Строка не заводится
        // заново, а растёт: число эпизодов за защёлкой и есть мера того, сколько народу
        // задето.
        Alert."Last Seen At" := Now;
        Alert.Episodes := Alert.Episodes + 1;
        IF Episode."Max Wait (ms)" > Alert."Max Wait (ms)" THEN
          Alert."Max Wait (ms)" := Episode."Max Wait (ms)";
        Alert.MODIFY;
        LatchedCount := LatchedCount + 1;
        EXIT(FALSE);
      END;

      Alert.RESET;
      Alert.INIT;
      Alert."Entry No." := NextNo;
      Alert."Raised At" := Now;
      Alert."Last Seen At" := Now;
      Alert."Head SPID" := Episode."Head SPID";
      Alert."NAV Table Name" := Episode."NAV Table Name";
      Alert."NAV Key No." := Episode."NAV Key No.";
      Alert.Episodes := 1;
      Alert."Max Wait (ms)" := Episode."Max Wait (ms)";
      Alert."Threshold (ms)" := ThresholdMs;
      Alert."Document No." := Episode."Document No.";
      Alert."User Name" := Episode."User Name";
      Alert."Episode Entry No." := Episode."Entry No.";
      Alert.Advice := Episode.Advice;
      Alert.Channel := Alert.Channel::JournalOnly;
      // Канал у одного прохода один, и выбор сделан вызывающим. Заметка ложится тем же
      // INSERT, что и строка: лишней записи не будет.
      //
      // При канале журнала событий заметка здесь не ставится вовсе - её перепишет
      // Announce после вставки, своим ответом. Письмо без группы - не ошибка и не повод
      // звать подписчика: письма не будет, и строка говорит об этом сама.
      IF NOT ToEventLog THEN
        IF ToMail AND (NotifGroupID > 0) THEN
          Alert."Channel Note" := COPYSTR(PendingTxt,1,MAXSTRLEN(Alert."Channel Note"))
        ELSE
          IF ToMail THEN
            Alert."Channel Note" := COPYSTR(NoGroupTxt,1,MAXSTRLEN(Alert."Channel Note"))
          ELSE
            Alert."Channel Note" := COPYSTR(OffTxt,1,MAXSTRLEN(Alert."Channel Note"));
      Alert.INSERT;
      // Наружу - ТОЛЬКО ПОСЛЕ строки. Отказ вставки (повтор номера, отказ прав) иначе
      // оставил бы в журнале событий Windows сообщение о беде, которой в журнале
      // уведомлений нет, и разбираться пришлось бы с внешним журналом, которого наш не
      // подтверждает. По той же причине и письмо берётся в очередь только после INSERT:
      // отправить то, чего в журнале нет, значило бы разойтись с ним молча.
      IF ToMail AND (NOT ToEventLog) AND (NotifGroupID > 0) THEN
        AddPendingMail(Alert."Entry No.",NotifGroupID,Episode."Started At",Episode."Culprit NAV Sessions");
      IF ToEventLog THEN BEGIN
        Announce(Alert,Episode,SourceName);
        Alert.MODIFY;
      END;
      RaisedCount := RaisedCount + 1;
      EXIT(TRUE);
    END;

    LOCAL PROCEDURE OnDeliberateShow@8(VAR Episode@1000000000 : Record 110231) : Boolean;
    VAR
      DemoRef@1000000001 : RecordRef;
    BEGIN
      // Показ по кнопке "Демо" создаёт НАСТОЯЩЕЕ ожидание на СВОЕЙ строке - и держит её
      // дольше порога уведомления: шесть секунд против пяти по умолчанию. Уведомление по
      // нему звало бы человека на нажатие кнопки, и это ровно тот случай, после которого
      // настоящие уведомления перестают читать.
      //
      // Поэтому эпизод показа не уведомляет ВООБЩЕ, а не уходит в журнал без канала:
      // пометить строку уведомления как "это был показ" нечем - такого поля в ней нет, - и
      // оставленная строка читалась бы потом как настоящая беда.
      //
      // Отметка тут - ТАБЛИЦА мишени, и она наша: показ спорит за строку своей таблицы, а
      // имя её берётся у самой таблицы, а не набирается строкой. Переименуют объект -
      // переедет и имя, и признак показа поедет вместе с ним.
      DemoRef.OPEN(DATABASE::"LockWatch Demo Row");
      EXIT(Parse.SameSqlName(Episode."NAV Table Name",DemoRef.NAME));
    END;

    [Integration]

    LOCAL PROCEDURE Announce@5(VAR Row@1000000000 : Record 110236;VAR Episode@1000000001 : Record 110231;SourceName@1000000002 : Text);
    VAR
      Message@1000000003 : Text;
    BEGIN
      // Канал наружу может не сработать, и молчащий канал, о котором никто не знает, хуже
      // отсутствующего. Поэтому ответ канала пишется В СТРОКУ, а не в никуда, и строка
      // остаётся в журнале уведомлений в любом случае.
      Message := BuildMessage(Row,Episode."Started At",Episode."Culprit NAV Sessions");
      IF NOT TryWriteEventLog(SourceName,Message) THEN BEGIN
        Row."Channel Note" := COPYSTR(STRSUBSTNO(WriteFailedTxt,GETLASTERRORTEXT),1,MAXSTRLEN(Row."Channel Note"));
        EXIT;
      END;
      Row.Channel := Row.Channel::EventLog;
      Row."Channel Note" := COPYSTR(STRSUBSTNO(WrittenTxt,SourceName),1,MAXSTRLEN(Row."Channel Note"));
    END;

    [TryFunction]
    LOCAL PROCEDURE TryWriteEventLog@6(SourceName@1000000000 : Text;Message@1000000001 : Text);
    VAR
      EventLog@1000000002 : DotNet "'System, Version=4.0.0.0, Culture=neutral, PublicKeyToken=b77a5c561934e089'.System.Diagnostics.EventLog";
      EntryType@1000000003 : DotNet "'System, Version=4.0.0.0, Culture=neutral, PublicKeyToken=b77a5c561934e089'.System.Diagnostics.EventLogEntryType";
    BEGIN
      // Источник событий заводит АДМИНИСТРАТОР, и заводится он один раз на машину. Писать
      // в чужой источник - скажем, в источник самой службы NAV - нельзя: мониторинг
      // припишет нашу запись платформе, и разбираться будут не с тем.
      //
      // Создавать источник отсюда мы не пробуем вовсе. Создание требует прав, которых у
      // службы нет и быть не должно, а неудачная попытка выглядит в журнале приложений
      // отказом безопасности - шумом, за который отвечать нам.
      IF NOT EventLog.SourceExists(SourceName) THEN
        ERROR(NoSourceTxt,SourceName);
      EntryType := EntryType.Warning;
      EventLog.WriteEntry(SourceName,Message,EntryType);
    END;

    LOCAL PROCEDURE AddPendingMail@12(EntryNo@1000000000 : Integer;GroupId@1000000001 : Integer;StartedAt@1000000002 : DateTime;Sessions@1000000003 : Text);
    BEGIN
      // Очередь хранит номер строки и всё, что письму понадобится ПОСЛЕ фиксации прохода.
      // Копируется здесь нарочно: к тому мигу эпизод может уехать в историю или быть
      // выброшен чисткой, а строка уведомления останется.
      //
      // Потолок массива равен потолку разбора, поэтому проверка здесь - страховка от
      // прямого вызова, а не рабочая ветка: без неё выход за границу массива уронил бы
      // проход. Выброшенное письмо не отправляется, но строка остаётся честной - в ней
      // написано, что она ждёт отправки, а доставка эту пометку не перепишет.
      IF MailPendingCount >= MaxPendingMails THEN
        EXIT;
      MailPendingCount := MailPendingCount + 1;
      MailEntryNo[MailPendingCount] := EntryNo;
      MailGroups[MailPendingCount] := GroupId;
      MailStartedAt[MailPendingCount] := StartedAt;
      MailSessions[MailPendingCount] := Sessions;
    END;

    LOCAL PROCEDURE MaxPendingMails@13() : Integer;
    BEGIN
      // Двадцать - тот же потолок, что MaxAlertsPerPass в разборе: больше строк один проход
      // не заводит. Число повторено здесь потому, что размер массива объявляется
      // константой времени компиляции, а имя у предела должно быть одно: при росте
      // потолка разбора правка будет в двух соседних местах, и забыть второе не даст эта
      // функция.
      EXIT(20);
    END;

    [External]
    PROCEDURE DeliverPending@9();
    VAR
      Row@1000000000 : Record 110236;
      Index@1000000001 : Integer;
      Subject@1000000002 : Text;
      Body@1000000003 : Text;
      ErrorText@1000000006 : Text;
      Note@1000000007 : Text;
      Handled@1000000004 : Boolean;
      Sent@1000000005 : Boolean;
      Changed@1000000008 : Boolean;
    BEGIN
      IF MailPendingCount = 0 THEN
        EXIT;
      FOR Index := 1 TO MailPendingCount DO BEGIN
        // Строку могли убрать со страницы уведомлений между фиксацией прохода и этим
        // вызовом. Письма без строки нет: журнал - запись о том, что ушло, и отправлять
        // то, чего в нём уже нет, значило бы разойтись с ним молча.
        IF Row.GET(MailEntryNo[Index]) THEN BEGIN
          Subject := STRSUBSTNO(MailSubjectTxt,Row."Max Wait (ms)",Row."NAV Table Name");
          Body := BuildMessage(Row,MailStartedAt[Index],MailSessions[Index]);
          Handled := FALSE;
          Sent := FALSE;
          ErrorText := '';
          IF TrySendMail(MailGroups[Index],Subject,Body,Handled,Sent,ErrorText) THEN BEGIN
            IF NOT Handled THEN
              Note := NoHandlerTxt
            ELSE
              IF NOT Sent THEN BEGIN
                Note := ErrorText;
                IF Note = '' THEN
                  Note := NotSentTxt;
              END ELSE BEGIN
                Row.Channel := Row.Channel::Mail;
                Note := SentTxt;
              END;
          END ELSE
            // Ошибка подписчика поймана TryFunction: строка прохода уже зафиксирована, и
            // откатывать её нечем и незачем. Причина остаётся в заметке словами.
            Note := GETLASTERRORTEXT;
          Row."Channel Note" := COPYSTR(Note,1,MAXSTRLEN(Row."Channel Note"));
          Row.MODIFY;
          Changed := TRUE;
        END;
      END;
      MailPendingCount := 0;
      // Второй COMMIT прохода - и только в том проходе, где строки ждали письма. Почта -
      // внешний вызов на секунды: держать её под транзакцией наблюдения нельзя, а отказ
      // SMTP не должен стоить журнала. Время прохода к этому мигу уже посчитано и
      // записано (Codeunit 110235), поэтому ожидание почты в Last Pass (ms) не попадает
      // и наблюдение из-за чужого SMTP не разрежается.
      IF Changed THEN
        COMMIT;
    END;

    [TryFunction]
    LOCAL PROCEDURE TrySendMail@10(NotifGroupID@1000000000 : Integer;Subject@1000000001 : Text;Body@1000000002 : Text;VAR Handled@1000000003 : Boolean;VAR Sent@1000000004 : Boolean;VAR ErrorText@1000000005 : Text);
    BEGIN
      // Только вызов события. Ни записи, ни COMMIT здесь нет: ошибка подписчика не
      // откатывает строки прохода, а наш перехват не должен сам ничего писать - иначе
      // отказ SMTP оставлял бы за собой половину заметки.
      OnSendAlertMail(NotifGroupID,Subject,Body,Handled,Sent,ErrorText);
    END;

    LOCAL PROCEDURE BuildMessage@11(VAR Row@1000000000 : Record 110236;StartedAt@1000000001 : DateTime;Sessions@1000000002 : Text) : Text;
    BEGIN
      // Один и тот же текст уходит и в журнал событий Windows, и в тело письма: два
      // сборщика разошлись бы при первой же правке, и каналы начали бы рассказывать
      // разное про одно событие. Время начала и сеансы виновника приходят снаружи: в
      // строке уведомления их нет, а письму они нужны.
      EXIT(STRSUBSTNO(EntryTxt,Row."Max Wait (ms)",Row."Threshold (ms)",Row."Head SPID",
                      Row."NAV Table Name",Row."NAV Key No.",Row."Document No.",
                      Row."User Name",FORMAT(StartedAt,0,0),Sessions,Row.Advice));
    END;

    BEGIN
    {
      Уведомление: порог из настройки наконец не только красит строку, но и зовёт человека.

      Пять решений, и каждое против одного и того же - против уведомления, которое
      перестают читать:

      1. Защёлка по ГОЛОВЕ цепочки, а не по эпизоду. Одна забытая транзакция в шторм
         держит сотни жертв; сотня строк уведомлений - это мусор. Пока голова та же и окно
         не вышло, строка одна, а число эпизодов за ней растёт полем.
      2. Уведомление только на ИДУЩИЙ эпизод. Смысл его - "происходит сейчас, вмешайся";
         звать человека к тому, что кончилось само, значит приучить его не читать.
      3. Ответ канала пишется в строку словами. Канал наружу может не сработать - нет
         источника событий, не выдали прав, не настроен SMTP, - и тогда уведомление всё
         равно остаётся здесь, а причина молчания видна. Молчащий канал, о котором никто
         не знает, хуже отсутствующего.
      4. Убирает уведомления ЧЕЛОВЕК кнопкой на странице уведомлений, и никто больше. Срез
         по сроку здесь был написан и снят: он удалял бы записи о разговоре с мониторингом
         по числу дней, которого никто не называл, и не звал его ни один проход.
      5. Показ по кнопке "Демо" уведомления не создаёт вовсе. Он держит строку СВОЕЙ
         таблицы дольше порога нарочно - шесть секунд против пяти по умолчанию, - и
         уведомление уходило бы наружу по нажатию кнопки. Признак показа - таблица мишени,
         взятая у самой таблицы: она наша, и спутать её с чужой нельзя.

      Наружу уведомление уходит в журнал событий Windows или почтой, и это выбор не от
      бедности. Своего канала в журнал событий у C/AL 2018 нет вовсе; почта требует
      настроенного SMTP, которого может не быть; а журнал событий Windows - ровно то, что
      мониторинг на сервере и без нас уже читает.

      Письмо LockWatch не отправляет сам: он поднимает событие, а отправляет подписчик
      установки. Так SMTP и его настройки остаются там, где им место, - у установки.

      И запись наружу идёт ПОСЛЕ вставки строки, а не до неё. Отказ вставки - повтор
      номера, отказ прав - оставил бы в чужом журнале сообщение о беде, которой в нашем
      журнале нет; разбирающийся искал бы её здесь и не нашёл.

      Письмо уходит ещё позже - после COMMIT прохода. Внешний вызов на секунды не должен
      держать транзакцию журнала, а отказ SMTP - стоить уже записанного прохода.
    }
    END.
