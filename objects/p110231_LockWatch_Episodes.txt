    LOCAL PROCEDURE ShowStatement@6(Blocker@1000000000 : Boolean);
    VAR
      Body@1000000001 : Text;
      Whose@1000000002 : Text;
    BEGIN
      Body := Watch.StatementText("Entry No.",Blocker);
      // Пустой текст объясняется словами, а не пустым окном. Причин у пустоты две, и обе
      // не ошибка: признак выключен, либо эпизод открылся раньше, чем его включили.
      IF Body = '' THEN
        ERROR(NoStatementErr);
      IF Blocker THEN
        Whose := STRSUBSTNO(BlockerTxt,"Blocker SPID")
      ELSE
        Whose := STRSUBSTNO(VictimTxt,"Victim SPID");
      MESSAGE(StatementMsg,Whose,Body);
    END;

    LOCAL PROCEDURE "------- Legal Hardcodes -------"@7();
    BEGIN
      // Разделитель, а не работа: ниже идут только величины, у которых есть имя.
      // Число в коде читается как случайное, а величины, зависящие от среды, живут
      // в настройке LockWatch Setup. Длина начала оператора к среде не относится.
    END;

    LOCAL PROCEDURE PreviewChars@8() : Integer;
    BEGIN
      // Сто знаков - это имя операции и первые поля. Дальше в операторе идёт то, что и
      // так видно по таблице и ключу, а колонка длиннее съела бы ширину у остальных.
      EXIT(100);
    END;

    BEGIN
    {
      Журнал эпизодов. Список, а не карточка: сравнивают здесь строки между собой, а не
      разглядывают одну.

      Порядок - по началу, сверху свежее. Отбор "только идущие" вынесен кнопкой, потому
      что вопрос "что происходит ПРЯМО СЕЙЧАС" задают первым и в спешке.

      Текст оператора показан прямо в списке началом, а целиком открывается по F4: за
      началом ходить в другой экран пришлось бы за каждой строкой, а тащить в список все
      шестнадцать тысяч знаков - значит утопить в них остальные колонки. Читаются блобы
      только у видимых строк, и пустая клетка объясняется словами при открытии.

      Колонки подписаны длинно и честно, и две подписи стоят того, чтобы их прочли:
      исход "не определён" - это ответ, а не пропуск; пустое имя пользователя - не сбой,
      а отсутствие права, которое никто не выдавал.
    }
    END.
