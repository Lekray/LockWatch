  CODE
  {
    VAR
      AlertMs@1000000000 : Integer;
      WaitStyle@1000000001 : Text[30];

    LOCAL PROCEDURE ReadAlertThreshold@1();
    VAR
      Setup@1000000000 : Record 110230;
    BEGIN
      // Порог читается ОДИН раз при открытии, а не на каждой строке: страница листает
      // тысячи строк, и чтение настройки на каждую было бы тысячей лишних обращений
      // ради числа, которое за время просмотра не меняется.
      Setup.FetchOnce;
      AlertMs := Setup."Alert Threshold (ms)";
    END;

    LOCAL PROCEDURE SetStyle@2();
    BEGIN
      // Порог ноль означает "не тревожить вовсе", и тогда не красится ничего: цвет не
      // должен появляться сам собой у того, кто тревогу выключил.
      WaitStyle := StandardStyle;
      IF (AlertMs > 0) AND ("Max Wait (ms)" >= AlertMs) THEN
        WaitStyle := AlarmStyle;
    END;

    LOCAL PROCEDURE StandardStyle@3() : Text[30];
    BEGIN
      EXIT('Standard');
    END;

    LOCAL PROCEDURE AlarmStyle@4() : Text[30];
    BEGIN
      // Имена стилей - платформенные, и написать их с ошибкой можно молча: неизвестный
      // стиль клиент рисует обычным, ничего не сказав. Поэтому они здесь по одному разу.
      EXIT('Unfavorable');
    END;

    BEGIN
    {
      Журнал эпизодов. Список, а не карточка: сравнивают здесь строки между собой, а не
      разглядывают одну.

      Порядок - по началу, сверху свежее. Отбор "только идущие" вынесен кнопкой, потому
      что вопрос "что происходит ПРЯМО СЕЙЧАС" задают первым и в спешке.

      Колонки подписаны длинно и честно, и три подписи стоят того, чтобы их прочли:
      голова цепочки - это не непосредственный блокирующий; исход "не определён" - это
      ответ, а не пропуск; пустое имя пользователя - не сбой, а отсутствие права, которое
      никто не выдавал.

      Совет и улика стоят РЯДОМ и разделены нарочно. Совет говорит, что делать, улика -
      по какому числу он выдан, и без второго первое было бы мнением. Пустой совет тоже
      ответ: там, где сказать нечего, вежливая общая фраза увела бы в сторону вернее,
      чем молчание.
    }
    END.
  }
