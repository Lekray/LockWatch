OBJECT Codeunit 110245 LockWatch Demo Hold
{
  OBJECT-PROPERTIES
  {
    Date=10.09.26;
    Time=12:00:00;
    Modified=Yes;
    Version List=LockWatch;
  }
  PROPERTIES
  {
    OnRun=BEGIN
            Hold;
          END;

  }
  CODE
  {
    VAR
      Demo@1000000000 : Record 110239;
      Control@1000000001 : Codeunit 110244;
      HeldTxt@1000000002 : TextConst 'ENU=Held by the demo;RUS=Держит показ';

    LOCAL PROCEDURE Hold@1();
    BEGIN
      // Держатель. Берёт строку показа в свою транзакцию и не отпускает её, пока не
      // выйдет время: замок живёт ровно столько, сколько живёт транзакция, и отпускается
      // он не строкой кода, а концом сессии.
      //
      // LOCKTABLE до чтения, а не после: прочитанное без него берётся без замка, и
      // MODIFY взял бы монопольный замок только на миг записи. Ждать за такой строкой
      // никто бы не успел.
      Demo.LOCKTABLE;
      // Строки нет - значит пульт до нас не дошёл. Заводить её здесь нельзя: две сессии,
      // заводящие одну и ту же строку наперегонки, дерутся уже за страницу индекса, и
      // показ показывал бы не то, что обещал.
      IF NOT Demo.GET(Control.TargetNo) THEN
        EXIT;
      Demo.Note := HeldTxt;
      Demo."Touched At" := CURRENTDATETIME;
      Demo.MODIFY;
      SLEEP(Control.HoldMs);
    END;

    BEGIN
    {
      Одна из двух сессий показа - та, что держит. Выключатель показа она НЕ спрашивает, и
      это не забывчивость: обещание "выключатель запрещает показ" держится в одном месте,
      у пульта. Продублируй его здесь - и проверка этого обещания станет пустой: снятый у
      пульта, выключатель сработал бы тут, и поломка прошла бы молча.
    }
    END.
  }
}
