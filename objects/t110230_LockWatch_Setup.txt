  CODE
  {
    VAR
      PeriodTooLongErr@1000000000 : TextConst 'ENU=The poll period must be shorter than the NAV lock timeout of %1 ms. An episode fits entirely inside that window, and a longer period would miss whole episodes without saying so.;RUS=Период опроса должен быть короче предела ожидания блокировки NAV в %1 мс. Эпизод целиком укладывается в это окно, и период длиннее терял бы эпизоды целиком, ничего об этом не сообщая.';

    [External]
    PROCEDURE FetchOnce@1();
    BEGIN
      // Настройка одна на базу. Заводится сама и с осмысленными значениями: пустая
      // запись превратила бы ловимое "записи нет" в TESTFIELD посреди фоновой задачи.
      IF GET THEN
        EXIT;
      INIT;
      "Primary Key" := '';
      Enabled := FALSE;
      "Poll Period (ms)" := DefaultPollPeriodMs;
      "Max Rows per Pass" := DefaultMaxRows;
      "Max Pass (ms)" := DefaultMaxPassMs;
      "Retention (Days)" := DefaultRetentionDays;
      "Alert Threshold (ms)" := DefaultAlertMs;
      "Collect Statement Values" := FALSE;
      "Deadlocks Enabled" := TRUE;
      "Coverage Enabled" := TRUE;
      // Имена от платформы по умолчанию ВЫКЛЮЧЕНЫ, и это не осторожность ради осторожности:
      // включение стоит правки конфига, перезапуска экземпляра, обоих прав SQL и постоянной
      // серверной сессии XE, переживающей снятие инструмента. Решение это заказчика.
      "Platform Names Enabled" := FALSE;
      "Alert Channel" := "Alert Channel"::Journal;
      "Alert Event Source" := DefaultEventSource;
      INSERT;
    END;

    LOCAL PROCEDURE CheckPollPeriod@2();
    VAR
      Parse@1000000000 : Codeunit 110230;
    BEGIN
      // Предел ожидания у NAV - 10 000 мс, и эпизод целиком укладывается в это окно.
      // Период, равный пределу или больше, означает журнал, который честно пуст на
      // сломанной системе. Это не настройка среды, это смысловая граница - в коде.
      IF "Poll Period (ms)" >= Parse.NavLockTimeoutMs THEN
        ERROR(PeriodTooLongErr,Parse.NavLockTimeoutMs);
    END;

    LOCAL PROCEDURE DefaultPollPeriodMs@3() : Integer;
    BEGIN
      // Три секунды. Заметно меньше предела в 10 000 мс, чтобы эпизод был увиден
      // хотя бы дважды, и заметно больше цены прохода (12 мс при двухстах ждущих),
      // чтобы наблюдение не стоило заметной доли ядра.
      EXIT(3000);
    END;

    LOCAL PROCEDURE DefaultMaxRows@4() : Integer;
    BEGIN
      // Совпадает с размерностью массивов разматывания цепочки в Codeunit 110230.
      // Больше пятисот ждущих - это не разбор, а шторм; проход метится "усечён".
      EXIT(500);
    END;

    LOCAL PROCEDURE DefaultMaxPassMs@5() : Integer;
    BEGIN
      // Проход обязан кончиться заведомо раньше следующего. Треть периода оставляет
      // запас и на NAV-половину, которая в шторм и есть самая дорогая часть.
      EXIT(1000);
    END;

    LOCAL PROCEDURE DefaultRetentionDays@6() : Integer;
    BEGIN
      // Срок объявленный, а не "сколько накопится": инструмент показывает чужие
      // учётные записи, и бессрочное хранение - отдельный разговор со службой
      // безопасности, которого мы не выиграем.
      EXIT(30);
    END;

    LOCAL PROCEDURE DefaultEventSource@8() : Text[50];
    BEGIN
      // Имя источника событий Windows. Заводит его администратор один раз на машину, и до
      // тех пор канал наружу молчит - но молчит СЛОВАМИ: причина пишется в строку тревоги.
      // Писать в чужой источник, скажем в источник самой службы NAV, нельзя: мониторинг
      // припишет запись платформе, и разбираться будут не с тем.
      EXIT('LockWatch');
    END;

    LOCAL PROCEDURE DefaultAlertMs@7() : Integer;
    BEGIN
      // Половина предела NAV. Эпизод, переваливший за пять секунд, уже заметен
      // человеку, но ещё не кончился отказом - тревога успевает быть полезной.
      EXIT(5000);
    END;

    BEGIN
    {
      Настройка. Здесь живёт только то, что зависит от СРЕДЫ: сервер SQL, период, потолки,
      срок хранения, порог тревоги, выключатели.

      Имя сервера SQL приходится спрашивать у человека, и это единственное имя, которое
      инструмент не может узнать сам: платформа отдаёт коду имя БАЗЫ (Active Session), а
      сервер в той же строке записан НАВовский, не серверный. Имя базы поэтому и не
      настраивается - ошибиться в нём значило бы наблюдать не за той базой. Смысловые границы - в коде функциями с
      доводом рядом, и предел ожидания NAV (10 000 мс) - одна из них: сюда он не
      попадает и настройкой не задаётся.

      Умолчания заведены не нулями. Пустая запись настройки переводит ловимое "записи
      нет" в TESTFIELD посреди фоновой задачи, а необработанное исключение на старте
      сессии роняет клиент насмерть, без окна с текстом.

      Галка сбора значений в операторах выключена нарочно и по умолчанию: для службы
      безопасности это выгрузка данных, а не структура, и согласование проходится
      один раз.
    }
    END.
