    [External]
    PROCEDURE FetchOnce@1();
    BEGIN
      // Чтение БЕЗ замка - им проход начинается. Замок здесь стоил бы дорого: держать его
      // пришлось бы через съём очереди, то есть через обращение к SQL, и кнопка "стоп"
      // ждала бы ровно столько, сколько ждёт сервер.
      IF GET THEN
        EXIT;
      INIT;
      "Primary Key" := '';
      INSERT;
    END;

    [External]
    PROCEDURE FetchToWrite@2();
    BEGIN
      // Перед записью строка перечитывается ПОД ЗАМКОМ. NAV сверяет версию строки при
      // записи, и без этого проход, начавшийся до нажатия "стоп", падал бы на своей же
      // последней строке - с сообщением про обновлённую страницу, которого никто не ждёт
      // от фоновой задачи. Замок держится до COMMIT, а он тут же следом.
      LOCKTABLE;
      IF GET THEN
        EXIT;
      INIT;
      "Primary Key" := '';
      INSERT;
    END;

    [External]
    PROCEDURE Say@3(Message@1000000000 : Text);
    BEGIN
      // Только слово, без чисел прохода: завод, остановка и падение задачи говорят о
      // СЕБЕ, а числа последнего прохода при этом остаются прежними - они не устарели.
      FetchToWrite;
      "Watchdog Message" := COPYSTR(Message,1,MAXSTRLEN("Watchdog Message"));
      MODIFY;
    END;

    [External]
    PROCEDURE Arm@6() : Integer;
    BEGIN
      // Метка поколения цепочки. Поднимает её ТОЛЬКО завод, и по ней проход, начавшийся до
      // нового завода, узнаёт, что цепочка уже не его. Число, а не время: два завода подряд
      // укладываются в одну миллисекунду, и метка временем их бы не различила.
      FetchToWrite;
      "Watch Generation" := "Watch Generation" + 1;
      MODIFY;
      EXIT("Watch Generation");
    END;

    [External]
    PROCEDURE Generation@7() : Integer;
    BEGIN
      // Строки ещё нет - значит и цепочки не было: ноль отличается от любого заведённого
      // поколения, и проход с нулём себя не продолжит. Заводить строку чтением нельзя.
      IF GET THEN
        EXIT("Watch Generation");
      EXIT(0);
    END;

    END;

    [External]
    PROCEDURE Dropped@8(VAR Episodes@1000000000 : Integer;VAR WaitMs@1000000001 : BigInteger);
    BEGIN
      // Счёт выброшенного отдаётся ПАРОЙ и одним заходом в базу: число эпизодов и их
      // ожидание - две половины одного ответа, и спрошенные порознь они разошлись бы на
      // строке, изменившейся между двумя чтениями.
      FetchOnce;
      Episodes := "Dropped Episodes";
      WaitMs := "Dropped Wait (ms)";
    END;

    PROCEDURE CoverageSince@5() : DateTime;
    BEGIN
      // Экран охвата спрашивает отметку, но заводить строку состояния он не должен: она
      // появляется от прохода, а не от того, что кто-то открыл экран.
      IF GET THEN
        EXIT("Coverage Since");
      EXIT(0DT);
    END;

    BEGIN
    {
      Состояние сторожа: когда был последний проход, чего он стоил, сколько строк принёс,
      что имеет сказать и до какого мига прочитаны буфер взаимоблокировок и счётчики
      охвата. Строка одна на компанию, и пишет её ТОЛЬКО фон - кроме одной кнопки
      "начать счёт заново" на экране охвата.

      Отдельная таблица заведена не ради порядка, а по вычитанной цене. Всё это лежало в
      строке настройки - там же, где галки и числа, которые правит человек. NAV сверяет
      версию строки при записи, а проход переписывал её каждые три секунды: страница
      настройки устаревала почти сразу после открытия, и любая правка возвращала "мы
      только что обновили эту страницу". Ошибка выглядела как каприз платформы, а была
      нашей - фон и человек писали одну строку (FINDINGS, раздел 33).

      Номера полей нарочно те же, что были в настройке: то же поле, тот же номер. На
      странице настройки они остались на своих местах и показываются вычисляемыми полями
      (FlowField Lookup) - вычисляемое поле не хранится, версию строки не трогает, и
      строка человека больше не меняется сама по себе.

      Строку заводит сам инструмент первым же проходом. Скрипту, которому нужно тронуть
      состояние ДО первого прохода, придётся завести её запросом со ВСЕМИ столбцами:
      умолчаний NAV в SQL не создаёт, а объявляет всё NOT NULL.
    }
    END.
