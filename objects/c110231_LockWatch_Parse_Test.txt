    END;

    LOCAL PROCEDURE CheckChain@23(VAR Passed@1000000000 : Integer;VAR Total@1000000001 : Integer;VAR Report@1000000002 : Text);
    VAR
      Waiter@1000000003 : ARRAY [500] OF Integer;
      Blocker@1000000004 : ARRAY [500] OF Integer;
      Head@1000000005 : Integer;
      Depth@1000000006 : Integer;
      Cycle@1000000007 : Boolean;
    BEGIN
      // Цепочка 3 -> 2 -> 1. Непосредственный виновник у сеанса 3 - это 2, но 2 сам
      // стоит в очереди за 1. Назвать виновным 2 значит назвать виновным пострадавшего.
      Waiter[1] := 3; Blocker[1] := 2;
      Waiter[2] := 2; Blocker[2] := 1;
      Head := Parse.ChainHead(Waiter,Blocker,2,3,Depth,Cycle);
      Check(Passed,Total,Report,ChkChainTxt,
        (Head = 1) AND (Depth = 2) AND (NOT Cycle),
        STRSUBSTNO(ChainTxt,Head,Depth,Cycle));
    END;

    LOCAL PROCEDURE CheckChainRing@24(VAR Passed@1000000000 : Integer;VAR Total@1000000001 : Integer;VAR Report@1000000002 : Text);
    VAR
      Waiter@1000000003 : ARRAY [500] OF Integer;
      Blocker@1000000004 : ARRAY [500] OF Integer;
      Head@1000000005 : Integer;
      Depth@1000000006 : Integer;
      Cycle@1000000007 : Boolean;
    BEGIN
      // Кольцо 1 -> 2 -> 1. Это взаимоблокировка на подходе либо уже разобранная
      // сервером. Разматывание обязано оборваться и сказать об этом, а не зациклиться:
      // фоновая задача, ушедшая в вечный цикл, держит слот планировщика насмерть.
      Waiter[1] := 1; Blocker[1] := 2;
      Waiter[2] := 2; Blocker[2] := 1;
      Head := Parse.ChainHead(Waiter,Blocker,2,1,Depth,Cycle);
      Check(Passed,Total,Report,ChkCycleTxt,
        Cycle AND (Depth <= Parse.MaxChainSteps + 1),
        STRSUBSTNO(CycleTxt,Cycle,Depth));

      // И второе утверждение, отдельное: обрыв - это ещё не ответ. Головы в кольце нет, и
      // возвращать начало нельзя: голова идёт в колонку виновника, начало - в колонку
      // ждущего, и обе колонки называют одного и того же человека, пострадавшего.
      // На прежнем коде голова равнялась 1, то есть самой жертве, и проверка красная.
      Check(Passed,Total,Report,ChkRingHeadTxt,
        (Head = 2) AND (Depth = 1),
        STRSUBSTNO(RingHeadTxt,Head,2,Depth));
    END;

    LOCAL PROCEDURE CheckVictims@25(VAR Passed@1000000000 : Integer;VAR Total@1000000001 : Integer;VAR Report@1000000002 : Text);
    VAR
      Waiter@1000000003 : ARRAY [500] OF Integer;
      Blocker@1000000004 : ARRAY [500] OF Integer;
      Head@1000000005 : ARRAY [500] OF Integer;
      Depth@1000000006 : ARRAY [500] OF Integer;
      Ring@1000000007 : ARRAY [500] OF Boolean;
      Behind@1000000008 : ARRAY [500] OF Integer;
    BEGIN
      // Три ждущих за одной головой и четвёртый - за своей. Отчёт, считающий первых троих
      // тремя бедами, отправит чинить три операции вместо одной. Считается всё разом:
      // головы, глубины, кольца и жертвы за головой - один проход на пачку, а не по
      // проходу на строку.
      //
      // Чужая голова здесь не для полноты картины, а ради самой проверки. На наборе с
      // ОДНОЙ головой за ней стоят все, и счёт жертв неотличим от счёта ждущих вовсе:
      // Behind := Count прошёл бы такую проверку зелёным, а на бою слил бы в одну беду
      // все споры сервера разом. Различает их только вторая голова: за ней жертва одна.
      Waiter[1] := 3; Blocker[1] := 2;
      Waiter[2] := 2; Blocker[2] := 1;
      Waiter[3] := 4; Blocker[3] := 1;
      Waiter[4] := 6; Blocker[4] := 5;
      Parse.Chains(Waiter,Blocker,4,Head,Depth,Ring,Behind);
      Check(Passed,Total,Report,ChkVictimsTxt,
        (Head[1] = 1) AND (Behind[1] = 3) AND (Behind[3] = 3) AND
        (Head[4] = 5) AND (Behind[4] = 1),
        STRSUBSTNO(VictimsTxt,Behind[1],Behind[4]));
    END;

    LOCAL PROCEDURE CheckAdviceEvidence@60(VAR Passed@1000000000 : Integer;VAR Total@1000000001 : Integer;VAR Report@1000000002 : Text);

    LOCAL PROCEDURE CrLf@54() : Text[2];
    VAR
      Cr@1000000000 : Char;
      Lf@1000000001 : Char;
    BEGIN
      Cr := 13;
      Lf := 10;
      EXIT(FORMAT(Cr) + FORMAT(Lf));
    END;

    BEGIN
    {
      Обкатка разбора. Базы не касается вовсе: всё, что она проверяет, - чистые
      функции над данными, и потому её можно гонять где угодно и сколько угодно.

      Три правила, по которым она сделана:

      1. Судит сама. Первой строкой "пройдено N из M", и при любом провале выходит
         ERROR, а не MESSAGE: запуск через Invoke-NAVCodeunit тогда отличает целое от
         сломанного по коду возврата, а не по чтению глазами.

      2. Каждая проверка непуста. Проверяемое место ломается нарочно, и проверка
         обязана покраснеть. Проверки, которая проходит всегда, здесь быть не должно.

      3. Данные взяты с натуры, а не выдуманы. Строки описания ресурса сняты с живого
         ожидания на стенде 07.09.2026, и потому проверяют настоящий формат, а не
         представление о нём.

      Проверено поломкой 07.09.2026: со снятой границей слова в PairValue прогон дал
      19 из тогдашних 20, и покраснела именно проверка границы - hobt 111 вместо
      72057594261667840. Проверять надо ПОСЛЕ перезапуска экземпляра службы: у неё свой
      кэш объектов, и без перезапуска сломанный разбор тоже отвечает целым.

      Проверено поломкой ещё раз 11.09.2026: счёт жертв за головой подменён на число
      ждущих вовсе (Behind := Count) - 24 из 25, и покраснел именно он: "за общей головой
      4, за чужой 4". До этого дня в опыте была ОДНА голова, за ней стояли все, и подмена
      проходила зелёной - набор с одной головой не отличает счёт жертв от счёта ждущих.
      Отличает вторая голова: за ней жертва одна.

      Самая ценная из проверок - на границу слова: в описании очереди рядом стоят
      hobtid=, id=lock... и associatedObjectId=, и разбор простой подстрокой подменит
      одно другим МОЛЧА, выдав правдоподобные числа. Такое не ловится ни компиляцией,
      ни глазами - только проверкой.
    }
    END.
