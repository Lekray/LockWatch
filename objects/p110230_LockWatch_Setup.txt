    LOCAL PROCEDURE SetWatchState@5();
    BEGIN
      // Одно слово на все строки ниже: галка, число задач и здоровье цепочки по отдельности
      // правдивы, но ответа "идёт ли наблюдение" не дают ни одна из них. Порядок проверок -
      // от того, что чинится кнопкой, к тому, что видно только по времени.
      WatchStateStyle := AlarmStyle;
      CALCFIELDS("Last Pass At");
      CASE TRUE OF
        NOT Enabled:
          BEGIN
            WatchState := StateStoppedTxt;
            WatchStateStyle := StoppedStyle;
          END;
        ScheduledTasks = 0:
          WatchState := StateNoTaskTxt;
        ScheduledTasks > 1:
          WatchState := STRSUBSTNO(StateManyTasksTxt,ScheduledTasks);
        ("Last Pass At" = 0DT) AND (WatchSilent = ''):
          BEGIN
            WatchState := StateFirstPassTxt;
            WatchStateStyle := PendingStyle;
          END;
        WatchSilent <> '':
          WatchState := StateSilentTxt;
        ELSE BEGIN
          WatchState := StateRunningTxt;
          WatchStateStyle := RunningStyle;
        END;
      END;
    END;

    LOCAL PROCEDURE "------- Legal Hardcodes -------"@2();
    BEGIN
      // Разделитель, а не работа: ниже идут только величины, у которых есть имя.
      // Число в коде читается как случайное, а величины, зависящие от среды, живут
      // в настройке LockWatch Setup. Имена стилей показа со средой не связаны.
    END;

    LOCAL PROCEDURE StandardStyle@3() : Text[30];
    BEGIN
      // Имя платформенного стиля. Написать его с ошибкой можно молча: неизвестный
      // стиль клиент рисует обычным, ничего не сказав.
      EXIT('Standard');
    END;

    LOCAL PROCEDURE AlarmStyle@4() : Text[30];
    BEGIN
      // Тот же довод, что и у обычного стиля: имя платформенное, и ошибка в нём молчит.
      EXIT('Unfavorable');
    END;

    LOCAL PROCEDURE RunningStyle@6() : Text[30];
    BEGIN
      // Тот же довод: имя платформенное. Зелёный - единственный цвет, который значит "всё идёт".
      EXIT('Favorable');
    END;

    LOCAL PROCEDURE StoppedStyle@7() : Text[30];
    BEGIN
      // Остановленный сторож - не беда, а выбор человека: жирным, но не красным.
      EXIT('Strong');
    END;

    LOCAL PROCEDURE PendingStyle@8() : Text[30];
    BEGIN
      // Заведён, но первого прохода ещё нет: секунды после завода, а не поломка.
      EXIT('Ambiguous');
    END;

    BEGIN
    {
      Настройка и пульт. Кнопок шесть, и три из них - про то, чтобы человек мог убедиться
      своими глазами, а не поверить на слово: один проход сейчас, журнал, таблицы
      контекста.

      Число запланированных задач вынесено на экран нарочно. Ноль при включённом стороже
      означает, что цепочка оборвалась, и это единственный видимый признак: сама по себе
      строка настройки при этом выглядит совершенно здоровой.

      Одной задачи, однако, мало для спокойствия: при выключенном планировщике экземпляра
      она стоит в очереди и не исполняется НИ РАЗУ - замерено 12.09.2026. Поэтому рядом
      стоит здоровье цепочки: оно считает молчание в периодах опроса и называет его словами.

      Всё, что пишет сюда сам инструмент, стоит только для чтения.
    }
    END.
  }
