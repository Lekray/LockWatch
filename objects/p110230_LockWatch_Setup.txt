    LOCAL PROCEDURE CheckPlatformRoad@1();
    BEGIN
      // Сам ответ и его запись в строку настройки живут в разборе дороги, а не здесь:
      // кнопка на странице и вызов снаружи обязаны делать РОВНО ОДНО И ТО ЖЕ, иначе
      // проверенным окажется не то, чем пользуются.
      Platform.CheckRoad;
      CurrPage.UPDATE(FALSE);
    END;

    BEGIN
    {
      Настройка и пульт. Кнопок пять, и три из них - про то, чтобы человек мог убедиться
      своими глазами, а не поверить на слово: один проход сейчас, журнал, таблицы контекста.

      Число запланированных задач вынесено на экран нарочно. Ноль при включённом стороже
      означает, что цепочка оборвалась, и это единственный видимый признак: сама по себе
      строка настройки при этом выглядит совершенно здоровой.
    }
    END.
