OBJECT Codeunit 110235 LockWatch Hold Bench
{
  OBJECT-PROPERTIES
  {
    Date=19.10.26;
    Time=12:00:00;
    Modified=Yes;
    Version List=LockWatch;
  }
  PROPERTIES
  {
  }
  CODE
  {
    VAR
      HeldTxt@1000000000 : TextConst 'ENU=LOCKHOLD user %1 session %2 ms %3 ownrows %4 laterows %5;RUS=LOCKHOLD user %1 session %2 ms %3 ownrows %4 laterows %5';
      NoRowErr@1000000001 : TextConst 'ENU=The bench row could not be put down: %1. A row with the bench number is already in the context table and must not survive a run - remove it before repeating the experiment.;RUS=Строку-носитель не удалось положить: %1. Строка с мерным номером уже есть в таблице контекста и переживать прогон не должна - уберите её перед повтором опыта.';

    [External]
    PROCEDURE Hold@1();
    VAR
      Mark@1000000002 : Record 110232;
      Locks@1000000003 : Record 2000000154;
      Waited@1000000004 : Integer;
      OwnRows@1000000005 : Integer;
      LateRows@1000000006 : Integer;
    BEGIN
      // Блокировку держит СЕССИЯ NAV, и в этом весь смысл. Штатная таблица блокировок
      // показывает только те транзакции, что платформа завела в свою карту, а туда
      // попадают лишь сессии NAV этого экземпляра. Соединение sqlcmd в ней не появится
      // никогда, и проверять дорогу к имени человека им нельзя вовсе.
      //
      // Носитель блокировки - строка таблицы контекста. Свой ключ у неё ручной, и мерный
      // номер настоящей таблице не принадлежит: столкнуться с настройкой заказчика неоткуда.
      Mark.INIT;
      Mark."Table No." := BenchTableNo;
      Mark."Table Name" := BenchTableName;
      Mark."Enabled" := FALSE;
      IF NOT Mark.INSERT THEN
        ERROR(NoRowErr,GETLASTERRORTEXT);
      // COMMIT здесь не делается нарочно: транзакция держится открытой, и вместе с ней
      // держится блокировка. Строка живёт только внутри этой транзакции - в конце вызова
      // её снимает DELETE, и в настройке заказчика от опыта не остаётся ничего.
      //
      // Штатная таблица читается ОТСЮДА, изнутри собственной открытой транзакции. Это и
      // есть решающий опыт: если держатель не видит даже СВОЕЙ блокировки, значит карта
      // транзакций пуста вовсе; если видит свою, а сторонний наблюдатель не видит ничего,
      // значит карта у каждой сессии своя, и дорога к чужому имени по ней не проходит.
      OwnRows := Locks.COUNT;
      Waited := HoldMs;
      SLEEP(Waited DIV SleepHalves);
      // Второй счёт - когда ждущий уже стоит в очереди. Первый мог оказаться слишком
      // ранним: карта наполняется на старте транзакции, а не на взятии блокировки.
      Locks.RESET;
      LateRows := Locks.COUNT;
      SLEEP(Waited DIV SleepHalves);
      MESSAGE(HeldTxt,USERID,SESSIONID,Waited,OwnRows,LateRows);
      // Строка-носитель уходит в той же транзакции. Блокировка при этом держится до её
      // конца - то есть до возврата из вызова, - а в таблице контекста не остаётся строки.
      Mark.DELETE;
    END;

    LOCAL PROCEDURE "------- Legal Hardcodes -------"@2();
    BEGIN
      // Разделитель, а не работа: ниже идут только величины, у которых есть имя.
      // Мерный объект не настраивается - его числа описывают поставленный опыт, а не
      // среду установки, - поэтому они лежат здесь, а не в настройке LockWatch Setup.
    END;

    LOCAL PROCEDURE HoldMs@3() : Integer;
    BEGIN
      // Минута. Больше предела ожидания NAV (10 000 мс) - чтобы жертва успела отвалиться
      // по нему и эпизод получил честный исход, - и заведомо больше, чем нужно прогону на
      // проход и нажатие кнопки. Меньше минуты не годится: на холодной службе первый
      // проход платит за сборку business assemblies, и окно съедается целиком.
      EXIT(60000);
    END;

    LOCAL PROCEDURE BenchTableNo@4() : Integer;
    BEGIN
      // Потолок целого NAV. Настоящей таблицы с таким номером нет и быть не может, а
      // отрицательный номер таблица контекста не примет: у поля стоит MinValue=1.
      EXIT(2147483647);
    END;

    LOCAL PROCEDURE BenchTableName@5() : Text;
    BEGIN
      // Метка строки-носителя. Нужна человеку, который застал опыт на середине: по ней
      // видно, откуда взялась строка в таблице контекста, если прогон оборвался.
      EXIT('LOCKWATCH-BENCH');
    END;

    LOCAL PROCEDURE SleepHalves@6() : Integer;
    BEGIN
      // Удержание делится на две половины: первый счёт берётся до прихода ждущего, второй
      // после. Две половины, а не одно ожидание целиком, - иначе второй счёт нечем
      // отличить от первого.
      EXIT(2);
    END;

    BEGIN
    {
      Держатель блокировки для проверки дороги к имени ЧЕЛОВЕКА.

      Нужен он ровно затем, что этой дороги нельзя проверить соединением sqlcmd: штатная
      таблица блокировок показывает лишь те транзакции, что платформа завела в свою карту,
      а туда попадают только сессии NAV этого экземпляра. Держатель поэтому - сессия NAV,
      запускаемая через Invoke-NAVCodeunit, а ждущим остаётся всё тот же sqlcmd.

      Ответ он отдаёт машиночитаемой строкой и на обоих языках одинаково: разбирать её
      прогону, а не читать человеку, и зависеть от языка службы разбор не должен.

      Объект мерный, как и Codeunit 110234: в бою он не нужен и вреда не приносит -
      строка-носитель живёт только внутри его собственной транзакции.
    }
    END.
  }
}
