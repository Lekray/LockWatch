    VAR
      Parse@1000000000 : Codeunit 110230;
      Coverage@1000000001 : Record 110235;
      AddedRows@1000000002 : Integer;
      ChangedRows@1000000003 : Integer;
      ResetRows@1000000004 : Integer;
      BaselineErr@1000000005 : TextConst 'ENU=The coverage count was not started over - the server counters could not be read - %1;RUS=Счёт охвата не начат заново - счётчики сервера не прочитаны - %1';

    [External]
    PROCEDURE Accumulate@1(VAR Reading@1000000000 : Record 110235) : Integer;
    VAR
      DeltaWaits@1000000001 : BigInteger;
      DeltaMs@1000000002 : BigInteger;
      DeltaEsc@1000000003 : BigInteger;
      ResetWaits@1000000004 : Boolean;
      ResetMs@1000000005 : Boolean;
      ResetEsc@1000000006 : Boolean;
      Fresh@1000000007 : Boolean;
      Changed@1000000008 : Boolean;
      Renamed@1000000009 : Boolean;
      Now@1000000010 : DateTime;
    BEGIN
      // Один проход по прочитанному, и запись ТОЛЬКО там, где что-то изменилось. Иначе
      // накопительный слой стал бы тем, чем ему быть запрещено: сотнями записей в базу
      // каждые три секунды на сервере, где ничего не происходит.
      AddedRows := 0;
      ChangedRows := 0;
      ResetRows := 0;
      Now := CURRENTDATETIME;
      Reading.RESET;
      IF NOT Reading.FINDSET THEN
        EXIT(0);
      REPEAT
        Fresh := NOT Coverage.GET(Reading."SQL Object Id",Reading."Index Id");
        IF Fresh THEN BEGIN
          Coverage.INIT;
          Coverage."SQL Object Id" := Reading."SQL Object Id";
          Coverage."Index Id" := Reading."Index Id";
          Coverage.INSERT;
          AddedRows := AddedRows + 1;
        END;

        Renamed := (Coverage."SQL Object Name" <> Reading."SQL Object Name") OR
                   (Coverage."NAV Table Name" <> Reading."NAV Table Name") OR
                   (Coverage."NAV Key No." <> Reading."NAV Key No.") OR
                   (Coverage."On SIFT" <> Reading."On SIFT");

        DeltaWaits := 0;
        DeltaMs := 0;
        DeltaEsc := 0;
        IF NOT Fresh THEN BEGIN
          // Первое чтение даёт только ОТМЕТКУ, а не прирост: счётчик сервера считает от
          // неизвестного момента, и записать его целиком в "насчитано" значило бы
          // приписать наблюдению всё, что случилось до его начала.
          DeltaWaits := Parse.CounterDelta(Coverage."Last Waits",Reading."Last Waits",ResetWaits);
          DeltaMs := Parse.CounterDelta(Coverage."Last Wait (ms)",Reading."Last Wait (ms)",ResetMs);
          DeltaEsc := Parse.CounterDelta(Coverage."Last Escalations",Reading."Last Escalations",ResetEsc);
          IF ResetWaits OR ResetMs OR ResetEsc THEN BEGIN
            // Обнуление считается отдельно и по ЛЮБОМУ из трёх счётчиков: пойти назад мог
            // один из них, а причина у всех одна - объект ушёл из кэша метаданных сервера.
            Coverage.Resets := Coverage.Resets + 1;
            ResetRows := ResetRows + 1;
          END;
        END;

        Changed := (DeltaWaits <> 0) OR (DeltaMs <> 0) OR (DeltaEsc <> 0);
        IF Fresh OR Changed OR Renamed THEN BEGIN
          Coverage."SQL Object Name" := Reading."SQL Object Name";
          Coverage."NAV Table Name" := Reading."NAV Table Name";
          Coverage."NAV Key No." := Reading."NAV Key No.";
          Coverage."On SIFT" := Reading."On SIFT";
          Coverage.Waits := Coverage.Waits + DeltaWaits;
          Coverage."Wait (ms)" := Coverage."Wait (ms)" + DeltaMs;
          Coverage.Escalations := Coverage.Escalations + DeltaEsc;
          Coverage."Last Waits" := Reading."Last Waits";
          Coverage."Last Wait (ms)" := Reading."Last Wait (ms)";
          Coverage."Last Escalations" := Reading."Last Escalations";
          IF Changed THEN
            Coverage."Last Change At" := Now;
          Coverage.MODIFY;
          IF Changed THEN
            ChangedRows := ChangedRows + 1;
        END;
      UNTIL Reading.NEXT = 0;
      EXIT(ChangedRows);
    END;

    [External]
    PROCEDURE PassCounts@2(VAR Added@1000000000 : Integer;VAR Changed@1000000001 : Integer;VAR Resets@1000000002 : Integer);
    BEGIN
      Added := AddedRows;
      Changed := ChangedRows;
      Resets := ResetRows;
    END;

    [External]
    PROCEDURE CounterTotal@3(VAR Waits@1000000000 : BigInteger;VAR WaitMs@1000000001 : BigInteger;VAR Escalations@1000000002 : BigInteger);
    BEGIN
      // Итог собирается ПЕРЕБОРОМ, и это то же решение, что и в сводке: SumIndexFields
      // здесь означал бы агрегат на таблице, в которую пишет фоновая задача. Строк тут
      // столько, сколько индексов хоть раз ждали, - десятки, а не тысячи.
      Waits := 0;
      WaitMs := 0;
      Escalations := 0;
      Coverage.RESET;
      IF Coverage.FINDSET THEN
        REPEAT
          Waits := Waits + Coverage.Waits;
          WaitMs := WaitMs + Coverage."Wait (ms)";
          Escalations := Escalations + Coverage.Escalations;
        UNTIL Coverage.NEXT = 0;
    END;

    [External]
    PROCEDURE JournalTotal@4(SinceAt@1000000000 : DateTime;VAR Episodes@1000000001 : Integer;VAR WaitMs@1000000002 : BigInteger);
    BEGIN
      // Названная половина охвата считается за ТОТ ЖЕ срок, с которого ведётся счёт
      // приростов. Сравнивать журнал за сутки со счётчиками за месяц значило бы получить
      // долю охвата в несколько процентов и напугать ею на ровном месте.
      //
      // Считается по журналу И по истории, и без всякой кнопки - в отличие от свода. Свод
      // отвечает на "что происходит", и "в последнее время" против "за всё время" - два
      // разных законных вопроса, оттого там кнопка. Охват отвечает на другое: "чего журнал
      // НЕ увидел". Эпизод, увиденный и потом уехавший в историю, увиден БЫЛ, и посчитать
      // один журнал значило бы объявить пропуском собственный переезд: доля падала бы сама
      // собой по сроку или по потолку, и падение читалось бы как испортившееся наблюдение.
      //
      // Обходов два - тип записи в параметре C/AL фиксирован, - а складывается эпизод в
      // одном месте: разойдясь, две половины охвата считали бы разные величины.
      Episodes := 0;
      WaitMs := 0;
      AddJournal(SinceAt,Episodes,WaitMs);
      AddHistory(SinceAt,Episodes,WaitMs);
      AddDropped(Episodes,WaitMs);
    END;

    LOCAL PROCEDURE AddJournal@7(SinceAt@1000000000 : DateTime;VAR Episodes@1000000001 : Integer;VAR WaitMs@1000000002 : BigInteger);
    VAR
      Episode@1000000003 : Record 110231;
    BEGIN
      Episode.RESET;
      Episode.SETCURRENTKEY("Started At");
      IF SinceAt <> 0DT THEN
        Episode.SETFILTER("Started At",'>=%1',SinceAt);
      IF Episode.FINDSET THEN
        REPEAT
          AddOne(Episodes,WaitMs,Episode."Max Wait (ms)");
        UNTIL Episode.NEXT = 0;
    END;

    LOCAL PROCEDURE AddHistory@8(SinceAt@1000000000 : DateTime;VAR Episodes@1000000001 : Integer;VAR WaitMs@1000000002 : BigInteger);
    VAR
      Moved@1000000003 : Record 110238;
    BEGIN
      // Тот же отбор по истории. Ключ "Started At" в ней есть: история заводилась с ключами
      // под чтение, и этот - первый из них.
      Moved.RESET;
      Moved.SETCURRENTKEY("Started At");
      IF SinceAt <> 0DT THEN
        Moved.SETFILTER("Started At",'>=%1',SinceAt);
      IF Moved.FINDSET THEN
        REPEAT
          AddOne(Episodes,WaitMs,Moved."Max Wait (ms)");
        UNTIL Moved.NEXT = 0;
    END;

    LOCAL PROCEDURE AddDropped@11(VAR Episodes@1000000000 : Integer;VAR WaitMs@1000000001 : BigInteger);
    VAR
      State@1000000002 : Record 110237;
      DroppedEpisodes@1000000003 : Integer;
      DroppedMs@1000000004 : BigInteger;
    BEGIN
      // Слагаемое третье, и без него охват врёт на себя. Эпизод, который сторож увидел и
      // САМ выбросил как слишком короткий, увиден БЫЛ: не считать его - значит объявить
      // пропуском собственную уборку. Довод тот же, по которому считается история, только
      // дверь другая: там строка переехала, здесь её нет вовсе, и остался от неё счёт.
      //
      // Отбора по сроку тут нет и быть не может. Счёт этот ведётся ОТ нажатия "начать счёт
      // заново" - то есть ровно от той отметки, с которой сравнивают, - а строк, чтобы
      // отобрать их по дате, уже не существует. Поэтому обнулять его обязано то же
      // нажатие, что ставит отметку, и делается это в одном месте (StartOver).
      State.Dropped(DroppedEpisodes,DroppedMs);
      Episodes := Episodes + DroppedEpisodes;
      WaitMs := WaitMs + DroppedMs;
    END;

    LOCAL PROCEDURE AddOne@9(VAR Episodes@1000000000 : Integer;VAR WaitMs@1000000001 : BigInteger;Ms@1000000002 : Integer);
    BEGIN
      // Общий кусок нарочно, хоть он и в две строки: это и есть то, что охват считает.
      Episodes := Episodes + 1;
      WaitMs := WaitMs + Ms;
    END;

    [External]
    PROCEDURE Share@5(CounterMs@1000000000 : BigInteger;JournalMs@1000000001 : BigInteger) : Decimal;
    BEGIN
      // Доля охвата в процентах. Ноль в знаменателе - это не ноль процентов, а "нечего
      // делить": счётчики ещё ничего не насчитали, и доля неизвестна. Показывать в этом
      // случае ноль значило бы сказать "не видно ничего" там, где не было ничего.
      IF CounterMs <= 0 THEN
        EXIT(0);
      EXIT(ROUND(JournalMs * 100 / CounterMs,0.1));
    END;

    LOCAL PROCEDURE ResetAll@6();
    BEGIN
      // Счёт начинается заново целиком: и приросты, и отметки последнего чтения. Оставить
      // отметки значило бы получить на следующем проходе прирост за всё время с прошлого
      // чтения и объявить его случившимся только что.
      //
      // Дверь наружу закрыта нарочно: снаружи видно одно StartOver. Половины счёта
      // обнуляются вместе или никак, а открытая половинчатая дорога - это обещание,
      // которое кто-нибудь однажды выполнит.
      Coverage.RESET;
      Coverage.DELETEALL;
    END;

    [External]
    PROCEDURE StartOver@10();
    VAR
      State@1000000000 : Record 110237;
      Sql@1000000001 : Codeunit 110234;
      Reading@1000000002 : TEMPORARY Record 110235;
      Rows@1000000003 : Integer;
    BEGIN
      // Счёт охвата начинается заново ЦЕЛИКОМ и в одном месте. Приросты и отметка "считается
      // с" - две половины одного числа: обнулить приросты, оставив отметку, значит сравнить
      // счётчики с этого мига против журнала за всё прошлое, а сбросить одну отметку -
      // наоборот. Лежала эта пара на странице, и дорога к ней была ровно одна - нажатие
      // руками. Обещание, к которому нет дороги из кода, проверяется глазами и ничем больше.
      //
      // Половин на деле ТРИ, и третья - точка отсчёта счётчиков. Снимал её прежде следующий
      // проход, то есть до периода опроса спустя, и эпизоды этого промежутка попадали в
      // названную половину, а в насчитанную не попадали вовсе: доля уходила выше правды и
      // оставалась такой НАВСЕГДА - обе половины копятся каждая от своего начала и сами
      // между собой не сходятся. Довод этот написан в самом проходе, где отметка ставится
      // ровно в тот проход, что делает первое чтение; нажатие его обходило. Кто начал счёт,
      // тот и снял точку отсчёта.
      //
      // Читается ДО уборки нарочно: не прочлось - не тронуто ничто. Половина начатого счёта
      // хуже неначатого, а отказ здесь называет причину словами сервера и соврать не может.
      IF NOT Sql.ReadCounters(0,Reading,Rows) THEN
        ERROR(BaselineErr,Sql.LastErrorText);
      ResetAll;
      // После уборки все строки свежие, а свежая даёт только ОТМЕТКУ и никакого прироста -
      // это и есть точка отсчёта. Выключенный накопительный слой ей не помеха: чтение
      // счётчиков само ничего не копит, и снятая точка дождётся дня, когда слой включат, -
      // обе половины останутся считанными от одного мига.
      Accumulate(Reading);
      // Строка состояния перечитывается ПОД ЗАМКОМ: ту же строку пишет каждый проход, и
      // разойтись версиями им нельзя (FINDINGS, раздел 69).
      State.FetchToWrite;
      State."Coverage Since" := CURRENTDATETIME;
      // Выброшенное обнуляется ВМЕСТЕ с отметкой, иначе доля уйдёт за сотню процентов:
      // журнальная половина считалась бы с этого мига плюс всё выброшенное за прошлую
      // жизнь. Это та же пара, что отметка и приросты, - и обнуляется она там же.
      State."Dropped Episodes" := 0;
      State."Dropped Wait (ms)" := 0;
      State.MODIFY;
      COMMIT;
    END;

    BEGIN
    {
      Накопительный слой: сколько ожиданий насчитал сам сервер против того, сколько назвал
      журнал. Сторожем этот слой быть не может - счётчик прирастает только в момент КОНЦА
      ожидания ([FINDINGS.md], раздел 12), - зато он единственный, кто видит ожидания
      короче периода опроса, а таких на живой базе большинство.

      Отсюда и весь смысл: доля охвата пишется на экран, вместо молчаливого обещания, что
      видно всё. Число это не украшение - оно говорит, можно ли верить сводке. Охват в
      девяносто процентов значит "сводка почти полна", охват в пять - "журнал видит хвост,
      а беда в другом месте".

      Считается всё приростами, а сырые значения счётчика хранятся только затем, чтобы
      было из чего вычитать. Обнуление счётчика (объект ушёл из кэша метаданных сервера)
      видно тем, что значение пошло НАЗАД, и считается отдельным полем: сумма, собранная
      поверх незамеченного обнуления, была бы неотличима от правды.

      Запись идёт только туда, где что-то изменилось. На тихом сервере это ноль строк за
      проход, и наблюдатель не платит за наблюдение ничем.
    }
    END.
