    VAR
      Journal@1000000000 : Codeunit 110232;
      WindowFrom@1000000001 : DateTime;
      WindowTo@1000000002 : DateTime;
      AlertMs@1000000003 : Integer;
      TotalEpisodes@1000000004 : Integer;
      OverThresholdStyle@1000000005 : Text[30];
      WithHistory@1000000006 : Boolean;
      CountedOverText@1000000007 : Text[50];
      JournalOnlyTxt@1000000008 : TextConst 'ENU=the journal;RUS=журналу';
      WithHistoryTxt@1000000009 : TextConst 'ENU=the journal and history;RUS=журналу и истории';

    LOCAL PROCEDURE SetWindow@1(Hours@1000000000 : Integer);
    VAR
      Back@1000000001 : Duration;
    BEGIN
      // Верхняя граница берётся ОДИН раз и запоминается, а не считается при каждом
      // обращении к строке: иначе окно уезжало бы, пока сводку разглядывают, и две
      // соседние колонки оказались бы посчитаны за разные промежутки.
      WindowTo := CURRENTDATETIME;
      Back := Hours * MsPerHour;
      WindowFrom := WindowTo - Back;
    END;

    LOCAL PROCEDURE SetWholeJournal@2();
    BEGIN
      // Пустая нижняя граница означает "весь журнал". Насколько это далеко, решает срок
      // хранения, а не сводка: показывать окно шире хранимого было бы обещанием пустоты.
      WindowTo := CURRENTDATETIME;
      WindowFrom := 0DT;
    END;

    LOCAL PROCEDURE Fill@3();
    VAR
      Setup@1000000000 : Record 110230;
    BEGIN
      // Порог берётся из настройки при каждом сведении, а не запоминается: его меняют
      // ровно тогда, когда сводка не сошлась с ощущением, и увидеть новый порог хотят
      // сразу же.
      Setup.FetchOnce;
      AlertMs := Setup."Alert Threshold (ms)";
      TotalEpisodes := Journal.Summarize(Rec,WindowFrom,WindowTo,AlertMs,WithHistory);
      // Объём пишется рядом с числом, а не подразумевается кнопкой: кнопку нажали минуту
      // назад, а число смотрят сейчас, и вспомнить, за что оно посчитано, будет неоткуда.
      IF WithHistory THEN
        CountedOverText := WithHistoryTxt
      ELSE
        CountedOverText := JournalOnlyTxt;
    END;

    LOCAL PROCEDURE Refill@4();
    BEGIN
      // Заполнение и обновление экрана разделены нарочно: при открытии страницы обновлять
      // ещё нечего, а из действия - обязательно, иначе строки сменились бы, а на экране
      // осталось прежнее.
      //
      // Курсор ставится на первую строку ДО обновления. Строки перед этим удалены все до
      // одной, и на тихом сервере новых не появилось вовсе - самый обычный случай для
      // здоровой базы, и оставлять экран на строке, которой больше нет, незачем.
      Fill;
      IF FINDFIRST THEN;
      CurrPage.UPDATE(FALSE);
    END;

    LOCAL PROCEDURE ShowEpisodes@5();
    VAR
      Episode@1000000000 : Record 110231;
      TableName@1000000001 : Text[250];
      KeyNo@1000000002 : Integer;
    BEGIN
      // Имя и ключ снимаются в свои переменные до отбора: в отборе слева поле чужой
      // записи, справа - поле нашей, и запись их одним именем читается как ошибка,
      // даже когда работает.
      TableName := "NAV Table Name";
      KeyNo := "NAV Key No.";
      Episode.SETRANGE("NAV Table Name",TableName);
      Episode.SETRANGE("NAV Key No.",KeyNo);
      IF WindowFrom = 0DT THEN
        Episode.SETFILTER("Started At",'<=%1',WindowTo)
      ELSE
        Episode.SETRANGE("Started At",WindowFrom,WindowTo);
      PAGE.RUN(PAGE::"LockWatch Episodes",Episode);
    END;

    LOCAL PROCEDURE ShowHistoryEpisodes@13();
    VAR
      Episode@1000000000 : Record 110238;
      TableName@1000000001 : Text[250];
      KeyNo@1000000002 : Integer;
    BEGIN
      // То же самое, но по истории. Отдельной кнопкой, а не догадкой по признаку "считали
      // ли с историей": в журнале и в истории лежат РАЗНЫЕ эпизоды одной и той же группы,
      // и человек сам решает, куда смотреть.
      TableName := "NAV Table Name";
      KeyNo := "NAV Key No.";
      Episode.SETRANGE("NAV Table Name",TableName);
      Episode.SETRANGE("NAV Key No.",KeyNo);
      IF WindowFrom = 0DT THEN
        Episode.SETFILTER("Started At",'<=%1',WindowTo)
      ELSE
        Episode.SETRANGE("Started At",WindowFrom,WindowTo);
      PAGE.RUN(PAGE::"LockWatch Episode History",Episode);
    END;

    [External]
    PROCEDURE CountHistory@14(Yes@1000000000 : Boolean);
    BEGIN
      // Открыть свод сразу с историей может тот, кто пришёл из неё. Ставится ДО запуска
      // страницы: после запуска первое заполнение уже случилось, и человек увидел бы
      // сначала одно число, а потом другое.
      WithHistory := Yes;
    END;

    LOCAL PROCEDURE SetStyle@10();
    BEGIN
      // Строка, где есть хоть один эпизод выше порога, красится. Порог ноль означает
      // "не тревожить вовсе", и тогда не красится ничего: цвет не должен появляться
      // сам собой у того, кто тревогу выключил.
      OverThresholdStyle := StandardStyle;
      IF (AlertMs > 0) AND ("Over Threshold" > 0) THEN
        OverThresholdStyle := AlarmStyle;
    END;

    LOCAL PROCEDURE StandardStyle@11() : Text[30];
    BEGIN
      EXIT('Standard');
    END;

    LOCAL PROCEDURE AlarmStyle@12() : Text[30];
    BEGIN
      // Имена стилей - платформенные, и написать их с ошибкой можно молча: неизвестный
      // стиль клиент рисует обычным, ничего не сказав. Поэтому они здесь по одному разу.
      EXIT('Unfavorable');
    END;

    LOCAL PROCEDURE DayHours@6() : Integer;
    BEGIN
      // Сутки. Окно по умолчанию: спрашивают обычно про смену или прошедшую ночь, а не
      // про месяц, и сводка за месяц на первом экране увела бы взгляд с того, что болит
      // сейчас, на то, что болело когда-то.
      EXIT(24);
    END;

    LOCAL PROCEDURE WeekHours@7() : Integer;
    BEGIN
      // Семь суток. Второе окно нужно ровно за тем, чтобы увидеть беду, которая случается
      // раз в неделю - закрытие периода, выгрузку, регламент, - и в суточном окне видна
      // только в тот самый день.
      EXIT(168);
    END;

    LOCAL PROCEDURE MsPerHour@8() : Integer;
    BEGIN
      EXIT(3600000);
    END;

    BEGIN
    {
      Сводка: какие таблицы и ключи ждут блокировок и на сколько. Это первый из трёх
      вопросов, ради которых инструмент затевался, и до сих пор на него отвечал только
      журнал - построчно, а значит никак: пятьсот строк одного шторма выглядят как
      пятьсот бед, хотя беда одна.

      Строки упорядочены по СУММАРНОМУ ожиданию, а не по числу эпизодов. Число эпизодов
      говорит, как часто задевали, сумма - во сколько это обошлось, и лечить начинают со
      второго. Колонка "из них выше порога" стоит рядом с числом эпизодов ровно затем,
      чтобы одно долгое ожидание не потерялось на фоне сотни коротких.

      Тип страницы - Worksheet, а не List, и это не украшение: шапка с окном обязана
      быть ВИДНА рядом с числами. Число без окна, за которое оно посчитано, - не число.

      Окно задаётся кнопками, а не полями ввода. Причина не в лени: верхняя граница
      берётся один раз и запоминается, и пока сводку разглядывают, числа не едут. Поле
      ввода с датой соблазняло бы править её по одному концу и сравнивать несравнимое.

      Строка сводки - не ответ, а вопрос: "почему эта таблица столько ждёт". Ответ лежит
      в эпизодах за ней, поэтому кнопка "Эпизоды" переносит туда ТОТ ЖЕ отбор - окно,
      таблицу и ключ, - а не открывает журнал целиком.

      Обещания "видно всё" сводка не даёт, и это принципиально: опрос очереди не видит
      ожиданий короче своего периода, а таких большинство. Сколько именно видно, говорит
      отдельный экран охвата, и кнопка туда стоит здесь же.
    }
    END.
