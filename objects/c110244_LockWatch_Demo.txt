OBJECT Codeunit 110244 LockWatch Demo
{
  OBJECT-PROPERTIES
  {
    Date=10.09.26;
    Time=12:00:00;
    Modified=Yes;
    Version List=LockWatch;
  }
  PROPERTIES
  {
    OnRun=BEGIN
            RunDemo;
          END;

  }
  CODE
  {
    VAR
      Setup@1000000000 : Record 110230;
      Demo@1000000001 : Record 110239;
      OffErr@1000000002 : TextConst 'ENU=The Demo button is switched off in the setup. While the switch is clear, the demo raises no lock.;RUS=Кнопка "Демо" выключена в настройке. Пока галка снята, показ блокировки не поднимает.';
      NoWatchErr@1000000003 : TextConst 'ENU=The watch is not armed: there would be nobody to write the episode down. Arm the watch and press again.;RUS=Сторож не заведён: записать эпизод будет некому. Заведите наблюдение и нажмите ещё раз.';
      NoSessionErr@1000000004 : TextConst 'ENU=A background session did not start, so there is nothing to argue with.;RUS=Фоновая сессия не завелась, спорить не с кем.';
      StartedTxt@1000000005 : TextConst 'ENU=The demo is running: two background sessions argue over one row for %1 seconds. Look at the journal in about that long.;RUS=Показ пошёл: две фоновые сессии спорят за одну строку %1 секунд. Через столько же смотрите журнал.';
      SlowPollTxt@1000000006 : TextConst 'ENU=Mind the poll period: %1 ms against the %2 ms of holding, so the pass may miss the wait entirely.;RUS=Учтите период опроса: %1 мс против %2 мс удержания - проход может не увидеть ожидания вовсе.';
      TargetTxt@1000000007 : TextConst 'ENU=The row the demo argues over;RUS=Строка, за которую спорит показ';

    [External]
    PROCEDURE RunDemo@1();
    VAR
      HoldSession@1000000010 : Integer;
      WaitSession@1000000011 : Integer;
      Note@1000000012 : Text;
    BEGIN
      Setup.FetchOnce;

      // Выключатель спрашивается ЗДЕСЬ - и только здесь. Соблазн повторить проверку в
      // держателе есть, и он вреден: проверка "выключатель запрещает показ" стала бы
      // ПУСТОЙ - снятый у кнопки, выключатель всё равно сработал бы у держателя, и
      // поломка прошла бы молча. Одно обещание - одно место, где оно держится.
      IF NOT Setup."Demo Enabled" THEN
        ERROR(OffErr);

      // Показ показывает СТОРОЖА, а не блокировку саму по себе. Без заведённого сторожа
      // ожидание случится и пройдёт, а в журнале не появится ничего - и человек унесёт
      // с показа вывод, что инструмент не работает. Отказ словами честнее пустого экрана.
      IF NOT Setup.Enabled THEN
        ERROR(NoWatchErr);

      EnsureTarget;

      // COMMIT обязателен, и причин ему две. Фоновая сессия - ДРУГАЯ транзакция: строки,
      // которую мы только что завели, она без фиксации не увидит и спорить будет не за
      // что. И вторая: пока своя транзакция открыта, строка-мишень заперта нами самими,
      // а значит держатель встал бы в очередь за собственной кнопкой.
      COMMIT;

      IF NOT STARTSESSION(HoldSession,CODEUNIT::"LockWatch Demo Hold",COMPANYNAME) THEN
        ERROR(NoSessionErr);

      // Держатель обязан взять строку ПЕРВЫМ. Заведи обе сессии разом - и кто кого ждёт,
      // решит случай: в половине случаев первой окажется ждущая, ожидание выйдет длиной
      // в её собственную запись, то есть почти нулевым, и показывать будет нечего.
      SLEEP(StartGapMs);

      IF NOT STARTSESSION(WaitSession,CODEUNIT::"LockWatch Demo Wait",COMPANYNAME) THEN
        ERROR(NoSessionErr);

      Note := STRSUBSTNO(StartedTxt,HoldMs DIV 1000);
      // Период опроса длиннее удержания - это не отказ, а предупреждение: показ состоится,
      // но проход может лечь мимо окна. Молчать об этом нельзя, иначе пустой журнал после
      // показа читается как поломка инструмента.
      IF Setup."Poll Period (ms)" >= HoldMs THEN
        Note := Note + ' ' + STRSUBSTNO(SlowPollTxt,Setup."Poll Period (ms)",HoldMs);
      IF GUIALLOWED THEN
        MESSAGE(Note);
    END;

    [External]
    PROCEDURE TargetNo@2() : Integer;
    BEGIN
      // Строка одна, и номер у неё постоянный. Спорить за РАЗНЫЕ строки значит не спорить
      // вовсе: обе сессии отработали бы по очереди, ожидания не возникло бы, и в журнал
      // не попало бы ничего.
      EXIT(1);
    END;

    [External]
    PROCEDURE HoldMs@3() : Integer;
    BEGIN
      // Шесть секунд, и обе границы у этого числа жёсткие.
      //
      // Снизу - период опроса по умолчанию, три секунды: держать короче периода значит
      // надеяться, что проход случится ровно в это окно. Шесть секунд - это два периода,
      // и хотя бы один проход в них попадает.
      //
      // Сверху - предел ожидания блокировки у NAV, десять секунд: держать дольше значит
      // уронить ждущую сессию по таймауту. Показ обязан кончаться ОБЫЧНЫМ эпизодом -
      // ожидание было и кончилось, - а не отказом: отказом он показывал бы не работу
      // инструмента, а собственную неаккуратность.
      EXIT(6000);
    END;

    [External]
    PROCEDURE StartGapMs@4() : Integer;
    BEGIN
      // Полсекунды. Столько нужно фоновой сессии, чтобы подняться и взять строку; меньше
      // - и порядок спора снова решает случай. Больше брать незачем: это время вычитается
      // из удержания, а его и так ровно столько, чтобы уложиться до предела NAV.
      EXIT(500);
    END;

    LOCAL PROCEDURE EnsureTarget@5();
    BEGIN
      // Строка заводится один раз за жизнь установки и дальше только переписывается.
      // Удалять её после показа незачем: по ней видно, чем показ кончился в прошлый раз.
      IF Demo.GET(TargetNo) THEN
        EXIT;
      Demo.INIT;
      Demo."Entry No." := TargetNo;
      Demo.Note := TargetTxt;
      Demo."Touched At" := CURRENTDATETIME;
      Demo.INSERT;
    END;

    BEGIN
    {
      Пульт показа. Кнопка "Демо" зовёт RunDemo, и он поднимает НАСТОЯЩЕЕ ожидание, какое
      случается на бою: две фоновые сессии спорят за одну строку, одна держит её шесть
      секунд, другая стоит за ней в очереди. Проход видит спор так же, как увидел бы спор
      двух людей, и записывает эпизод сам - нажимать больше ничего не нужно.

      Поддельного показа тут нет НАРОЧНО. Строку в журнал можно было бы просто вписать, и
      это была бы ложь ровно в том месте, ради которого инструмент ставят: человек увидел
      бы не работу наблюдения, а картинку, нарисованную самим наблюдением.

      Ролей три, и каждая в своём объекте, чтобы не разбирать в OnRun, кем сессия пришла:
      здесь пульт, в 110245 держатель, в 110246 ждущий. STARTSESSION умеет звать только
      OnRun, поэтому иначе пришлось бы передавать роль записью-параметром - и читать этот
      кодюнит стало бы вдвое дороже, чем он того стоит.
    }
    END.
  }
}
