    PROCEDURE ReadQueue@1(MaxRows@1000000010 : Integer;WantStatements@1000000011 : Boolean;VAR Observed@1000000012 : TEMPORARY Record 110231;VAR RowCount@1000000013 : Integer;VAR ObservedAt@1000000014 : DateTime;VAR OtherDatabase@1000000015 : Integer) : Boolean;
    VAR
      Done@1000000016 : Boolean;
    BEGIN
      // Один запрос на проход, и он же приносит ЧАСЫ. Часы нужны серверные: NAV и SQL
      // могут стоять на разных машинах, а в журнале рядом лежат начало эпизода (считанное
      // назад от наблюдения) и начало транзакции виновника (взятое у SQL). Смешать две
      // шкалы значит получить эпизод, начавшийся раньше собственной причины.
      LastError := '';
      RowCount := 0;
      OtherDatabase := 0;
      ObservedAt := 0DT;
      Observed.RESET;
      Observed.DELETEALL;
      // Потолок спрашивается у разбора, и он же решает признак "усечено" в проходе:
      // пределов два (настройка и пятьсот строк разбора), и фактический - меньший.
      MaxRows := Watch.QueueRowLimit(MaxRows);
      Done := TryReadQueue(MaxRows,WantStatements,Observed,RowCount,ObservedAt,OtherDatabase);
      IF NOT Done THEN
        // Отказ съёма не роняет фоновую задачу: прав может не быть, сервер может быть занят,
        // и в обоих случаях правильный ответ - записать причину и жить дальше. Молча
        // проглоченная ошибка тут страшнее отказа: журнал выглядел бы просто пустым.
        LastError := COPYSTR(GETLASTERRORTEXT,1,ErrorTextMaxChars);
      // Соединение закрывается на ОБОИХ путях, и текст ошибки берётся ДО закрытия: CloseSession
      // идёт через TryFunction, глотает собственный отказ, и он переписал бы GETLASTERRORTEXT.
      CloseSession;
      EXIT(Done);
    END;

    [TryFunction]
    LOCAL PROCEDURE CloseSession@2();
    BEGIN
      // Закрывается всё, что открывал съём, и на ОБОИХ путях - и на удачном, и на ошибочном.
      // try/finally в C/AL нет: отказ внутри съёма выбрасывает из функции МИМО обеих строк
      // Close, и незакрытая сессия SQL остаётся висеть на сервере - по одной на каждый отказ,
      // то есть на каждый проход, пока беда не кончится. Поэтому сессия и чтение живут не в
      // теле съёма, а у объекта: закрывать их обязан тот, кто отказ пережил.
      //
      // Сам Close - в TryFunction нарочно. У отказавшего съёма чтение открыть не успевало
      // вовсе (отказ пришёл на Open или на ExecuteReader), и закрывать там нечего; молчать
      // про это честно, а второй отказ поверх первого ничего бы не добавил.
      IF NOT ISNULL(SqlReader) THEN
        SqlReader.Close;
      IF NOT ISNULL(SqlConn) THEN
        SqlConn.Close;
    END;

    [TryFunction]
    LOCAL PROCEDURE TryReadQueue@6(MaxRows@1000000019 : Integer;WantStatements@1000000020 : Boolean;VAR Observed@1000000021 : TEMPORARY Record 110231;VAR RowCount@1000000022 : Integer;VAR ObservedAt@1000000023 : DateTime;VAR OtherDatabase@1000000024 : Integer);
    VAR
      SqlCmd@1000000025 : DotNet "'System.Data, Version=2.0.0.0, Culture=neutral, PublicKeyToken=b77a5c561934e089'.System.Data.SqlClient.SqlCommand";
      OurDatabaseId@1000000026 : Integer;
      ObjectId@1000000027 : Integer;
      DatabaseId@1000000028 : Integer;
      Kind@1000000029 : Text;
      ResourceText@1000000030 : Text;
      VictimBody@1000000031 : Text;
      BlockerBody@1000000032 : Text;
      HobtId@1000000033 : BigInteger;
      Parsed@1000000034 : Boolean;
      Nameless@1000000035 : Boolean;
      OutStr@1000000036 : OutStream;
    BEGIN
      SqlConn := SqlConn.SqlConnection(ConnString);
      SqlConn.Open;
      SqlCmd := SqlConn.CreateCommand;
      SqlCmd.CommandTimeout := ReadTimeoutSec;
      SqlCmd.CommandText := QueueSql(MaxRows,WantStatements);
      SqlReader := SqlCmd.ExecuteReader;

      // Первый набор - часы и номер базы. Он приходит ДАЖЕ когда ждущих нет: без него
      // проход, заставший тишину, не знал бы, каким моментом закрывать вчерашние эпизоды.
      IF NOT SqlReader.Read THEN
        ERROR(NoClockErr);
      ObservedAt := SqlMoment(SqlReader.GetInt32(QueueColumnObservedDay),
                              SqlReader.GetInt32(QueueColumnObservedMs));
      OurDatabaseId := SqlReader.GetInt32(QueueColumnDatabaseId);
      SqlReader.NextResult;

      WHILE SqlReader.Read DO BEGIN
        ResourceText := ReaderText(SqlReader,QueueColumnResourceDescription);
        Parsed := Watch.ParseWaitResource(ResourceText,Kind,HobtId,ObjectId,DatabaseId);
        // Чужая база на том же сервере - не наша беда и не наш журнал. Считается
        // отдельно: пустая колонка без объяснения читается как "блокировок не было".
        IF Parsed AND (DatabaseId <> OurDatabaseId) AND (OurDatabaseId <> 0) THEN
          OtherDatabase := OtherDatabase + 1
        ELSE BEGIN
          RowCount := RowCount + 1;

          Observed.INIT;
          Observed."Entry No." := RowCount;
          Observed."Victim SPID" := SqlReader.GetInt32(QueueColumnVictimSpid);
          Observed."Victim Transaction Id" := ReaderBigInt(SqlReader,QueueColumnVictimTran);
          Observed."Victim Is NAV" := NavFlag(ReaderText(SqlReader,QueueColumnVictimProgram));
          Observed."Blocker SPID" := ReaderInt(SqlReader,QueueColumnBlockerSpid);
          Observed."Blocker Transaction Id" := ReaderBigInt(SqlReader,QueueColumnBlockerTran);
          // Кто держит - тем, что о нём знает сам сервер. Соединение с sys.dm_exec_sessions
          // для виновника уже сделано, и логин и узел из ТОЙ ЖЕ строки не стоят ничего.
          // Для чужого соединения - утилиты, задания, чьего-то SSMS - это единственный
          // возможный ответ на "кто": для самой сессии NAV логин здесь общий, учётной
          // записи службы, и человеком он не притворяется.
          Observed."Blocker Login" :=
            COPYSTR(ReaderText(SqlReader,QueueColumnBlockerLogin),1,
                    MAXSTRLEN(Observed."Blocker Login"));
          Observed."Blocker Host" :=
            COPYSTR(ReaderText(SqlReader,QueueColumnBlockerHost),1,
                    MAXSTRLEN(Observed."Blocker Host"));
          Observed."Blocker Program" :=
            COPYSTR(ReaderText(SqlReader,QueueColumnBlockerProgram),1,
                    MAXSTRLEN(Observed."Blocker Program"));
          // Колонки 5, 6 и 23 (узел, процесс и логин жертвы), а также 3, 15 и 16 (тип
          // ожидания и начало транзакции виновника) остаются в запросе, но в строку не
          // ложатся: номера остальных столбцов от этого не съезжают. Программа жертвы
          // читается колонкой 7 один раз - по ней решается, наш ли это сеанс.
          Observed."Last Wait (ms)" := SqlReader.GetInt32(QueueColumnWaitMs);
          Observed."Resource Kind" := COPYSTR(UPPERCASE(Kind),1,MAXSTRLEN(Observed."Resource Kind"));
          Observed."Hobt Id" := HobtId;
          Observed."Database Id" := DatabaseId;
          Observed."SQL Object Id" := ObjectId;
          Observed."NAV Key No." := Watch.UnknownKeyNo;
          // У ресурса, за которым нет НИ hobt, НИ номера объекта (pagelock, ridlock), назвать
          // таблицу NAV нечем: в каталог за таким ресурсом не сходить вовсе. Рядом ложится
          // его СЫРОЕ описание: пустая колонка читалась бы как "сервер таблицы не назвал",
          // хотя сервер назвал ровно то, что сказал. Сырое описание тут ещё и РАЗЛИЧИТЕЛЬ:
          // у двух таких ожиданий одной жертвы (разные страницы, разные файлы) различающего
          // имени, кроме него, нет, а ключ эпизода его спрашивает (SQL Object Name в
          // FindOpen журнала) - иначе оба ожидания слились бы в один эпизод, правдоподобно
          // и молча.
          Nameless := Parsed AND (HobtId = 0) AND (ObjectId = 0);
          IF (NOT Parsed) OR Nameless THEN
            // Ресурс, который мы не поняли, показывается СЫРЫМ. Выбросить строку значило
            // бы соврать тишиной, а показать пустую - соврать дважды.
            Observed."SQL Object Name" := COPYSTR(ResourceText,1,MAXSTRLEN(Observed."SQL Object Name"));
          VictimBody := '';
          BlockerBody := '';
          IF WantStatements THEN BEGIN
            // Текст кладётся во ВРЕМЕННУЮ строку, и она же довезёт его до журнала. Отдельным
            // запросом за ним ходить нельзя: к моменту второго запроса виновник успевает
            // послать другой оператор, и в журнал лёг бы не тот, который взял блокировку.
            VictimBody := Watch.StatementWithMark(ReaderText(SqlReader,QueueColumnVictimStatement),
                                                  ReaderInt(SqlReader,QueueColumnVictimStatementLength));
            BlockerBody := Watch.StatementWithMark(ReaderText(SqlReader,QueueColumnBlockerStatement),
                                                   ReaderInt(SqlReader,QueueColumnBlockerStatementLength));
            IF BlockerBody <> '' THEN
              // Начало запроса ложится и в короткое поле: в списке эпизодов колонка должна
              // что-то показывать, не заставляя открывать каждую строку по очереди.
              Observed."Blocker Statement" :=
                COPYSTR(Watch.Flatten(BlockerBody),1,MAXSTRLEN(Observed."Blocker Statement"));
          END;
          // Поток в поле-БЛОБ открывается ДО вставки строки, и это не мелочь: на временной
          // записи блоб, записанный после INSERT и закреплённый MODIFY, до читателя не
          // доезжает вовсе. Измерено на себе - HASVALUE у принимающей стороны отвечал "нет"
          // при полностью заполненной строке.
          IF VictimBody <> '' THEN BEGIN
            Observed."Victim Statement".CREATEOUTSTREAM(OutStr,TEXTENCODING::UTF8);
            OutStr.WRITETEXT(VictimBody);
          END;
          IF BlockerBody <> '' THEN BEGIN
            Observed."Blocker Statement Text".CREATEOUTSTREAM(OutStr,TEXTENCODING::UTF8);
            OutStr.WRITETEXT(BlockerBody);
          END;
          Observed.INSERT;
        END;
      END;
      SqlReader.Close;
      SqlConn.Close;
    END;

    [TryFunction]
    LOCAL PROCEDURE TryResolveDocuments@9(VAR Buffer@1000000048 : TEMPORARY Record 110231;VAR Context@1000000049 : TEMPORARY Record 110232);
    VAR
      SqlCmd@1000000050 : DotNet "'System.Data, Version=2.0.0.0, Culture=neutral, PublicKeyToken=b77a5c561934e089'.System.Data.SqlClient.SqlCommand";
      LockHobt@1000000051 : ARRAY [40] OF BigInteger;
      LockObject@1000000052 : ARRAY [40] OF Text;
      LockIndex@1000000053 : ARRAY [40] OF Text;
      LockHash@1000000054 : ARRAY [40] OF Text;
      HobtList@1000000055 : Text;
      SpidList@1000000056 : Text;
      LockSpid@1000000057 : ARRAY [40] OF Integer;
      Locks@1000000058 : Integer;
      Slot@1000000059 : Integer;
    BEGIN
      // Одно чтение dm_tran_locks на весь проход. В ответ едут НЕВЫПОЛНЕННЫЕ просьбы
      // жертв: они называют строку, за которую идёт спор, - ровно тот вопрос, ради
      // которого инструмент и заводился. Хэш строки берётся у САМОЙ ЖЕРТВЫ, а не у
      // виновника: виновник держит столько строк, сколько тронул, и выбранная наугад
      // назвала бы чужой документ уверенным тоном.
      //
      // Скан представления дорог: 45,6 мс при шестидесяти тысячах блокировок, и отбор в
      // WHERE не помогает вовсе. Поэтому он один на проход и только тогда, когда есть
      // что разбирать.
      BuildLockLists(Buffer,HobtList,SpidList);
      // Отметка обрезки - своя у каждого разбора, и обнуляется она ДО решения "разбирать
      // нечего": разбирать и вправду нечего, а вчерашнее "обрезано" осталось бы стоять.
      LocksTruncated := FALSE;
      IF (HobtList = '') OR (SpidList = '') THEN
        EXIT;

      SqlConn := SqlConn.SqlConnection(ConnString);
      SqlConn.Open;
      SqlCmd := SqlConn.CreateCommand;
      SqlCmd.CommandTimeout := ReadTimeoutSec;
      SqlCmd.CommandText := LocksOfInterestSql(HobtList,SpidList);
      SqlReader := SqlCmd.ExecuteReader;
      WHILE SqlReader.Read DO BEGIN
        Locks := Locks + 1;
        IF Locks <= MaxLocks THEN BEGIN
          LockSpid[Locks] := ReaderInt(SqlReader,LocksColumnSpid);
          LockHobt[Locks] := SqlReader.GetInt64(LocksColumnHobt);
          LockObject[Locks] := ReaderText(SqlReader,LocksColumnObjectName);
          LockIndex[Locks] := ReaderText(SqlReader,LocksColumnIndexName);
          LockHash[Locks] := Watch.TrimLockHash(ReaderText(SqlReader,LocksColumnLockHash));
        END;
      END;
      LocksTruncated := Locks > MaxLocks;
      // Лишняя строка - улика, а не строка ответа: в массивах её нет, и в разбор
      // она ехать не должна вовсе - там цикл идёт до этого числа.
      IF LocksTruncated THEN
        Locks := MaxLocks;
      SqlReader.Close;

      Buffer.RESET;
      IF Buffer.FINDSET(TRUE) THEN
        REPEAT
          Slot := FindWaitedLock(LockSpid,LockHobt,Locks,Buffer."Victim SPID",Buffer."Hobt Id");
          IF Slot > 0 THEN
            ByContendedRow(SqlConn,Buffer,Context,LockObject[Slot],LockIndex[Slot],LockHash[Slot]);
        UNTIL Buffer.NEXT = 0;
      Buffer.RESET;
      SqlConn.Close;
    END;

    END;

    LOCAL PROCEDURE FindContext@11(VAR Context@1000000069 : TEMPORARY Record 110232;NavTableName@1000000070 : Text) : Boolean;
    BEGIN
      // Имя таблицы в журнале снято с SQL, а в списке контекста лежит имя объекта NAV.
      // Сравниваются они по тому же правилу, по какому строится образец столбца: разойтись
      // им позволено только там, где платформа могла заменить знак.
      Context.RESET;
      IF Context.FINDSET THEN
        REPEAT
          IF Watch.SameSqlName(NavTableName,Context."Table Name") THEN
            EXIT(TRUE);
        UNTIL Context.NEXT = 0;
      EXIT(FALSE);
    END;

    LOCAL PROCEDURE FindWaitedLock@12(VAR LockSpid@1000000071 : ARRAY [40] OF Integer;VAR LockHobt@1000000072 : ARRAY [40] OF BigInteger;Locks@1000000073 : Integer;Spid@1000000074 : Integer;HobtId@1000000075 : BigInteger) : Integer;
    VAR
      Index@1000000076 : Integer;
    BEGIN
      // Спорная строка ищется по hobt И по номеру сеанса жертвы. Невыполненная просьба
      // жертвы на этом hobt ровно одна, и она называет ту строку, за которую идёт спор.
      // Ждущая задача в dm_os_waiting_tasks хэша не несёт вовсе - только hobt, - поэтому
      // спрашивать приходится dm_tran_locks.
      IF (Spid = 0) OR (HobtId = 0) THEN
        EXIT(0);
      FOR Index := 1 TO Locks DO
        IF (LockSpid[Index] = Spid) AND (LockHobt[Index] = HobtId) THEN
          EXIT(Index);
      EXIT(0);
    END;

    LOCAL PROCEDURE BuildLockLists@13(VAR Buffer@1000000077 : TEMPORARY Record 110231;VAR HobtList@1000000078 : Text;VAR SpidList@1000000079 : Text);
    VAR
      Seen@1000000080 : Integer;
    BEGIN
      // Два списка на один запрос: номера сеансов жертв вместе с hobt - для спорных
      // строк. Отбор по номеру сеанса обязателен: без него в ответ попали бы
      // невыполненные просьбы всей базы.
      Setup.FetchOnce;
      HobtList := '';
      SpidList := '';
      Buffer.RESET;
      IF Buffer.FINDSET THEN

    LOCAL PROCEDURE RowByHashSql@15(ObjectName@1000000086 : Text;IndexName@1000000087 : Text;LockHash@1000000088 : Text;ColumnPattern@1000000089 : Text) : Text;
    VAR
      Sql@1000000090 : Text;
    BEGIN
      // Имя столбца не вычисляется, а СПРАШИВАЕТСЯ у сервера по образцу: правило, по
      // которому платформа переводит имя поля в имя столбца, нигде не объявлено, и
      // повторять его по памяти значит однажды промахнуться молча.
      //
      // Двух подходящих столбцов быть не должно: если их два, отвечать нельзя вовсе -
      // выбранный наугад дал бы правдоподобный, но чужой номер. Отсюда TOP 2 и проверка.
      Sql :=
        'DECLARE @col sysname, @hits int, @sql nvarchar(max);' +
        'SELECT @hits ' + Eq + ' COUNT(*) FROM (SELECT ' + TopTwo + ' c.name FROM sys.columns c' +
        ' WHERE c.object_id ' + Eq + ' OBJECT_ID(' + Q + Bracket(ObjectName) + Q + ')' +
        ' AND c.name LIKE ' + Q + ColumnPattern + Q + ') x;' +
        'SELECT ' + TopOne + ' @col ' + Eq + ' c.name FROM sys.columns c' +
        ' WHERE c.object_id ' + Eq + ' OBJECT_ID(' + Q + Bracket(ObjectName) + Q + ')' +
        ' AND c.name LIKE ' + Q + ColumnPattern + Q + ';' +
        'IF @col IS NULL OR @hits <> 1 BEGIN SELECT CONVERT(' + NVarChar(DocumentNoWidth) + ',NULL); END' +
        ' ELSE BEGIN' +
        ' SET @sql ' + Eq + ' N' + Q + 'SELECT ' + TopOne + ' CONVERT(' + NVarChar(DocumentNoWidth) + ',' + Q +
        ' + QUOTENAME(@col) + N' + Q + ') FROM ' + Bracket(ObjectName) +
        ' WITH (INDEX(' + Bracket(IndexName) + '), NOLOCK) WHERE ' + LockResColumn +
        ' ' + Eq + ' @h' + Q + ';' +
        ' EXEC sp_executesql @sql, N' + Q + '@h ' + NVarChar(LockHashWidth) + Q + ', @h ' + Eq + ' ' +
        Q + LockHash + Q + '; END;';
      EXIT(Sql);
    END;

    [External]
    PROCEDURE ReadDeadlocks@20(SinceAt@1000000096 : DateTime;MaxRows@1000000097 : Integer;WantStatements@1000000098 : Boolean;VAR Found@1000000099 : TEMPORARY Record 110231;VAR BufferPresent@1000000100 : Boolean;VAR OldestAt@1000000101 : DateTime;VAR Rows@1000000102 : Integer;VAR Counted@1000000103 : BigInteger) : Boolean;

    [TryFunction]
    LOCAL PROCEDURE TryReadDeadlocks@21(SinceAt@1000000105 : DateTime;MaxRows@1000000106 : Integer;WantStatements@1000000107 : Boolean;VAR Found@1000000108 : TEMPORARY Record 110231;VAR BufferPresent@1000000109 : Boolean;VAR OldestAt@1000000110 : DateTime;VAR Rows@1000000111 : Integer;VAR Counted@1000000112 : BigInteger);
    VAR
      SqlCmd@1000000113 : DotNet "'System.Data, Version=2.0.0.0, Culture=neutral, PublicKeyToken=b77a5c561934e089'.System.Data.SqlClient.SqlCommand";
      EventAt@1000000114 : DateTime;
      ObjectName@1000000115 : Text;
      IndexName@1000000116 : Text;
      CompanyName@1000000117 : Text;
      TableName@1000000118 : Text;
      SiftKeyNo@1000000119 : Integer;
      KeyNo@1000000120 : Integer;
      StatementOut@1000000121 : OutStream;
      StatementBody@1000000122 : Text;
    BEGIN
      SqlConn := SqlConn.SqlConnection(ConnString);
      SqlConn.Open;
      SqlCmd := SqlConn.CreateCommand;
      SqlCmd.CommandTimeout := ReadTimeoutSec;
      SqlCmd.CommandText := DeadlocksSql(MomentDays(SinceAt),MomentMs(SinceAt),MaxRows,WantStatements);
      SqlReader := SqlCmd.ExecuteReader;

      // Первый набор - про сам буфер, а не про события: работает ли сессия system_health
      // и каким событием буфер начинается. Без этих двух ответов выключенная сессия и
      // потерянные события выглядят одинаково - как "взаимоблокировок не было".
      IF SqlReader.Read THEN BEGIN
        BufferPresent := ReaderInt(SqlReader,DeadlockColumnBufferPresent) <> 0;
        OldestAt := SqlMoment(ReaderInt(SqlReader,DeadlockColumnOldestDay),
                            ReaderInt(SqlReader,DeadlockColumnOldestMs));
        // Счёт кругов у самого сервера. Едет он ТУТ ЖЕ, четвёртым столбцом первого набора,
        // а не отдельным запросом: соединение и так открыто, а лишний заход к серверу стоит
        // дороже самого счёта. Права он не требует новых - всё тот же VIEW SERVER STATE.
        Counted := ReaderBigInt(SqlReader,DeadlockColumnServerCount);
      END;
      SqlReader.NextResult;

      WHILE SqlReader.Read AND (Rows < MaxRows) DO BEGIN
        Rows := Rows + 1;
        EventAt := SqlMoment(ReaderInt(SqlReader,DeadlockColumnEventDay),
                           ReaderInt(SqlReader,DeadlockColumnEventMs));
        Found.INIT;
        Found."Entry No." := Rows;
        Found."Victim SPID" := ReaderInt(SqlReader,DeadlockColumnVictimSpid);
        // Колонки 3 и 4 (узел и процесс жертвы) остаются в запросе ради нумерации.
        Found."Victim Is NAV" := NavFlag(ReaderText(SqlReader,DeadlockColumnVictimProgram));
        Found."Victim Transaction Id" := ReaderBigInt(SqlReader,DeadlockColumnVictimTran);
        Found."Last Wait (ms)" := ReaderInt(SqlReader,DeadlockColumnWaitMs);
        Found."Max Wait (ms)" := Found."Last Wait (ms)";
        Found."Blocker SPID" := ReaderInt(SqlReader,DeadlockColumnBlockerSpid);
        // Граф взаимоблокировки называет участников сам: узел, логин и программу он держит
        // прямо в узле процесса. Программа едет в свою колонку - по ней видно, чей это сеанс.
        Found."Blocker Program" :=
          COPYSTR(ReaderText(SqlReader,DeadlockColumnBlockerProgram),1,
                  MAXSTRLEN(Found."Blocker Program"));
        Found."Blocker Host" :=
          COPYSTR(ReaderText(SqlReader,DeadlockColumnBlockerHost),1,
                  MAXSTRLEN(Found."Blocker Host"));
        Found."Blocker Login" :=
          COPYSTR(ReaderText(SqlReader,DeadlockColumnBlockerLogin),1,
                  MAXSTRLEN(Found."Blocker Login"));
        // Колонки 5, 11, 12, 18 да 19 и 26 (программа, начало транзакции, режим, тип
        // ожидания и логин жертвы) остаются в запросе ради нумерации, но в строку не
        // ложатся: столбцы 16, 17 и 20 читаются по прежним номерам.
        Found."Blocker Transaction Id" := ReaderBigInt(SqlReader,DeadlockColumnBlockerTran);
        Found."Resource Kind" := COPYSTR(UPPERCASE(ReaderText(SqlReader,DeadlockColumnResourceKind)),1,
                                         MAXSTRLEN(Found."Resource Kind"));
        Found."Hobt Id" := ReaderBigInt(SqlReader,DeadlockColumnHobtId);
        Found."Database Id" := ReaderInt(SqlReader,DeadlockColumnDatabaseId);
        ObjectName := ReaderText(SqlReader,DeadlockColumnObjectName);
        IndexName := ReaderText(SqlReader,DeadlockColumnIndexName);
        Found."Deadlock Id" := Watch.TextFingerprint(ReaderText(SqlReader,DeadlockColumnDeadlockId));
        Found."Blocker Statement" := COPYSTR(ReaderText(SqlReader,DeadlockColumnBlockerStatement),1,
                                             MAXSTRLEN(Found."Blocker Statement"));
        StatementBody := Watch.StatementWithMark(ReaderText(SqlReader,DeadlockColumnStatementBody),
                                                 ReaderInt(SqlReader,DeadlockColumnStatementLength));

        Found.Outcome := Watch.OutcomeDeadlocked;

        // Эпизод приезжает УЖЕ ЗАКОНЧЕННЫМ: круг разорвал сервер, и конец известен точно.
        // Начало считается назад от конца по времени ожидания жертвы - тем же правилом,
        // каким считается начало обычного эпизода.
        Found."Ended At" := EventAt;
        Found."Last Seen At" := EventAt;
        Found."Started At" := Watch.EpisodeStart(EventAt,Found."Last Wait (ms)");
        Found.Open := FALSE;

        KeyNo := Watch.UnknownKeyNo;
        IF Watch.SplitSqlObjectName(ObjectName,CompanyName,TableName,SiftKeyNo) THEN BEGIN
          IF SiftKeyNo <> Watch.UnknownKeyNo THEN
            KeyNo := SiftKeyNo
          ELSE
            KeyNo := Watch.KeyNoFromIndexName(IndexName);
          NameRow(Found,ObjectName,TableName,KeyNo,SiftKeyNo <> Watch.UnknownKeyNo);
        END ELSE
          // Имя, которое не разобралось, показывается СЫРЫМ и метится непонятым. Пустая
          // колонка на его месте читалась бы как "таблица неизвестна серверу".
          UnnamedRow(Found,ObjectName);
        IF StatementBody <> '' THEN BEGIN
          Found."Blocker Statement Text".CREATEOUTSTREAM(StatementOut,TEXTENCODING::UTF8);
          StatementOut.WRITETEXT(StatementBody);
        END;
        Found.INSERT;
      END;
      SqlReader.Close;
      SqlConn.Close;
    END;

    LOCAL PROCEDURE DeadlocksSql@22(SinceDays@1000000123 : Integer;SinceMs@1000000124 : Integer;MaxRows@1000000125 : Integer;WantStatements@1000000126 : Boolean) : Text;
    VAR
      Sql@1000000127 : Text;
      Events@1000000128 : Text;
      Statement@1000000129 : Text;
      TranStarted@1000000130 : Text;
    BEGIN
      // Граф разбирает СЕРВЕР, а в C/AL приезжают готовые столбцы. Тащить XML в NAV
      // значило бы завести второй разбор имён и держать его в согласии с первым.
      //
      // Кольцевой буфер превращается в XML ОДИН раз, в переменную: обращений к нему два -
      // за границей буфера и за самими событиями, - а разбор многомегабайтной строки
      // стоит столько, что платить за него дважды нельзя.
      //
      // QUOTED_IDENTIFIER включается явно. Драйвер и так включает его на соединении, но
      // без него ЛЮБОЙ метод XML отказывает, и отказ называет не причину, а список
      // требований к настройкам сеанса.
      //
      // Отметка собирается через datetime2(3), а не через datetime. У datetime шаг 3,33 мс,
      // и он округляет ВВЕРХ: собранная им отметка оказывается на полтора миллисекунды
      // позже настоящей, а событие ровно на границе при этом пропускается молча. Ради
      // него отбор и сделан нестрогим, так что округление обесценило бы всю защиту.
      Events := Q + '/RingBufferTarget/event[@name' + Eq + '"xml_deadlock_report"]' + Q;
      // nvarchar(max) метод .value() не принимает вовсе, и это не наш выбор потолка:
      // сервер САМ обрезает входной буфер на четырёх тысячах знаков, поэтому 4000 здесь
      // ничего не теряет. Отметка усечения на этой дороге поэтому и не срабатывает.
      IF WantStatements THEN
        Statement := ' LTRIM(CONVERT(nvarchar(250),bp.n' + XVal('(inputbuf)[1]','nvarchar(500)') + ')),' +
                     ' LTRIM(bp.n' + XVal('(inputbuf)[1]','nvarchar(4000)') + '),' +
                     ' CONVERT(int,ISNULL(LEN(LTRIM(bp.n' + XVal('(inputbuf)[1]','nvarchar(4000)') + ')),0))'
      ELSE
        // Текст оператора едет со ЗНАЧЕНИЯМИ - теми самыми данными, которых мы обещали не
        // собирать. Он приходит только под галкой, выключенной по умолчанию.
        Statement := ' SPACE(0), SPACE(0), 0';

      // Атрибут графа сервер пишет по местным часам, а отдать его надо по UTC - той же
      // шкалой, в какой NAV хранит DateTime. Выражение собирается ОДИН раз и подставляется
      // трижды: разъехавшись между вхождениями, оно дало бы сутки вместо миллисекунд.
      TranStarted := 'DATEADD(minute,@shift,bp.n' + XVal('@lasttranstarted','datetime2(3)') + ')';
      Sql :=
        'SET QUOTED_IDENTIFIER ON;' +
        'DECLARE @x xml, @since datetime2(3), @shift int;' +
        'SET @shift ' + Eq + ' DATEDIFF(minute,GETDATE(),GETUTCDATE());' +
        'SET @since ' + Eq + ' DATEADD(ms,' + FORMAT(SinceMs) +
        ',CONVERT(datetime2(3),CONVERT(datetime,' + FORMAT(SinceDays) + ')));' +
        'SELECT @x ' + Eq + ' CONVERT(xml,t.target_data)' +
        ' FROM sys.dm_xe_session_targets t' +
        ' JOIN sys.dm_xe_sessions s ON s.address ' + Eq + ' t.event_session_address' +
        ' WHERE s.name ' + Eq + ' ' + Q + HealthSessionName + Q +
        ' AND t.target_name ' + Eq + ' ' + Q + RingBufferTargetName + Q + ';';

      // Время события в буфере лежит по UTC - и УЖЕ в той шкале, в какой NAV хранит свои
      // отметки, поэтому сдвигать его больше не надо. Прежде оно переводилось в местное,
      // чтобы сойтись с остальным журналом; теперь весь журнал перешёл на UTC, и перевод
      // стал бы ошибкой на разницу поясов.
      //
      // Сдвиг, однако, нужен - но в ДРУГУЮ сторону и для другого. Атрибуты самого графа
      // (@lasttranstarted) сервер пишет по МЕСТНЫМ часам, и к UTC их приводит он же.
      Sql := Sql +
        'SELECT CASE WHEN @x IS NULL THEN 0 ELSE 1 END,' +
        ' ISNULL(DATEDIFF(day,CONVERT(datetime,0),MIN(v.EventAt)),0),' +
        ' ISNULL(DATEDIFF(ms,CONVERT(date,MIN(v.EventAt)),MIN(v.EventAt)),0),' +
        // Круги сервер считает сам, и счёт этот от кольца не зависит вовсе: кольцо может
        // замолчать целиком, а счётчик продолжит расти. Имя счётчика кончается на "/sec",
        // но величина накопительная - это давняя странность имён у счётчиков SQL Server.
        // Отбор по имени нестрогий: counter_name объявлен nchar(128) и добит пробелами.
        ' ISNULL((SELECT ' + TopOne + ' c.cntr_value FROM sys.dm_os_performance_counters c' +
        ' WHERE c.counter_name LIKE ' + Q + DeadlockCounterName + Wildcard + Q +
        ' AND c.instance_name ' + Eq + ' ' + Q + TotalInstanceName + Q + '),0)' +
        ' FROM (SELECT n.e' + XVal('@timestamp','datetime2(3)') +
        ' AS EventAt FROM @x.nodes(' + Events + ') AS n(e)) v;';

      Sql := Sql +
        'WITH ev AS (SELECT n.e' + XVal('@timestamp','datetime2(3)') +
        ' AS EventAt, n.e.query(' + Q + 'data[@name' + Eq + '"xml_report"]/value/deadlock' +
        Q + ') AS G FROM @x.nodes(' + Events + ') AS n(e)),' +
        ' vic AS (SELECT ev.EventAt, ev.G, v.n' + XVal('@id','varchar(60)') + ' AS VictimId' +
        ' FROM ev CROSS APPLY ev.G.nodes(' + Q + '/deadlock/victim-list/victimProcess' + Q +
        ') AS v(n)),';

      // Ресурс берётся не любой из графа, а ТОТ, в очереди к которому стоит жертва: в
      // графе их столько, сколько сторон у круга, и любой другой назвал бы чужую таблицу.
      Sql := Sql +
        ' res AS (SELECT vic.EventAt, vic.G, vic.VictimId,' +
        ' r.n' + XVal('local-name(.)','varchar(30)') + ' AS Kind,' +
        ' r.n' + XVal('@hobtid','bigint') + ' AS HobtId,' +
        ' r.n' + XVal('@dbid','int') + ' AS DbId,' +
        ' r.n' + XVal('@objectname','nvarchar(300)') + ' AS ObjectName,' +
        ' r.n' + XVal('@indexname','nvarchar(300)') + ' AS IndexName,' +
        ' r.n' + XVal('(owner-list/owner/@id)[1]','varchar(60)') + ' AS OwnerId,' +
        ' r.n' + XVal('(owner-list/owner/@mode)[1]','varchar(10)') + ' AS OwnerMode,' +
        ' r.n' + XVal('(waiter-list/waiter/@mode)[1]','varchar(10)') + ' AS WaiterMode' +
        ' FROM vic CROSS APPLY vic.G.nodes(' + Q + '/deadlock/resource-list/*[waiter-list/waiter/@id' +
        Eq + 'sql:column("vic.VictimId")]' + Q + ') AS r(n))';

      Sql := Sql +
        ' SELECT TOP (' + FORMAT(MaxRows) + ')' +
        ' DATEDIFF(day,CONVERT(datetime,0),res.EventAt),' +
        ' DATEDIFF(ms,CONVERT(date,res.EventAt),res.EventAt),' +
        ' vp.n' + XVal('@spid','int') + ',' +
        ' vp.n' + XVal('@hostname','nvarchar(80)') + ',' +
        ' vp.n' + XVal('@hostpid','int') + ',' +
        ' vp.n' + XVal('@clientapp','nvarchar(120)') + ',' +
        ' vp.n' + XVal('@xactid','bigint') + ',' +
        ' vp.n' + XVal('@waittime','int') + ',' +
        ' bp.n' + XVal('@spid','int') + ',' +
        ' bp.n' + XVal('@clientapp','nvarchar(120)') + ',' +
        ' bp.n' + XVal('@xactid','bigint') + ',' +
        ' DATEDIFF(day,CONVERT(datetime,0),' + TranStarted + '),' +
        ' DATEDIFF(ms,CONVERT(date,' + TranStarted + '),' + TranStarted + '),' +
        ' res.Kind, res.HobtId, res.DbId, ISNULL(PARSENAME(res.ObjectName,1),res.ObjectName),' +
        ' res.IndexName, res.OwnerMode, res.WaiterMode,';

      // Отпечаток графа собирается из того, что у события не меняется между чтениями:
      // время, обе транзакции, hobt и ресурс, которого ждала жертва. Место в буфере в
      // отпечаток не входит - оно меняется при каждом чтении, а граф нет.
      Sql := Sql +
        ' CONVERT(varchar(30),res.EventAt,126) + ' + Q + '|' + Q +
        ' + CONVERT(varchar(20),vp.n' + XVal('@xactid','bigint') + ') + ' + Q + '|' + Q +
        ' + CONVERT(varchar(20),bp.n' + XVal('@xactid','bigint') + ') + ' + Q + '|' + Q +
        ' + CONVERT(varchar(20),res.HobtId) + ' + Q + '|' + Q +
        ' + vp.n' + XVal('@waitresource','varchar(120)') + ',' +
        Statement +
        // Дописано ПОСЛЕ Statement нарочно: он всегда ровно три столбца - и под галкой на
        // тексты, и без неё, - значит хвост неподвижен и ни один прежний номер не поехал.
        ', bp.n' + XVal('@hostname','nvarchar(80)') + ',' +
        ' bp.n' + XVal('@loginname','nvarchar(128)') + ',' +
        ' vp.n' + XVal('@loginname','nvarchar(128)') +
        ' FROM res' +
        ' CROSS APPLY res.G.nodes(' + Q + '/deadlock/process-list/process[@id' + Eq +
        'sql:column("res.VictimId")]' + Q + ') AS vp(n)' +
        ' CROSS APPLY res.G.nodes(' + Q + '/deadlock/process-list/process[@id' + Eq +
        'sql:column("res.OwnerId")]' + Q + ') AS bp(n)' +
        // Отбор нестрогий нарочно: два события в одну миллисекунду при строгом отборе
        // потеряли бы второе молча. Повтор снимает отпечаток, потеря не снимается ничем.
        ' WHERE res.EventAt >' + Eq + ' @since' +
        // Чужая база на том же сервере не должна занимать строки потолка: TOP считается
        // уже после этого отбора, и взаимоблокировки другого предприятия в наш журнал не
        // попадут (а при потолке ещё и вытеснили бы свои).
        ' AND res.DbId ' + Eq + ' DB_ID()' +
        ' ORDER BY res.EventAt;';
      EXIT(Sql);
    END;

    LOCAL PROCEDURE UnnamedRow@24(VAR Row@1000000136 : Record 110231;ObjectName@1000000137 : Text);

    LOCAL PROCEDURE MomentDays@25(At@1000000138 : DateTime) : Integer;
    VAR
      Elapsed@1000000139 : Duration;
      Days@1000000140 : Decimal;
    BEGIN
      // Отметка едет в запрос теми же двумя целыми, какими время приходит обратно. Дата
      // строкой разбиралась бы по языку сервера и ошиблась бы не отказом, а сдвигом -
      // тем самым, из-за которого события пропускались бы молча.
      //
      // Считается всё РАЗНИЦЕЙ МОМЕНТОВ, а не вычитанием дат: "дата минус дата" в C/AL
      // берёт вторую дату за число дней и на любой настоящей дате роняет проход с
      // "The date is not valid". Ошибка приходит не при компиляции, а на первом же
      // непустом значении отметки - то есть на втором проходе, а не на первом.
      IF At = 0DT THEN
        EXIT(0);
      Elapsed := At - CREATEDATETIME(SqlEpoch,0T) - SessionShift;
      Days := ROUND(Elapsed / Watch.MsPerDay,1,'<');
      EXIT(Days);
    END;

    LOCAL PROCEDURE MomentMs@26(At@1000000141 : DateTime) : Integer;

      Elapsed@1000000142 : Duration;
    BEGIN
      IF At = 0DT THEN

      Elapsed := At - CREATEDATETIME(SqlEpoch + MomentDays(At),0T) - SessionShift;

    LOCAL PROCEDURE ApplyName@28(VAR Observed@1000000145 : TEMPORARY Record 110231;Source@1000000146 : Integer;Id@1000000147 : BigInteger;ObjectName@1000000148 : Text;IndexName@1000000149 : Text);
    VAR
      CompanyName@1000000150 : Text;
      TableName@1000000151 : Text;
      SiftKeyNo@1000000152 : Integer;
      KeyNo@1000000153 : Integer;
      SqlObjectId@1000000154 : Integer;
      Parsed@1000000155 : Boolean;
    BEGIN
      // Неразобранное имя - не повод бросить строку. Здесь стоял выход, и строка после
      // него оставалась без имени ВОВСЕ - даже без пометки "не опознано". Читается это
      // как "сервер таблицы не назвал", хотя сервер её назвал: не понял её разбор.
      // Ловится на блокировке платформенной таблицы $ndo$ - их на базу около дюжины, и
      // таблицами NAV они не являются, поэтому разбор честно отвечает "не моё".
      Parsed := Watch.SplitSqlObjectName(ObjectName,CompanyName,TableName,SiftKeyNo);
      KeyNo := Watch.UnknownKeyNo;
      // Правил два, и одно теряет самый частый случай блокировок NAV молча: у таблицы
      // номер ключа несёт имя ИНДЕКСА, у SIFT-представления - имя самого объекта.
      IF Parsed THEN BEGIN
        IF SiftKeyNo <> Watch.UnknownKeyNo THEN
          KeyNo := SiftKeyNo
        ELSE
          KeyNo := Watch.KeyNoFromIndexName(IndexName);
      END;

      Observed.RESET;
      IF Source = SourceHobt THEN
        Observed.SETRANGE("Hobt Id",Id)
      ELSE BEGIN
        SqlObjectId := Id;
        Observed.SETRANGE("Hobt Id",0);
        Observed.SETRANGE("SQL Object Id",SqlObjectId);
      END;
      IF Observed.FINDSET(TRUE) THEN
        REPEAT
          IF Parsed THEN
            NameRow(Observed,ObjectName,TableName,KeyNo,SiftKeyNo <> Watch.UnknownKeyNo)
          ELSE
            UnnamedRow(Observed,ObjectName);
          Observed.MODIFY;
        UNTIL Observed.NEXT = 0;
      Observed.RESET;
    END;

    LOCAL PROCEDURE BuildLists@29(VAR Observed@1000000156 : TEMPORARY Record 110231;VAR HobtList@1000000157 : Text;VAR ObjectList@1000000158 : Text);
    BEGIN
      // Повторы в списке не убираем: в наборе SQL одному hobt отвечает одна строка, и
      // ответ приходит различённым сам собой. Отдельный проход ради уникальности стоил бы
      // дороже, чем лишние знаки в тексте запроса.
      HobtList := '';
      ObjectList := '';
      Observed.RESET;

              ObjectList := AppendId(ObjectList,FORMAT(Observed."SQL Object Id"));
        UNTIL Observed.NEXT = 0;
      Observed.RESET;
    END;

    LOCAL PROCEDURE QueueSql@31(MaxRows@1000000161 : Integer;WantStatements@1000000162 : Boolean) : Text;
    VAR
      Sql@1000000163 : Text;
      Cols@1000000164 : Text;
      Joins@1000000165 : Text;
      Cap@1000000166 : Text;
    BEGIN
      // Текст запроса стоит четырёх присоединений к горячему запросу прохода, и потому он
      // ВЫКЛЮЧАЕМ. Выключенный не платит ничем: колонки остаются на месте пустыми, а
      // присоединений в запросе нет вовсе - читателю не приходится знать, какой сегодня
      // набор столбцов.
      //
      // Жертва ждёт, то есть у неё есть исполняемый запрос, и текст берётся по нему с
      // точными границами оператора. Виновник чаще СПИТ - в этом вся беда NAV-блокировок, -
      // и запроса у него нет; берётся последний, что он послал, из его соединения. Это и
      // есть тот оператор, который взял блокировку.
      Cap := FORMAT(Watch.MaxStatementChars);
      IF WantStatements THEN BEGIN
        Cols :=
          ', LEFT(vst.t,' + Cap + '), CONVERT(int,ISNULL(LEN(vst.t),0)),' +
          ' LEFT(bst.t,' + Cap + '), CONVERT(int,ISNULL(LEN(bst.t),0))';
        Joins :=
          ' OUTER APPLY (SELECT ' + TopOne + ' r.sql_handle AS h, r.statement_start_offset AS so,' +
          ' r.statement_end_offset AS eo FROM sys.dm_exec_requests r' +
          ' WHERE r.session_id ' + Eq + ' wt.session_id) vq' +
          ' OUTER APPLY sys.dm_exec_sql_text(vq.h) vtx' +
          ' OUTER APPLY (SELECT LTRIM(SUBSTRING(vtx.text,(vq.so/2)+1,' +
          ' ((CASE vq.eo WHEN -1 THEN DATALENGTH(vtx.text) ELSE vq.eo END - vq.so)/2)+1)) AS t) vst' +
          ' OUTER APPLY (SELECT ' + TopOne + ' c.most_recent_sql_handle AS h FROM sys.dm_exec_connections c' +
          ' WHERE c.session_id ' + Eq + ' wt.blocking_session_id) bq' +
          ' OUTER APPLY sys.dm_exec_sql_text(bq.h) btx' +
          ' OUTER APPLY (SELECT LTRIM(btx.text) AS t) bst';
      END ELSE BEGIN
        Cols := ', SPACE(0), 0, SPACE(0), 0';
        Joins := '';
      END;
      // Каждое целое приведено к int нарочно, и это не перестраховка: номер сеанса и
      // номер базы в системных представлениях - smallint, а GetInt32 на smallint падает
      // с "Specified cast is not valid". Отказ приходит на ПЕРВОЙ строке очереди, то есть
      // ровно тогда, когда блокировка наконец случилась, - и выглядит как "инструмент не
      // видит блокировок". Тип столбца в DMV проверять раньше, чем себя.
      //
      // Порядок по убыванию длительности не украшение: при усечении в журнал попадают
      // САМЫЕ ДОЛГИЕ ожидания, а не первые попавшиеся. Усечённый проход честнее метится
      // отдельно, но и то, что в него попало, должно быть тем, за чем пришли.
      // ШКАЛА ВРЕМЕНИ ЗДЕСЬ - UTC, и другой она быть не может. Часы серверной сессии NAV
      // идут по UTC, в колонку момент ложится как есть, а в пояс читателя его переводит
      // клиент; отдай сюда местное - и человек увидит БУДУЩЕЕ ровно на разницу поясов.
      Sql :=
        'DECLARE @now datetime2 = SYSUTCDATETIME();' +
        'SELECT DATEDIFF(day,CONVERT(datetime,0),@now),' +
        ' DATEDIFF(ms,CONVERT(date,@now),@now), CONVERT(int,DB_ID());' +
        'SELECT TOP (' + FORMAT(MaxRows) + ')' +
        ' CONVERT(int,wt.session_id), CONVERT(int,wt.blocking_session_id),' +
        ' CONVERT(int,CASE WHEN wt.wait_duration_ms > ' + FORMAT(IntegerCeiling) +
        ' THEN ' + FORMAT(IntegerCeiling) + ' ELSE wt.wait_duration_ms END),' +
        ' wt.wait_type, wt.resource_description,' +
        ' vs.host_name, vs.host_process_id, vs.program_name,' +
        ' vt.transaction_id, bt.transaction_id,' +
        ' bs.program_name' +
        Cols +
        // Дописано ПОСЛЕ Cols нарочно, и это не небрежность. Cols - всегда ровно четыре
        // столбца (при выключенных текстах он отдаёт четыре пустышки), поэтому хвост
        // неподвижен и ни один прежний номер колонки не поехал. Сдвиг на единицу в этом
        // же запросе уже стоил нам молчаливой поломки - заметка о ней выше по коду.
        ', bs.login_name, bs.host_name, vs.login_name' +
        ' FROM sys.dm_os_waiting_tasks wt' +
        ' JOIN sys.dm_exec_sessions vs ON vs.session_id = wt.session_id' +
        ' LEFT JOIN sys.dm_tran_session_transactions vt ON vt.session_id = wt.session_id' +
        ' LEFT JOIN sys.dm_exec_sessions bs ON bs.session_id = wt.blocking_session_id' +
        ' LEFT JOIN sys.dm_tran_session_transactions bt' +
        ' ON bt.session_id = wt.blocking_session_id' +
        Joins +
        ' WHERE wt.wait_type LIKE ' + Q + LockWaitPattern + Q +
        ' AND wt.session_id <> @@SPID' +
        ' ORDER BY wt.wait_duration_ms DESC;';
      EXIT(Sql);
    END;

    LOCAL PROCEDURE SqlMoment@35(Days@1000000174 : Integer;Milliseconds@1000000175 : Integer) : DateTime;
    VAR
      Offset@1000000176 : Duration;
    BEGIN
      // Время приходит ДВУМЯ целыми, а не строкой, и это не причуда: разбор строки даты
      // зависит от языка сеанса и от языка сервера, и ошибётся он не отказом, а сдвигом.
      // Сутки от начала отсчёта SQL плюс миллисекунды внутри суток врать не умеют.
      //
      // Части приезжают по UTC, а CREATEDATETIME читает дату и время в поясе СЕССИИ. В
      // фоновой сессии это UTC, и поправка не нужна. В сессии клиента - пояс его машины:
      // проход, запущенный кнопкой ("Один проход сейчас", "Демо"), без поправки собрал бы на
      // московском клиенте момент на три часа раньше настоящего, и сторож назвал бы идущую
      // цепочку замолчавшей 180 минут назад (FINDINGS, раздел 102).
      IF Days <= 0 THEN
        EXIT(0DT);
      Offset := Milliseconds;
      EXIT(CREATEDATETIME(SqlEpoch + Days,0T) + Offset + SessionShift);
    END;

    LOCAL PROCEDURE SessionShift@36() : Duration;
    VAR
      TypeHelper@1000000177 : Codeunit 10;
      UtcClock@1000000178 : DotNet "'mscorlib, Version=4.0.0.0, Culture=neutral, PublicKeyToken=b77a5c561934e089'.System.DateTime";
      Moment@1000000179 : DateTime;
      Shift@1000000180 : Duration;
      Minutes@1000000181 : BigInteger;
    BEGIN
      // Насколько CREATEDATETIME этой сессии отстаёт от UTC. Пояс сессии МЕРИТСЯ, а не
      // угадывается: от настройки службы он зависит по-разному у разных сессий. На стенде
      // фоновая сессия шла по UTC, а в установке с поясом сервера - по поясу машины службы,
      // и эпизоды фонового прохода ложились на три часа раньше (19.10.2026). Клиентская
      // сессия идёт по поясу машины пользователя, которого служба не знает вовсе.
      //
      // Мера одна на все случаи: настоящий момент UTC минус тот же момент, собранный
      // CREATEDATETIME из своих же UTC-даты и времени. Два чтения часов расходятся на
      // миллисекунды, и потому разница округляется до минуты - пояса минутами и меряются.
      // Время дня собирается из часов, минут, секунд и миллисекунд: свойство TimeOfDay
      // отдаёт TimeSpan, и цепочку к его свойствам компилятор C/AL не принимает.
      Moment := TypeHelper.GetCurrUTCDateTime;
      UtcClock := UtcClock.UtcNow;
      Shift := Moment - CREATEDATETIME(DMY2DATE(UtcClock.Day,UtcClock.Month,UtcClock.Year),0T) -
        ((UtcClock.Hour * 60 + UtcClock.Minute) * 60 + UtcClock.Second) * 1000 - UtcClock.Millisecond;
      Minutes := ROUND(Shift / Watch.MsPerMinute,1);
      Shift := Minutes * Watch.MsPerMinute;
      EXIT(Shift);
    END;

    LOCAL PROCEDURE "------- Legal Hardcodes -------"@45();
    BEGIN
      // Разделитель, а не работа: ниже идут только величины, у которых есть имя.
      // Число в коде читается как случайное, а величины, зависящие от среды, живут
      // в настройке LockWatch Setup - таймауты SQL читаются оттуда, - здесь же пределы
      // самого инструмента, имена платформы и ординалы столбцов ответа.
      //
