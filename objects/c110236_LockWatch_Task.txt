      Parse@1000000006 : Codeunit 110230;
      NoSchedulerErr@1000000001 : TextConst 'ENU=The service instance runs with the task scheduler switched off (EnableTaskScheduler in the service configuration). Пока он выключен, сторож завестись не может.';
      StartedTxt@1000000002 : TextConst 'ENU=The watch is armed, the first pass is due shortly.;RUS=Сторож заведён, первый проход вот-вот.';

      DisabledTxt@1000000004 : TextConst 'ENU=The watch is switched off in the setup, the chain of passes ends here.;RUS=Сторож выключен в настройке, цепочка проходов на этом кончается.';
      PassFailedTxt@1000000005 : TextConst 'ENU=The pass fell over and was picked up by the task: %1;RUS=Проход упал и был подхвачен задачей: %1';

    [External]
    PROCEDURE StartWatch@1();
    BEGIN
      // Завод сторожа идемпотентен: сначала снимаются все свои задачи, потом ставится одна.
      // Две цепочки проходов, идущие вперемежку, - это не удвоенное наблюдение, а два
      // наблюдателя, закрывающих эпизоды друг другу, и разобрать потом такой журнал нельзя.
      Setup.FetchOnce;
      // Язык журнала, если его не задавали, берётся у того, кто сторожа ЗАВЁЛ: читать
      // журнал будут такие же люди, а фоновой задаче взять язык неоткуда. Сессия без окна
      // (прогон, командлет) его не назначает - у неё язык службы, а не человека.
      IF GUIALLOWED AND (Setup."Journal Language ID" = 0) THEN
        Setup."Journal Language ID" := GLOBALLANGUAGE;
      IF NOT TASKSCHEDULER.CANCREATETASK THEN
        ERROR(NoSchedulerErr);
      // Поколение поднимается ПЕРВЫМ, до снятия задач, и порядок тут значащий. Строка
      // идущей задачи в очереди ЕСТЬ, но CANCELTASK её не останавливает: она доходит до
      // конца и перевзводит себя уже после снятия - цепочек выходило ДВЕ, обе живые, обе
      // перевзводятся (замер 09.09.2026: два задания с разницей в две секунды, состояние
      // устойчивое; при запертом журнале строка идущей задачи после снятия на месте). Подними
      // поколение ПОСЛЕ снятия - и проход, успевший закончиться в этот промежуток,
      // поставил бы задачу, которую снимать уже некому.
      //
      // Видно новое число станет на COMMIT, и до него читающий его проход просто подождёт
      // на строке состояния: очередь на одной строке честнее гонки на двух.
      Watchdog.Arm;
      CancelOurTasks;
      Setup.Enabled := TRUE;
      Setup.MODIFY;
      Watchdog.Say(StartedTxt);
      COMMIT;
      Schedule(FirstDelayMs);
    END;

    [External]
    PROCEDURE StopWatch@2();
    BEGIN
      Setup.FetchOnce;
      Setup.Enabled := FALSE;
      Setup.MODIFY;
      Watchdog.Say(StoppedTxt);
      COMMIT;
      // Снимаем задачу И гасим выключатель. Одного выключателя мало: уже поставленная
      // задача всё равно проснётся, и человек увидит проход после того, как нажал "стоп".
      CancelOurTasks;
    END;

    [External]
    PROCEDURE RunOnce@3();
    VAR
      Pass@1000000000 : Codeunit 110235;
    BEGIN
      // Вручную - напрямую и БЕЗ перехвата: человек, нажавший кнопку, должен увидеть
      // отказ окном, а не вычитать его потом из сообщения сторожа.
      Pass.RunPass;
    END;

    [External]
    PROCEDURE RefreshContextNames@12() : Integer;
    VAR
      ContextTable@1000000000 : Record 110232;
      Before@1000000001 : Text;
      Changed@1000000002 : Integer;
    BEGIN
      // Перечитать имена таблиц и полей по номерам. Нужно ровно там, где строку завели
      // мимо экрана, и ещё там, где таблицу или поле переименовали: в настройке осталось
      // прежнее имя, а съём сверяет его с именем из SQL и после переименования не сойдётся.
      // Возвращается число ИЗМЕНЁННЫХ строк, а не всех: "перечитано 0" и "перечитано 7" -
      // разные ответы, и второй значит, что настройка разошлась с базой и это заметили.
      IF ContextTable.FINDSET(TRUE) THEN
        REPEAT
          Before := ContextTable."Table Name" + '|' + ContextTable."Document Field Name";
          ContextTable.FillNames;
          IF Before <> ContextTable."Table Name" + '|' + ContextTable."Document Field Name" THEN BEGIN
            ContextTable.MODIFY;
            Changed := Changed + 1;
          END;
        UNTIL ContextTable.NEXT = 0;
      EXIT(Changed);
    END;

    [External]
    PROCEDURE ScheduledCount@4() : Integer;
    VAR
      ScheduledTask@1000000000 : Record 2000000175;
    BEGIN
      ScheduledTask.SETRANGE("Run Codeunit",CODEUNIT::"LockWatch Task");
      ScheduledTask.SETRANGE(Company,COMPANYNAME);
      EXIT(ScheduledTask.COUNT);
    END;

    LOCAL PROCEDURE RunAndReschedule@5();
    VAR
      Generation@1000000000 : Integer;
    BEGIN
      Setup.FetchOnce;
      // Задача пишет свои слова тем же языком, что и проход: сообщение "сторож выключен",
      // пришедшее по-английски в русский журнал, выглядит чужим - и им и является.
      Parse.UseJournalLanguage(Setup."Journal Language ID");
      IF NOT Setup.Enabled THEN BEGIN
        // Выключатель сняли между проходами. Цепочка обрывается сама и говорит об этом:
        // сторож, замолчавший без объяснения, неотличим от сторожа, который умер.
        Watchdog.Say(DisabledTxt);
        COMMIT;
        EXIT;
      END;

      // Поколение своей цепочки запоминается ДО прохода: пока он идёт, сторожа могут
      // завести заново, и тогда эта цепочка уже не та, что должна продолжаться.
      Generation := Watchdog.Generation;

      // Проход зовётся через CODEUNIT.RUN нарочно, И ОТВЕТ ЕГО ИСПОЛЬЗУЕТСЯ: без этого
      // ошибка не перехватывается вовсе. Своя ошибка внутри прохода откатит его
      // транзакцию, но НЕ уронит задачу - иначе цепочка оборвалась бы на первой же
      // неожиданности, и наблюдение кончилось бы молча, посреди ночи.
      //
      // Проверено замером 10.09.2026, а не рассуждением: журнал держат замком дольше
      // предела ожидания NAV, проход срывается по-настоящему, сторож пишет "проход упал
      // и был подхвачен задачей", и проходы идут дальше. Снять перехват - и цепочка
      // умирает молча: ни слова в журнале сторожа, ни задачи в планировщике.
      IF NOT CODEUNIT.RUN(CODEUNIT::"LockWatch Pass") THEN
        NoteFailure(GETLASTERRORTEXT);

      Setup.FetchOnce;
      Reschedule(Generation);
    END;

    LOCAL PROCEDURE Reschedule@6(Generation@1000000002 : Integer);
    VAR
      DelayMs@1000000000 : Integer;
      Floor@1000000001 : Integer;
    BEGIN
      // Обратная связь работает на РАЗРЕЖЕНИЕ, а не на учащение. Правило простое и
      // проверяемое: наблюдатель не берёт себе больше одной десятой времени. Если проход
      // стоил 450 мс, следующий будет не раньше чем через 4500, каким бы ни был период.
      // Учащение при занятости было бы разгоном самого себя: в шторм проход дорожает, а
      // от учащения дорожал бы ещё, и наблюдатель добил бы сервер, который пришёл лечить.
      // Выключатель перечитывается ЗДЕСЬ, а не только в начале задачи. Между этими двумя
      // мигами проходит целый проход, и человек успевает нажать "стоп". Снять идущую задачу
      // остановке нечем, а перевзвод тут же поставил бы новую - сторож воскресал бы сам,
      // ровно один раз, и объяснить это было бы нечем.
      //
      // Окно длиной в проход открыто миллисекунды из трёх секунд, и попасть в него СНОМ
      // нельзя - потому долго и считалось, что прогоном это не ловится. Ловится: проход
      // держат замком на ЖУРНАЛЕ, который он читает и пишет каждым проходом, и окно
      // становится каким угодно широким. Так проверены ОБА случая - и завод поверх идущего
      // прохода, и остановка. Опасение, что остановку пришлось бы держать вдвое дольше,
      // замер снял: сама она укладывается в 466 мс, ставить ей нечего.
      IF NOT Setup.Enabled THEN BEGIN
        Watchdog.Say(DisabledTxt);
        COMMIT;
        EXIT;
      END;
      // Пока шёл проход, цепочку завели заново. Остановить эту задачу завод не мог - снятие
      // на идущую не действует, - и перевзвод создал бы ВТОРУЮ цепочку. Две
      // цепочки вперемежку - это не удвоенное наблюдение, а два наблюдателя, закрывающих
      // эпизоды друг другу, и разобрать потом такой журнал нельзя.
      IF Watchdog.Generation <> Generation THEN BEGIN
        Watchdog.Say(SupersededTxt);
        COMMIT;
        EXIT;
      END;
      DelayMs := Setup."Poll Period (ms)";
      IF DelayMs <= 0 THEN
        DelayMs := FirstDelayMs;
      Floor := Watchdog.LastPassMs * MaxDutyFactor;
      IF Floor > DelayMs THEN
        DelayMs := Floor;
      Schedule(DelayMs);
    END;

    LOCAL PROCEDURE Schedule@7(DelayMs@1000000000 : Integer);
    VAR
      Delay@1000000001 : Duration;
    BEGIN
      IF NOT TASKSCHEDULER.CANCREATETASK THEN
        EXIT;
      Delay := DelayMs;
      TASKSCHEDULER.CREATETASK(CODEUNIT::"LockWatch Task",0,TRUE,COMPANYNAME,CURRENTDATETIME + Delay);
    END;

    LOCAL PROCEDURE CancelOurTasks@8() : Integer;
    VAR
      ScheduledTask@1000000000 : Record 2000000175;
      Cancelled@1000000001 : Integer;
    BEGIN
      // Таблица задач общая на базу (DataPerCompany No), поэтому отбор идёт и по компании:
      // снять чужую задачу той же компании нельзя, а снять свою в другой компании - тем более.
      ScheduledTask.SETRANGE("Run Codeunit",CODEUNIT::"LockWatch Task");
      ScheduledTask.SETRANGE(Company,COMPANYNAME);
      IF ScheduledTask.FINDSET THEN
        REPEAT
          IF TASKSCHEDULER.TASKEXISTS(ScheduledTask.ID) THEN BEGIN
            TASKSCHEDULER.CANCELTASK(ScheduledTask.ID);
            Cancelled := Cancelled + 1;
          END;
        UNTIL ScheduledTask.NEXT = 0;
      EXIT(Cancelled);
    END;

    LOCAL PROCEDURE NoteFailure@9(ErrorText@1000000000 : Text);
    BEGIN
      // Обрезает длинный текст ошибки сама таблица состояния: у неё это поле, ей и знать
      // его длину. Здесь бы пришлось повторять MAXSTRLEN чужого поля.
      Watchdog.Say(STRSUBSTNO(PassFailedTxt,ErrorText));
      COMMIT;
    END;

    LOCAL PROCEDURE FirstDelayMs@10() : Integer;
    BEGIN
      // Секунда после нажатия. Достаточно, чтобы транзакция завода успела закрыться, и
      // достаточно мало, чтобы человек увидел первый проход, не уходя со страницы.
      EXIT(1000);
    END;

    LOCAL PROCEDURE MaxDutyFactor@11() : Integer;
    BEGIN
      // Доля времени, которую наблюдателю позволено занимать: одна десятая. Число это -
      // не про удобство, а про обещание заказчику. Всё, что дороже, само себя разредит.
      EXIT(10);
    END;

    BEGIN
    {
      Фоновая задача. Своя самоперевзводящаяся цепочка TASKSCHEDULER, а не запись в очередь
      заданий: очередь на целевом контуре доработана заказчиком, и вешаться в неё значит
      зависеть от чужих правок и делить с ними ошибки.

      Цепочка держится на трёх решениях, и каждое - про то, чтобы наблюдение не кончилось
      незаметно:

      1. Проход зовётся через CODEUNIT.RUN. Ошибка внутри откатит его транзакцию, но не
         уронит задачу: без этого цепочка оборвалась бы на первой неожиданности, ночью и
         молча, а утром журнал выглядел бы как "блокировок не было".
      2. Выключенный сторож обрывает цепочку САМ и пишет об этом. Замолчавший без
         объяснения неотличим от умершего. Выключатель перечитывается ДВАЖДЫ - и перед
         проходом, и перед перевзводом: между ними проходит целый проход, и нажатое в это
         время "стоп" иначе отменялось бы перевзводом.
      3. Завод идемпотентен: сначала снимаются свои задачи, потом ставится одна. Две
         цепочки вперемежку - это два наблюдателя, закрывающих эпизоды друг другу.

      Обратная связь - только на разрежение. Наблюдатель не берёт себе больше одной
      десятой времени, и в шторм, когда проход дорожает, он сам отходит в сторону.
      Учащение при занятости было бы разгоном самого себя.

      Планировщик может быть выключен в конфигурации службы, и тогда завод отказывает
      ВНЯТНО, называя параметр. Молчаливо не завестись - худший из возможных ответов:
      кнопка нажата, сообщение бодрое, наблюдения нет.
    }
    END.
