    [External]
    PROCEDURE Mark@1(ContextTableNo@1000000000 : Integer;DocumentNo@1000000001 : Code[20]);
    BEGIN
      // Отметка ставится подписчиком на событие таблицы контекста, то есть ВНУТРИ
      // транзакции пишущего. Этим она и ценна: транзакция виновника держит на этой
      // строке монопольную блокировку, и по номеру транзакции её можно найти со
      // стороны SQL. Это не корреляция по времени, а факт.
      //
      // Строка СВОЯ у каждого сеанса. Поэтому сама отметка ни с кем не конкурирует и
      // чужой блокировки не создаёт, а вставки в конец индекса, за который дрались бы
      // все сессии сразу, здесь не возникает вовсе: после первого раза это MODIFY.
      IF GET(SERVICEINSTANCEID,SESSIONID) THEN BEGIN
        FillFrom(ContextTableNo,DocumentNo);
        MODIFY;
      END ELSE BEGIN
        INIT;
        "Server Instance Id" := SERVICEINSTANCEID;
        "Session Id" := SESSIONID;
        FillFrom(ContextTableNo,DocumentNo);
        INSERT;
      END;
    END;

    [External]
    PROCEDURE Release@2();
    BEGIN
      // Снимать отметку после каждой операции не нужно: она всё равно перезапишется
      // следующей записью того же сеанса, а лишняя запись стоит чужой транзакции.
      // А вот при выходе сеанса убрать её стоит - иначе журнал назовёт документ по
      // отметке, оставшейся от работы, которая давно кончилась.
      IF GET(SERVICEINSTANCEID,SESSIONID) THEN
        DELETE;
    END;

    [External]
    PROCEDURE SweepDead@4(MaxRows@1000000000 : Integer) : Integer;
    VAR
      Session@1000000001 : Record 2000000110;
      Doomed@1000000002 : Record 110233;
      Victim@1000000007 : Record 110233;
      Threshold@1000000003 : DateTime;
      Age@1000000004 : Duration;
      Instance@1000000005 : Integer;
      Removed@1000000006 : Integer;
    BEGIN
      // Отметку не убирает НИКТО: Release снимает только свою и зовётся обкаткой. А ключ у
      // строки - экземпляр службы плюс номер сеанса, и меняются оба: замерено 12.09.2026,
      // что экземпляр получает новый номер при каждом запуске службы - 1 534 разных за
      // двенадцать дней, - а сеансы после запуска нумеруются заново. Разных пар за те же
      // дни вышло 82 057, и столько же строк осталось бы в таблице навсегда.
      //
      // Цена этого не в объёме, а в своей же дороге: обратный поиск отметки идёт по
      // %%lockres%%, а индекса по псевдостолбцу не бывает - просматривается индекс целиком.
      // Замерено: 0,005 мс на пустой таблице и 0,026 мс при 82 000 строк.
      IF MaxRows <= 0 THEN
        EXIT(0);
      // Кэш службы отдаёт эту таблицу такой, какой она была, а пишут в неё ЧУЖИЕ сессии -
      // и в кластере даже чужие экземпляры, чья запись этот кэш не сбрасывает. Замерено
      // 12.09.2026: проход не увидел тридцати лежащих в таблице строк вовсе и убрал ноль,
      // а блокирующее чтение те же строки видело - оно идёт в базу мимо кэша.
      //
      // Отсюда и место вызова: уборка идёт ДО первой записи прохода. Внутри начатой
      // транзакции этой функции делать нечего, а снаружи она стоит один заход в базу.
      SELECTLATESTVERSION;
      Age := KeepMarkMs;
      Threshold := CURRENTDATETIME - Age;
      RESET;
      IF NOT FINDFIRST THEN
        EXIT(0);
      REPEAT
        Instance := "Server Instance Id";
        Session.RESET;
        Session.SETRANGE("Server Instance ID",Instance);
        // Живым считается экземпляр, у которого платформа помнит хоть один сеанс, и свой
        // собственный. Таблица сеансов переживает смерть сеанса - на стенде в ней нашлись
        // строки одиннадцатидневной давности от давно мёртвого экземпляра, - и ошибается
        // она в безопасную сторону: лишняя строка значит "оставить", а не "снести живое".
        IF (Instance <> SERVICEINSTANCEID) AND Session.ISEMPTY THEN BEGIN
          Doomed.RESET;
          Doomed.SETRANGE("Server Instance Id",Instance);
          // Возраст спрашивается у КАЖДОЙ строки, а не у экземпляра. Свежая отметка - это
          // сеанс, писавший только что, и замок на ней может ещё держаться; ждать его
          // проходу нечем, предела ожидания у C/AL здесь нет. Суточная не держится ничем.
          Doomed.SETFILTER("Marked At",'<%1',Threshold);
          // Читается БЕЗ замка, а запирается только то, что уже отобрано. FINDSET(TRUE)
          // здесь запирал группу экземпляра целиком - вместе со свежими строками, которые
          // фильтр по возрасту потом выбрасывал, - и проход умирал на чужом замке. Первый
          // же прогон прохода это и поймал: "a record in the LockWatch Context Mark table
          // was locked by another user".
          IF Doomed.FINDSET THEN
            REPEAT
              IF Victim.GET(Doomed."Server Instance Id",Doomed."Session Id") THEN BEGIN
                Victim.DELETE;
                Removed := Removed + 1;
              END;
            UNTIL (Removed >= MaxRows) OR (Doomed.NEXT = 0);
        END;
        SETFILTER("Server Instance Id",'>%1',Instance);
      UNTIL (Removed >= MaxRows) OR (NOT FINDFIRST);
      EXIT(Removed);
    END;

    LOCAL PROCEDURE KeepMarkMs@5() : Duration;
    BEGIN
      // Сутки. Отметка живёт до следующей записи того же сеанса, и суточная - это почти
      // наверняка след работы, которая кончилась. Меньше нельзя: сеанс, оставленный на
      // ночь, пишет редко, и уборка отняла бы у него отметку прямо посреди работы. Больше
      // незачем: мусор этот не нужен уже через час, а ждать дольше значит носить его.
      EXIT(86400000);
    END;

    LOCAL PROCEDURE FillFrom@3(ContextTableNo@1000000000 : Integer;DocumentNo@1000000001 : Code[20]);
    BEGIN
      "User Id" := COPYSTR(USERID,1,MAXSTRLEN("User Id"));
      "Company Name" := COPYSTR(COMPANYNAME,1,MAXSTRLEN("Company Name"));
      "Table No." := ContextTableNo;
      "Document No." := DocumentNo;
      "Marked At" := CURRENTDATETIME;
    END;

    BEGIN
    {
      Отметка "этот сеанс работает вот с этим документом". Единственное место во всём
      инструменте, куда пишут ЧУЖИЕ сессии, а не фоновая задача, - и потому единственное,
      где цена записи ложится на прикладную работу заказчика.

      Убирает отметки проход, и только мёртвые: экземпляр службы, которого платформа уже не
      помнит, плюс возраст строки. Снимать свою отметку после каждой операции не надо - она
      перезапишется следующей записью, - а вот после смерти сеанса её не снимет уже ничто.

      Ключ составной, и это не украшение: SESSIONID уникален только ВНУТРИ экземпляра
      службы. На контуре с двумя экземплярами два разных человека получили бы один и тот
      же номер сеанса, и отметка одного молча перетирала бы отметку другого - а журнал
      с уверенностью называл бы чужой документ. Отсюда SERVICEINSTANCEID в ключе; он же,
      кстати, число, а не имя: имени экземпляра C/AL не отдаёт вовсе, оно лежит в
      Table 2000000110 "Active Session".

      Три ограничения, каждое вписано нарочно:

      1. Строка своя у каждого сеанса. Ни спора за строку, ни спора за последнюю
         страницу индекса, которая была бы одна на всю систему при сквозном номере.
      2. Ни одного SumIndexFields и ни одного вторичного ключа: каждый лишний индекс -
         это ещё одна запись в чужой транзакции.
      3. Отметка добавляет ОДНУ запись в транзакцию, которая и так пишет в таблицу
         контекста. Меряется это до включения на бою, а не после.

      COMMIT внутри прикладной операции закрывает транзакцию и рвёт мост к SQL. Здесь
      это лечится само: подписчик срабатывает на КАЖДОЙ записи, поэтому первая же запись
      после COMMIT кладёт отметку в новую транзакцию.
    }
    END.
  }
