    LOCAL PROCEDURE ReadBlob@1(Blocker@1000000000 : Boolean) : Text;
    VAR
      InStr@1000000001 : InStream;
      Body@1000000002 : BigText;
      Result@1000000003 : Text;
    BEGIN
      // Блоб сам не приезжает: выборка его не читает, и без CALCFIELDS выгрузка увезла бы
      // пустоту вместо текста запроса - молча, потому что пустой текст не отличить от
      // выключенного сбора операторов. Это самое ценное, что есть в старом эпизоде.
      IF Blocker THEN BEGIN
        Episode.CALCFIELDS("Blocker Statement Text");
        IF NOT Episode."Blocker Statement Text".HASVALUE THEN
          EXIT('');
        Episode."Blocker Statement Text".CREATEINSTREAM(InStr,TEXTENCODING::UTF8);
      END ELSE BEGIN
        Episode.CALCFIELDS("Victim Statement");
        IF NOT Episode."Victim Statement".HASVALUE THEN
          EXIT('');
        Episode."Victim Statement".CREATEINSTREAM(InStr,TEXTENCODING::UTF8);
      END;
      Body.READ(InStr);
      Body.GETSUBTEXT(Result,1,Body.LENGTH);
      EXIT(Result);
    END;

    LOCAL PROCEDURE WriteBlob@2(Blocker@1000000000 : Boolean;Body@1000000001 : Text);
    VAR
      OutStr@1000000002 : OutStream;
    BEGIN
      // Пишется ДО сохранения строки: элементы разбираются по порядку, а запись в таблицу
      // случается на закрытии табличного элемента, и блоб уезжает в базу вместе с ней.
      IF Body = '' THEN
        EXIT;
      IF Blocker THEN BEGIN
        Episode."Blocker Statement Text".CREATEOUTSTREAM(OutStr,TEXTENCODING::UTF8);
        OutStr.WRITETEXT(Body);
      END ELSE BEGIN
        Episode."Victim Statement".CREATEOUTSTREAM(OutStr,TEXTENCODING::UTF8);
        OutStr.WRITETEXT(Body);
      END;
    END;

    BEGIN
    {
      Выгрузка истории в файл и возврат её из файла. Нужна она ради одного случая, и случай
      этот неизбежен: снятие инструмента сносит историю вместе со всеми его таблицами. Год
      наблюдений уходит вместе с объектами, и записи об этом в документе мало - нужна кнопка.

      Направление ОБА нарочно. Выгрузка без возврата - это не сохранение, а утешение:
      проверить её можно только тем, что она вставляется обратно.

      AutoReplace=Yes: строка с тем же номером записи в истории - это тот же самый эпизод,
      номера выдаются один раз и живут с ним. Возврат поверх существующей строки обязан её
      заменить, а не отказаться на ключе.

      Текст запроса едет отдельными элементами, привязанными к переменным: типа данных под
      BLOB у XMLport нет вовсе. Молча выбросить его было бы худшим из решений - выгрузка
      выглядела бы полной.
    }
    END.
