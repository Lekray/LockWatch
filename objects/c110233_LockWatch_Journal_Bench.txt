    [External]
    PROCEDURE Bench@1();
    VAR
      Passed@1000000000 : Integer;
      Total@1000000001 : Integer;
      Report@1000000002 : Text;
    BEGIN
      // Мерный прогон судит сам, как и обкатка: цена, которую никто не сравнил с
      // объявленным потолком, - это не замер, а число в отчёте.
      RunBench(Passed,Total,Report);
      Report := STRSUBSTNO(VerdictTxt,Passed,Total) + CrLf + CrLf + Report;
      IF Passed < Total THEN
        ERROR(FailedRunErr,Report);
      MESSAGE(Report);
    END;

    LOCAL PROCEDURE RunBench@2(VAR Passed@1000000000 : Integer;VAR Total@1000000001 : Integer;VAR Report@1000000002 : Text);
    VAR
      Setup@1000000003 : Record 110230;
      Waiters@1000000004 : Integer;
      Opened@1000000005 : Integer;
      Updated@1000000006 : Integer;
      Closed@1000000007 : Integer;
      Restarted@1000000008 : Integer;
      OpenMs@1000000009 : Integer;
      UpdateMs@1000000010 : Integer;
      CloseMs@1000000011 : Integer;
      SweepMs@1000000012 : Integer;
      Granted@1000000013 : Integer;
      StillOpen@1000000014 : Integer;
      Backdated@1000000015 : Integer;
      Removed@1000000016 : Integer;
      Worst@1000000017 : Integer;
      Line@1000000018 : TEMPORARY Record 110234;
      Lines@1000000019 : Integer;
      Summed@1000000020 : Integer;
      OverLow@1000000021 : Integer;
      OverHigh@1000000022 : Integer;
      FutureLines@1000000023 : Integer;
      TotalWait@1000000024 : BigInteger;
      Reading@1000000025 : TEMPORARY Record 110235;
      Cov@1000000026 : Record 110235;
      CovAdded@1000000027 : Integer;
      CovChanged@1000000028 : Integer;
      CovResets@1000000029 : Integer;
      AlertEp@1000000030 : Record 110231;
      Alert@1000000031 : Record 110236;
      AlertRaised@1000000032 : Integer;
      AlertLatched@1000000033 : Integer;
      AlertRows@1000000034 : Integer;
      AlertEpisodes@1000000035 : Integer;
    BEGIN
      Setup.FetchOnce;
      IF Setup.Enabled THEN
        ERROR(EnabledErr);
      RefuseIfJournalUsed;

      Waiters := Parse.MaxWaitersPerPass;

      OpenMs := PassOfWaiters(Waiters,FirstWaitMs,Opened,Updated,Closed,Restarted);
      Check(Passed,Total,Report,ChkOpenTxt,
            (Opened = Waiters) AND (Updated = 0) AND (Closed = 0),
            STRSUBSTNO(OpenTxt,Opened,Waiters,Updated,Closed));

      UpdateMs := PassOfWaiters(Waiters,SecondWaitMs,Opened,Updated,Closed,Restarted);
      Check(Passed,Total,Report,ChkUpdateTxt,
            (Updated = Waiters) AND (Opened = 0) AND (Restarted = 0) AND (BenchRowCount = Waiters),
            STRSUBSTNO(UpdateTxt,Updated,Opened,BenchRowCount));

      CloseMs := PassOfWaiters(0,0,Opened,Updated,Closed,Restarted);
      CountOutcomes(Granted,StillOpen);
      Check(Passed,Total,Report,ChkCloseTxt,
            (Closed = Waiters) AND (Granted = Waiters) AND (StillOpen = 0),
            STRSUBSTNO(CloseTxt,Closed,Waiters,Granted,StillOpen));

      // Сводка проверяется ЗДЕСЬ, до состаривания и уборки: журнал в этот миг содержит
      // ровно то, что мерный прогон в него положил, - пятьсот закрытых эпизодов одной
      // таблицы и одного ключа. Обращений к SQL в сводке нет, поэтому и проверяется она
      // без стенда с блокировками.
      Summed := Journal.Summarize(Line,0DT,CURRENTDATETIME,AlertUnderTest);
      Lines := Line.COUNT;
      Line.FINDFIRST;
      TotalWait := Line."Total Wait (ms)";
      OverLow := Line."Over Threshold";
      Check(Passed,Total,Report,ChkSummaryTxt,
            (Summed = Waiters) AND (Lines = 1) AND (Line.Episodes = Waiters) AND
            (TotalWait = Waiters * SecondWaitMs) AND (Line."Max Wait (ms)" = SecondWaitMs),
            STRSUBSTNO(SummaryTxt,Lines,Line.Episodes,Waiters,TotalWait,Line."Max Wait (ms)"));

      // Порог и окно проверяются тем, что их СДВИГАЮТ. Одна сводка ничего не доказывает:
      // колонка "выше порога", равная числу эпизодов, одинаково выглядит и у настоящего
      // счёта, и у того, который порога не читает вовсе.
      Journal.Summarize(Line,0DT,CURRENTDATETIME,SecondWaitMs + 1);
      Line.FINDFIRST;
      OverHigh := Line."Over Threshold";
      FutureLines := Journal.Summarize(Line,CURRENTDATETIME,CURRENTDATETIME,AlertUnderTest);
      Lines := Line.COUNT;
      Check(Passed,Total,Report,ChkWindowTxt,
            (OverLow = Waiters) AND (OverHigh = 0) AND (FutureLines = 0) AND (Lines = 0),
            STRSUBSTNO(WindowTxt,OverLow,Waiters,OverHigh,Lines));

      // Накопительный слой проверяется без единого обращения к SQL: съём отдаёт буфер, а
      // накопитель считает по нему приросты - значит, буфер можно подать руками. Строка
      // берётся с ОТРИЦАТЕЛЬНЫМ номером объекта: в SQL номера положительны, и столкнуться
      // с настоящей строкой такая не может.
      MakeCounterReading(Reading,100,5000,1);
      Coverage.Accumulate(Reading);
      MakeCounterReading(Reading,150,8000,1);
      Coverage.Accumulate(Reading);
      Cov.GET(BenchObjectId,BenchIndexId);
      Check(Passed,Total,Report,ChkCoverageTxt,
            (Cov.Waits = 50) AND (Cov."Wait (ms)" = 3000) AND (Cov.Escalations = 0),
            STRSUBSTNO(CoverageTxt,Cov.Waits,50,Cov."Wait (ms)",3000,Cov.Escalations));

      // Обнуление счётчика сервером выглядит как значение МЕНЬШЕ прежнего. Разность в этом
      // случае была бы отрицательной, и итог поехал бы назад - молча и правдоподобно.
      MakeCounterReading(Reading,10,200,0);
      Coverage.Accumulate(Reading);
      Coverage.PassCounts(CovAdded,CovChanged,CovResets);
      Cov.GET(BenchObjectId,BenchIndexId);
      Check(Passed,Total,Report,ChkCoverageResetTxt,
            (CovResets = 1) AND (Cov.Resets = 1) AND (Cov.Waits = 60) AND (Cov."Wait (ms)" = 3200),
            STRSUBSTNO(CoverageResetTxt,Cov.Resets,Cov.Waits,60,Cov."Wait (ms)",3200));
      Cov.DELETE;

      // Защёлка тревоги проверяется без базы SQL и без единой блокировки: она работает над
      // строкой эпизода, а строку можно подать руками. Головы берутся отрицательные - в
      // настоящем spid отрицательных не бывает, и столкнуться с чужой тревогой нельзя.
      Alerts.BeginPass;
      MakeAlertEpisode(AlertEp,AlertHeadA,AlertWaitMs);
      Alerts.Raise(AlertEp,AlertThresholdMs,FALSE,'');
      Alerts.Raise(AlertEp,AlertThresholdMs,FALSE,'');
      Alerts.Raise(AlertEp,AlertThresholdMs,FALSE,'');
      MakeAlertEpisode(AlertEp,AlertHeadB,AlertWaitMs);
      Alerts.Raise(AlertEp,AlertThresholdMs,FALSE,'');
      // Ниже порога тревоги нет вовсе, и это тоже часть защёлки: порог, который не
      // отсекает, превращает тревогу в поток.
      MakeAlertEpisode(AlertEp,AlertHeadC,AlertThresholdMs - 1);
      Alerts.Raise(AlertEp,AlertThresholdMs,FALSE,'');
      Alerts.PassCounts(AlertRaised,AlertLatched);
      AlertRows := AlertRowCount;
      Alert.RESET;
      Alert.SETRANGE("Head SPID",AlertHeadA);
      IF Alert.FINDFIRST THEN
        AlertEpisodes := Alert.Episodes;
      Check(Passed,Total,Report,ChkLatchTxt,
            (AlertRaised = 2) AND (AlertLatched = 2) AND (AlertRows = 2) AND (AlertEpisodes = 3),
            STRSUBSTNO(LatchTxt,AlertRaised,2,AlertLatched,2,AlertRows,AlertEpisodes));

      // Окно защёлки проверяется тем, что его СДВИГАЮТ: строка состаривается за окно, и
      // следующая тревога по той же голове обязана завестись заново. Без этой проверки
      // защёлка, залипшая навсегда, выглядела бы точно так же.
      AgeAlerts;
      MakeAlertEpisode(AlertEp,AlertHeadA,AlertWaitMs);
      Alerts.Raise(AlertEp,AlertThresholdMs,FALSE,'');
      AlertRows := AlertRowCount;
      Check(Passed,Total,Report,ChkLatchWindowTxt,
            AlertRows = 3,
            STRSUBSTNO(LatchWindowTxt,AlertRows,3));
      SweepAlerts;

      Backdated := Backdate(BackdateRows,BackdateDays);
      Removed := Journal.TrimOldEntries(RetentionUnderTest,Waiters);
      Check(Passed,Total,Report,ChkTrimTxt,
            (Removed = Backdated) AND (Backdated = BackdateRows) AND (BenchRowCount = Waiters - BackdateRows),
            STRSUBSTNO(TrimTxt,Removed,Backdated,BenchRowCount,Waiters));

      SweepMs := Sweep;

      Worst := OpenMs;
      IF UpdateMs > Worst THEN
        Worst := UpdateMs;
      IF CloseMs > Worst THEN
        Worst := CloseMs;
      Check(Passed,Total,Report,ChkCostTxt,Worst <= CeilingMs,STRSUBSTNO(CostTxt,Worst,CeilingMs));

      Report := Report + CrLf +
        Timing(OpenPhaseTxt,OpenMs,Waiters) +
        Timing(UpdatePhaseTxt,UpdateMs,Waiters) +
        Timing(ClosePhaseTxt,CloseMs,Waiters) +
        Timing(SweepPhaseTxt,SweepMs,Waiters - BackdateRows);
    END;

    LOCAL PROCEDURE PassOfWaiters@3(Waiters@1000000000 : Integer;WaitMs@1000000001 : Integer;VAR Opened@1000000002 : Integer;VAR Updated@1000000003 : Integer;VAR Closed@1000000004 : Integer;VAR Restarted@1000000005 : Integer) : Integer;
    VAR
      Observed@1000000006 : Record 110231;
      Started@1000000007 : DateTime;
      Elapsed@1000000008 : Duration;
      Index@1000000009 : Integer;
      Milliseconds@1000000010 : Integer;
    BEGIN
      // Меряется ровно то, что делает фоновая задача после съёма: проход целиком, вместе
      // с COMMIT. Без COMMIT замер был бы вдвое красивее и вдвое бесполезнее - запись в
      // журнал транзакций и есть то, чем проход платит.
      Started := CURRENTDATETIME;
      Journal.BeginPass(Started,PeriodMs,AlertUnderTest);
      FOR Index := 1 TO Waiters DO BEGIN
        MakeObserved(Observed,Index,WaitMs);
        Journal.Observe(Observed);
      END;
      Journal.FinishPass;
      COMMIT;
      Elapsed := CURRENTDATETIME - Started;
      Milliseconds := Elapsed;
      Journal.PassCounts(Opened,Updated,Closed,Restarted);
      EXIT(Milliseconds);
    END;

    LOCAL PROCEDURE MakeObserved@4(VAR Observed@1000000000 : Record 110231;Index@1000000001 : Integer;WaitMs@1000000002 : Integer);
    BEGIN
      // Строка сделана похожей на настоящую не для красоты: цена записи зависит от того,
      // сколько байт уезжает в SQL, а строковые поля здесь - самые широкие в таблице.
      // Замер на пустой строке был бы вдвое бодрее того, что случится на бою.
      Observed.INIT;
      Observed."Victim SPID" := -Index;
      Observed."Victim Transaction Id" := 4000000000 + Index;
      Observed."Victim Is NAV" := Observed."Victim Is NAV"::Yes;
      Observed."Victim Host" := 'STAND-HOST';
      Observed."Victim Process Id" := 4000 + Index;
      Observed."Blocker SPID" := -Index - 1;
      Observed."Blocker Transaction Id" := 5000000000 + Index;
      Observed."Head SPID" := -1;
      Observed."Chain Depth" := 2;
      Observed."Victims Behind Head" := Index;
      Observed."Resource Kind" := 'KEYLOCK';
      Observed."Hobt Id" := SampleHobt + Index;
      Observed."Database Id" := 7;
      Observed."Held Mode" := 'X';
      Observed."Wait Type" := 'LCK_M_U';
      Observed."SQL Object Name" := 'Company Name$Sales Line';
      Observed."NAV Table Name" := 'Sales Line';
      Observed."NAV Key No." := 1;
      Observed."Last Wait (ms)" := WaitMs;
      Observed."Blocker Tran Began At" := CURRENTDATETIME;
      Observed."Blocker Statement" := 'UPDATE "Company Name$Sales Line" SET "Quantity" ' +
        'WHERE "Document Type" AND "Document No_" AND "Line No_"';
    END;

    LOCAL PROCEDURE SampleHobt@20() : BigInteger;
    VAR
      Value@1000000000 : BigInteger;
    BEGIN
      // Числовой литерал в C/AL - это Decimal, а Decimal не переваривает больше
      // 999 999 999 999 999,99. Настоящий hobt длиннее, и написанный литералом он не
      // компилируется вовсе. Единственный способ внести его в код - через EVALUATE.
      EVALUATE(Value,'72057594261667840');
      EXIT(Value);
    END;

    LOCAL PROCEDURE Backdate@5(Rows@1000000000 : Integer;Days@1000000001 : Integer) : Integer;
    VAR
      Moved@1000000002 : Integer;
      Back@1000000003 : Duration;
    BEGIN
      // Состаривание идёт правкой поля, а не ожиданием месяца.
      Back := Days * Parse.MsPerDay;
      Episode.RESET;
      Episode.SETFILTER("Victim SPID",'<%1',0);
      IF Episode.FINDSET(TRUE) THEN
        REPEAT
          Episode."Started At" := CURRENTDATETIME - Back;
          Episode.MODIFY;
          Moved := Moved + 1;
        UNTIL (Moved >= Rows) OR (Episode.NEXT = 0);
      COMMIT;
      EXIT(Moved);
    END;

    LOCAL PROCEDURE Sweep@6() : Integer;
    VAR
      Started@1000000000 : DateTime;
      Elapsed@1000000001 : Duration;
      Milliseconds@1000000002 : Integer;
    BEGIN
      // Уборка за собой обязательна и делается по СВОЕЙ метке: у строк мерного прогона
      // номер сессии отрицательный, а настоящий spid отрицательным не бывает. Строка,
      // оставшаяся в журнале после замера, - это ложь в отчёте заказчику.
      Started := CURRENTDATETIME;
      Episode.RESET;
      Episode.SETFILTER("Victim SPID",'<%1',0);
      Episode.DELETEALL;
      COMMIT;
      Elapsed := CURRENTDATETIME - Started;
      Milliseconds := Elapsed;
      EXIT(Milliseconds);
    END;

    LOCAL PROCEDURE RefuseIfJournalUsed@7();
    VAR
      Foreign@1000000000 : Integer;
    BEGIN
      Episode.RESET;
      Episode.SETFILTER("Victim SPID",'>=%1',0);
      Foreign := Episode.COUNT;
      IF Foreign > 0 THEN
        ERROR(DirtyErr,Foreign);
      // Свои строки от прошлого прогона, не дожившего до уборки, - не отказ, а мусор.
      Sweep;
    END;

    LOCAL PROCEDURE BenchRowCount@8() : Integer;
    BEGIN
      Episode.RESET;
      Episode.SETFILTER("Victim SPID",'<%1',0);
      EXIT(Episode.COUNT);
    END;

    LOCAL PROCEDURE CountOutcomes@9(VAR Granted@1000000000 : Integer;VAR StillOpen@1000000001 : Integer);
    BEGIN
      Episode.RESET;
      Episode.SETFILTER("Victim SPID",'<%1',0);
      Episode.SETRANGE(Outcome,Episode.Outcome::Granted);
      Granted := Episode.COUNT;
      Episode.SETRANGE(Outcome);
      Episode.SETRANGE(Open,TRUE);
      StillOpen := Episode.COUNT;
    END;

    LOCAL PROCEDURE Check@10(VAR Passed@1000000000 : Integer;VAR Total@1000000001 : Integer;VAR Report@1000000002 : Text;What@1000000003 : Text;Ok@1000000004 : Boolean;Detail@1000000005 : Text);
    VAR
      Mark@1000000006 : Text;
    BEGIN
      Total := Total + 1;
      IF Ok THEN BEGIN
        Passed := Passed + 1;
        Mark := PassedTxt;
      END ELSE
        Mark := FailedTxt;
      Report := Report + Mark + ' ' + What + CrLf + '       ' + Detail + CrLf;
    END;

    LOCAL PROCEDURE Timing@11(Phase@1000000000 : Text;Milliseconds@1000000001 : Integer;Rows@1000000002 : Integer) : Text;
    VAR
      PerRow@1000000003 : Integer;
    BEGIN
      IF Rows > 0 THEN
        PerRow := ROUND(Milliseconds * 1000 / Rows,1);
      EXIT(STRSUBSTNO(TimingTxt,Phase,Milliseconds,Rows,PerRow) + CrLf);
    END;

    LOCAL PROCEDURE PeriodMs@12() : Integer;
    BEGIN
      // Тот же период, что и в умолчании настройки: от него зависит суждение об исходе,
      // и мерить проход при другом значило бы мерить не тот проход.
      EXIT(3000);
    END;

    LOCAL PROCEDURE FirstWaitMs@13() : Integer;
    BEGIN
      EXIT(1200);
    END;

    LOCAL PROCEDURE SecondWaitMs@14() : Integer;
    BEGIN
      // Больше первого - иначе счётчик ожидания пойдёт назад, и проход честно решит, что

    LOCAL PROCEDURE AgeAlerts@27();
    VAR
      Alert@1000000000 : Record 110236;
      Back@1000000001 : Duration;
    BEGIN
      // Состаривание правкой поля, а не ожиданием пяти минут. Сдвиг берётся с запасом в
      // одну секунду: ровно на границе окна проверка проверяла бы часы, а не защёлку.
      Back := Parse.AlertLatchMs + 1000;
      Alert.RESET;
      Alert.SETFILTER("Head SPID",'<%1',0);
      IF Alert.FINDSET(TRUE) THEN
        REPEAT
          Alert."Last Seen At" := CURRENTDATETIME - Back;
          Alert.MODIFY;
        UNTIL Alert.NEXT = 0;
    END;

      COMMIT;
    END;

    LOCAL PROCEDURE MakeCounterReading@22(VAR Reading@1000000000 : TEMPORARY Record 110235;Waits@1000000001 : Integer;WaitMs@1000000002 : Integer;Escalations@1000000003 : Integer);
    BEGIN
      Reading.RESET;
      Reading.DELETEALL;
      Reading.INIT;
      Reading."SQL Object Id" := BenchObjectId;
      Reading."Index Id" := BenchIndexId;
      Reading."SQL Object Name" := 'Company Name$Sales Line';
      Reading."NAV Table Name" := 'Sales Line';
      Reading."NAV Key No." := 1;
      // В буфере лежат СЫРЫЕ значения счётчика сервера. Приростом их делает накопитель,
      // и подавать сюда сразу прирост значило бы проверять не то, что работает на бою.
      Reading."Last Waits" := Waits;
      Reading."Last Wait (ms)" := WaitMs;
      Reading."Last Escalations" := Escalations;
      Reading.INSERT;
    END;

    LOCAL PROCEDURE AlertUnderTest@21() : Integer;
