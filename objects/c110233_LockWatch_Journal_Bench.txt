    [External]
    PROCEDURE Bench@1();
    VAR
      Passed@1000000000 : Integer;
      Total@1000000001 : Integer;
      Report@1000000002 : Text;
    BEGIN
      // Мерный прогон судит сам, как и обкатка: цена, которую никто не сравнил с
      // объявленным потолком, - это не замер, а число в отчёте.
      RunBench(Passed,Total,Report);
      Report := STRSUBSTNO(VerdictTxt,Passed,Total) + CrLf + CrLf + Report;
      IF Passed < Total THEN
        ERROR(FailedRunErr,Report);
      MESSAGE(Report);
    END;

    LOCAL PROCEDURE RunBench@2(VAR Passed@1000000000 : Integer;VAR Total@1000000001 : Integer;VAR Report@1000000002 : Text);
    VAR
      Setup@1000000003 : Record 110230;
      Waiters@1000000004 : Integer;
      Opened@1000000005 : Integer;
      Updated@1000000006 : Integer;
      Closed@1000000007 : Integer;
      Restarted@1000000008 : Integer;
      OpenMs@1000000009 : Integer;
      UpdateMs@1000000010 : Integer;
      CloseMs@1000000011 : Integer;
      SweepMs@1000000012 : Integer;
      Granted@1000000013 : Integer;
      StillOpen@1000000014 : Integer;
      Backdated@1000000015 : Integer;
      Removed@1000000016 : Integer;
      InHistoryBefore@1000000047 : Integer;
      HeldNo@1000000065 : Integer;
      OldestClosed@1000000066 : Integer;
      RowsBefore@1000000067 : Integer;
      MovedByCeiling@1000000068 : Integer;
      CovEpBefore@1000000069 : Integer;
      CovEpAfter@1000000070 : Integer;
      CovMsBefore@1000000071 : BigInteger;
      CovMsAfter@1000000072 : BigInteger;
      MirrorNote@1000000053 : Text;
      SummedJournal@1000000062 : Integer;
      SummedBoth@1000000063 : Integer;
      InHistoryNow@1000000064 : Integer;
      Worst@1000000017 : Integer;
      Line@1000000018 : TEMPORARY Record 110234;
      Lines@1000000019 : Integer;
      Summed@1000000020 : Integer;
      OverLow@1000000021 : Integer;
      OverHigh@1000000022 : Integer;
      FutureLines@1000000023 : Integer;
      TotalWait@1000000024 : BigInteger;
      Reading@1000000025 : TEMPORARY Record 110235;
      Cov@1000000026 : Record 110235;
      CovAdded@1000000027 : Integer;
      CovChanged@1000000028 : Integer;
      CovResets@1000000029 : Integer;
      AlertEp@1000000030 : Record 110231;
      Alert@1000000031 : Record 110236;
      AlertRaised@1000000032 : Integer;
      AlertLatched@1000000033 : Integer;
      AlertRows@1000000034 : Integer;
      AlertEpisodes@1000000035 : Integer;
      NamePending@1000000036 : TEMPORARY Record 110231;
      NameA@1000000037 : Integer;
      NameB@1000000038 : Integer;
      NamePickedFirst@1000000039 : Integer;
      NamePickedAgain@1000000040 : Integer;
      NameAttemptsA@1000000041 : Integer;
      NameAttemptsB@1000000042 : Integer;
      NamePickedNamed@1000000043 : Integer;
      NamePickedCeiling@1000000044 : Integer;
    BEGIN
      Setup.FetchOnce;
      IF Setup.Enabled THEN
        ERROR(EnabledErr);
      RefuseIfJournalUsed;

      Waiters := Parse.MaxWaitersPerPass;

      OpenMs := PassOfWaiters(Waiters,FirstWaitMs,Opened,Updated,Closed,Restarted);
      Check(Passed,Total,Report,ChkOpenTxt,
            (Opened = Waiters) AND (Updated = 0) AND (Closed = 0),
            STRSUBSTNO(OpenTxt,Opened,Waiters,Updated,Closed));

      UpdateMs := PassOfWaiters(Waiters,SecondWaitMs,Opened,Updated,Closed,Restarted);
      Check(Passed,Total,Report,ChkUpdateTxt,
            (Updated = Waiters) AND (Opened = 0) AND (Restarted = 0) AND (BenchRowCount = Waiters),
            STRSUBSTNO(UpdateTxt,Updated,Opened,BenchRowCount));

      CloseMs := PassOfWaiters(0,0,Opened,Updated,Closed,Restarted);
      CountOutcomes(Granted,StillOpen);
      Check(Passed,Total,Report,ChkCloseTxt,
            (Closed = Waiters) AND (Granted = Waiters) AND (StillOpen = 0),
            STRSUBSTNO(CloseTxt,Closed,Waiters,Granted,StillOpen));

      // Сводка проверяется ЗДЕСЬ, до состаривания и уборки: журнал в этот миг содержит
      // ровно то, что мерный прогон в него положил, - пятьсот закрытых эпизодов одной
      // таблицы и одного ключа. Обращений к SQL в сводке нет, поэтому и проверяется она
      // без стенда с блокировками.
      Summed := Journal.Summarize(Line,0DT,CURRENTDATETIME,AlertUnderTest,FALSE);
      Lines := Line.COUNT;
      Line.FINDFIRST;
      TotalWait := Line."Total Wait (ms)";
      OverLow := Line."Over Threshold";
      Check(Passed,Total,Report,ChkSummaryTxt,
            (Summed = Waiters) AND (Lines = 1) AND (Line.Episodes = Waiters) AND
            (TotalWait = Waiters * SecondWaitMs) AND (Line."Max Wait (ms)" = SecondWaitMs),
            STRSUBSTNO(SummaryTxt,Lines,Line.Episodes,Waiters,TotalWait,Line."Max Wait (ms)"));

      // Порог и окно проверяются тем, что их СДВИГАЮТ. Одна сводка ничего не доказывает:
      // колонка "выше порога", равная числу эпизодов, одинаково выглядит и у настоящего
      // счёта, и у того, который порога не читает вовсе.
      Journal.Summarize(Line,0DT,CURRENTDATETIME,SecondWaitMs + 1,FALSE);
      Line.FINDFIRST;
      OverHigh := Line."Over Threshold";
      FutureLines := Journal.Summarize(Line,CURRENTDATETIME,CURRENTDATETIME,AlertUnderTest,FALSE);
      Lines := Line.COUNT;
      Check(Passed,Total,Report,ChkWindowTxt,
            (OverLow = Waiters) AND (OverHigh = 0) AND (FutureLines = 0) AND (Lines = 0),
            STRSUBSTNO(WindowTxt,OverLow,Waiters,OverHigh,Lines));

      // Накопительный слой проверяется без единого обращения к SQL: съём отдаёт буфер, а
      // накопитель считает по нему приросты - значит, буфер можно подать руками. Строка
      // берётся с ОТРИЦАТЕЛЬНЫМ номером объекта: в SQL номера положительны, и столкнуться
      // с настоящей строкой такая не может.
      MakeCounterReading(Reading,100,5000,1);
      Coverage.Accumulate(Reading);
      MakeCounterReading(Reading,150,8000,1);
      Coverage.Accumulate(Reading);
      Cov.GET(BenchObjectId,BenchIndexId);
      Check(Passed,Total,Report,ChkCoverageTxt,
            (Cov.Waits = 50) AND (Cov."Wait (ms)" = 3000) AND (Cov.Escalations = 0),
            STRSUBSTNO(CoverageTxt,Cov.Waits,50,Cov."Wait (ms)",3000,Cov.Escalations));

      // Обнуление счётчика сервером выглядит как значение МЕНЬШЕ прежнего. Разность в этом
      // случае была бы отрицательной, и итог поехал бы назад - молча и правдоподобно.
      MakeCounterReading(Reading,10,200,0);
      Coverage.Accumulate(Reading);
      Coverage.PassCounts(CovAdded,CovChanged,CovResets);
      Cov.GET(BenchObjectId,BenchIndexId);
      Check(Passed,Total,Report,ChkCoverageResetTxt,
            (CovResets = 1) AND (Cov.Resets = 1) AND (Cov.Waits = 60) AND (Cov."Wait (ms)" = 3200),
            STRSUBSTNO(CoverageResetTxt,Cov.Resets,Cov.Waits,60,Cov."Wait (ms)",3200));
      Cov.DELETE;

      // Защёлка тревоги проверяется без базы SQL и без единой блокировки: она работает над
      // строкой эпизода, а строку можно подать руками. Головы берутся отрицательные - в
      // настоящем spid отрицательных не бывает, и столкнуться с чужой тревогой нельзя.
      Alerts.BeginPass;
      MakeAlertEpisode(AlertEp,AlertHeadA,AlertWaitMs);
      Alerts.Raise(AlertEp,AlertThresholdMs,FALSE,'');
      Alerts.Raise(AlertEp,AlertThresholdMs,FALSE,'');
      Alerts.Raise(AlertEp,AlertThresholdMs,FALSE,'');
      MakeAlertEpisode(AlertEp,AlertHeadB,AlertWaitMs);
      Alerts.Raise(AlertEp,AlertThresholdMs,FALSE,'');
      // Ниже порога тревоги нет вовсе, и это тоже часть защёлки: порог, который не
      // отсекает, превращает тревогу в поток.
      MakeAlertEpisode(AlertEp,AlertHeadC,AlertThresholdMs - 1);
      Alerts.Raise(AlertEp,AlertThresholdMs,FALSE,'');
      Alerts.PassCounts(AlertRaised,AlertLatched);
      AlertRows := AlertRowCount;
      Alert.RESET;
      Alert.SETRANGE("Head SPID",AlertHeadA);
      IF Alert.FINDFIRST THEN
        AlertEpisodes := Alert.Episodes;
      Check(Passed,Total,Report,ChkLatchTxt,
            (AlertRaised = 2) AND (AlertLatched = 2) AND (AlertRows = 2) AND (AlertEpisodes = 3),
            STRSUBSTNO(LatchTxt,AlertRaised,2,AlertLatched,2,AlertRows,AlertEpisodes));

      // Окно защёлки проверяется тем, что его СДВИГАЮТ: строка состаривается за окно, и
      // следующая тревога по той же голове обязана завестись заново. Без этой проверки
      // защёлка, залипшая навсегда, выглядела бы точно так же.
      AgeAlerts;
      MakeAlertEpisode(AlertEp,AlertHeadA,AlertWaitMs);
      Alerts.Raise(AlertEp,AlertThresholdMs,FALSE,'');
      AlertRows := AlertRowCount;
      Check(Passed,Total,Report,ChkLatchWindowTxt,
            AlertRows = 3,
            STRSUBSTNO(LatchWindowTxt,AlertRows,3));
      SweepAlerts;

      // Переезд проверяется с ОБЕИХ сторон: сколько ушло из журнала и сколько доехало до
      // истории. Половина проверки - "из журнала пропало" - прошла бы и на простом
      // удалении, то есть на потере, ради недопущения которой всё и затевалось.
      Backdated := Backdate(BackdateRows,BackdateDays);
      InHistoryBefore := MovedRowCount;
      Removed := Journal.MoveOldEntries(RetentionUnderTest,0,Waiters);
      Check(Passed,Total,Report,ChkTrimTxt,
            (Removed = Backdated) AND (Backdated = BackdateRows) AND
            (BenchRowCount = Waiters - BackdateRows) AND
            (MovedRowCount - InHistoryBefore = Backdated),
            STRSUBSTNO(TrimTxt,Removed,Backdated,BenchRowCount,Waiters,MovedRowCount - InHistoryBefore));

      // Второй предел - потолок СТРОК, и нужен он ровно там, где срок бессилен: в шторм
      // всё, что лежит в журнале, моложе срока, и одним сроком горячую таблицу не удержать.
      //
      // Срок здесь передаётся нулём нарочно: проверка обязана мерить потолок, а не то, что
      // за неё доделал возраст. Потолок ставится НИЖЕ текущего числа строк на известную
      // величину, и уехать обязано ровно это число - не больше и не меньше.
      //
      // Самая старая строка перед этим метится идущей. Она первый кандидат на переезд по
      // возрасту - и обязана остаться: увезти открытый эпизод значило бы разорвать одну
      // строку на две таблицы. Проверка без неё прошла бы и на переезде, который этого
      // не различает.
      HeldNo := HoldOldest;
      RowsBefore := JournalRowCount;
      OldestClosed := OldestClosedNo;
      // Охват снимается ДО переезда и ПОСЛЕ. Переезд - это перекладывание строки из одной
      // таблицы в другую, и охват от него меняться не смеет: эпизод, увиденный и потом
      // уехавший, увиден был. Мерка сильна именно равенством: доля, посчитанная по одному
      // журналу, падала бы здесь сама собой и выглядела бы как испортившееся наблюдение.
      Coverage.JournalTotal(0DT,CovEpBefore,CovMsBefore);
      MovedByCeiling := Journal.MoveOldEntries(0,RowsBefore - CeilingSurplus,Waiters);
      Check(Passed,Total,Report,ChkCeilingTxt,
            (MovedByCeiling = CeilingSurplus) AND
            (JournalRowCount = RowsBefore - CeilingSurplus) AND
            (NOT InJournal(OldestClosed)) AND InHistory(OldestClosed) AND
            InJournal(HeldNo),
            STRSUBSTNO(CeilingTxt,MovedByCeiling,CeilingSurplus,JournalRowCount,
                       RowsBefore - CeilingSurplus,HeldNo));

      // Переезд состоялся - без него равенство ниже прошло бы и на охвате по одному журналу.
      Coverage.JournalTotal(0DT,CovEpAfter,CovMsAfter);
      Check(Passed,Total,Report,ChkCovMoveTxt,
            (CovEpAfter = CovEpBefore) AND (CovMsAfter = CovMsBefore) AND (MovedByCeiling > 0),
            STRSUBSTNO(CovMoveTxt,CovEpBefore,CovEpAfter,CovMsBefore,CovMsAfter,MovedByCeiling));

      // Зеркало сверяется КАЖДЫМ прогоном, а не глазами при правке. Список полей истории
      // списан с журнала, и разойтись он может только молча: добавленное поле просто не
      // доедет при переезде, и заметят это через месяц, когда понадобится.
      MirrorNote := MirrorGap;
      Check(Passed,Total,Report,ChkMirrorTxt,
            MirrorNote = '',
            MirrorNote + MirrorOkNote(MirrorNote));

      // Свод по истории. Проверяется он ЗДЕСЬ, а не рядом с остальными проверками свода:
      // это единственный миг прогона, когда в истории заведомо что-то лежит - строки
      // только что переехали туда сами.
      //
      // Разница считается против ЧИСЛА строк истории за окно, а не против числа
      // переехавших: на стенде в истории могут лежать и настоящие эпизоды, и свод обязан
      // сложить их тоже. Сравнение с числом переехавших прошло бы только на пустой истории
      // и краснело бы у всякого, кто уже наблюдал.
      SummedJournal := Journal.Summarize(Line,0DT,CURRENTDATETIME,0,FALSE);
      SummedBoth := Journal.Summarize(Line,0DT,CURRENTDATETIME,0,TRUE);
      InHistoryNow := HistoryRowsUpTo(CURRENTDATETIME);
      Check(Passed,Total,Report,ChkSummaryHistoryTxt,
            (SummedBoth - SummedJournal = InHistoryNow) AND (InHistoryNow > 0),
            STRSUBSTNO(SummaryHistoryTxt,SummedJournal,SummedBoth,InHistoryNow));

      SweepMs := Sweep;

      // Имена в истории. Проверяется на СВОИХ двух эпизодах и на пустом журнале - тот
      // самый миг после уборки, когда в журнале нет ничего чужого и счёт выбранных строк
      // можно сравнивать с числом, а не с "примерно столько".
      //
      // Смысл всех четырёх: разбор идёт не один раз. Отметку виновник кладёт тогда, когда
      // трогает таблицу документа, а это бывает и позже начала ожидания; жертва
      // отмечается, только когда сама что-то пишет. Один вопрос на эпизод оставлял в
      // истории пустую колонку там, где ответ появлялся секундой позже.
      NamePass(NameIndexA,NameIndexB);
      NameA := EntryOfIndex(NameIndexA);
      NameB := EntryOfIndex(NameIndexB);
      NamePickedFirst := Journal.PickUnresolved(NamePending,Parse.MaxDocumentsPerPass);
      Journal.SetDocument(NameA,'','',Parse.DocSourceNone,NoAnswerReason,'');
      Journal.SetDocument(NameB,'','',Parse.DocSourceNone,NoAnswerReason,'');
      NamePickedAgain := Journal.PickUnresolved(NamePending,Parse.MaxDocumentsPerPass);
      Check(Passed,Total,Report,ChkNameAgainTxt,
            (NamePickedFirst = 2) AND (NamePickedAgain = 2),
            STRSUBSTNO(NameAgainTxt,NamePickedFirst,NamePickedAgain));

      NamePass(NameIndexA,NameIndexB);
      Episode.GET(NameA);
      NameAttemptsA := Episode."Name Attempts";
      Journal.SetDocument(NameA,NameDocument,NameCulprit,Parse.DocSourceFromMark,'',NameVictim);
      Episode.GET(NameA);
      Check(Passed,Total,Report,ChkNameLateTxt,
            (NameAttemptsA = 1) AND (Episode."User Name" = NameCulprit) AND
            (Episode."Victim User Name" = NameVictim) AND (Episode."Document No." = NameDocument),
            STRSUBSTNO(NameLateTxt,NameAttemptsA,Episode."User Name",
                       Episode."Victim User Name",Episode."Document No."));

      Journal.SetDocument(NameA,'','',Parse.DocSourceNone,NoAnswerReason,'');
      Episode.GET(NameA);
      Journal.PickUnresolved(NamePending,Parse.MaxDocumentsPerPass);
      NamePickedNamed := 0;
      IF NamePending.GET(NameA) THEN
        NamePickedNamed := 1;
      Check(Passed,Total,Report,ChkNameHeldTxt,
            (Episode."User Name" = NameCulprit) AND (Episode."Victim User Name" = NameVictim) AND
            (Episode."Document No." = NameDocument) AND (NamePickedNamed = 0),
            STRSUBSTNO(NameHeldTxt,Episode."User Name",Episode."Victim User Name",
                       Episode."Document No.",NamePickedNamed));

      // Второй эпизод так и остался без ответа: на нём и проверяется потолок. Без него
      // вопрос повторялся бы столько же, сколько живёт эпизод, а живёт он и по часу.
      Episode.GET(NameB);
      WHILE Episode."Name Attempts" < Parse.MaxNameAttempts DO BEGIN
        Journal.SetDocument(NameB,'','',Parse.DocSourceNone,NoAnswerReason,'');
        Episode.GET(NameB);
      END;
      NameAttemptsB := Episode."Name Attempts";
      Journal.PickUnresolved(NamePending,Parse.MaxDocumentsPerPass);
      NamePickedCeiling := 0;
      IF NamePending.GET(NameB) THEN
        NamePickedCeiling := 1;
      Check(Passed,Total,Report,ChkNameCeilingTxt,
            (NameAttemptsB = Parse.MaxNameAttempts) AND (NamePickedCeiling = 0),
            STRSUBSTNO(NameCeilingTxt,NameAttemptsB,Parse.MaxNameAttempts,NamePickedCeiling));

      Sweep;

      Worst := OpenMs;
      IF UpdateMs > Worst THEN
        Worst := UpdateMs;
      IF CloseMs > Worst THEN
        Worst := CloseMs;
      Check(Passed,Total,Report,ChkCostTxt,Worst <= CeilingMs,STRSUBSTNO(CostTxt,Worst,CeilingMs));

      Report := Report + CrLf +
        Timing(OpenPhaseTxt,OpenMs,Waiters) +
        Timing(UpdatePhaseTxt,UpdateMs,Waiters) +
        Timing(ClosePhaseTxt,CloseMs,Waiters) +
        Timing(SweepPhaseTxt,SweepMs,Waiters - BackdateRows - CeilingSurplus);
    END;

    LOCAL PROCEDURE PassOfWaiters@3(Waiters@1000000000 : Integer;WaitMs@1000000001 : Integer;VAR Opened@1000000002 : Integer;VAR Updated@1000000003 : Integer;VAR Closed@1000000004 : Integer;VAR Restarted@1000000005 : Integer) : Integer;
    VAR
      Observed@1000000006 : Record 110231;
      Started@1000000007 : DateTime;
      Elapsed@1000000008 : Duration;
      Index@1000000009 : Integer;
      Milliseconds@1000000010 : Integer;
    BEGIN
      // Меряется ровно то, что делает фоновая задача после съёма: проход целиком, вместе
      // с COMMIT. Без COMMIT замер был бы вдвое красивее и вдвое бесполезнее - запись в
      // журнал транзакций и есть то, чем проход платит.
      Started := CURRENTDATETIME;
      Journal.BeginPass(Started,PeriodMs,AlertUnderTest);
      FOR Index := 1 TO Waiters DO BEGIN
        MakeObserved(Observed,Index,WaitMs);
        Journal.Observe(Observed);
      END;
      Journal.FinishPass;
      COMMIT;
      Elapsed := CURRENTDATETIME - Started;
      Milliseconds := Elapsed;
      Journal.PassCounts(Opened,Updated,Closed,Restarted);
      EXIT(Milliseconds);
    END;

    LOCAL PROCEDURE MakeObserved@4(VAR Observed@1000000000 : Record 110231;Index@1000000001 : Integer;WaitMs@1000000002 : Integer);
    BEGIN
      // Строка сделана похожей на настоящую не для красоты: цена записи зависит от того,
      // сколько байт уезжает в SQL, а строковые поля здесь - самые широкие в таблице.
      // Замер на пустой строке был бы вдвое бодрее того, что случится на бою.
      Observed.INIT;
      Observed."Victim SPID" := -Index;
      Observed."Victim Transaction Id" := 4000000000 + Index;
      Observed."Victim Is NAV" := Observed."Victim Is NAV"::Yes;
      Observed."Victim Host" := 'STAND-HOST';
      Observed."Victim Process Id" := 4000 + Index;
      Observed."Blocker SPID" := -Index - 1;
      Observed."Blocker Transaction Id" := 5000000000 + Index;
      Observed."Head SPID" := -1;
      Observed."Chain Depth" := 2;
      Observed."Victims Behind Head" := Index;
      Observed."Resource Kind" := 'KEYLOCK';
      Observed."Hobt Id" := SampleHobt + Index;
      Observed."Database Id" := 7;
      Observed."Held Mode" := 'X';
      Observed."Wait Type" := 'LCK_M_U';
      Observed."SQL Object Name" := 'Company Name$Sales Line';
      Observed."NAV Table Name" := 'Sales Line';
      Observed."NAV Key No." := 1;
      Observed."Last Wait (ms)" := WaitMs;
      Observed."Blocker Tran Began At" := CURRENTDATETIME;
      Observed."Blocker Statement" := 'UPDATE "Company Name$Sales Line" SET "Quantity" ' +
        'WHERE "Document Type" AND "Document No_" AND "Line No_"';
    END;

    LOCAL PROCEDURE SampleHobt@20() : BigInteger;
    VAR
      Value@1000000000 : BigInteger;
    BEGIN
      // Числовой литерал в C/AL - это Decimal, а Decimal не переваривает больше
      // 999 999 999 999 999,99. Настоящий hobt длиннее, и написанный литералом он не
      // компилируется вовсе. Единственный способ внести его в код - через EVALUATE.
      EVALUATE(Value,'72057594261667840');
      EXIT(Value);
    END;

    LOCAL PROCEDURE Backdate@5(Rows@1000000000 : Integer;Days@1000000001 : Integer) : Integer;
    VAR
      Moved@1000000002 : Integer;
      Back@1000000003 : Duration;
    BEGIN
      // Состаривание идёт правкой поля, а не ожиданием месяца.
      Back := Days * Parse.MsPerDay;
      Episode.RESET;
      Episode.SETFILTER("Victim SPID",'<%1',0);
      IF Episode.FINDSET(TRUE) THEN
        REPEAT
          Episode."Started At" := CURRENTDATETIME - Back;
          Episode.MODIFY;
          Moved := Moved + 1;
        UNTIL (Moved >= Rows) OR (Episode.NEXT = 0);
      COMMIT;
      EXIT(Moved);
    END;

    LOCAL PROCEDURE Sweep@6() : Integer;
    VAR
      Started@1000000000 : DateTime;
      Elapsed@1000000001 : Duration;
      Milliseconds@1000000002 : Integer;
    BEGIN
      // Уборка за собой обязательна и делается по СВОЕЙ метке: у строк мерного прогона
      // номер сессии отрицательный, а настоящий spid отрицательным не бывает. Строка,
      // оставшаяся в журнале после замера, - это ложь в отчёте заказчику.
      Started := CURRENTDATETIME;
      Episode.RESET;
      Episode.SETFILTER("Victim SPID",'<%1',0);
      Episode.DELETEALL;
      // Своё убирается и из ИСТОРИИ - по той же метке. Мерный прогон гонит туда строки
      // нарочно, а история чужого удаления не терпит: тронуть в ней можно только то, что
      // сам же и положил, и отличается это отрицательным номером сессии.
      Moved.RESET;
      Moved.SETFILTER("Victim SPID",'<%1',0);
      Moved.DELETEALL;
      COMMIT;
      Elapsed := CURRENTDATETIME - Started;
      Milliseconds := Elapsed;
      EXIT(Milliseconds);
    END;

    LOCAL PROCEDURE HoldOldest@52() : Integer;

    LOCAL PROCEDURE MirrorGap@43() : Text;
    VAR
      JournalRef@1000000000 : RecordRef;
      HistoryRef@1000000001 : RecordRef;
      Here@1000000002 : FieldRef;
      There@1000000003 : FieldRef;
      Index@1000000004 : Integer;
    BEGIN
      // История обязана принимать строку журнала целиком, и списки полей обязаны совпадать.
      // Список списан с журнала слово в слово, но разойтись он может молча - на первом же
      // поле, добавленном в журнал и забытом здесь. Сверяются номер, имя, тип и длина:
      // одного имени мало (Text50 против Text250 прошло бы), одного типа тоже (обрезание
      // при переезде выглядело бы как усечённый текст, а не как поломка).
      JournalRef.OPEN(DATABASE::"LockWatch Episode");
      HistoryRef.OPEN(DATABASE::"LockWatch Episode History");
      IF JournalRef.FIELDCOUNT <> HistoryRef.FIELDCOUNT THEN
        EXIT(STRSUBSTNO(MirrorCountTxt,JournalRef.FIELDCOUNT,HistoryRef.FIELDCOUNT));
      FOR Index := 1 TO JournalRef.FIELDCOUNT DO BEGIN
        Here := JournalRef.FIELDINDEX(Index);
        IF NOT HistoryRef.FIELDEXIST(Here.NUMBER) THEN
          EXIT(STRSUBSTNO(MirrorMissingTxt,Here.NUMBER,Here.NAME));
        There := HistoryRef.FIELD(Here.NUMBER);
        IF (Here.NAME <> There.NAME) OR (FORMAT(Here.TYPE) <> FORMAT(There.TYPE)) OR
           (Here.LENGTH <> There.LENGTH) THEN
          EXIT(STRSUBSTNO(MirrorDifferTxt,Here.NUMBER,
                          Here.NAME + ' ' + FORMAT(Here.TYPE) + FORMAT(Here.LENGTH),
                          There.NAME + ' ' + FORMAT(There.TYPE) + FORMAT(There.LENGTH)));
      END;
      EXIT('');
    END;

    LOCAL PROCEDURE MirrorOkNote@44(Gap@1000000000 : Text) : Text;
    BEGIN

      // как недоделанная проверка.
      IF Gap = '' THEN
        EXIT(MirrorOkTxt);
      EXIT('');
    END;

    LOCAL PROCEDURE NamePass@34(FirstIndex@1000000000 : Integer;SecondIndex@1000000001 : Integer);

    BEGIN
      // Оба ждущих обязаны приходить КАЖДЫМ проходом: FinishPass закрывает эпизод,

    LOCAL PROCEDURE NameIndexA@36() : Integer;
    BEGIN

    BEGIN
      Reading.RESET;
      Reading.DELETEALL;
      Reading.INIT;
      Reading."SQL Object Id" := BenchObjectId;
      Reading."Index Id" := BenchIndexId;
      Reading."SQL Object Name" := 'Company Name$Sales Line';
      Reading."NAV Table Name" := 'Sales Line';
      Reading."NAV Key No." := 1;
      // В буфере лежат СЫРЫЕ значения счётчика сервера. Приростом их делает накопитель,
      // и подавать сюда сразу прирост значило бы проверять не то, что работает на бою.
      Reading."Last Waits" := Waits;
      Reading."Last Wait (ms)" := WaitMs;
      Reading."Last Escalations" := Escalations;
      Reading.INSERT;
    END;
