    [External]
    PROCEDURE BeginPass@1(ObservedAt@1000000000 : DateTime;PeriodMs@1000000001 : Integer;AlertMs@1000000012 : Integer);
    BEGIN
      // Момент прохода берётся ОДИН на весь проход и приходит снаружи - часами SQL,
      // тем же запросом, что отдал очередь. Свои часы здесь не годятся: между съёмом
      // и записью проходят миллисекунды, и они легли бы в начало каждого эпизода.
      PassAt := ObservedAt;
      PassPeriodMs := PeriodMs;
      // Порог тревоги нужен не журналу, а советчику: он единственное, что советчик берёт
      // из настройки, и приходит он сюда тем же путём, что и период - снаружи и один раз
      // на проход, а не чтением настройки на каждую строку.
      PassAlertMs := AlertMs;
      OpenedCount := 0;
      UpdatedCount := 0;
      ClosedCount := 0;
      RestartedCount := 0;
      PassStarted := TRUE;
    END;

    [External]
    PROCEDURE Observe@2(VAR Observed@1000000000 : Record 110231);
    BEGIN
      // Одна строка очереди. Эпизод ведётся ОДНОЙ строкой журнала, которая обновляется:
      // снимки на каждый проход дали бы в шторм по полтысячи вставок каждые три секунды,
      // и наблюдатель стал бы дороже наблюдаемого.
      IF NOT PassStarted THEN
        ERROR(NoPassErr);
      IF FindOpen(Observed) THEN BEGIN
        IF Parse.IsNewEpisode(Journal."Last Wait (ms)",Observed."Last Wait (ms)") THEN BEGIN
          // Жертва дождалась и тут же встала снова на тот же ресурс: сессия та же,
          // транзакция та же, ресурс тот же. По ключу это одна строка, а на деле два
          // эпизода, и единственная улика - счётчик ожидания пошёл назад.
          CloseRow;
          RestartedCount := RestartedCount + 1;
          OpenRow(Observed);
        END ELSE
          UpdateRow(Observed);
      END ELSE
        OpenRow(Observed);
    END;

    [External]
    PROCEDURE FinishPass@3() : Integer;
    BEGIN
      IF NOT PassStarted THEN
        ERROR(NoPassErr);
      // Кто был открыт и в этом проходе не встретился - тот из очереди ушёл. Сюда же
      // попадают эпизоды, оставшиеся открытыми от прошлой жизни задачи: она могла быть
      // снята посреди прохода, и вечно открытая строка - это утечка, а не наблюдение.
      //
      // Идём по СКВОЗНОМУ ключу, хотя отбор - по признаку "открыт". Причина в том, что
      // закрытие меняет сам признак, а он первое поле того ключа, по которому читать было
      // бы удобнее: строка ушла бы из-под курсора, и остаток набора либо пропустился бы,
      // либо прочёлся дважды. Открытых эпизодов единицы, сортировка здесь ничего не стоит.
      Journal.RESET;
      Journal.SETCURRENTKEY("Entry No.");
      Journal.SETRANGE(Open,TRUE);
      Journal.SETFILTER("Last Seen At",'<%1',PassAt);
      IF Journal.FINDSET(TRUE) THEN
        REPEAT
          CloseRow;
        UNTIL Journal.NEXT = 0;
      PassStarted := FALSE;
      EXIT(ClosedCount);
    END;

    [External]
    PROCEDURE PassCounts@4(VAR Opened@1000000000 : Integer;VAR Updated@1000000001 : Integer;VAR Closed@1000000002 : Integer;VAR Restarted@1000000003 : Integer);
    BEGIN
      Opened := OpenedCount;
      Updated := UpdatedCount;
      Closed := ClosedCount;
      Restarted := RestartedCount;
    END;

    [External]
    PROCEDURE PickUnresolved@12(VAR Buffer@1000000000 : TEMPORARY Record 110231;MaxRows@1000000001 : Integer) : Integer;
    VAR
      Picked@1000000002 : Integer;
    BEGIN
      // Кого разбирать до номера документа. Отбор строгий нарочно: только ИДУЩИЕ эпизоды,
      // потому что мост к документу - это блокировка, которую виновник держит ПРЯМО СЕЙЧАС.
      // На закрытом эпизоде разбирать уже нечего: ожидание кончилось ровно потому, что
      // блокировку отпустили, и в dm_tran_locks её больше нет.
      //
      // "Причина не заполнена" - это отметка "ещё не пробовали". Без неё эпизод, у которого
      // документа нет и не будет, разбирался бы заново каждый проход, и разбор из редкого
      // стал бы постоянным.
      Buffer.RESET;
      Buffer.DELETEALL;
      Journal.RESET;
      Journal.SETCURRENTKEY(Open,"Victim SPID","Victim Transaction Id","Hobt Id");
      Journal.SETRANGE(Open,TRUE);
      Journal.SETFILTER("Blocker Transaction Id",'<>%1',0);
      Journal.SETRANGE("Document Source",Journal."Document Source"::None);
      Journal.SETRANGE("No Document Reason",'');
      IF Journal.FINDSET THEN
        REPEAT
          Buffer := Journal;
          Buffer.INSERT;
          Picked := Picked + 1;
        UNTIL (Picked >= MaxRows) OR (Journal.NEXT = 0);
      Journal.RESET;
      EXIT(Picked);
    END;

    [External]
    PROCEDURE SetDocument@13(EntryNo@1000000000 : Integer;DocumentNo@1000000001 : Code[20];UserName@1000000002 : Code[50];Source@1000000003 : Integer;Reason@1000000004 : Text);
    BEGIN
      Journal.RESET;
      IF NOT Journal.GET(EntryNo) THEN
        EXIT;
      Journal."Document No." := DocumentNo;
      Journal."Document Source" := Source;
      Journal."No Document Reason" := COPYSTR(Reason,1,MAXSTRLEN(Journal."No Document Reason"));
      IF UserName <> '' THEN BEGIN
        // Имя приходит из отметки контекста, то есть от самой транзакции виновника, а не
        // от платформы. Это НЕ то имя, которое обещано в разделе про EnableDeadlockMonitoring:
        // здесь оно про виновника и получено без единого нового права.
        Journal."User Name" := UserName;
        Journal."User Source" := Journal."User Source"::FromMark;
      END;
      Journal.MODIFY;
    END;

    [External]
    PROCEDURE AddDeadlock@14(VAR Row@1000000000 : Record 110231) : Boolean;
    VAR
      Existing@1000000001 : Record 110231;
    BEGIN
      // Один и тот же граф виден при КАЖДОМ чтении буфера, пока не вытеснится, поэтому
      // отметка "прочитано до" - только половина защиты. Вторая половина здесь: отпечаток
      // графа. Отметка от повторов не спасает нарочно - события, попавшие в одну
      // миллисекунду, читаются заново, чтобы не потерять второе из них.
      IF Row."Deadlock Id" = '' THEN
        EXIT(FALSE);
      Existing.RESET;
      Existing.SETCURRENTKEY("Deadlock Id");
      Existing.SETRANGE("Deadlock Id",Row."Deadlock Id");
      IF NOT Existing.ISEMPTY THEN
        EXIT(FALSE);
      // Строка приезжает ЗАКОНЧЕННОЙ: у взаимоблокировки нет "идёт сейчас", сервер уже
      // разорвал круг. Поэтому она не открывается и ведению эпизода не подлежит вовсе.
      Journal := Row;
      Journal."Entry No." := NextNo;
      Journal.Open := FALSE;
      ApplyAdvice(Journal);
      Journal.INSERT;
      CopyStatements(Row,Journal);
      EXIT(TRUE);
    END;

    LOCAL PROCEDURE CopyStatements@17(VAR From@1000000000 : Record 110231;VAR Into@1000000001 : Record 110231);
    VAR
      InStr@1000000002 : InStream;
      OutStr@1000000003 : OutStream;
      Body@1000000004 : BigText;
      Line@1000000006 : Text;
      Touched@1000000005 : Boolean;
    BEGIN
      // Текст запроса переносится ОДИН раз, при открытии эпизода, и дальше не трогается.
      // Довод не в экономии: строка эпизода обновляется каждым проходом, и переписывать
      // текст заново значило бы менять его под читателем. Виновник за три секунды успевает
      // послать другой оператор, и в журнале оказался бы не тот, который взял блокировку.
      //
      // CALCFIELDS у источника не зовётся нарочно: источник - ВРЕМЕННАЯ запись, значение
      // у неё уже в памяти, и считать там нечего.
      // CALCFIELDS обязателен ДАЖЕ на временной записи, и это ровно наоборот тому, чего
      // ждёшь: у временной таблицы значения и так в памяти. Без него HASVALUE отвечает
      // "нет" при полностью заполненном источнике, текст молча не переносится, а на экране
      // это выглядит как "запрос не сняли". Измерено на себе 08.09.2026.
      From.CALCFIELDS("Victim Statement","Blocker Statement Text");
      IF From."Victim Statement".HASVALUE THEN BEGIN
        From."Victim Statement".CREATEINSTREAM(InStr,TEXTENCODING::UTF8);
        Body.READ(InStr);
        Body.GETSUBTEXT(Line,1,Body.LENGTH);
        IF Line <> '' THEN BEGIN
          Into."Victim Statement".CREATEOUTSTREAM(OutStr,TEXTENCODING::UTF8);
          OutStr.WRITETEXT(Line);
          Touched := TRUE;
        END;
      END;
      IF From."Blocker Statement Text".HASVALUE THEN BEGIN
        CLEAR(Body);
        Line := '';
        From."Blocker Statement Text".CREATEINSTREAM(InStr,TEXTENCODING::UTF8);
        Body.READ(InStr);
        Body.GETSUBTEXT(Line,1,Body.LENGTH);
        IF Line <> '' THEN BEGIN
          Into."Blocker Statement Text".CREATEOUTSTREAM(OutStr,TEXTENCODING::UTF8);
          OutStr.WRITETEXT(Line);
          Touched := TRUE;
        END;
      END;
      // Лишний MODIFY там, где текста нет вовсе, - это цена ВЫКЛЮЧЕННОГО выключателя, а
      // выключенное платить не должно.
      IF Touched THEN
        Into.MODIFY;
    END;

    [External]
    PROCEDURE StatementText@18(EntryNo@1000000000 : Integer;Blocker@1000000001 : Boolean) : Text;
    VAR
      Row@1000000002 : Record 110231;
      InStr@1000000003 : InStream;
      Body@1000000004 : BigText;
      Result@1000000005 : Text;
    BEGIN
      IF NOT Row.GET(EntryNo) THEN
        EXIT('');
      IF Blocker THEN BEGIN
        Row.CALCFIELDS("Blocker Statement Text");
        IF NOT Row."Blocker Statement Text".HASVALUE THEN
          EXIT('');
        Row."Blocker Statement Text".CREATEINSTREAM(InStr,TEXTENCODING::UTF8);
      END ELSE BEGIN
        Row.CALCFIELDS("Victim Statement");
        IF NOT Row."Victim Statement".HASVALUE THEN
          EXIT('');
        Row."Victim Statement".CREATEINSTREAM(InStr,TEXTENCODING::UTF8);
      END;
      Body.READ(InStr);
      Body.GETSUBTEXT(Result,1,Body.LENGTH);
      EXIT(Result);
    END;

    LOCAL PROCEDURE ApplyAdvice@11(VAR Row@1000000000 : Record 110231);
    VAR
      AdviceText@1000000001 : Text;
      Evidence@1000000002 : Text;
    BEGIN
      // Совет - чистая функция от полей строки, и живёт он в разборе: журнал только
      // относит ему то, что уже знает, и кладёт ответ обратно. Никакого своего суждения
      // здесь нет, и потому советчик проверяется без базы вовсе.
      Parse.Advise(Row.Class,Row.Outcome,Row."Max Wait (ms)",PassAlertMs,
                   Row."Victims Behind Head",Row."On SIFT",Row."NAV Key No.",
                   Row."Head Is Foreign",AdviceText,Evidence);
      Row.Advice := COPYSTR(AdviceText,1,MAXSTRLEN(Row.Advice));
      Row."Advice Evidence" := COPYSTR(Evidence,1,MAXSTRLEN(Row."Advice Evidence"));
    END;

    [External]
    PROCEDURE PickAlerting@16(VAR Buffer@1000000000 : TEMPORARY Record 110231;AlertMs@1000000001 : Integer;MaxRows@1000000002 : Integer) : Integer;
    VAR
      Picked@1000000003 : Integer;
    BEGIN
      // На тревогу разбираются только ИДУЩИЕ эпизоды. Смысл тревоги - "происходит сейчас,
      // вмешайся"; звать человека к тому, что кончилось само, значит приучить его тревогу
      // не читать, и тогда она не сработает в тот раз, когда была нужна.
      Buffer.RESET;
      Buffer.DELETEALL;
      IF (AlertMs <= 0) OR (MaxRows <= 0) THEN
        EXIT(0);
      Journal.RESET;
      Journal.SETCURRENTKEY(Open,"Victim SPID","Victim Transaction Id","Hobt Id");
      Journal.SETRANGE(Open,TRUE);
      Journal.SETFILTER("Max Wait (ms)",'>=%1',AlertMs);
      IF Journal.FINDSET THEN
        REPEAT
          Buffer := Journal;
          Buffer.INSERT;
          Picked := Picked + 1;
        UNTIL (Picked >= MaxRows) OR (Journal.NEXT = 0);
      Journal.RESET;
      EXIT(Picked);
    END;

    [External]
    PROCEDURE Summarize@15(VAR Line@1000000000 : Record 110234;FromAt@1000000001 : DateTime;ToAt@1000000002 : DateTime;AlertMs@1000000003 : Integer) : Integer;
    VAR
      Episode@1000000004 : Record 110231;
      Counted@1000000005 : Integer;
      WaitMs@1000000006 : Integer;
    BEGIN
      // Сводка считается ПЕРЕБОРОМ, а не агрегатом базы. FlowField со счётом по журналу
      // означал бы SumIndexFields на таблице, куда фоновая задача пишет каждые три
      // секунды, - то есть наблюдатель сам стал бы тем, за что дерутся. Сводку открывают
      // руками и изредка, и платить за неё каждой записью эпизода незачем.
      Line.RESET;
      Line.DELETEALL;

      Episode.RESET;
      Episode.SETCURRENTKEY("NAV Table Name","Started At");
      // Пустая нижняя граница означает "весь журнал". Диапазон с пустым концом выглядел
      // бы тем же самым, но означал бы отбор ПО пустому значению, а не его отсутствие.
      IF FromAt = 0DT THEN
        Episode.SETFILTER("Started At",'<=%1',ToAt)
      ELSE
        Episode.SETRANGE("Started At",FromAt,ToAt);
      IF Episode.FINDSET THEN
        REPEAT
          Counted := Counted + 1;
          WaitMs := Episode."Max Wait (ms)";
          IF NOT Line.GET(Episode."NAV Table Name",Episode."NAV Key No.") THEN BEGIN
            Line.INIT;
            Line."NAV Table Name" := Episode."NAV Table Name";
            Line."NAV Key No." := Episode."NAV Key No.";
            Line.INSERT;
          END;
          Line.Episodes := Line.Episodes + 1;
          Line."Total Wait (ms)" := Line."Total Wait (ms)" + WaitMs;
          IF (AlertMs > 0) AND (WaitMs >= AlertMs) THEN
            Line."Over Threshold" := Line."Over Threshold" + 1;
          IF Episode."On SIFT" THEN
            Line."On SIFT" := TRUE;
          IF Episode."Started At" > Line."Last At" THEN
            Line."Last At" := Episode."Started At";
          // Совет строки - совет самого долгого эпизода группы. Первый эпизод берётся
          // безусловно, а дальше только строго длиннее: при равенстве побеждает первый
          // встреченный, и строка сводки не меняется от порядка чтения.
          IF (Line.Episodes = 1) OR (WaitMs > Line."Max Wait (ms)") THEN BEGIN
            Line."Max Wait (ms)" := WaitMs;
            Line.Advice := Episode.Advice;
          END;
          Line.MODIFY;
        UNTIL Episode.NEXT = 0;

      // Порядок - часть смысла сводки, а не украшение страницы: первым идёт то, во что
      // ожидание обошлось дороже всего. Поэтому он ставится здесь, а не только видом
      // страницы, который перебор строк по временной записи мог бы и сбросить.
      Line.RESET;
      Line.SETCURRENTKEY("Total Wait (ms)");
      Line.ASCENDING(FALSE);
      EXIT(Counted);
    END;

    [External]
    PROCEDURE TrimOldEntries@5(RetentionDays@1000000000 : Integer;MaxRows@1000000001 : Integer) : Integer;
    VAR
      Threshold@1000000002 : DateTime;
      Age@1000000003 : Duration;
      Removed@1000000004 : Integer;
    BEGIN
      // Срок хранения объявленный, и чистка порционная: журнал показывает чужие учётные
      // записи, а разовое DELETEALL на месяц накопленного - это блокировка собственной
      // таблицы посреди прохода, то есть ровно то, что мы ловим.
      IF (RetentionDays <= 0) OR (MaxRows <= 0) THEN
        EXIT(0);
      Age := RetentionDays * Parse.MsPerDay;
      Threshold := CURRENTDATETIME - Age;

      Journal.RESET;
      Journal.SETCURRENTKEY("Started At");
      Journal.SETRANGE(Open,FALSE);
      Journal.SETFILTER("Started At",'<%1',Threshold);
      IF Journal.FINDSET(TRUE) THEN
        REPEAT
          Journal.DELETE;
          Removed := Removed + 1;
        UNTIL (Removed >= MaxRows) OR (Journal.NEXT = 0);
      EXIT(Removed);
    END;

    LOCAL PROCEDURE FindOpen@6(VAR Observed@1000000000 : Record 110231) : Boolean;
    BEGIN
      Journal.RESET;
      Journal.SETCURRENTKEY(Open,"Victim SPID","Victim Transaction Id","Hobt Id");
      Journal.SETRANGE(Open,TRUE);
      Journal.SETRANGE("Victim SPID",Observed."Victim SPID");
      Journal.SETRANGE("Victim Transaction Id",Observed."Victim Transaction Id");
      Journal.SETRANGE("Hobt Id",Observed."Hobt Id");
      // Ожидание объекта целиком hobt не несёт ВОВСЕ: у всех таких строк он ноль. По одному
      // ключу два разных ожидания одной сессии слились бы в один эпизод - правдоподобно и
      // молча. Ресурс дофильтровывается мимо индекса: после ключа тут считанные строки.
      Journal.SETRANGE("Resource Kind",Observed."Resource Kind");
      Journal.SETRANGE("SQL Object Name",Observed."SQL Object Name");
      EXIT(Journal.FINDFIRST);
    END;

    LOCAL PROCEDURE OpenRow@7(VAR Observed@1000000000 : Record 110231);
    BEGIN
      Journal := Observed;
      Journal."Entry No." := NextNo;
      Journal."Started At" := Parse.EpisodeStart(PassAt,Observed."Last Wait (ms)");
      Journal."Last Seen At" := PassAt;
      Journal."Max Wait (ms)" := Observed."Last Wait (ms)";
      Journal."Ended At" := 0DT;
      Journal.Outcome := Journal.Outcome::Undetermined;
      Journal.Open := TRUE;
      ApplyAdvice(Journal);
      Journal.INSERT;
      CopyStatements(Observed,Journal);
      OpenedCount := OpenedCount + 1;
    END;

    LOCAL PROCEDURE UpdateRow@8(VAR Observed@1000000000 : Record 110231);
    VAR
      Keep@1000000001 : Record 110231;
    BEGIN
      // Наблюдаемое перекрывает строку ЦЕЛИКОМ, кроме того, что устанавливается один раз.
      // Так новое поле съёма начинает обновляться само, а не забывается молча; список
      // "устанавливается один раз" короткий и лежит здесь, на виду.
      //
      // Обновлять есть что: голова цепочки за три секунды меняется, глубина растёт, класс
      // эпизода может смениться с обычного на эскалацию. Строка обязана показывать то, что
      // есть СЕЙЧАС, а не то, что было в первом наблюдении.
      Keep := Journal;
      Journal := Observed;
      Journal."Entry No." := Keep."Entry No.";
      Journal."Started At" := Keep."Started At";
      Journal."Ended At" := Keep."Ended At";
      Journal.Outcome := Keep.Outcome;
      Journal."Document No." := Keep."Document No.";
      Journal."Document Source" := Keep."Document Source";
      Journal."No Document Reason" := Keep."No Document Reason";
      Journal."User Name" := Keep."User Name";
      Journal."User Source" := Keep."User Source";
      // Совета в этом списке нет нарочно. Всё остальное здесь - то, что установлено ОДИН
      // раз и повторно не выясняется, а совет обязан идти за строкой: ожидание растёт,
      // класс меняется, и совет, поставленный при открытии, к закрытию был бы про другое.
      Journal.Open := TRUE;
      Journal."Last Seen At" := PassAt;
      Journal."Max Wait (ms)" := Keep."Max Wait (ms)";
      IF Observed."Last Wait (ms)" > Journal."Max Wait (ms)" THEN
        Journal."Max Wait (ms)" := Observed."Last Wait (ms)";
      ApplyAdvice(Journal);
      Journal.MODIFY;
      UpdatedCount := UpdatedCount + 1;
    END;

    LOCAL PROCEDURE CloseRow@9();
    VAR
      Waited@1000000000 : Duration;
      VictimIsNav@1000000001 : Boolean;
    BEGIN
      // Мига конца мы не знаем: жертва исчезла где-то между прошлым наблюдением и этим,
      // и назвать его точнее нельзя. Зато длительность из DMV точна, поэтому концом
      // объявляется последний миг, когда ожидание ТОЧНО ещё шло. Насколько твёрд вывод,
      // говорит исход - у него три значения, и "не определён" честнее правдоподобной цифры.
      VictimIsNav := Journal."Victim Is NAV" = Journal."Victim Is NAV"::Yes;
      Waited := Journal."Max Wait (ms)";
      Journal."Ended At" := Journal."Started At" + Waited;
      // Значения исхода в разборе - те же числа, что и в списке опций поля, и это не
      // совпадение: разбор о таблице ничего не знает и знать не должен, а порядок опций
      // держится проверкой в обкатке.
      Journal.Outcome := Parse.Outcome(Journal."Last Wait (ms)",PassPeriodMs,VictimIsNav);
      Journal.Open := FALSE;
      // Совет пересчитывается ПОСЛЕ исхода, а не до: "отказ по пределу" - это отдельный
      // совет, и до закрытия эпизода он неизвестен.
      ApplyAdvice(Journal);
      Journal.MODIFY;
      ClosedCount := ClosedCount + 1;
    END;

    LOCAL PROCEDURE NextNo@10() : Integer;
    VAR
      Last@1000000000 : Record 110231;
    BEGIN
      // Номер снимается с конца ОДИН раз за жизнь экземпляра кодюнита, а не на каждую
      // строку: в шторм это полтысячи лишних чтений последней страницы за проход.
      // Так можно потому, что пишет сюда одна фоновая задача, и спорить за номер не с кем.
      IF NextEntryNo = 0 THEN BEGIN
        Last.RESET;
        IF Last.FINDLAST THEN
          NextEntryNo := Last."Entry No.";
      END;
      NextEntryNo := NextEntryNo + 1;
      EXIT(NextEntryNo);
    END;

    BEGIN
    {
      NAV-половина прохода: ведение эпизода в журнале. Съёма с SQL здесь нет ни строки -
      на вход подаётся уже разобранная строка очереди, и потому эта половина меряется и
      проверяется БЕЗ единого ожидания на стенде (Codeunit 110233).

      Цена именно этой половины и была последним, что жило на обещании. Съём с DMV стоит
      12 мс при двухстах ждущих и от нас не зависит; а вот поиск эпизода, запись и COMMIT
      на сотни строк - зависит целиком, и в шторм это самая дорогая часть прохода.

      Три решения, которые эту цену и держат:

      1. Эпизод - строка, которая ОБНОВЛЯЕТСЯ. Полтысячи вставок каждые три секунды
         превратили бы наблюдателя в причину.
      2. Номер записи берётся с конца один раз за жизнь задачи, а не на строку.
      3. COMMIT здесь не делается вовсе. Он один на проход и принадлежит вызывающему:
         COMMIT на строку - это полтысячи записей в журнал транзакций за проход.

      Закрытие эпизода - единственное место, где легко соврать красиво. Мига конца мы не
      знаем; знаем последнюю длительность из DMV. Поэтому концом объявляется последний
      миг, когда ожидание точно ещё шло, а вся неуверенность вынесена в исход отдельным
      полем с тремя значениями.
    }
    END.
  }
}
