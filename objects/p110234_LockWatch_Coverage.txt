  CODE
  {
    VAR
      Coverage@1000000000 : Codeunit 110238;
      CoverageSince@1000000001 : DateTime;
      CounterWaits@1000000002 : BigInteger;
      CounterMs@1000000003 : BigInteger;
      CounterEscalations@1000000004 : BigInteger;
      JournalEpisodes@1000000005 : Integer;
      JournalMs@1000000006 : BigInteger;
      SharePct@1000000007 : Decimal;
      ShareStyle@1000000008 : Text[30];
      StartOverQst@1000000009 : TextConst 'ENU=Drop the coverage counted so far and start over from now?;RUS=Сбросить насчитанный охват и начать счёт заново с этого мига?';

    LOCAL PROCEDURE ReadTotals@1();
    VAR
      State@1000000000 : Record 110237;
    BEGIN
      CoverageSince := State.CoverageSince;
      Coverage.CounterTotal(CounterWaits,CounterMs,CounterEscalations);
      Coverage.JournalTotal(CoverageSince,JournalEpisodes,JournalMs);
      SharePct := Coverage.Share(CounterMs,JournalMs);
      SetShareStyle;
    END;

    LOCAL PROCEDURE SetShareStyle@2();
    BEGIN
      // Красится НИЗКАЯ доля, и порог у неё свой - не порог тревоги из настройки. Тот про
      // длительность одного эпизода, этот про то, можно ли вообще верить сводке.
      ShareStyle := StandardStyle;
      IF (CounterMs > 0) AND (SharePct < LowShare) THEN
        ShareStyle := AlarmStyle;
    END;

    LOCAL PROCEDURE StartOver@3();
    VAR
      State@1000000000 : Record 110237;
    BEGIN
      IF NOT CONFIRM(StartOverQst,FALSE) THEN
        EXIT;
      // Отметка и приросты сбрасываются ВМЕСТЕ. Оставить отметку значило бы сравнивать
      // счётчики с этого мига против журнала за всё прошлое, а сбросить одну её - наоборот.
      Coverage.ResetAll;
      // Единственная кнопка, которой человеку позволено тронуть состояние сторожа. Под
      // замком - потому что проход пишет ту же строку, и разойтись версиями им нельзя.
      State.FetchToWrite;
      State."Coverage Since" := CURRENTDATETIME;
      State.MODIFY;
      COMMIT;
      ReadTotals;
      CurrPage.UPDATE(FALSE);
    END;

    LOCAL PROCEDURE ShowEpisodes@4();
    VAR
      Episode@1000000000 : Record 110231;
      TableName@1000000001 : Text[250];
      KeyNo@1000000002 : Integer;
    BEGIN
      TableName := "NAV Table Name";
      KeyNo := "NAV Key No.";
      Episode.SETRANGE("NAV Table Name",TableName);
      Episode.SETRANGE("NAV Key No.",KeyNo);
      IF CoverageSince <> 0DT THEN
        Episode.SETFILTER("Started At",'>=%1',CoverageSince);
      PAGE.RUN(PAGE::"LockWatch Episodes",Episode);
    END;

    LOCAL PROCEDURE ShowHistoryEpisodes@8();
    VAR
      Episode@1000000000 : Record 110238;
      TableName@1000000001 : Text[250];
      KeyNo@1000000002 : Integer;
    BEGIN
      // Отдельной кнопкой, а не догадкой: в журнале и в истории лежат РАЗНЫЕ эпизоды одной
      // и той же пары "таблица плюс ключ", и человек сам решает, куда смотреть.
      TableName := "NAV Table Name";
      KeyNo := "NAV Key No.";
      Episode.SETRANGE("NAV Table Name",TableName);
      Episode.SETRANGE("NAV Key No.",KeyNo);
      IF CoverageSince <> 0DT THEN
        Episode.SETFILTER("Started At",'>=%1',CoverageSince);
      PAGE.RUN(PAGE::"LockWatch Episode History",Episode);
    END;

    LOCAL PROCEDURE LowShare@5() : Integer;
    BEGIN
      // Половина. Ниже неё журнал называет меньше половины ожидания, и сводкой по нему
      // уже нельзя пользоваться как картиной целого - только как списком самых долгих
      // случаев. Число это про доверие к экрану, а не про среду, и потому живёт в коде.
      EXIT(50);
    END;

    LOCAL PROCEDURE StandardStyle@6() : Text[30];
    BEGIN
      EXIT('Standard');
    END;

    LOCAL PROCEDURE AlarmStyle@7() : Text[30];
    BEGIN
      EXIT('Unfavorable');
    END;

    BEGIN
    {
      Охват: сколько ожиданий насчитал сам сервер против того, сколько назвал журнал.
      Экран этот заведён не для красоты, а против молчаливого обещания: сводка без охвата
      утверждает, что видно всё, а это неверно и было неверно с самого начала.

      Опрос очереди видит только те ожидания, которые дожили до следующего прохода.
      Ожидание в двести миллисекунд при периоде в три секунды не увидит никто, и таких
      на живой базе большинство. Поэтому доля НАЗВАННОГО почти никогда не будет
      стопроцентной, и низкая доля сама по себе не поломка - поломкой было бы её не знать.

      Сравнивать надо обе пары чисел, и они не об одном и том же. Число ожиданий против
      числа эпизодов говорит, сколько ожиданий было слишком коротким, чтобы их увидеть;
      миллисекунды против миллисекунд - сколько в этом было боли. Обычная картина -
      "ожиданий видно десятую часть, а боли девять десятых": длинные ожидания опрос ловит
      почти все, а коротких много и они дёшевы.

      Сторожем этот слой быть не может: счётчик прирастает только в момент КОНЦА ожидания
      ([FINDINGS.md], раздел 12), и триггер по нему сработал бы тогда, когда в очереди
      уже никого нет.
    }
    END.
