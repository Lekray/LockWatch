OBJECT Codeunit 110246 LockWatch Demo Wait
{
  OBJECT-PROPERTIES
  {
    Date=10.09.26;
    Time=12:00:00;
    Modified=Yes;
    Version List=LockWatch;
  }
  PROPERTIES
  {
    OnRun=BEGIN
            Want;
          END;

  }
  CODE
  {
    VAR
      Demo@1000000000 : Record 110239;
      Control@1000000001 : Codeunit 110244;
      WantedTxt@1000000002 : TextConst 'ENU=Waited for it and got it;RUS=Дождался и взял';

    LOCAL PROCEDURE Want@1();
    BEGIN
      // Ждущий. Просит ту же строку и встаёт за ней в очередь сервера - вот это ожидание
      // и видит проход, вот оно и попадает в журнал.
      //
      // Ждёт он ВНУТРИ GET, а не в отдельной строке кода: LOCKTABLE переводит следующее
      // чтение в чтение с замком, и сервер задерживает его до тех пор, пока держатель не
      // отпустит своё. Никакого цикла ожидания тут нет и быть не должно.
      Demo.LOCKTABLE;
      IF NOT Demo.GET(Control.TargetNo) THEN
        EXIT;
      // Запись после ожидания - не украшение: по ней видно, что ждущий ДОЖДАЛСЯ, а не
      // сорвался по пределу ожидания NAV. Отличить одно от другого по журналу нельзя:
      // и там и там эпизод закрыт.
      Demo.Note := WantedTxt;
      Demo."Touched At" := CURRENTDATETIME;
      Demo.MODIFY;
    END;

    BEGIN
    {
      Вторая сессия показа - та, что ждёт. Своего предела ожидания она не ставит и ставить
      не должна: у неё предел NAV, десять секунд, и показ нарочно держит строку меньше -
      ждущий обязан ДОЖДАТЬСЯ. Сорвись он по таймауту, показ учил бы человека тому, чего
      на исправной системе не происходит.
    }
    END.
  }
}
