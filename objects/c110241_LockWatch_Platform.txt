    [External]
    PROCEDURE NameUser@1(EntryNo@1000000000 : Integer;VAR Reason@1000000001 : Text) : Boolean;
    VAR
      Episode@1000000002 : Record 110231;
      Locks@1000000003 : Record 2000000154;
    BEGIN
      // Дорога к имени ЧЕЛОВЕКА от самой платформы. Мост - номер транзакции виновника: он
      // уже лежит в строке журнала, снятый со стороны SQL, и он же стоит первым полем в
      // штатной таблице. Совпадение это не случайное: обе стороны берут его у одного и
      // того же request_owner_id.
      //
      // Читается таблица ТОЛЬКО по кнопке и только на один разбираемый случай. Провайдер у
      // неё EagerVirtual: любое чтение, даже с отбором, материализует ВСЕ блокировки
      // экземпляра и обходит все живые сессии - 157 мс при тридцати тысячах блокировок,
      // то есть дороже всего именно в шторм (FINDINGS, раздел 2).
      Reason := '';
      IF NOT Episode.GET(EntryNo) THEN
        ERROR(NoEntryErr,EntryNo);
      IF Episode."Blocker Transaction Id" = 0 THEN BEGIN
        Reason := NoTranTxt;
        EXIT(FALSE);
      END;

      Locks.SETRANGE("Transaction ID",Episode."Blocker Transaction Id");
      IF NOT Locks.FINDFIRST THEN BEGIN
        Reason := STRSUBSTNO(NoRowTxt,Episode."Blocker Transaction Id");
        EXIT(FALSE);
      END;

      Episode."User Name" := COPYSTR(Locks."User Name",1,MAXSTRLEN(Episode."User Name"));
      Episode."User Source" := Episode."User Source"::FromPlatform;
      // Колонки AL отвечают НЕ на тот вопрос, на который их хочется прочесть: платформа
      // берёт их из живого стека виновника В МОМЕНТ ЧТЕНИЯ, а не оттуда, где он взял
      // блокировку. Длинная проводка уходит через десятки кодюнитов, и два соседних
      // чтения одного эпизода назовут разные объекты. Поэтому поле подписано честно.
      Episode."Blocker AL Scope" := COPYSTR(Locks."AL Method Scope",1,MAXSTRLEN(Episode."Blocker AL Scope"));
      Episode.MODIFY;
      EXIT(TRUE);
    END;

    [External]
    PROCEDURE RoadStatus@2(VAR Reason@1000000000 : Text) : Boolean;
    VAR
      Locks@1000000001 : Record 2000000154;
      ViewServerState@1000000002 : Boolean;
      AlterEventSession@1000000003 : Boolean;
      MonitorRunning@1000000004 : Boolean;
      LockWaits@1000000005 : Integer;
      Rows@1000000006 : Integer;
    BEGIN
      // Отказ этой дороги НЕВИДИМ: не выдали право - платформа пишет одну строку
      // трассировки уровня Warning, и таблица ведёт себя в точности как при отсутствии
      // блокировок. Экран будет пуст и на здоровой системе, и на сломанной, и различить
      // их можно только этой проверкой.
      Reason := '';
      IF NOT Sql.PlatformRoad(ViewServerState,AlterEventSession,MonitorRunning,LockWaits) THEN BEGIN
        Reason := STRSUBSTNO(DeadTxt,STRSUBSTNO(NoSqlTxt,Sql.LastErrorText));
        EXIT(FALSE);
      END;
      IF NOT (ViewServerState AND AlterEventSession) THEN BEGIN
        Reason := STRSUBSTNO(DeadTxt,STRSUBSTNO(NoRightsTxt,ViewServerState,AlterEventSession));
        EXIT(FALSE);
      END;
      IF NOT MonitorRunning THEN BEGIN
        Reason := STRSUBSTNO(DeadTxt,NoMonitorTxt);
        EXIT(FALSE);
      END;

      Rows := Locks.COUNT;
      // Перекрёстная проверка, и она здесь главная: права на месте, сессия работает, а
      // таблица пуста ПРИ НЕПУСТОЙ ОЧЕРЕДИ - значит карта транзакций не наполнялась, и
      // ключ был снят в момент первого обращения к базе. Без этой проверки инструмент
      // сказал бы "всё хорошо" и показал пустую колонку.
      IF (Rows = 0) AND (LockWaits > 0) THEN BEGIN
        Reason := STRSUBSTNO(DeadTxt,STRSUBSTNO(DeadMapTxt,LockWaits));
        EXIT(FALSE);
      END;
      IF Rows = 0 THEN
        Reason := QuietTxt
      ELSE
        Reason := STRSUBSTNO(AliveTxt,Rows);
      EXIT(TRUE);
    END;

    [External]
    PROCEDURE CheckRoad@3();
    VAR
      Setup@1000000000 : Record 110230;
      Reason@1000000001 : Text;
      Alive@1000000002 : Boolean;
    BEGIN
      // Ответ кладётся В СТРОКУ настройки, а не только в окно: окно закроют и забудут, а
      // строка останется рядом с выключателем и будет видна тому, кто придёт после.
      Setup.FetchOnce;
      Alive := RoadStatus(Reason);
      Setup."Platform Road Status" := COPYSTR(Reason,1,MAXSTRLEN(Setup."Platform Road Status"));
      Setup.MODIFY;
      COMMIT;
      MESSAGE(StatusMsg,Alive,Reason);
    END;

    [External]
    PROCEDURE NameOpen@4();
    VAR
      Episode@1000000000 : Record 110231;
      Setup@1000000001 : Record 110230;
      Reason@1000000002 : Text;
      LastReason@1000000003 : Text;
      Named@1000000004 : Integer;
      Tried@1000000005 : Integer;
    BEGIN
      // Назвать всех, кто ждёт ПРЯМО СЕЙЧАС. Читается штатная таблица по одному разу на
      // эпизод, и это дорого - но идущих эпизодов единицы, а кнопку жмут руками и в тот
      // момент, когда уже разбираются. В фоновом проходе этого не делается никогда.
      Setup.FetchOnce;
      IF NOT Setup."Platform Names Enabled" THEN
        ERROR(OffErr);
      Episode.RESET;
      Episode.SETCURRENTKEY(Open,"Victim SPID","Victim Transaction Id","Hobt Id");
      Episode.SETRANGE(Open,TRUE);
      Episode.SETFILTER("Blocker Transaction Id",'<>%1',0);
      IF Episode.FINDSET THEN
        REPEAT
          Tried := Tried + 1;
          IF NameUser(Episode."Entry No.",Reason) THEN
            Named := Named + 1
          ELSE
            LastReason := Reason;
        UNTIL Episode.NEXT = 0;
      COMMIT;
      MESSAGE(NamedMsg,Named,Tried,LastReason);
    END;

    [External]
    PROCEDURE ShowLocks@5();
    VAR
      Locks@1000000000 : Record 2000000154;
      Line@1000000001 : Text;
      Shown@1000000002 : Integer;
    BEGIN
      // Что платформа видит СЕЙЧАС. Нужно это ровно тогда, когда имя не назвалось: пустая
      // штатная таблица и таблица, где нет нужной транзакции, - две разные беды, и лечатся
      // они по-разному. Без этого списка их не различить.
      Locks.RESET;
      IF Locks.FINDSET THEN
        REPEAT
          Shown := Shown + 1;
          IF Shown <= ShowLimit THEN
            Line := Line + FORMAT(Locks."Transaction ID") + ' ' + Locks."User Name" + ' ' +
                    COPYSTR(Locks."Object Name",1,40) + CrLf;
        UNTIL Locks.NEXT = 0;
      MESSAGE(LocksMsg,Shown,Line);
    END;

    LOCAL PROCEDURE ShowLimit@6() : Integer;
    BEGIN
      // Двадцать строк. Больше в окно всё равно не читают, а таблица в шторм отдаёт
      // тридцать тысяч - и собрать их в одну строку значило бы уронить клиент.
      EXIT(20);
    END;

    VAR
      Cr@1000000000 : Char;
      Lf@1000000001 : Char;
    BEGIN
      Cr := 13;
      Lf := 10;
      EXIT(FORMAT(Cr) + FORMAT(Lf));
    END;

      Код при этом оставлен, и не из упрямства: сборки платформы разные, а сторож ниже
