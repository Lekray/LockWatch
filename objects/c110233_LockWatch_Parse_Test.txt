      Head@1000000163 : Integer;
      Depth@1000000164 : Integer;
      Cycle@1000000165 : Boolean;
    BEGIN
      // Кольцо 1 -> 2 -> 1. Это взаимоблокировка на подходе либо уже разобранная
      // сервером. Разматывание обязано оборваться и сказать об этом, а не зациклиться:
      // фоновая задача, ушедшая в вечный цикл, держит слот планировщика насмерть.
      Waiter[1] := ChainSessionA; Blocker[1] := ChainSessionB;
      Waiter[2] := ChainSessionB; Blocker[2] := ChainSessionA;
      Head := Watch.ChainHead(Waiter,Blocker,ChainRows,ChainSessionA,Depth,Cycle);
      // Глубина после обрыва приходит ЕДИНИЦЕЙ, а не числом пройденных шагов: головы в
      // кольце нет вовсе, и объявленный ответ - непосредственный блокирующий. Сравнение с
      // потолком шагов стояло здесь прежде и покраснеть не могло: единица меньше потолка у
      // всякого кода, который вообще вернулся, - а об измерении потолка речь ниже.
      Check(Passed,Total,Report,ChkCycleTxt,
        Cycle AND (Depth = 1),
        STRSUBSTNO(CycleTxt,Cycle,Depth));

      // И второе утверждение, отдельное: обрыв - это ещё не ответ. Головы в кольце нет, и
      // возвращать начало нельзя: голова идёт в колонку виновника, начало - в колонку
      // ждущего, и обе колонки называют одного и того же человека, пострадавшего.
      // На прежнем коде голова равнялась 1, то есть самой жертве, и проверка красная.
      Check(Passed,Total,Report,ChkRingHeadTxt,
        (Head = ChainSessionB) AND (Depth = 1),
        STRSUBSTNO(RingHeadTxt,Head,ChainSessionB,Depth));

      // А это кольцо, в которое НАЧАЛО НЕ ВХОДИТ: 1 ждёт 3, 3 ждёт 2, 2 ждёт 3. Ветка
      // "вернулись в начало" сюда не заходит НИКОГДА - разматывание кончается только
      // потолком шагов. Такого набора в обкатке не было, и потолок не мерился ничем: в
      // кольце выше обрыв приходит возвратом в начало, и до потолка дело не доходит.
      //
      // Числа пройденных шагов наружу не отдаётся: глубину на обрыве разбор тоже
      // объявляет единицей. Мерится поэтому то, что видно: набор обязан ВЕРНУТЬСЯ (убери
      // потолок - этот REPEAT не кончится ничем, и прогон повиснет вместо красного
      // отчёта), назвать кольцо и ответить непосредственным блокирующим, а не жертвой.
      Waiter[1] := ChainSessionA; Blocker[1] := ChainSessionC;
      Waiter[2] := ChainSessionC; Blocker[2] := ChainSessionB;
      Waiter[3] := ChainSessionB; Blocker[3] := ChainSessionC;
      Head := Watch.ChainHead(Waiter,Blocker,RingRows,ChainSessionA,Depth,Cycle);
      Check(Passed,Total,Report,ChkCeilRingTxt,
        Cycle AND (Head = ChainSessionC) AND (Depth = 1) AND (Head <> ChainSessionA),
        STRSUBSTNO(ChainTxt,Head,Depth,Cycle));
    END;

    LOCAL PROCEDURE CheckChainCeiling@75(VAR Passed@1000000166 : Integer;VAR Total@1000000167 : Integer;VAR Report@1000000168 : Text);

    LOCAL PROCEDURE CheckWatchSilent@72(VAR Passed@1000000240 : Integer;VAR Total@1000000241 : Integer;VAR Report@1000000242 : Text);
    VAR
      Asked@1000000243 : DateTime;
      Ceiling@1000000244 : Duration;
      Gap@1000000245 : Duration;
      Fresh@1000000246 : Text;
      Silent@1000000247 : Text;
      Never@1000000248 : Text;
      Stopped@1000000249 : Text;
      JustArmed@1000000261 : Text;
      Rearmed@1000000262 : Text;
      ArmedLongAgo@1000000263 : Text;
    BEGIN
      // Цепочка, стоящая в очереди и не исполняющаяся ни разу, - состояние измеренное, а не
      // выдуманное (FINDINGS, раздел 63). Сказать о нём некому: строку сторожа пишет проход,
      // а прохода нет. Говорит поэтому страница, и спрашиваются тут все четыре ответа.

      // Свежий: молчание РОВНО в порог - ещё не беда. Граница обязана лежать там, где
      // названа, иначе уведомление создаётся на ровном месте.
      //
      // Молчит: за порогом слово обязано быть, и в нём обязано стоять ЧИСЛО минут - без
      // числа "давно" не отличить от "только что".
      //
      // Ни разу: прохода не было вовсе, и это ДРУГАЯ беда - цепочка не пошла с самого
      // завода. Слово обязано быть своё.
      //
      // Остановлен: галку сняли нарочно, и молчание тут законно. Без этой половины сторож
      // кричал бы после каждой остановки и перестал бы читаться до того, как окажется прав.
      //
      // Только заведён: прохода нет, а завод был ровно порог назад - первый проход ещё
      // впереди, и тревога в этот миг была бы ложной. Перезаведён: прежний проход давний,

      Asked := SilentObservedAt;
      Ceiling := Watch.SilenceCeilingMs(PollPeriodUnderTest);
      Gap := SilentMinutesUnderTest * Watch.MsPerMinute;
      Fresh := Watch.WatchSilentNote(TRUE,Asked - Ceiling,0DT,Asked,PollPeriodUnderTest);
      Silent := Watch.WatchSilentNote(TRUE,Asked - Gap,0DT,Asked,PollPeriodUnderTest);
      Never := Watch.WatchSilentNote(TRUE,0DT,0DT,Asked,PollPeriodUnderTest);
      Stopped := Watch.WatchSilentNote(FALSE,Asked - Gap,0DT,Asked,PollPeriodUnderTest);
      JustArmed := Watch.WatchSilentNote(TRUE,0DT,Asked - Ceiling,Asked,PollPeriodUnderTest);
      Rearmed := Watch.WatchSilentNote(TRUE,Asked - Gap,Asked - Ceiling,Asked,PollPeriodUnderTest);
      ArmedLongAgo := Watch.WatchSilentNote(TRUE,0DT,Asked - Gap,Asked,PollPeriodUnderTest);
      Check(Passed,Total,Report,ChkSilentTxt,
        (Fresh = '') AND (Stopped = '') AND (Never <> '') AND (Silent <> Never) AND
        (STRPOS(Silent,FORMAT(SilentMinutesUnderTest)) > 0) AND
        (JustArmed = '') AND (Rearmed = '') AND (ArmedLongAgo = Never),
        STRSUBSTNO(SilentTxt,Fresh,Silent,Never,Stopped,JustArmed,Rearmed,ArmedLongAgo));
    END;

    LOCAL PROCEDURE SqlNamePart@142(Name@1000000250 : Text) : Text;
    VAR
      Part@1000000251 : Text;
      Index@1000000252 : Integer;
      Ch@1000000253 : Text[1];
    BEGIN
      // Имя объекта или поля NAV платформа переводит в имя SQL заменой точки, косой

    LOCAL PROCEDURE CrLf@54() : Text[2];
    VAR
      Cr@1000000259 : Char;
      Lf@1000000260 : Char;
    BEGIN
      Cr := 13;
      Lf := 10;
      EXIT(FORMAT(Cr) + FORMAT(Lf));
    END;

    BEGIN
    {
      Обкатка чистых функций сторожа. Базы не касается вовсе: всё, что она проверяет, -
      счёт и разбор над данными, а не обращение к базе.
      функции над данными, и потому её можно гонять где угодно и сколько угодно.

      Три правила, по которым она сделана:

      1. Судит сама. Первой строкой "пройдено N из M", и при любом провале выходит
         ERROR, а не MESSAGE: запуск через Invoke-NAVCodeunit тогда отличает целое от
         сломанного по коду возврата, а не по чтению глазами.

      2. Каждая проверка непуста. Проверяемое место ломается нарочно, и проверка
         обязана покраснеть. Проверки, которая проходит всегда, здесь быть не должно.

      3. Данные взяты с натуры, а не выдуманы. Строки описания ресурса сняты с живого
         ожидания на стенде 07.09.2026, и потому проверяют настоящий формат, а не
         представление о нём.

      Проверено поломкой 07.09.2026: со снятой границей слова в PairValue прогон дал
      19 из тогдашних 20, и покраснела именно проверка границы - hobt 111 вместо
      72057594261667840. Проверять надо ПОСЛЕ перезапуска экземпляра службы: у неё свой
      кэш объектов, и без перезапуска сломанный разбор тоже отвечает целым.

      Проверено поломкой ещё раз 11.09.2026: счёт жертв за головой подменён на число
      ждущих вовсе (Behind := Count) - 24 из 25, и покраснел именно он: "за общей головой
      4, за чужой 4". До этого дня в опыте была ОДНА голова, за ней стояли все, и подмена
      проходила зелёной - набор с одной головой не отличает счёт жертв от счёта ждущих.
      Отличает вторая голова: за ней жертва одна.

      Самая ценная из проверок - на границу слова: в описании очереди рядом стоят
      hobtid=, id=lock... и associatedObjectId=, и разбор простой подстрокой подменит
      одно другим МОЛЧА, выдав правдоподобные числа. Такое не ловится ни компиляцией,
      ни глазами - только проверкой.
    }
    END.
