    END;

    LOCAL PROCEDURE RunBench@2(VAR Passed@1000000051 : Integer;VAR Total@1000000052 : Integer;VAR Report@1000000053 : Text);
    VAR
      Setup@1000000054 : Record 110230;
      NamePending@1000000055 : TEMPORARY Record 110231;
      Waiters@1000000056 : Integer;
      Opened@1000000057 : Integer;
      Updated@1000000058 : Integer;
      Closed@1000000059 : Integer;
      Restarted@1000000060 : Integer;
      OpenMs@1000000061 : Integer;
      UpdateMs@1000000062 : Integer;
      CloseMs@1000000063 : Integer;
      SweepMs@1000000064 : Integer;
      Granted@1000000065 : Integer;
      StillOpen@1000000066 : Integer;
      Backdated@1000000067 : Integer;
      Removed@1000000068 : Integer;
      DeletedByCeiling@1000000069 : Integer;
      HeldNo@1000000070 : Integer;
      OldestClosed@1000000071 : Integer;
      RowsBefore@1000000072 : Integer;
      NameA@1000000073 : Integer;
      NameB@1000000074 : Integer;
      NamePickedFirst@1000000075 : Integer;
      NamePickedAgain@1000000076 : Integer;
      NameBufferFirst@1000000077 : Integer;
      NameBufferAgain@1000000078 : Integer;
      NameAttemptsA@1000000079 : Integer;
      NameAttemptsB@1000000080 : Integer;
      NamePickedAfter@1000000081 : Integer;
      NamePickedCeiling@1000000082 : Integer;
      PickedAfter@1000000083 : Integer;
      Rounds@1000000084 : Integer;
      RegLongOutcome@1000000085 : Integer;
      RegShortOutcome@1000000086 : Integer;
      RegHugeOutcome@1000000087 : Integer;
      ThrStoredWait@1000000088 : Integer;
      Worst@1000000089 : Integer;
      SweepRows@1000000090 : Integer;
      NameAInBuffer@1000000091 : Boolean;
      NameBInBuffer@1000000092 : Boolean;
      ThrFound@1000000093 : Boolean;
      RegLongGap@1000000094 : Duration;
      RegHugeGap@1000000095 : Duration;
      ThrGap@1000000096 : Duration;
      ThrStarted@1000000097 : DateTime;
    BEGIN
      Setup.FetchOnce;
      IF Setup.Enabled THEN
        ERROR(EnabledErr);
      RefuseIfJournalUsed;

      Waiters := Watch.MaxWaitersPerPass;

      // Первым утверждением - что результат ВООБЩЕ не пуст. Число ждущих - константа, а не
      // результат, и пустой журнал сошёлся бы сам с собой: ноль открытых равен нулю
      // ожидаемых, ноль строк равно нулю. "Открыто ноль" от "открыто пятьсот" отличается
      // только этой проверкой.
      //
      // Счётчик прохода и число строк в журнале - РАЗНЫЕ величины, и спрашиваются обе.
      // Счётчик наращивает тот же код, что и вставляет: уйди вставка - он всё равно
      // скажет "открыто пятьсот", и проверка с одним счётчиком осталась бы зелёной над
      // пустым журналом. "Ровно по эпизоду на ждущего" - утверждение про строки.
      OpenMs := PassOfWaiters(Waiters,FirstWaitMs,Opened,Updated,Closed,Restarted);
      Check(Passed,Total,Report,ChkOpenTxt,
            (Opened > 0) AND (BenchRowCount > 0) AND (Opened = Waiters) AND (Updated = 0) AND
            (Closed = 0) AND (BenchRowCount = Waiters),
            STRSUBSTNO(OpenTxt,Opened,Waiters,Updated,Closed,BenchRowCount));

      // Половина "не плодит новые" держится на числе строк, и она здесь была. Половина
      // "обновляет" держалась на счётчике - а он растёт и при MODIFY, которого не было:
      // проход, потерявший запись, отчитался бы о пятистах обновлениях над нетронутыми
      // строками. Спрашивается поэтому само значение: второе ожидание больше первого, и
      // доехать оно обязано до КАЖДОЙ строки.
      UpdateMs := PassOfWaiters(Waiters,SecondWaitMs,Opened,Updated,Closed,Restarted);
      Check(Passed,Total,Report,ChkUpdateTxt,
            (Updated = Waiters) AND (Opened = 0) AND (Restarted = 0) AND
            (BenchRowCount = Waiters) AND (BenchRowsWithWait(SecondWaitMs) = Waiters),
            STRSUBSTNO(UpdateTxt,Updated,Opened,BenchRowCount,BenchRowsWithWait(SecondWaitMs)));

      CloseMs := PassOfWaiters(0,0,Opened,Updated,Closed,Restarted);
      CountOutcomes(Granted,StillOpen);
      Check(Passed,Total,Report,ChkCloseTxt,
            (Closed = Waiters) AND (Granted = Waiters) AND (StillOpen = 0),
            STRSUBSTNO(CloseTxt,Closed,Waiters,Granted,StillOpen));

      // Исход считается по настоящему промежутку между последним наблюдением и
      // закрывающим проходом, а не по периоду из настройки. Долгий промежуток при
      // коротком ожидании не должен читаться как "предоставлено", короткий промежуток
      // должен, а месяц и больше не должен переполнять разрядность.
      RegLongGap := Watch.NavLockTimeoutMs + RegGapSurplusMs;
      RegLongOutcome := RegEpisode(RegLongVictimSpid,SecondWaitMs,RegLongGap);
      Check(Passed,Total,Report,ChkCloseGapLongTxt,
            RegLongOutcome = Watch.OutcomeUnknown,
            STRSUBSTNO(CloseGapLongTxt,RegLongOutcome,RegLongGap));
      RegShortOutcome := RegEpisode(RegShortVictimSpid,SecondWaitMs,RegShortAgeMs);
      Check(Passed,Total,Report,ChkCloseGapShortTxt,
            RegShortOutcome = Watch.OutcomeGranted,
            STRSUBSTNO(CloseGapShortTxt,RegShortOutcome,RegShortAgeMs));
      RegHugeGap := RegHugeGapDays * Watch.MsPerDay;
      RegHugeOutcome := RegEpisode(RegHugeVictimSpid,SecondWaitMs,RegHugeGap);
      Check(Passed,Total,Report,ChkCloseGapHugeTxt,
            RegHugeOutcome = Watch.OutcomeUnknown,
            STRSUBSTNO(CloseGapHugeTxt,RegHugeOutcome,RegHugeGapDays));

      // Срок хранения удаляет эпизод НАВСЕГДА: журнал - единственное хранилище, и
      // состаривание идёт правкой поля, а не ожиданием месяца.
      Backdated := Backdate(BackdateRows,BackdateDays);
      Removed := Watch.MoveOldEntries(RetentionUnderTest,0,Waiters);
      Check(Passed,Total,Report,ChkTrimTxt,
            (Removed = Backdated) AND (Backdated = BackdateRows) AND
            (BenchRowCount = Waiters - BackdateRows),
            STRSUBSTNO(TrimTxt,Removed,Backdated,BenchRowCount,Waiters));

      // Второй предел - потолок СТРОК, и нужен он ровно там, где срок бессилен: в шторм
      // всё, что лежит в журнале, моложе срока, и одним сроком горячую таблицу не удержать.
      //
      // Срок здесь передаётся нулём нарочно: проверка обязана мерить потолок, а не то, что
      // за неё доделал возраст. Потолок ставится НИЖЕ текущего числа строк на известную
      // величину, и уйти обязано ровно это число - не больше и не меньше.
      //
      // Самая старая строка перед этим метится идущей. Она первый кандидат на удаление по
      // возрасту - и обязана остаться: стереть открытый эпизод значило бы разорвать
      // наблюдение, а следующий проход, не найдя строки, завёл бы эпизод заново.
      // Проверка без неё прошла бы и на удалении, которое этого не различает.
      HeldNo := HoldOldest;
      RowsBefore := JournalRowCount;
      OldestClosed := OldestClosedNo;
      DeletedByCeiling := Watch.MoveOldEntries(0,RowsBefore - CeilingSurplus,Waiters);
      Check(Passed,Total,Report,ChkCeilingTxt,
            (DeletedByCeiling = CeilingSurplus) AND
            (JournalRowCount = RowsBefore - CeilingSurplus) AND
            (NOT InJournal(OldestClosed)) AND InJournal(HeldNo),
            STRSUBSTNO(CeilingTxt,DeletedByCeiling,CeilingSurplus,JournalRowCount,
                       RowsBefore - CeilingSurplus,HeldNo));

      // Строк в уборке - столько, сколько их в журнале ПЕРЕД ней, и число это берётся
      // счётом, а не вычитанием сцен: сцены меняются, и вычитание тихо устаревает.
      SweepRows := BenchRowCount;
      SweepMs := Sweep;

      // Имена. Проверяется на СВОИХ двух эпизодах и на пустом журнале - тот самый миг
      // после уборки, когда в журнале нет ничего чужого и счёт выбранных строк можно
      // сравнивать с числом, а не с "примерно столько".
      //
      // Смысл всех четырёх: разбор идёт не один раз. Отметку виновник кладёт тогда, когда
      // трогает таблицу документа, а это бывает и позже начала ожидания; жертва
      // отмечается, только когда сама что-то пишет. Один вопрос на эпизод оставлял в
      // журнале пустую колонку там, где ответ появлялся секундой позже.
      NamePass(NameIndexA,NameIndexB);
      NameA := EntryOfIndex(NameIndexA);
      NameB := EntryOfIndex(NameIndexB);
      // Спрашивается не только ЧИСЛО выбранных, но и сам буфер. Счёт растёт отдельно от
      // вставки в него: уйди вставка - разбор всё равно ответит "двое", а разбирать
      // проходу будет нечего, и имена перестанут доезжать до журнала молча.
      NamePickedFirst := Watch.PickUnresolved(NamePending,Watch.MaxDocumentsPerPass);
      NameBufferFirst := NamePending.COUNT;
      Watch.SetDocument(NameA,'',Watch.DocSourceNone,NoAnswerReason);
      Watch.SetDocument(NameB,'',Watch.DocSourceNone,NoAnswerReason);
      NamePickedAgain := Watch.PickUnresolved(NamePending,Watch.MaxDocumentsPerPass);
      NameBufferAgain := NamePending.COUNT;
      Check(Passed,Total,Report,ChkNameAgainTxt,
            (NamePickedFirst = NamePickedExpected) AND
            (NamePickedAgain = NamePickedExpected) AND
            (NameBufferFirst = NamePickedExpected) AND
            (NameBufferAgain = NamePickedExpected) AND
            NamePending.GET(NameA) AND NamePending.GET(NameB),
            STRSUBSTNO(NameAgainTxt,NamePickedFirst,NamePickedAgain,
                       NameBufferFirst,NameBufferAgain));

      // Счёт попыток обязан пережить следующий съём: наблюдаемая строка приезжает без
      // него, и ноль здесь значил бы "не спрашивали ни разу" - то есть вечное
      // переспрашивание. Поздний документ обязан доехать до строки, а причина, с которой
      // эпизод жил до него, - очиститься: она была объяснением пустой клетки.
      NamePass(NameIndexA,NameIndexB);
      Episode.GET(NameA);
      NameAttemptsA := Episode."Name Attempts";
      Watch.SetDocument(NameA,NameDocument,Watch.DocSourceFromHash,'');
      Episode.GET(NameA);
      Check(Passed,Total,Report,ChkNameLateTxt,
            (NameAttemptsA = 1) AND (Episode."Document No." = NameDocument) AND
            (Episode."Document Source" = Episode."Document Source"::FromHash) AND
            (Episode."No Document Reason" = ''),
            STRSUBSTNO(NameLateTxt,NameAttemptsA,Episode."Document No.",
                       FORMAT(Episode."Document Source",0,0)));

      // Поздний пустой ответ ничего не стирает: документ назван, источник остаётся, а
      // причина не пишется - у названного эпизода она была бы не объяснением, а
      // противоречием соседней колонке.
      Watch.SetDocument(NameA,'',Watch.DocSourceNone,NoAnswerReason);
      Episode.GET(NameA);
      NameAttemptsA := Episode."Name Attempts";
      // Названный эпизод больше не спрашивают: имя виновника и имя жертвы у него уже есть.
      // Имена ставит проход, а не разбор документа, поэтому здесь они кладутся прямо.
      Episode."User Name" := NameCulprit;
      Episode."Victim User Name" := NameVictim;
      Episode.MODIFY;
      NamePickedAfter := Watch.PickUnresolved(NamePending,Watch.MaxDocumentsPerPass);
      NameBInBuffer := NamePending.GET(NameB);
      NameAInBuffer := NamePending.GET(NameA);
      Check(Passed,Total,Report,ChkNameHeldTxt,
            (Episode."Document No." = NameDocument) AND
            (Episode."Document Source" = Episode."Document Source"::FromHash) AND
            (Episode."No Document Reason" = '') AND (NameAttemptsA = 3) AND
            (NamePickedAfter = 1) AND NameBInBuffer AND (NOT NameAInBuffer),
            STRSUBSTNO(NameHeldTxt,Episode."Document No.",
                       FORMAT(Episode."Document Source",0,0),NameAttemptsA,
                       NamePickedAfter,NameBInBuffer));

      // Второй эпизод так и остался без ответа: на нём и проверяется потолок. Без него
      // вопрос повторялся бы столько же, сколько живёт эпизод, а живёт он и по часу.
      //
      // Кругов ровно столько, сколько нужно, чтобы упереться в потолок, и число их - СВОЁ.
      // Цикл по счётчику проверяемого кода не кончился бы никогда ровно в том случае, ради
      // которого он написан: сломай SetDocument счёт попыток - и прогон ушёл бы в вечный
      // цикл вместо красного отчёта. Кончается он и тогда, когда проход перестаёт видеть
      // эпизод, - а это и есть проверяемое место.
      Rounds := 0;
      REPEAT
        Rounds := Rounds + 1;
        Watch.SetDocument(NameB,'',Watch.DocSourceNone,NoAnswerReason);
        PickedAfter := PickNameNow(NameB);
      UNTIL (PickedAfter = 0) OR (Rounds >= NameAttemptRoundCeiling);
      Episode.GET(NameB);
      NameAttemptsB := Episode."Name Attempts";
      NamePickedCeiling := PickedAfter;
      Check(Passed,Total,Report,ChkNameCeilingTxt,
            (NameAttemptsB = Watch.MaxNameAttempts) AND (NamePickedCeiling = 0) AND
            (Rounds < NameAttemptRoundCeiling),
            STRSUBSTNO(NameCeilingTxt,NameAttemptsB,Watch.MaxNameAttempts,NamePickedCeiling));

      Sweep;

      // ---------- порог записи в журнал ----------
      // Четыре сцены, и каждая непустая по-своему: порог, пропускающий короткое; порог,
      // отвергающий равное себе; порог, мешающий обновлять уже открытое; и ноль, который
      // обязан открывать всё. Идут они ПОСЛЕ последней уборки: журнал пуст, и число строк
      // после сцены - это ровно то, что сцена записала.
      //
      // Строка, уже открытая, порогом не задета: обновление и закрытие идут своей дорогой.
      // Проверяется это растущим ожиданием (1200 -> 1500) - счётчик, пошедший назад, был бы
      // новым эпизодом, и сцена мерила бы не порог, а признак перезапуска.
      ThrStarted := CURRENTDATETIME;
      ThrGap := PeriodMs;

      OneWaitPass(ThresholdIndex,FirstWaitMs,MinWaitUnderTest,ThrStarted,Opened,Updated);
      Check(Passed,Total,Report,ChkMinWaitNewTxt,
            (Opened = 0) AND (BenchRowCount = 0),
            STRSUBSTNO(MinWaitNewTxt,MinWaitUnderTest,FirstWaitMs,Opened,BenchRowCount));
      Sweep;

      OneWaitPass(ThresholdIndex,MinWaitUnderTest,MinWaitUnderTest,ThrStarted,Opened,Updated);
      Check(Passed,Total,Report,ChkMinWaitBoundTxt,
            (Opened = 1) AND (BenchRowCount = 1) AND
            (BenchRowsWithWait(MinWaitUnderTest) = 1),
            STRSUBSTNO(MinWaitBoundTxt,MinWaitUnderTest,MinWaitUnderTest,Opened,BenchRowCount));
      Sweep;

      // Первый проход открывает строку без порога, второй приходит с порогом выше нового
      // ожидания. Обновление обязано случиться: иначе длинный эпизод, пойманный однажды,
      // пропадал бы на первом же коротком проходе.
      OneWaitPass(ThresholdIndex,FirstWaitMs,NoMinWait,ThrStarted,Opened,Updated);
      OneWaitPass(ThresholdIndex,UpdateWaitUnderTest,MinWaitUnderTest,
                  ThrStarted + ThrGap,Opened,Updated);
      Episode.RESET;
      Episode.SETRANGE("Victim SPID",-ThresholdIndex);
      ThrFound := Episode.FINDFIRST;
      ThrStoredWait := -1;
      IF ThrFound THEN
        ThrStoredWait := Episode."Max Wait (ms)";
      Check(Passed,Total,Report,ChkMinWaitUpdateTxt,
            (Updated = 1) AND (Opened = 0) AND ThrFound AND
            (ThrStoredWait = UpdateWaitUnderTest) AND (BenchRowCount = 1),
            STRSUBSTNO(MinWaitUpdateTxt,MinWaitUnderTest,UpdateWaitUnderTest,Updated,Opened,ThrStoredWait));
      Sweep;

      OneWaitPass(ThresholdIndex,TinyWaitUnderTest,NoMinWait,ThrStarted,Opened,Updated);
      Check(Passed,Total,Report,ChkMinWaitOffTxt,
            (Opened = 1) AND (BenchRowCount = 1) AND
            (BenchRowsWithWait(TinyWaitUnderTest) = 1),
            STRSUBSTNO(MinWaitOffTxt,TinyWaitUnderTest,Opened,BenchRowCount));
      Sweep;

      // ---------- смена виновника и усечённый проход ----------
      // Две беды, которые видно ТОЛЬКО на сцене: обе живут в ведении уже открытого
      // эпизода, и обе молчат на полном съёме с одним ждущим.
      CulpritScene(Passed,Total,Report);
      CutoffScene(Passed,Total,Report);

      Worst := OpenMs;
      IF UpdateMs > Worst THEN
        Worst := UpdateMs;
      IF CloseMs > Worst THEN
        Worst := CloseMs;
      // Спрашиваются ОБЕ границы: потолок и пол. Нижняя тут не украшение - замер,
      // показавший ноль, означает не мгновенный проход, а непрочитанные часы, и проверка
      // с одним потолком на ноль смотрела бы зелёным.
      Check(Passed,Total,Report,ChkCostTxt,
            (Worst <= CeilingMs) AND (Worst >= CostFloorMs),
            STRSUBSTNO(CostTxt,Worst,CeilingMs,CostFloorMs));

      Report := Report + CrLf +
        Timing(OpenPhaseTxt,OpenMs,Waiters) +
        Timing(UpdatePhaseTxt,UpdateMs,Waiters) +
        Timing(ClosePhaseTxt,CloseMs,Waiters) +
        Timing(SweepPhaseTxt,SweepMs,SweepRows);
    END;

    LOCAL PROCEDURE PassOfWaiters@3(Waiters@1000000098 : Integer;
                                    WaitMs@1000000099 : Integer;
                                    VAR Opened@1000000100 : Integer;
                                    VAR Updated@1000000101 : Integer;
                                    VAR Closed@1000000102 : Integer;
                                    VAR Restarted@1000000103 : Integer) : Integer;
    VAR
      Observed@1000000104 : Record 110231;
      Started@1000000105 : DateTime;
      Elapsed@1000000106 : Duration;
      Index@1000000107 : Integer;
      Milliseconds@1000000108 : Integer;
    BEGIN
      // Меряется ровно то, что делает фоновая задача после съёма: проход целиком, вместе
      // с COMMIT. Без COMMIT замер был бы вдвое красивее и вдвое бесполезнее - запись в
      // журнал транзакций и есть то, чем проход платит.
      Started := CURRENTDATETIME;
      // Порог записи здесь ноль: замер и его проверки держатся на том, что каждый ждущий
      // даёт строку. Сам порог проверяется отдельными сценами ниже.
      Watch.BeginPass(Started,NoMinWait);
      FOR Index := 1 TO Waiters DO BEGIN
        MakeObserved(Observed,Index,WaitMs);
        Watch.Observe(Observed);
      END;
      Watch.FinishPass;
      COMMIT;
      Elapsed := CURRENTDATETIME - Started;
      Milliseconds := Elapsed;
      Watch.PassCounts(Opened,Updated,Closed,Restarted);
      EXIT(Milliseconds);
    END;

    LOCAL PROCEDURE OneWaitPass@4(Index@1000000109 : Integer;
                                  WaitMs@1000000110 : Integer;
                                  MinWaitMs@1000000111 : Integer;
                                  ObservedAt@1000000112 : DateTime;
                                  VAR Opened@1000000113 : Integer;
                                  VAR Updated@1000000114 : Integer);
    VAR
      Observed@1000000115 : Record 110231;
      Closed@1000000116 : Integer;
      Restarted@1000000117 : Integer;
    BEGIN
      // Один ждущий, один проход: наименьшее, из чего виден порог записи. Счётчики
      // возвращаются наружу целиком, чтобы проверка спрашивала и открытие, и обновление,
      // а не одно из двух.
      Watch.BeginPass(ObservedAt,MinWaitMs);
      MakeObserved(Observed,Index,WaitMs);
      Watch.Observe(Observed);
      Watch.FinishPass;
      Watch.PassCounts(Opened,Updated,Closed,Restarted);
    END;

    LOCAL PROCEDURE RegEpisode@5(VictimSpid@1000000118 : Integer;
                                 WaitMs@1000000119 : Integer;
                                 AgeMs@1000000120 : Duration) : Integer;
    VAR
      Observed@1000000121 : Record 110231;
      Found@1000000122 : Record 110231;
      FoundOutcome@1000000123 : Integer;
    BEGIN
      // Регрессионный эпизод: открыть, отодвинуть последнее наблюдение на настоящий
      // промежуток, закрыть проходом и вернуть исход. Строка удаляется, чтобы остальные
      // проверки видели только своих ожидающих.
      FoundOutcome := -1;
      Watch.BeginPass(CURRENTDATETIME,NoMinWait);
      MakeObserved(Observed,VictimSpid,WaitMs);
      Watch.Observe(Observed);
      Watch.FinishPass;
      COMMIT;
      Found.RESET;
      Found.SETRANGE("Victim SPID",-VictimSpid);
      IF Found.FINDFIRST THEN BEGIN
        Found."Last Seen At" := CURRENTDATETIME - AgeMs;
        Found.MODIFY;
      END;
      COMMIT;
      Watch.BeginPass(CURRENTDATETIME,NoMinWait);
      Watch.FinishPass;
      COMMIT;
      Found.RESET;
      Found.SETRANGE("Victim SPID",-VictimSpid);
      IF Found.FINDFIRST THEN BEGIN
        IF Found.Outcome = Found.Outcome::Granted THEN
          FoundOutcome := Watch.OutcomeGranted
        ELSE IF Found.Outcome = Found.Outcome::TimedOut THEN
          FoundOutcome := Watch.OutcomeTimedOut
        ELSE IF Found.Outcome = Found.Outcome::Deadlocked THEN
          FoundOutcome := Watch.OutcomeDeadlocked
        ELSE
          FoundOutcome := Watch.OutcomeUnknown;
        Found.DELETE;
      END;
      COMMIT;
      EXIT(FoundOutcome);
    END;

    LOCAL PROCEDURE MakeObserved@6(VAR Observed@1000000124 : Record 110231;
                                   Index@1000000125 : Integer;
                                   WaitMs@1000000126 : Integer);
    BEGIN
      // Строка сделана похожей на настоящую не для красоты: цена записи зависит от того,
      // сколько байт уезжает в SQL, а строковые поля здесь - самые широкие в таблице.
      // Замер на пустой строке был бы вдвое бодрее того, что случится на бою.
      Observed.INIT;
      Observed."Victim SPID" := -Index;
      Observed."Victim Transaction Id" := ObservedTransactionBase + Index;
      Observed."Victim Is NAV" := Observed."Victim Is NAV"::Yes;
      Observed."Blocker SPID" := -Index - 1;
      Observed."Blocker Transaction Id" := ObservedBlockerTransactionBase + Index;
      Observed."Resource Kind" := ObservedResourceKind;
      Observed."Hobt Id" := SampleHobt + Index;
      Observed."Database Id" := ObservedDatabaseId;
      Observed."SQL Object Name" := ObservedSqlObjectName;
      Observed."NAV Table Name" := ObservedTableName;
      Observed."NAV Key No." := ObservedKeyNo;
      Observed."Last Wait (ms)" := WaitMs;
      Observed."Blocker Statement" := ObservedUpdateStatement;
    END;

    LOCAL PROCEDURE Backdate@7(Rows@1000000127 : Integer;Days@1000000128 : Integer) : Integer;
    VAR
      Back@1000000129 : Duration;
      Done@1000000130 : Integer;
    BEGIN
      // Состаривание идёт правкой поля, а не ожиданием месяца.
      Back := Days * Watch.MsPerDay;
      Episode.RESET;
      Episode.SETFILTER("Victim SPID",'<%1',0);
      IF Episode.FINDSET(TRUE) THEN
        REPEAT
          Episode."Started At" := CURRENTDATETIME - Back;
          Episode.MODIFY;
          Done := Done + 1;
        UNTIL (Done >= Rows) OR (Episode.NEXT = 0);
      COMMIT;
      EXIT(Done);
    END;

    LOCAL PROCEDURE Sweep@8() : Integer;
    VAR
      Started@1000000131 : DateTime;
      Elapsed@1000000132 : Duration;
      Milliseconds@1000000133 : Integer;
    BEGIN
      // Уборка за собой обязательна и делается по СВОЕЙ метке: у строк мерного прогона
      // номер сессии отрицательный, а настоящий spid отрицательным не бывает. Строка,
      // оставшаяся в журнале после замера, - это ложь в отчёте заказчику.
      Started := CURRENTDATETIME;
      Episode.RESET;
      Episode.SETFILTER("Victim SPID",'<%1',0);
      Episode.DELETEALL;
      COMMIT;
      Elapsed := CURRENTDATETIME - Started;
      Milliseconds := Elapsed;
      EXIT(Milliseconds);
    END;

    LOCAL PROCEDURE RefuseIfJournalUsed@9();

      Sweep;
    END;

    LOCAL PROCEDURE HoldOldest@13() : Integer;
    VAR
      Held@1000000136 : Record 110231;
    BEGIN
      // Самая старая строка метится ИДУЩЕЙ. Открытый эпизод ведёт текущий проход, и стереть
      // его значило бы разорвать наблюдение, а следующий проход, не найдя её в журнале,

      Held.Open := TRUE;
      Held.MODIFY;
      COMMIT;
      EXIT(Held."Entry No.");
    END;

    LOCAL PROCEDURE Check@17(VAR Passed@1000000142 : Integer;
                             VAR Total@1000000143 : Integer;
                             VAR Report@1000000144 : Text;
                             What@1000000145 : Text;
                             Ok@1000000146 : Boolean;
                             Detail@1000000147 : Text);

    LOCAL PROCEDURE Timing@18(Phase@1000000149 : Text;Milliseconds@1000000150 : Integer;Rows@1000000151 : Integer) : Text;
    VAR
      PerRow@1000000152 : Integer;
      Exact@1000000153 : Decimal;
    BEGIN
      // Деление ДРОБНОЕ, а печатается ЦЕЛОЕ - и то, и другое нарочно. Целочисленное деление
      // отбрасывало бы доли ДО округления, и ROUND рядом стоял бы украшением: отчёт обещал
      // бы десятые доли, которых в нём не считалось. А показывать десятые доли
      // микросекунды незачем: таких различий не видно ни на одной машине.
      IF Rows > 0 THEN BEGIN
        Exact := Milliseconds * MicrosPerMs;
        Exact := Exact / Rows;
        PerRow := ROUND(Exact,PerRowPrecision);
      END;
      EXIT(STRSUBSTNO(TimingTxt,Phase,Milliseconds,Rows,PerRow) + CrLf);
    END;

    LOCAL PROCEDURE NamePass@19(FirstIndex@1000000154 : Integer;SecondIndex@1000000155 : Integer);
    VAR
      Observed@1000000156 : Record 110231;
    BEGIN
      // Оба ждущих обязаны приходить КАЖДЫМ проходом: FinishPass закрывает эпизод,
      // которого в проходе не было, а закрытый эпизод не разбирается вовсе.
      Watch.BeginPass(CURRENTDATETIME,NoMinWait);
      MakeObserved(Observed,FirstIndex,FirstWaitMs);
      Watch.Observe(Observed);
      MakeObserved(Observed,SecondIndex,FirstWaitMs);
      Watch.Observe(Observed);
      Watch.FinishPass;
    END;

    LOCAL PROCEDURE PickNameNow@21(EntryNo@1000000158 : Integer) : Integer;
    VAR
      Buffer@1000000159 : TEMPORARY Record 110231;
    BEGIN
      // Спрашивается не только счёт выбранных, но и сам эпизод: проход мог ответить "взял"
      // и не положить в буфер ничего. Выбор идёт тем же вызовом, что и в проходе.
      Watch.PickUnresolved(Buffer,Watch.MaxDocumentsPerPass);
      IF Buffer.GET(EntryNo) THEN
        EXIT(1);
      EXIT(0);
    END;

    LOCAL PROCEDURE SqlNamePart@22(Name@1000000160 : Text) : Text;
    VAR
      Part@1000000161 : Text;
      Index@1000000162 : Integer;
      Ch@1000000163 : Text[1];
    BEGIN
      // Имя объекта NAV платформа переводит в имя SQL заменой точки, косой черты,
      // обратной косой черты и кавычки подчёркиванием (FINDINGS, раздел 5). Вход опыта
      // обязан собираться тем же переводом, каким его собрала бы платформа.
      FOR Index := 1 TO STRLEN(Name) DO BEGIN
        Ch := COPYSTR(Name,Index,1);
        IF (Ch = '.') OR (Ch = '/') OR (Ch = '\') OR (Ch = '"') THEN
          Part := Part + '_'
        ELSE
          Part := Part + Ch;
      END;
      EXIT(Part);
    END;

    LOCAL PROCEDURE ObservedTableName@23() : Text;
    VAR
      RecRef@1000000164 : RecordRef;
    BEGIN
      // Настоящее имя таблицы, а не догадка: имя для входа опыта и имя для разбора -
      // одно и то же, и переименование таблицы не оставит опыт врать о ней.
      RecRef.OPEN(DATABASE::"Sales Line");
      EXIT(RecRef.NAME);
    END;

    LOCAL PROCEDURE ObservedSqlObjectName@24() : Text;
    BEGIN

      // литерала: мерный прогон обязан идти в любой компании стенда.
      EXIT(SqlNamePart(COMPANYNAME) + SqlObjectSeparator + SqlNamePart(ObservedTableName));
    END;

    LOCAL PROCEDURE ObservedUpdateStatement@25() : Text;

    LOCAL PROCEDURE OpenEpisode@70(Index@1000000181 : Integer;WaitMs@1000000182 : Integer;

    LOCAL PROCEDURE CulpritScene@71(VAR Passed@1000000187 : Integer;
                                    VAR Total@1000000188 : Integer;
                                    VAR Report@1000000189 : Text);
    VAR
      Snap@1000000190 : TEMPORARY Record 110231;
      Row@1000000191 : Record 110231;
      EntryNo@1000000192 : Integer;
      At@1000000193 : DateTime;
      Gap@1000000194 : Duration;
      Act@1000000195 : DateTime;
      TextSeen@1000000196 : Text;
    BEGIN
      // Сцена смены виновника. Эпизод ОДИН на все три прохода: сессия, транзакция и ресурс
      // жертвы не меняются, ожидание только растёт - по ключу это тот же эпизод.
      Gap := CulpritGapMs;
      At := CURRENTDATETIME;
      EntryNo := OpenEpisode(CulpritIndex,CulpritFirstWaitMs,CulpritTextOld,At);
      Row.GET(EntryNo);
      Row."User Name" := NameCulprit;
      Row."User Source" := Row."User Source"::FromPlatform;
      Row.MODIFY;
      COMMIT;

      // Второй проход: виновник ТОТ ЖЕ, а текст в съёме другой. Имя и текст обязаны устоять:
      // оба ставятся один раз, и до смены виновника переписывать их нечем.
      Act := At + Gap;
      Watch.BeginPass(Act,NoMinWait);
      MakeObserved(Snap,CulpritIndex,CulpritSecondWaitMs);
      SetBlockerText(Snap,CulpritTextOther);
      Watch.Observe(Snap);
      Watch.FinishPass;
      COMMIT;
      Row.GET(EntryNo);
      TextSeen := Watch.StatementText(EntryNo,TRUE);
      Check(Passed,Total,Report,ChkCulpritKeepTxt,
            (Row."User Name" = NameCulprit) AND
            (Row."User Source" = Row."User Source"::FromPlatform) AND
            (TextSeen = CulpritTextOld),
            STRSUBSTNO(CulpritKeepTxt,Row."User Name",FORMAT(Row."User Source",0,0),TextSeen,
                       NameCulprit,CulpritTextOld));

      // Третий проход: виновник СМЕНИЛСЯ - другой сеанс и другая транзакция. Имя прежнего
      // обязано исчезнуть: иначе строка назвала бы новый сеанс именем старого. Текст берётся
      // из этого же съёма, а не остаётся прежним.
      Gap := CulpritThirdGapMs;
      Act := At + Gap;
      Watch.BeginPass(Act,NoMinWait);
      MakeObserved(Snap,CulpritIndex,CulpritThirdWaitMs);
      Snap."Blocker SPID" := -CulpritIndex - 1 - CulpritBlockerShift;
      Snap."Blocker Transaction Id" := ObservedBlockerTransactionBase + CulpritIndex +
                                       CulpritTransactionShift;
      SetBlockerText(Snap,CulpritTextNew);
      // Строка съёма вставляется во временную таблицу, как это делает съём очереди: перенос
      // текста читает блоб через CALCFIELDS, а у невставленной временной строки он пуст.
      Snap.INSERT;
      Watch.Observe(Snap);
      Watch.FinishPass;
      COMMIT;
      Row.GET(EntryNo);
      TextSeen := Watch.StatementText(EntryNo,TRUE);
      Check(Passed,Total,Report,ChkCulpritTxt,
            (Row."User Name" = '') AND
            (Row."User Source" = Row."User Source"::None) AND
            (TextSeen = CulpritTextNew),
            STRSUBSTNO(CulpritTxt,Row."User Name",FORMAT(Row."User Source",0,0),TextSeen,
                       CulpritTextNew));
      Sweep;
    END;

    LOCAL PROCEDURE CutoffScene@72(VAR Passed@1000000198 : Integer;
                                   VAR Total@1000000199 : Integer;
                                   VAR Report@1000000200 : Text);
    VAR
      Snap@1000000201 : TEMPORARY Record 110231;
      Row@1000000202 : Record 110231;
      At@1000000203 : DateTime;
      Gap@1000000204 : Duration;
      Act@1000000205 : DateTime;
      ShortNo@1000000206 : Integer;
      LongNo@1000000207 : Integer;
      CtrlShortNo@1000000208 : Integer;
      CtrlLongNo@1000000209 : Integer;
      ShortOpen@1000000210 : Integer;
      LongOpen@1000000211 : Integer;
      CtrlOpen@1000000212 : Integer;
    BEGIN
      // Усечённый проход. Оба эпизода открыты одним съёмом и в следующем не встречаются:
      // съём упёрся в потолок, и за ним мог остаться кто угодно. Порог 5000 мс, промежуток
      // 1000 мс: короткий ждал 1000 и набрал бы 2000 - он мог не попасть в съём, закрывать
      // его нельзя. Долгий ждал 4500 и набрал бы 5500 - он попал бы в съём, значит его
      // отсутствие и есть уход из очереди.
      Gap := CutoffGapMs;
      At := CURRENTDATETIME;
      Watch.BeginPass(At,NoMinWait);
      MakeObserved(Snap,CutoffShortIndex,CutoffShortWaitMs);
      Watch.Observe(Snap);
      MakeObserved(Snap,CutoffLongIndex,CutoffLongWaitMs);
      Watch.Observe(Snap);
      Watch.FinishPass;
      COMMIT;
      ShortNo := EntryOfIndex(CutoffShortIndex);
      LongNo := EntryOfIndex(CutoffLongIndex);

      Act := At + Gap;
      Watch.BeginPass(Act,NoMinWait);
      Watch.MarkIncomplete(CutoffMs);
      Watch.FinishPass;
      COMMIT;
      Row.GET(ShortNo);
      ShortOpen := 0;
      IF Row.Open THEN
        ShortOpen := 1;
      Row.GET(LongNo);
      LongOpen := 0;
      IF Row.Open THEN
        LongOpen := 1;
      Check(Passed,Total,Report,ChkCutoffShortTxt,ShortOpen = 1,
            STRSUBSTNO(CutoffShortTxt,CutoffMs,CutoffGapMs,CutoffShortWaitMs,ShortOpen));
      Check(Passed,Total,Report,ChkCutoffLongTxt,LongOpen = 0,
            STRSUBSTNO(CutoffLongTxt,CutoffMs,CutoffGapMs,CutoffLongWaitMs,LongOpen));

      // Контроль: тот же уход из очереди, но съём полный. Усечения нет - закрыты оба.
      // Без него обе проверки выше прошли бы и на коде, который не закрывает никогда.
      At := Act + Gap;
      Watch.BeginPass(At,NoMinWait);
      MakeObserved(Snap,CutoffCtrlShortIndex,CutoffShortWaitMs);
      Watch.Observe(Snap);
      MakeObserved(Snap,CutoffCtrlLongIndex,CutoffLongWaitMs);
      Watch.Observe(Snap);
      Watch.FinishPass;
      COMMIT;
      CtrlShortNo := EntryOfIndex(CutoffCtrlShortIndex);
      CtrlLongNo := EntryOfIndex(CutoffCtrlLongIndex);
      Act := At + Gap;
      Watch.BeginPass(Act,NoMinWait);
      Watch.FinishPass;
      COMMIT;
      CtrlOpen := 0;
      Row.GET(CtrlShortNo);
      IF Row.Open THEN
        CtrlOpen := CtrlOpen + 1;
      Row.GET(CtrlLongNo);
      IF Row.Open THEN
        CtrlOpen := CtrlOpen + 1;
      Check(Passed,Total,Report,ChkCutoffFullTxt,CtrlOpen = 0,
            STRSUBSTNO(CutoffFullTxt,CtrlOpen,CutoffPairSize));
      Sweep;
    END;
