    LOCAL PROCEDURE ShowEpisode@1();
    VAR
      Episode@1000000000 : Record 110231;
    BEGIN
      IF NOT Episode.GET("Episode Entry No.") THEN
        EXIT;
      Episode.SETRECFILTER;
      PAGE.RUN(PAGE::"LockWatch Episodes",Episode);
    END;

    LOCAL PROCEDURE ShowHead@2();
    VAR
      Episode@1000000000 : Record 110231;
      Spid@1000000001 : Integer;
    BEGIN
      // Отбор по голове, а не по одному эпизоду: за тревогой стоит столько эпизодов,
      // сколько жертв держала эта голова, и увидеть их надо всех.
      Spid := "Head SPID";
      Episode.SETCURRENTKEY("Head SPID",Open);
      Episode.SETRANGE("Head SPID",Spid);
      PAGE.RUN(PAGE::"LockWatch Episodes",Episode);
    END;

    BEGIN
    {
      Тревоги. Строка на голову цепочки за окно, а не на эпизод, и это главное, что нужно
      знать при чтении: число в колонке "эпизодов за защёлкой" и есть мера того, скольких
      задело, а строк при этом одна.

      Колонка "что сказал канал" стоит рядом с колонкой "куда ушла" нарочно. Канал наружу
      может не сработать - источника событий на машине нет, прав не выдали, - и тогда
      тревога всё равно здесь, а причина молчания записана словами. Пустая колонка на её
      месте читалась бы как "всё ушло куда надо".
    }
    END.
  }
