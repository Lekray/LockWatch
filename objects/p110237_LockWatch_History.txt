    LOCAL PROCEDURE ShowStatement@1(Blocker@1000000000 : Boolean);
    VAR
      Viewer@1000000001 : Page 110236;
      Body@1000000002 : Text;
      Whose@1000000003 : Text;
    BEGIN
      Body := Journal.HistoryStatementText("Entry No.",Blocker);
      // Пустой текст объясняется словами, а не пустым окном. Причин у пустоты две, и обе
      // не ошибка: признак выключен, либо эпизод открылся раньше, чем его включили.
      IF Body = '' THEN
        ERROR(NoStatementErr);
      IF Blocker THEN
        Whose := STRSUBSTNO(BlockerTxt,"Blocker SPID")
      ELSE
        Whose := STRSUBSTNO(VictimTxt,"Victim SPID");
      Viewer.SetStatement(Whose,Body);
      Viewer.RUNMODAL;
    END;

    LOCAL PROCEDURE PurgeShown@2();
    VAR
      Doomed@1000000000 : Record 110238;
      Rows@1000000001 : Integer;
    BEGIN
      // Единственное место во всём инструменте, где история удаляется, и нажать его должен
      // человек. Отбор берётся С ЭКРАНА: что человек видит, то и уйдёт. Кнопка "удалить
      // старше N дней" спрашивала бы число вслепую, и промах в ней виден только потом.
      //
      // Число называется ДО вопроса, а не после: "удалить показанное" без счёта - это
      // предложение согласиться неизвестно на что.
      Doomed.COPYFILTERS(Rec);
      Rows := Doomed.COUNT;
      IF Rows = 0 THEN BEGIN
        MESSAGE(NothingShownTxt);
        EXIT;
      END;
      IF NOT CONFIRM(ConfirmPurgeTxt,FALSE,Rows) THEN
        EXIT;
      Doomed.DELETEALL;
      CurrPage.UPDATE(FALSE);
      MESSAGE(PurgedTxt,Rows);
    END;

    BEGIN
    {
      История отдельной страницей, а не отбором на журнале, потому что смотрят их по
      разным поводам. Журнал открывают с вопросом "что происходит и что было только что";
      историю - с вопросом "было ли это раньше и повторяется ли". Колонки здесь под второй
      вопрос: когда, сколько, за что, кто - и ни одной про "идёт ли сейчас", потому что в
      истории не идёт ничего.

      Удаление тут одно и ручное. Всё остальное в инструменте историю только пополняет.
    }
    END.
