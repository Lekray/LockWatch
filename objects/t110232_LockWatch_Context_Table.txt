    LOCAL PROCEDURE SetNameFromTable@1();
    VAR
      AllObjWithCaption@1000000000 : Record 2000000058;
    BEGIN
      // Имя подставляем сами: строка настройки, в которой стоит один номер, читается
      // как "неизвестно что" и проверить её глазами нельзя.
      "Table Name" := '';
      IF "Table No." = 0 THEN
        EXIT;
      IF AllObjWithCaption.GET(AllObjWithCaption."Object Type"::Table,"Table No.") THEN
        "Table Name" := COPYSTR(AllObjWithCaption."Object Caption",1,MAXSTRLEN("Table Name"));
    END;

    BEGIN
    {
      Список "таблица плюс поле, несущее номер документа". Инструмент не знает, что за
      документ у заказчика; он знает, что у таблицы T поле F - это его номер. Поэтому
      номера доработанных объектов заказчика живут ЗДЕСЬ, в данных, а не в коде: код
      уезжает в публичный репозиторий, данные - нет.

      Поле "Переходник заведён" заполняет не человек, а сторож. Номер таблицы в подписке
      на событие - константа времени компиляции (см. FINDINGS, раздел 10), поэтому на
      каждую таблицу контекста нужен свой маленький объект-переходник. Строка настройки
      без переходника отвечает за РАЗБОР, но съёма не даёт, и колонка документа окажется
      пустой. Сторож обязан на это ругаться: пустая колонка без объяснения - худшее из
      возможного, потому что выглядит как "блокировок по документам не было".
    }
    END.
  }
