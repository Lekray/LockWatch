    PROCEDURE FillNames@3();
    BEGIN
      // Имена подставляет проверка поля, а она срабатывает только когда номер вводят

      SetNameFromTable;
    END;

    LOCAL PROCEDURE SetNameFromTable@1();
    VAR
      AllObjWithCaption@1000000000 : Record 2000000058;
    BEGIN
      // Имя подставляем сами: строка настройки, в которой стоит один номер, читается
      // как "неизвестно что" и проверить её глазами нельзя.
      "Table Name" := '';
      IF "Table No." = 0 THEN
        EXIT;
      // Берётся ИМЯ объекта, а не подпись. Подпись переведена на язык пользователя, а
      // сверять её придётся с именем таблицы в SQL - там лежит имя, и только оно.
      // 2000000058 - штатная "AllObjWithCaption": таблица ПЛАТФОРМЫ, по роду и номеру
      // объекта отдающая его имя. Стоит здесь номером, как и всякий чужой объект: имени
      // этой таблицы в наших именах нет и быть не может.
      IF AllObjWithCaption.GET(AllObjWithCaption."Object Type"::Table,"Table No.") THEN
        "Table Name" := COPYSTR(AllObjWithCaption."Object Name",1,MAXSTRLEN("Table Name"));
      SetFieldName;
    END;

    LOCAL PROCEDURE SetFieldName@2();
    VAR
      Field@1000000001 : Record 2000000041;
    BEGIN
      // Имя поля нужно съёму: по нему строится образец имени столбца SQL. Номер поля для
      // этого не годится вовсе - в SQL номеров полей NAV нет.
      // 2000000041 - штатная "Field": та же дорога от номера к имени, что и у соседней
      // процедуры, только для полей таблицы, и так же таблица платформы.
      "Document Field Name" := '';
      IF ("Table No." = 0) OR ("Document Field No." = 0) THEN
        EXIT;
      IF Field.GET("Table No.","Document Field No.") THEN
        "Document Field Name" := COPYSTR(Field.FieldName,1,MAXSTRLEN("Document Field Name"));
    END;

    BEGIN
    {
      Список "таблица плюс поле, несущее номер документа". Инструмент не знает, что за
      документ у заказчика; он знает, что у таблицы T поле F - это его номер. Поэтому
      номера доработанных объектов заказчика живут ЗДЕСЬ, в данных, а не в коде: код
      уезжает в публичный репозиторий, данные - нет.

      Строка настройки говорит разбору, где искать номер документа: спорную строку SQL
      называет физическим адресом, а не номером, и без этой строки документ остаётся
      ненайденным. Колонка документа в журнале тогда пуста, и рядом стоит поле "Почему
      документа нет": пустая колонка без объяснения читалась бы как "блокировок по
      документам не было".
    }
    END.
