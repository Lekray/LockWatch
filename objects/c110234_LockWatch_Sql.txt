    LOCAL PROCEDURE TryReadQueue@4(MaxRows@1000000000 : Integer;WantStatements@1000000019 : Boolean;VAR Observed@1000000001 : TEMPORARY Record 110231;VAR Waiter@1000000002 : ARRAY [500] OF Integer;VAR Blocker@1000000003 : ARRAY [500] OF Integer;VAR BlockerIsNav@1000000004 : ARRAY [500] OF Boolean;VAR RowCount@1000000005 : Integer;VAR ObservedAt@1000000006 : DateTime;VAR OtherDatabase@1000000007 : Integer);
    VAR
      SqlConn@1000000008 : DotNet "'System.Data, Version=2.0.0.0, Culture=neutral, PublicKeyToken=b77a5c561934e089'.System.Data.SqlClient.SqlConnection";
      SqlCmd@1000000009 : DotNet "'System.Data, Version=2.0.0.0, Culture=neutral, PublicKeyToken=b77a5c561934e089'.System.Data.SqlClient.SqlCommand";
      SqlReader@1000000010 : DotNet "'System.Data, Version=2.0.0.0, Culture=neutral, PublicKeyToken=b77a5c561934e089'.System.Data.SqlClient.SqlDataReader";
      OurDatabaseId@1000000011 : Integer;
      Kind@1000000012 : Text;
      HobtId@1000000013 : BigInteger;
      ObjectId@1000000014 : Integer;
      DatabaseId@1000000015 : Integer;
      HeldMode@1000000016 : Text;
      ResourceText@1000000017 : Text;
      Parsed@1000000018 : Boolean;
      OutStr@1000000020 : OutStream;
      VictimBody@1000000021 : Text;
      BlockerBody@1000000022 : Text;
    BEGIN
      SqlConn := SqlConn.SqlConnection(ConnString);
      SqlConn.Open;
      SqlCmd := SqlConn.CreateCommand;
      SqlCmd.CommandTimeout := ReadTimeoutSec;
      SqlCmd.CommandText := QueueSql(MaxRows,WantStatements);
      SqlReader := SqlCmd.ExecuteReader;

      // Первый набор - часы и номер базы. Он приходит ДАЖЕ когда ждущих нет: без него
      // проход, заставший тишину, не знал бы, каким моментом закрывать вчерашние эпизоды.
      IF NOT SqlReader.Read THEN
        ERROR(NoClockErr);
      ObservedAt := SqlMoment(SqlReader.GetInt32(0),SqlReader.GetInt32(1));
      OurDatabaseId := SqlReader.GetInt32(2);
      SqlReader.NextResult;

      WHILE SqlReader.Read DO BEGIN
        ResourceText := ReaderText(SqlReader,4);
        Parsed := Parse.ParseWaitResource(ResourceText,Kind,HobtId,ObjectId,DatabaseId,HeldMode);
        // Чужая база на том же сервере - не наша беда и не наш журнал. Считается
        // отдельно: пустая колонка без объяснения читается как "блокировок не было".
        IF Parsed AND (DatabaseId <> OurDatabaseId) AND (OurDatabaseId <> 0) THEN
          OtherDatabase := OtherDatabase + 1
        ELSE BEGIN
          RowCount := RowCount + 1;
          Waiter[RowCount] := SqlReader.GetInt32(0);
          Blocker[RowCount] := ReaderInt(SqlReader,1);
          BlockerIsNav[RowCount] := IsNavProgram(ReaderText(SqlReader,11));

          Observed.INIT;
          Observed."Entry No." := RowCount;
          Observed."Victim SPID" := Waiter[RowCount];
          Observed."Victim Transaction Id" := ReaderBigInt(SqlReader,8);
          Observed."Victim Is NAV" := NavFlag(ReaderText(SqlReader,7));
          Observed."Victim Host" := COPYSTR(ReaderText(SqlReader,5),1,MAXSTRLEN(Observed."Victim Host"));
          Observed."Victim Process Id" := ReaderInt(SqlReader,6);
          Observed."Blocker SPID" := Blocker[RowCount];
          Observed."Blocker Transaction Id" := ReaderBigInt(SqlReader,9);
          // Кто держит - тем, что о нём знает сам сервер. Соединение с sys.dm_exec_sessions
          // для виновника уже сделано, и программа его уже читается колонкой 11 (по ней
          // решается, наш ли это сеанс); логин и узел из ТОЙ ЖЕ строки не стоят ничего.
          // Для чужого соединения - утилиты, задания, чьего-то SSMS - это единственный
          // возможный ответ на "кто": отметку контекста кладёт только сессия NAV. Для самой
          // сессии NAV логин здесь общий, учётной записи службы, и человеком он не
          // притворяется - на то и отдельные поля, а не колонка "Пользователь".
          Observed."Blocker Login" :=
            COPYSTR(ReaderText(SqlReader,21),1,MAXSTRLEN(Observed."Blocker Login"));
          Observed."Blocker Host" :=
            COPYSTR(ReaderText(SqlReader,22),1,MAXSTRLEN(Observed."Blocker Host"));
          Observed."Blocker Program" :=
            COPYSTR(ReaderText(SqlReader,11),1,MAXSTRLEN(Observed."Blocker Program"));
          // Жертву сервер знает ровно так же, и соединение с её сеансом в запросе стоит
          // ПЕРВЫМ - им берутся узел, процесс и программа. Логин из той же строки не стоит
          // ничего, а без него жертва названа хуже виновника: у виновника три ответа, у
          // жертвы был один узел. Программа уже читается колонкой 7 (по ней решается, наш
          // ли это сеанс), и выбрасывать её после одной проверки было бы расточительством.
          Observed."Victim Login" :=
            COPYSTR(ReaderText(SqlReader,23),1,MAXSTRLEN(Observed."Victim Login"));
          Observed."Victim Program" :=
            COPYSTR(ReaderText(SqlReader,7),1,MAXSTRLEN(Observed."Victim Program"));
          // Колонки 15 и 16, а не 14 и 15. На единицу раньше лежит признак "у виновника
          // есть исполняемый запрос", и он уходил в ДНИ: журнал показывал начало
          // транзакции 1900-01-02, а класс эпизода считался по миллисекундам вместо
          // признака. Ни одна проверка этого не ловила - обе величины целые, обе
          // правдоподобные, и отказа не было ни разу.
          Observed."Blocker Tran Began At" := SqlMoment(ReaderInt(SqlReader,15),ReaderInt(SqlReader,16));
          Observed."Wait Type" := COPYSTR(ReaderText(SqlReader,3),1,MAXSTRLEN(Observed."Wait Type"));
          Observed."Last Wait (ms)" := SqlReader.GetInt32(2);
          Observed."Resource Kind" := COPYSTR(UPPERCASE(Kind),1,MAXSTRLEN(Observed."Resource Kind"));
          Observed."Held Mode" := COPYSTR(HeldMode,1,MAXSTRLEN(Observed."Held Mode"));
          Observed."Hobt Id" := HobtId;
          Observed."Database Id" := DatabaseId;
          Observed."SQL Object Id" := ObjectId;
          Observed."NAV Key No." := Parse.UnknownKeyNo;
          Observed."Resource Unresolved" := NOT Parsed;
          IF NOT Parsed THEN
            // Ресурс, который мы не поняли, показывается СЫРЫМ. Выбросить строку значило
            // бы соврать тишиной, а показать пустую - соврать дважды.
            Observed."SQL Object Name" := COPYSTR(ResourceText,1,MAXSTRLEN(Observed."SQL Object Name"));
          Observed.Class := EpisodeClassOf(SqlReader);
          VictimBody := '';
          BlockerBody := '';
          IF WantStatements THEN BEGIN
            // Текст кладётся во ВРЕМЕННУЮ строку, и она же довезёт его до журнала. Отдельным
            // запросом за ним ходить нельзя: к моменту второго запроса виновник успевает
            // послать другой оператор, и в журнал лёг бы не тот, который взял блокировку.
            VictimBody := Parse.StatementWithMark(ReaderText(SqlReader,17),ReaderInt(SqlReader,18));
            BlockerBody := Parse.StatementWithMark(ReaderText(SqlReader,19),ReaderInt(SqlReader,20));
            IF BlockerBody <> '' THEN
              // Начало запроса ложится и в короткое поле: в списке эпизодов колонка должна
              // что-то показывать, не заставляя открывать каждую строку по очереди.
              Observed."Blocker Statement" :=
                COPYSTR(Parse.Flatten(BlockerBody),1,MAXSTRLEN(Observed."Blocker Statement"));
          END;
          // Поток в поле-БЛОБ открывается ДО вставки строки, и это не мелочь: на временной
          // записи блоб, записанный после INSERT и закреплённый MODIFY, до читателя не
          // доезжает вовсе. Измерено на себе - HASVALUE у принимающей стороны отвечал "нет"
          // при полностью заполненной строке.
          IF VictimBody <> '' THEN BEGIN
            Observed."Victim Statement".CREATEOUTSTREAM(OutStr,TEXTENCODING::UTF8);
            OutStr.WRITETEXT(VictimBody);
          END;
          IF BlockerBody <> '' THEN BEGIN
            Observed."Blocker Statement Text".CREATEOUTSTREAM(OutStr,TEXTENCODING::UTF8);
            OutStr.WRITETEXT(BlockerBody);
          END;
          Observed.INSERT;
        END;
      END;
      SqlReader.Close;
      SqlConn.Close;
    END;

    LOCAL PROCEDURE TryResolveDocuments@31(VAR Buffer@1000000000 : TEMPORARY Record 110231;VAR Context@1000000014 : TEMPORARY Record 110232);
    VAR
      SqlConn@1000000001 : DotNet "'System.Data, Version=2.0.0.0, Culture=neutral, PublicKeyToken=b77a5c561934e089'.System.Data.SqlClient.SqlConnection";
      SqlCmd@1000000002 : DotNet "'System.Data, Version=2.0.0.0, Culture=neutral, PublicKeyToken=b77a5c561934e089'.System.Data.SqlClient.SqlCommand";

      HobtList@1000000011 : Text;
      SpidList@1000000017 : Text;
      MarkSpidList@1000000018 : Text;
      Slot@1000000012 : Integer;
      Answered@1000000013 : Boolean;
    BEGIN
      // Одно чтение dm_tran_locks на весь проход, и оно приносит блокировки для ОБЕИХ
      // дорог сразу, хотя берёт их с РАЗНЫХ сторон: за какую строку идёт спор, знает сама
      // жертва - её просьба ещё не выполнена; чем занят виновник, видно по тому, что он
      // держит. Читать представление дважды значило бы платить дважды за один и тот же
      // скан, а он дорог: 45,6 мс при шестидесяти тысячах блокировок, и отбор в WHERE не
      // помогает вовсе.
      BuildLockLists(Buffer,TranList,HobtList,SpidList,MarkSpidList);
      IF (TranList = '') AND (SpidList = '') AND (MarkSpidList = '') THEN
        EXIT;

      SqlConn := SqlConn.SqlConnection(ConnString);
      SqlConn.Open;
      SqlCmd := SqlConn.CreateCommand;
      SqlCmd.CommandTimeout := ReadTimeoutSec;
      SqlCmd.CommandText := LocksOfInterestSql(TranList,HobtList,SpidList,MarkSpidList);
      SqlReader := SqlCmd.ExecuteReader;
      WHILE SqlReader.Read AND (Locks < MaxLocks) DO BEGIN
        Locks := Locks + 1;
        LockKind[Locks] := ReaderInt(SqlReader,0);
        LockSpid[Locks] := ReaderInt(SqlReader,1);
        LockTran[Locks] := SqlReader.GetInt64(2);
        LockHobt[Locks] := SqlReader.GetInt64(3);
        LockObject[Locks] := ReaderText(SqlReader,4);
        LockIndex[Locks] := ReaderText(SqlReader,5);
        LockHash[Locks] := Parse.TrimLockHash(ReaderText(SqlReader,6));
      END;
      SqlReader.Close;

      Buffer.RESET;
      IF Buffer.FINDSET(TRUE) THEN
        REPEAT
          // Дороги отвечают на РАЗНЫЕ вопросы, и потому спрашиваются обе. Первая говорит,
          // за какую строку идёт спор, - ровно тот вопрос, ради которого инструмент и
          // заводился. Вторая говорит, КТО виновник и чем он занят.
          //
          // Раньше вторая шла только там, где промолчала первая, и это стоило человека:
          // имя есть ТОЛЬКО у отметки, а на таблице, объявленной таблицей документа,
          // первая дорога отвечает всегда - и журнал оставался с документом, но без
          // человека. Стоит исправление одного запроса и только там, где отметка есть:
          // без переходника блокировок на таблице отметок не бывает вовсе, и вторая
          // дорога до SQL не доходит.
          Answered := FALSE;
          Slot := FindWaitedLock(LockKind,LockSpid,LockHobt,Locks,Buffer."Victim SPID",Buffer."Hobt Id");
          IF Slot > 0 THEN
            Answered := ByContendedRow(SqlConn,Buffer,Context,LockObject[Slot],LockIndex[Slot],LockHash[Slot]);
          Slot := FindMarkLock(LockKind,LockSpid,LockTran,LockObject,Locks,
                               Buffer."Head SPID",Buffer."Blocker Transaction Id");
          IF Slot > 0 THEN
            ByContextMark(SqlConn,Buffer,LockObject[Slot],LockIndex[Slot],LockHash[Slot],Answered);

          // Третья дорога отвечает на вопрос "КОГО задержали". Отметка есть и у жертвы -
          // своя, положенная её же транзакцией до того, как она встала в очередь, - и
          // читается она тем же приёмом. Без неё журнал называет пострадавшего номером
          // сеанса и узлом, то есть не называет: по номеру сеанса человека не найти, а
          // через сутки его не найти и подавно.
          Slot := FindMarkBySpid(LockKind,LockSpid,LockObject,Locks,Buffer."Victim SPID");
          IF Slot > 0 THEN
            ByVictimMark(SqlConn,Buffer,LockObject[Slot],LockIndex[Slot],LockHash[Slot]);
        UNTIL Buffer.NEXT = 0;
      Buffer.RESET;
      SqlConn.Close;
    END;

        EXIT(FALSE);
      SqlCmd := SqlConn.CreateCommand;
      SqlCmd.CommandTimeout := ReadTimeoutSec;
      SqlCmd.CommandText := MarkRowSql(ObjectName,IndexName,LockHash);
      SqlReader := SqlCmd.ExecuteReader;
      IF SqlReader.Read THEN BEGIN
        DocumentNo := ReaderText(SqlReader,0);
        UserId := ReaderText(SqlReader,1);
      END;
      SqlReader.Close;
      IF (DocumentNo = '') AND (UserId = '') THEN
        EXIT(FALSE);

      // Имя человека берётся ВСЕГДА, даже когда документ уже назван первой дорогой:
      // другой дороги к имени у инструмента нет вовсе. Отметку положил сам виновник в
      // своей же транзакции - это его имя, а не догадка по времени.
      IF UserId <> '' THEN
        Buffer."User Name" := COPYSTR(UserId,1,MAXSTRLEN(Buffer."User Name"));

      IF DocumentNo <> '' THEN BEGIN
        IF NOT DocumentKnown THEN BEGIN
          Buffer."Document No." := COPYSTR(DocumentNo,1,MAXSTRLEN(Buffer."Document No."));
          Buffer."Document Source" := Parse.DocSourceFromMark;
        END ELSE BEGIN
          // Обе дороги назвали документ. Совпали - остаётся спорная строка: она и есть
          // ответ на вопрос "за что спор". Разошлись - об этом сказано вслух, отдельным
          // значением: виновник держит одну строку, а спор идёт за другую.
          IF Buffer."Document No." <> COPYSTR(DocumentNo,1,MAXSTRLEN(Buffer."Document No.")) THEN
            Buffer."Document Source" := Parse.DocSourceDisagree;
        END;
      END;
      Buffer.MODIFY;
      EXIT(TRUE);
    END;

    LOCAL PROCEDURE FindWaitedLock@35(VAR LockKind@1000000006 : ARRAY [40] OF Integer;VAR LockSpid@1000000007 : ARRAY [40] OF Integer;VAR LockHobt@1000000001 : ARRAY [40] OF BigInteger;Locks@1000000002 : Integer;Spid@1000000003 : Integer;HobtId@1000000004 : BigInteger) : Integer;
    VAR
      Index@1000000005 : Integer;
    BEGIN
      // Хэш спорной строки берётся у САМОЙ ЖЕРТВЫ, а не у виновника, и это разница между
      // верным ответом и правдоподобным. Виновник держит столько строк, сколько тронул:
      // измерено на стенде - две тронутые строки дают на одном hobt две выданные
      // блокировки, и выбранная из них наугад назвала бы чужой документ уверенным тоном.
      // Невыполненная просьба жертвы на том же hobt ровно одна, и она называет ту строку,
      // за которую идёт спор. Ждущая задача в dm_os_waiting_tasks хэша не несёт вовсе -
      // только hobt, - поэтому спрашивать приходится dm_tran_locks.
      IF (Spid = 0) OR (HobtId = 0) THEN
        EXIT(0);
      FOR Index := 1 TO Locks DO
        IF (LockKind[Index] = KindWaited) AND (LockSpid[Index] = Spid) AND (LockHobt[Index] = HobtId) THEN
          EXIT(Index);
      EXIT(0);
    END;

    LOCAL PROCEDURE FindMarkLock@36(VAR LockKind@1000000007 : ARRAY [40] OF Integer;VAR LockSpid@1000000008 : ARRAY [40] OF Integer;VAR LockTran@1000000000 : ARRAY [40] OF BigInteger;VAR LockObject@1000000001 : ARRAY [40] OF Text;Locks@1000000002 : Integer;HeadSpid@1000000009 : Integer;TranId@1000000003 : BigInteger) : Integer;
    VAR
      Index@1000000004 : Integer;
    BEGIN
      // Отметку виновник ДЕРЖИТ, а не просит: она поставлена его же записью внутри той же
      // транзакции. Невыполненная просьба на таблице отметок значила бы обратное - что
      // виновник сам чего-то ждёт, - и документом из неё отвечать нельзя.
      //
      // Спрашивается она у ГОЛОВЫ цепочки, а не у непосредственного блокирующего, и это
      // не мелочь. В очереди A -> B -> C непосредственный виновник у C - это B, который
      // сам стоит за A: у B своя отметка, и она про то, чем занят ПОСТРАДАВШИЙ. Назвать
      // его тем, кто держит, - ошибка того же рода, что назвать виновным пострадавшего,
      // только правдоподобнее: имя-то настоящее, живого человека. Держит строку голова,
      // она же и оставила отметку.
      //
      // Проверено очередью из трёх жертв: на прежнем коде имя получал ровно ОДИН эпизод
      // из трёх - тот, чей непосредственный виновник и был головой.
      Index := FindMarkBySpid(LockKind,LockSpid,LockObject,Locks,HeadSpid);
      IF Index > 0 THEN
        EXIT(Index);
      // Головы не знаем - остаётся непосредственный блокирующий. На цепочке глубиной в
      // единицу это одно и то же лицо, а глубже без головы отвечать всё равно нечем.
      IF TranId <> 0 THEN
        FOR Index := 1 TO Locks DO
          IF (LockKind[Index] = KindGranted) AND (LockTran[Index] = TranId) THEN
            IF IsMarkTable(LockObject[Index]) THEN
              EXIT(Index);
      EXIT(0);
    END;

    LOCAL PROCEDURE FindMarkBySpid@29(VAR LockKind@1000000000 : ARRAY [40] OF Integer;VAR LockSpid@1000000001 : ARRAY [40] OF Integer;VAR LockObject@1000000002 : ARRAY [40] OF Text;Locks@1000000003 : Integer;Spid@1000000004 : Integer) : Integer;
    VAR
      Index@1000000005 : Integer;
    BEGIN
      // Выданная блокировка на таблице отметок у названного сеанса. Выданная, а не
      // просимая: отметку СВОЮ сеанс держит, а не ждёт.
      IF Spid = 0 THEN
        EXIT(0);
      FOR Index := 1 TO Locks DO
        IF (LockKind[Index] = KindGranted) AND (LockSpid[Index] = Spid) THEN
          IF IsMarkTable(LockObject[Index]) THEN
            EXIT(Index);
      EXIT(0);
    END;

    LOCAL PROCEDURE IsMarkTable@28(Name@1000000000 : Text) : Boolean;
    VAR
      Suffix@1000000001 : Text;
    BEGIN
      // Имя таблицы приходит с ПРЕФИКСОМ компании, и сравнивать его целиком нельзя:
      // компаний в базе много, а таблица отметок у каждой своя.
      Suffix := MarkTableSuffix;
      IF STRLEN(Name) < STRLEN(Suffix) THEN
        EXIT(FALSE);
      EXIT(COPYSTR(Name,STRLEN(Name) - STRLEN(Suffix) + 1) = Suffix);
    END;

    LOCAL PROCEDURE BuildLockLists@37(VAR Buffer@1000000000 : TEMPORARY Record 110231;VAR TranList@1000000001 : Text;VAR HobtList@1000000002 : Text;VAR SpidList@1000000004 : Text;VAR MarkSpidList@1000000005 : Text);
    VAR

    BEGIN
      // Четыре списка на один запрос: транзакции виновников и сеансы, чьи ОТМЕТКИ нужны
      // (голова цепочки и сама жертва), - для имён; номера сеансов жертв вместе с hobt -
      // для спорных строк. Отбор по номеру сеанса обязателен: без него в ответ попали бы
      // невыполненные просьбы всей базы.
      TranList := '';
      HobtList := '';
      SpidList := '';
      MarkSpidList := '';

      IF Buffer.FINDSET THEN

          IF Buffer."Blocker Transaction Id" <> 0 THEN BEGIN
            TranList := AppendId(TranList,FORMAT(Buffer."Blocker Transaction Id"));
            IF Buffer."Head SPID" <> 0 THEN
              MarkSpidList := AppendId(MarkSpidList,FORMAT(Buffer."Head SPID"));
            IF Buffer."Victim SPID" <> 0 THEN

              HobtList := AppendId(HobtList,FORMAT(Buffer."Hobt Id"));
              SpidList := AppendId(SpidList,FORMAT(Buffer."Victim SPID"));
            END;
            Seen := Seen + 1;
          END;
        UNTIL (Seen >= Parse.MaxDocumentsPerPass) OR (Buffer.NEXT = 0);
      Buffer.RESET;
    END;

    LOCAL PROCEDURE LocksOfInterestSql@38(TranList@1000000000 : Text;HobtList@1000000001 : Text;SpidList@1000000004 : Text;MarkSpidList@1000000007 : Text) : Text;

    LOCAL PROCEDURE RowByHashSql@39(ObjectName@1000000000 : Text;IndexName@1000000001 : Text;LockHash@1000000002 : Text;ColumnPattern@1000000003 : Text) : Text;
    VAR
      Sql@1000000004 : Text;
    BEGIN
      // Имя столбца не вычисляется, а СПРАШИВАЕТСЯ у сервера по образцу: правило, по
      // которому платформа переводит имя поля в имя столбца, нигде не объявлено, и
      // повторять его по памяти значит однажды промахнуться молча.
      //
      // Двух подходящих столбцов быть не должно: если их два, отвечать нельзя вовсе -
      // выбранный наугад дал бы правдоподобный, но чужой номер. Отсюда TOP 2 и проверка.
      Sql :=
        'DECLARE @col sysname, @hits int, @sql nvarchar(max);' +
        'SELECT @hits ' + Eq + ' COUNT(*) FROM (SELECT TOP 2 c.name FROM sys.columns c' +
        ' WHERE c.object_id ' + Eq + ' OBJECT_ID(' + Q + Bracket(ObjectName) + Q + ')' +
        ' AND c.name LIKE ' + Q + ColumnPattern + Q + ') x;' +
        'SELECT TOP 1 @col ' + Eq + ' c.name FROM sys.columns c' +
        ' WHERE c.object_id ' + Eq + ' OBJECT_ID(' + Q + Bracket(ObjectName) + Q + ')' +
        ' AND c.name LIKE ' + Q + ColumnPattern + Q + ';' +
        'IF @col IS NULL OR @hits <> 1 BEGIN SELECT CONVERT(nvarchar(20),NULL); END' +
        ' ELSE BEGIN' +
        ' SET @sql ' + Eq + ' N' + Q + 'SELECT TOP 1 CONVERT(nvarchar(20),' + Q +
        ' + QUOTENAME(@col) + N' + Q + ') FROM ' + Bracket(ObjectName) +
        ' WITH (INDEX(' + Bracket(IndexName) + '), NOLOCK) WHERE ' + LockResColumn +
        ' ' + Eq + ' @h' + Q + ';' +
        ' EXEC sp_executesql @sql, N' + Q + '@h nvarchar(64)' + Q + ', @h ' + Eq + ' ' +
        Q + LockHash + Q + '; END;';
      EXIT(Sql);
    END;

    LOCAL PROCEDURE TryReadDeadlocks@51(SinceAt@1000000000 : DateTime;MaxRows@1000000001 : Integer;WantStatements@1000000002 : Boolean;VAR Found@1000000003 : TEMPORARY Record 110231;VAR BufferPresent@1000000004 : Boolean;VAR OldestAt@1000000005 : DateTime;VAR Rows@1000000006 : Integer;VAR Counted@1000000019 : BigInteger);

      SqlConn@1000000007 : DotNet "'System.Data, Version=2.0.0.0, Culture=neutral, PublicKeyToken=b77a5c561934e089'.System.Data.SqlClient.SqlConnection";

    BEGIN

      SqlConn.Open;
      SqlCmd := SqlConn.CreateCommand;
      SqlCmd.CommandTimeout := ReadTimeoutSec;
      SqlCmd.CommandText := DeadlocksSql(MomentDays(SinceAt),MomentMs(SinceAt),MaxRows,WantStatements);
      SqlReader := SqlCmd.ExecuteReader;

      // Первый набор - про сам буфер, а не про события: работает ли сессия system_health
      // и каким событием буфер начинается. Без этих двух ответов выключенная сессия и
      // потерянные события выглядят одинаково - как "взаимоблокировок не было".
      IF SqlReader.Read THEN BEGIN
        BufferPresent := ReaderInt(SqlReader,0) <> 0;
        OldestAt := SqlMoment(ReaderInt(SqlReader,1),ReaderInt(SqlReader,2));
        // Счёт кругов у самого сервера. Едет он ТУТ ЖЕ, четвёртым столбцом первого набора,
        // а не отдельным запросом: соединение и так открыто, а лишний заход к серверу стоит
        // дороже самого счёта. Права он не требует новых - всё тот же VIEW SERVER STATE.
        Counted := ReaderBigInt(SqlReader,3);
      END;
      SqlReader.NextResult;

      WHILE SqlReader.Read AND (Rows < MaxRows) DO BEGIN

        EventAt := SqlMoment(ReaderInt(SqlReader,0),ReaderInt(SqlReader,1));
        Found.INIT;
        Found."Entry No." := Rows;
        Found."Victim SPID" := ReaderInt(SqlReader,2);
        Found."Victim Host" := COPYSTR(ReaderText(SqlReader,3),1,MAXSTRLEN(Found."Victim Host"));

        Found."Blocker SPID" := ReaderInt(SqlReader,8);
        Found."Head Is Foreign" := NOT IsNavProgram(ReaderText(SqlReader,9));
        // Граф взаимоблокировки называет участников сам: узел, логин и программу он держит
        // прямо в узле процесса. Программа тут уже читается - по ней решается, наш ли это
        // сеанс, - и выбрасывать её после одной проверки было бы расточительством.
        Found."Blocker Program" :=
          COPYSTR(ReaderText(SqlReader,9),1,MAXSTRLEN(Found."Blocker Program"));
        Found."Blocker Host" :=
          COPYSTR(ReaderText(SqlReader,24),1,MAXSTRLEN(Found."Blocker Host"));
        Found."Blocker Login" :=
          COPYSTR(ReaderText(SqlReader,25),1,MAXSTRLEN(Found."Blocker Login"));
        // Жертву граф называет тем же узлом процесса, из которого уже взяты машина, процесс
        // и программа. Логин оттуда же и стоит столько же - нисколько.
        Found."Victim Login" :=
          COPYSTR(ReaderText(SqlReader,26),1,MAXSTRLEN(Found."Victim Login"));
        Found."Victim Program" :=
          COPYSTR(ReaderText(SqlReader,5),1,MAXSTRLEN(Found."Victim Program"));
        Found."Blocker Transaction Id" := ReaderBigInt(SqlReader,10);
        Found."Blocker Tran Began At" := SqlMoment(ReaderInt(SqlReader,11),ReaderInt(SqlReader,12));
        Found."Resource Kind" := COPYSTR(UPPERCASE(ReaderText(SqlReader,13)),1,MAXSTRLEN(Found."Resource Kind"));
        Found."Hobt Id" := ReaderBigInt(SqlReader,14);
        Found."Database Id" := ReaderInt(SqlReader,15);
        ObjectName := ReaderText(SqlReader,16);
        IndexName := ReaderText(SqlReader,17);
        Found."Held Mode" := COPYSTR(ReaderText(SqlReader,18),1,MAXSTRLEN(Found."Held Mode"));
        Found."Wait Type" := COPYSTR(LockWaitPrefix + ReaderText(SqlReader,19),1,MAXSTRLEN(Found."Wait Type"));
        Found."Deadlock Id" := Parse.TextFingerprint(ReaderText(SqlReader,20));
        Found."Blocker Statement" := COPYSTR(ReaderText(SqlReader,21),1,MAXSTRLEN(Found."Blocker Statement"));
        StatementBody := Parse.StatementWithMark(ReaderText(SqlReader,22),ReaderInt(SqlReader,23));

        // Цепочка из двоих, и она ЗАМКНУТА - этим взаимоблокировка и отличается от
        // очереди. Голова здесь не "тот, кто ни на кого не ждёт": ждут оба, и признак
        // кольца тут не догадка разбора, а факт из графа.
        Found."Head SPID" := Found."Blocker SPID";
        Found."Chain Depth" := 1;
        Found."Chain Ring" := TRUE;
        Found."Victims Behind Head" := 1;
        Found.Class := Parse.ClassDeadlock;
        Found.Outcome := Parse.OutcomeDeadlocked;

        // Эпизод приезжает УЖЕ ЗАКОНЧЕННЫМ: круг разорвал сервер, и конец известен точно.
        // Начало считается назад от конца по времени ожидания жертвы - тем же правилом,
        // каким считается начало обычного эпизода.
        Found."Ended At" := EventAt;
        Found."Last Seen At" := EventAt;
        Found."Started At" := Parse.EpisodeStart(EventAt,Found."Last Wait (ms)");
        Found.Open := FALSE;

        KeyNo := Parse.UnknownKeyNo;
        IF Parse.SplitSqlObjectName(ObjectName,CompanyName,TableName,SiftKeyNo) THEN BEGIN
          IF SiftKeyNo <> Parse.UnknownKeyNo THEN
            KeyNo := SiftKeyNo
          ELSE
            KeyNo := Parse.KeyNoFromIndexName(IndexName);
          NameRow(Found,ObjectName,TableName,KeyNo,SiftKeyNo <> Parse.UnknownKeyNo);
        END ELSE
          // Имя, которое не разобралось, показывается СЫРЫМ и метится непонятым. Пустая
          // колонка на его месте читалась бы как "таблица неизвестна серверу".
          UnnamedRow(Found,ObjectName);
        IF StatementBody <> '' THEN BEGIN
          Found."Blocker Statement Text".CREATEOUTSTREAM(StatementOut,TEXTENCODING::UTF8);
          StatementOut.WRITETEXT(StatementBody);
        END;
        Found.INSERT;
      END;
      SqlReader.Close;
      SqlConn.Close;
    END;

    LOCAL PROCEDURE DeadlocksSql@52(SinceDays@1000000000 : Integer;SinceMs@1000000001 : Integer;MaxRows@1000000002 : Integer;WantStatements@1000000003 : Boolean) : Text;
    VAR
      Sql@1000000004 : Text;
      Events@1000000005 : Text;
      Statement@1000000006 : Text;
      TranStarted@1000000007 : Text;
    BEGIN
      // Граф разбирает СЕРВЕР, а в C/AL приезжают готовые столбцы. Тащить XML в NAV
      // значило бы завести второй разбор имён и держать его в согласии с первым.
      //
      // Кольцевой буфер превращается в XML ОДИН раз, в переменную: обращений к нему два -
      // за границей буфера и за самими событиями, - а разбор многомегабайтной строки
      // стоит столько, что платить за него дважды нельзя.
      //
      // QUOTED_IDENTIFIER включается явно. Драйвер и так включает его на соединении, но
      // без него ЛЮБОЙ метод XML отказывает, и отказ называет не причину, а список
      // требований к настройкам сеанса.
      //
      // Отметка собирается через datetime2(3), а не через datetime. У datetime шаг 3,33 мс,
      // и он округляет ВВЕРХ: собранная им отметка оказывается на полтора миллисекунды
      // позже настоящей, а событие ровно на границе при этом пропускается молча. Ради
      // него отбор и сделан нестрогим, так что округление обесценило бы всю защиту.
      Events := Q + '/RingBufferTarget/event[@name' + Eq + '"xml_deadlock_report"]' + Q;
      // nvarchar(max) метод .value() не принимает вовсе, и это не наш выбор потолка:
      // сервер САМ обрезает входной буфер на четырёх тысячах знаков, поэтому 4000 здесь
      // ничего не теряет. Отметка усечения на этой дороге поэтому и не срабатывает.
      IF WantStatements THEN
        Statement := ' LTRIM(CONVERT(nvarchar(250),bp.n' + XVal('(inputbuf)[1]','nvarchar(500)') + ')),' +
                     ' LTRIM(bp.n' + XVal('(inputbuf)[1]','nvarchar(4000)') + '),' +
                     ' CONVERT(int,ISNULL(LEN(LTRIM(bp.n' + XVal('(inputbuf)[1]','nvarchar(4000)') + ')),0))'
      ELSE
        // Текст оператора едет со ЗНАЧЕНИЯМИ - теми самыми данными, которых мы обещали не
        // собирать. Он приходит только под галкой, выключенной по умолчанию.
        Statement := ' SPACE(0), SPACE(0), 0';

      // Атрибут графа сервер пишет по местным часам, а отдать его надо по UTC - той же
      // шкалой, в какой NAV хранит DateTime. Выражение собирается ОДИН раз и подставляется
      // трижды: разъехавшись между вхождениями, оно дало бы сутки вместо миллисекунд.
      TranStarted := 'DATEADD(minute,@shift,bp.n' + XVal('@lasttranstarted','datetime2(3)') + ')';
      Sql :=
        'SET QUOTED_IDENTIFIER ON;' +
        'DECLARE @x xml, @since datetime2(3), @shift int;' +
        'SET @shift ' + Eq + ' DATEDIFF(minute,GETDATE(),GETUTCDATE());' +
        'SET @since ' + Eq + ' DATEADD(ms,' + FORMAT(SinceMs) +
        ',CONVERT(datetime2(3),CONVERT(datetime,' + FORMAT(SinceDays) + ')));' +
        'SELECT @x ' + Eq + ' CONVERT(xml,t.target_data)' +
        ' FROM sys.dm_xe_session_targets t' +
        ' JOIN sys.dm_xe_sessions s ON s.address ' + Eq + ' t.event_session_address' +
        ' WHERE s.name ' + Eq + ' ' + Q + 'system_health' + Q +
        ' AND t.target_name ' + Eq + ' ' + Q + 'ring_buffer' + Q + ';';

      // Время события в буфере лежит по UTC - и УЖЕ в той шкале, в какой NAV хранит свои
      // отметки, поэтому сдвигать его больше не надо. Прежде оно переводилось в местное,
      // чтобы сойтись с остальным журналом; теперь весь журнал перешёл на UTC, и перевод
      // стал бы ошибкой на разницу поясов.
      //
      // Сдвиг, однако, нужен - но в ДРУГУЮ сторону и для другого. Атрибуты самого графа
      // (@lasttranstarted) сервер пишет по МЕСТНЫМ часам, и к UTC их приводит он же.
      Sql := Sql +
        'SELECT CASE WHEN @x IS NULL THEN 0 ELSE 1 END,' +
        ' ISNULL(DATEDIFF(day,CONVERT(datetime,0),MIN(v.EventAt)),0),' +
        ' ISNULL(DATEDIFF(ms,CONVERT(date,MIN(v.EventAt)),MIN(v.EventAt)),0),' +
        // Круги сервер считает сам, и счёт этот от кольца не зависит вовсе: кольцо может
        // замолчать целиком, а счётчик продолжит расти. Имя счётчика кончается на "/sec",
        // но величина накопительная - это давняя странность имён у счётчиков SQL Server.
        // Отбор по имени нестрогий: counter_name объявлен nchar(128) и добит пробелами.
        ' ISNULL((SELECT TOP 1 c.cntr_value FROM sys.dm_os_performance_counters c' +
        ' WHERE c.counter_name LIKE ' + Q + 'Number of Deadlocks/sec%' + Q +
        ' AND c.instance_name ' + Eq + ' ' + Q + '_Total' + Q + '),0)' +
        ' FROM (SELECT n.e' + XVal('@timestamp','datetime2(3)') +
        ' AS EventAt FROM @x.nodes(' + Events + ') AS n(e)) v;';

      Sql := Sql +
        'WITH ev AS (SELECT n.e' + XVal('@timestamp','datetime2(3)') +
        ' AS EventAt, n.e.query(' + Q + 'data[@name' + Eq + '"xml_report"]/value/deadlock' +
        Q + ') AS G FROM @x.nodes(' + Events + ') AS n(e)),' +
        ' vic AS (SELECT ev.EventAt, ev.G, v.n' + XVal('@id','varchar(60)') + ' AS VictimId' +
        ' FROM ev CROSS APPLY ev.G.nodes(' + Q + '/deadlock/victim-list/victimProcess' + Q +
        ') AS v(n)),';

      // Ресурс берётся не любой из графа, а ТОТ, в очереди к которому стоит жертва: в
      // графе их столько, сколько сторон у круга, и любой другой назвал бы чужую таблицу.
      Sql := Sql +
        ' res AS (SELECT vic.EventAt, vic.G, vic.VictimId,' +
        ' r.n' + XVal('local-name(.)','varchar(30)') + ' AS Kind,' +
        ' r.n' + XVal('@hobtid','bigint') + ' AS HobtId,' +
        ' r.n' + XVal('@dbid','int') + ' AS DbId,' +
        ' r.n' + XVal('@objectname','nvarchar(300)') + ' AS ObjectName,' +
        ' r.n' + XVal('@indexname','nvarchar(300)') + ' AS IndexName,' +
        ' r.n' + XVal('(owner-list/owner/@id)[1]','varchar(60)') + ' AS OwnerId,' +
        ' r.n' + XVal('(owner-list/owner/@mode)[1]','varchar(10)') + ' AS OwnerMode,' +
        ' r.n' + XVal('(waiter-list/waiter/@mode)[1]','varchar(10)') + ' AS WaiterMode' +
        ' FROM vic CROSS APPLY vic.G.nodes(' + Q + '/deadlock/resource-list/*[waiter-list/waiter/@id' +
        Eq + 'sql:column("vic.VictimId")]' + Q + ') AS r(n))';

      Sql := Sql +
        ' SELECT TOP (' + FORMAT(MaxRows) + ')' +
        ' DATEDIFF(day,CONVERT(datetime,0),res.EventAt),' +
        ' DATEDIFF(ms,CONVERT(date,res.EventAt),res.EventAt),' +
        ' vp.n' + XVal('@spid','int') + ',' +
        ' vp.n' + XVal('@hostname','nvarchar(80)') + ',' +
        ' vp.n' + XVal('@hostpid','int') + ',' +
        ' vp.n' + XVal('@clientapp','nvarchar(120)') + ',' +
        ' vp.n' + XVal('@xactid','bigint') + ',' +
        ' vp.n' + XVal('@waittime','int') + ',' +
        ' bp.n' + XVal('@spid','int') + ',' +
        ' bp.n' + XVal('@clientapp','nvarchar(120)') + ',' +
        ' bp.n' + XVal('@xactid','bigint') + ',' +
        ' DATEDIFF(day,CONVERT(datetime,0),' + TranStarted + '),' +
        ' DATEDIFF(ms,CONVERT(date,' + TranStarted + '),' + TranStarted + '),' +
        ' res.Kind, res.HobtId, res.DbId, ISNULL(PARSENAME(res.ObjectName,1),res.ObjectName),' +
        ' res.IndexName, res.OwnerMode, res.WaiterMode,';

      // Отпечаток графа собирается из того, что у события не меняется между чтениями:
      // время, обе транзакции, hobt и ресурс, которого ждала жертва. Место в буфере в
      // отпечаток не входит - оно меняется при каждом чтении, а граф нет.
      Sql := Sql +
        ' CONVERT(varchar(30),res.EventAt,126) + ' + Q + '|' + Q +
        ' + CONVERT(varchar(20),vp.n' + XVal('@xactid','bigint') + ') + ' + Q + '|' + Q +
        ' + CONVERT(varchar(20),bp.n' + XVal('@xactid','bigint') + ') + ' + Q + '|' + Q +
        ' + CONVERT(varchar(20),res.HobtId) + ' + Q + '|' + Q +
        ' + vp.n' + XVal('@waitresource','varchar(120)') + ',' +
        Statement +
        // Дописано ПОСЛЕ Statement нарочно: он всегда ровно три столбца - и под галкой на
        // тексты, и без неё, - значит хвост неподвижен и ни один прежний номер не поехал.
        ', bp.n' + XVal('@hostname','nvarchar(80)') + ',' +
        ' bp.n' + XVal('@loginname','nvarchar(128)') + ',' +
        ' vp.n' + XVal('@loginname','nvarchar(128)') +
        ' FROM res' +
        ' CROSS APPLY res.G.nodes(' + Q + '/deadlock/process-list/process[@id' + Eq +
        'sql:column("res.VictimId")]' + Q + ') AS vp(n)' +
        ' CROSS APPLY res.G.nodes(' + Q + '/deadlock/process-list/process[@id' + Eq +
        'sql:column("res.OwnerId")]' + Q + ') AS bp(n)' +
        // Отбор нестрогий нарочно: два события в одну миллисекунду при строгом отборе
        // потеряли бы второе молча. Повтор снимает отпечаток, потеря не снимается ничем.
        ' WHERE res.EventAt >' + Eq + ' @since' +
        ' ORDER BY res.EventAt;';
      EXIT(Sql);
    END;

    LOCAL PROCEDURE NameRow@53(VAR Row@1000000000 : Record 110231;ObjectName@1000000001 : Text;TableName@1000000002 : Text;KeyNo@1000000003 : Integer;OnSift@1000000004 : Boolean);
    BEGIN
      // Одно место, где разобранное имя SQL ложится в строку. Дороги сюда две - очередь
      // приходит с hobt, граф взаимоблокировки с готовым именем, - но раскладывать имя
      // обязаны одинаково, иначе сводка сравнивала бы несравнимое.
      Row."SQL Object Name" := COPYSTR(ObjectName,1,MAXSTRLEN(Row."SQL Object Name"));
      Row."NAV Table Name" := COPYSTR(TableName,1,MAXSTRLEN(Row."NAV Table Name"));
      Row."NAV Key No." := KeyNo;
      Row."On SIFT" := OnSift;
    END;

    LOCAL PROCEDURE UnnamedRow@64(VAR Row@1000000000 : Record 110231;ObjectName@1000000001 : Text);
    BEGIN
      // Место, парное первому: имя, которое разбор НЕ понял, показывается сырым и метится
      // непонятым. Дороги сюда те же две, и расходиться им нельзя - а они расходились:
      // граф показывал сырое имя, очередь молчала, и строка выходила без имени вовсе.
      //
      // Пустое имя здесь не мелочь. У ожидания ОБЪЕКТА целиком hobt равен нулю, и имя
      // остаётся единственным, чем два разных ожидания одной транзакции отличаются друг
      // от друга (см. FindOpen в журнале). Пустое у обоих - и различить эпизоды нечем:
      // порознь их держит один лишь признак "счётчик ожидания пошёл назад", а он о другом
      // и срабатывает не всегда.
      Row."SQL Object Name" := COPYSTR(ObjectName,1,MAXSTRLEN(Row."SQL Object Name"));
      Row."NAV Table Name" := '';
      Row."NAV Key No." := Parse.UnknownKeyNo;
      Row."On SIFT" := FALSE;
      Row."Resource Unresolved" := TRUE;
    END;

    LOCAL PROCEDURE MomentDays@54(At@1000000000 : DateTime) : Integer;
    VAR
      Elapsed@1000000001 : Duration;
      Days@1000000002 : Decimal;
    BEGIN
      // Отметка едет в запрос теми же двумя целыми, какими время приходит обратно. Дата
      // строкой разбиралась бы по языку сервера и ошиблась бы не отказом, а сдвигом -
      // тем самым, из-за которого события пропускались бы молча.
      //
      // Считается всё РАЗНИЦЕЙ МОМЕНТОВ, а не вычитанием дат: "дата минус дата" в C/AL
      // берёт вторую дату за число дней и на любой настоящей дате роняет проход с
      // "The date is not valid". Ошибка приходит не при компиляции, а на первом же
      // непустом значении отметки - то есть на втором проходе, а не на первом.
      IF At = 0DT THEN
        EXIT(0);
      Elapsed := At - CREATEDATETIME(SqlEpoch,0T);
      Days := ROUND(Elapsed / Parse.MsPerDay,1,'<');
      EXIT(Days);
    END;

    LOCAL PROCEDURE MomentMs@55(At@1000000000 : DateTime) : Integer;
    VAR
      Elapsed@1000000001 : Duration;
    BEGIN
      IF At = 0DT THEN
        EXIT(0);
      Elapsed := At - CREATEDATETIME(DT2DATE(At),0T);
      EXIT(Elapsed);
    END;

    LOCAL PROCEDURE LockWaitPrefix@56() : Text;
    BEGIN
      // Очередь называет ожидание как LCK_M_U, а граф - одной буквой режима. В журнале
      // колонка одна, и заполнять её двумя разными словарями значило бы сделать отбор по
      // ней бесполезным.
      EXIT('LCK_M_');
    END;

    LOCAL PROCEDURE XVal@57(Path@1000000000 : Text;Kind@1000000001 : Text) : Text;
    BEGIN
      // .value('путь','тип') - самая частая конструкция этого запроса, и собирать её
      // руками два десятка раз значит два десятка раз расставить кавычки.
      EXIT('.value(' + Q + Path + Q + ',' + Q + Kind + Q + ')');
    END;

    LOCAL PROCEDURE ApplyName@6(VAR Observed@1000000000 : TEMPORARY Record 110231;Source@1000000001 : Integer;Id@1000000002 : BigInteger;ObjectName@1000000003 : Text;IndexName@1000000004 : Text);
    VAR
      CompanyName@1000000005 : Text;
      TableName@1000000006 : Text;
      SiftKeyNo@1000000007 : Integer;
      KeyNo@1000000008 : Integer;
      SqlObjectId@1000000009 : Integer;
      Parsed@1000000010 : Boolean;
    BEGIN
      // Неразобранное имя - не повод бросить строку. Здесь стоял выход, и строка после
      // него оставалась без имени ВОВСЕ - даже без пометки "не опознано". Читается это
      // как "сервер таблицы не назвал", хотя сервер её назвал: не понял её разбор.
      // Ловится на блокировке платформенной таблицы $ndo$ - их на базу около дюжины, и
      // таблицами NAV они не являются, поэтому разбор честно отвечает "не моё".
      Parsed := Parse.SplitSqlObjectName(ObjectName,CompanyName,TableName,SiftKeyNo);
      KeyNo := Parse.UnknownKeyNo;
      // Правил два, и одно теряет самый частый случай блокировок NAV молча: у таблицы
      // номер ключа несёт имя ИНДЕКСА, у SIFT-представления - имя самого объекта.
      IF Parsed THEN BEGIN
        IF SiftKeyNo <> Parse.UnknownKeyNo THEN
          KeyNo := SiftKeyNo
        ELSE
          KeyNo := Parse.KeyNoFromIndexName(IndexName);
      END;

      Observed.RESET;
      IF Source = SourceHobt THEN
        Observed.SETRANGE("Hobt Id",Id)
      ELSE BEGIN
        SqlObjectId := Id;
        Observed.SETRANGE("Hobt Id",0);
        Observed.SETRANGE("SQL Object Id",SqlObjectId);
      END;
      IF Observed.FINDSET(TRUE) THEN
        REPEAT
          IF Parsed THEN
            NameRow(Observed,ObjectName,TableName,KeyNo,SiftKeyNo <> Parse.UnknownKeyNo)
          ELSE
            UnnamedRow(Observed,ObjectName);
          Observed.MODIFY;
        UNTIL Observed.NEXT = 0;
      Observed.RESET;
    END;

    LOCAL PROCEDURE BuildLists@7(VAR Observed@1000000000 : TEMPORARY Record 110231;VAR HobtList@1000000001 : Text;VAR ObjectList@1000000002 : Text);
    BEGIN
      // Повторы в списке не убираем: в наборе SQL одному hobt отвечает одна строка, и
      // ответ приходит различённым сам собой. Отдельный проход ради уникальности стоил бы
      // дороже, чем лишние знаки в тексте запроса.
      HobtList := '';
      ObjectList := '';
      Observed.RESET;
      IF Observed.FINDSET THEN
        REPEAT
          IF Observed."Hobt Id" <> 0 THEN
            HobtList := AppendId(HobtList,FORMAT(Observed."Hobt Id"))
          ELSE
            IF Observed."SQL Object Id" <> 0 THEN
              ObjectList := AppendId(ObjectList,FORMAT(Observed."SQL Object Id"));
        UNTIL Observed.NEXT = 0;
      Observed.RESET;
    END;

    LOCAL PROCEDURE AppendId@8(List@1000000000 : Text;Value@1000000001 : Text) : Text;
    BEGIN
      IF List = '' THEN
        EXIT(Value);
      EXIT(List + ',' + Value);
    END;

    LOCAL PROCEDURE QueueSql@9(MaxRows@1000000000 : Integer;WantStatements@1000000002 : Boolean) : Text;
    VAR
      Sql@1000000001 : Text;
      Cols@1000000003 : Text;
      Joins@1000000004 : Text;
      Cap@1000000005 : Text;
      TranBegan@1000000006 : Text;
      LastEnded@1000000007 : Text;
    BEGIN
      // Текст запроса стоит четырёх присоединений к горячему запросу прохода, и потому он
      // ВЫКЛЮЧАЕМ. Выключенный не платит ничем: колонки остаются на месте пустыми, а
      // присоединений в запросе нет вовсе - читателю не приходится знать, какой сегодня
      // набор столбцов.
      //
      // Жертва ждёт, то есть у неё есть исполняемый запрос, и текст берётся по нему с
      // точными границами оператора. Виновник чаще СПИТ - в этом вся беда NAV-блокировок, -
      // и запроса у него нет; берётся последний, что он послал, из его соединения. Это и
      // есть тот оператор, который взял блокировку.
      Cap := FORMAT(Parse.MaxStatementChars);
      IF WantStatements THEN BEGIN
        Cols :=
          ', LEFT(vst.t,' + Cap + '), CONVERT(int,ISNULL(LEN(vst.t),0)),' +
          ' LEFT(bst.t,' + Cap + '), CONVERT(int,ISNULL(LEN(bst.t),0))';
        Joins :=
          ' OUTER APPLY (SELECT TOP 1 r.sql_handle AS h, r.statement_start_offset AS so,' +
          ' r.statement_end_offset AS eo FROM sys.dm_exec_requests r' +
          ' WHERE r.session_id ' + Eq + ' wt.session_id) vq' +
          ' OUTER APPLY sys.dm_exec_sql_text(vq.h) vtx' +
          ' OUTER APPLY (SELECT LTRIM(SUBSTRING(vtx.text,(vq.so/2)+1,' +
          ' ((CASE vq.eo WHEN -1 THEN DATALENGTH(vtx.text) ELSE vq.eo END - vq.so)/2)+1)) AS t) vst' +
          ' OUTER APPLY (SELECT TOP 1 c.most_recent_sql_handle AS h FROM sys.dm_exec_connections c' +
          ' WHERE c.session_id ' + Eq + ' wt.blocking_session_id) bq' +
          ' OUTER APPLY sys.dm_exec_sql_text(bq.h) btx' +
          ' OUTER APPLY (SELECT LTRIM(btx.text) AS t) bst';
      END ELSE BEGIN
        Cols := ', SPACE(0), 0, SPACE(0), 0';
        Joins := '';
      END;
      // Каждое целое приведено к int нарочно, и это не перестраховка: номер сеанса и
      // номер базы в системных представлениях - smallint, а GetInt32 на smallint падает
      // с "Specified cast is not valid". Отказ приходит на ПЕРВОЙ строке очереди, то есть
      // ровно тогда, когда блокировка наконец случилась, - и выглядит как "инструмент не
      // видит блокировок". Тип столбца в DMV проверять раньше, чем себя.
      //
      // Порядок по убыванию длительности не украшение: при усечении в журнал попадают
      // САМЫЕ ДОЛГИЕ ожидания, а не первые попавшиеся. Усечённый проход честнее метится
      // отдельно, но и то, что в него попало, должно быть тем, за чем пришли.
      // ШКАЛА ВРЕМЕНИ ЗДЕСЬ - UTC, и другой она быть не может. Часы серверной сессии NAV
      // идут по UTC, в колонку момент ложится как есть, а в пояс читателя его переводит
      // клиент; отдай сюда местное - и человек увидит БУДУЩЕЕ ровно на разницу поясов.
      // Замер 10.09.2026 на стенде с поясом +3: отметка последнего прохода легла в колонку
      // местным временем. Рядом, в той же строке состояния, лежали отметки, записанные
      // CURRENTDATETIME, - те были верны, и две шкалы расходились на три часа внутри
      // ОДНОЙ строки.
      //
      // Отсюда два выражения вместо одного. Сами МОМЕНТЫ берутся с UTC-часов сервера, а
      // столбцы, которые сервер ведёт по МЕСТНЫМ часам, приводятся к UTC тем же сервером:
      // это last_request_end_time и transaction_begin_time. Смешать шкалы в одном
      // вычитании - и длительность вырастет на разницу поясов, то есть простой виновника
      // в три часа возникнет на ровном месте.
      TranBegan := 'DATEADD(minute,@shift,ba.transaction_begin_time)';
      LastEnded := 'DATEADD(minute,@shift,bs.last_request_end_time)';
      Sql :=
        'DECLARE @now datetime2 = SYSUTCDATETIME(),' +
        ' @shift int = DATEDIFF(minute,GETDATE(),GETUTCDATE());' +
        'SELECT DATEDIFF(day,CONVERT(datetime,0),@now),' +
        ' DATEDIFF(ms,CONVERT(date,@now),@now), CONVERT(int,DB_ID());' +
        'SELECT TOP (' + FORMAT(MaxRows) + ')' +
        ' CONVERT(int,wt.session_id), CONVERT(int,wt.blocking_session_id),' +
        ' CONVERT(int,CASE WHEN wt.wait_duration_ms > 2147483647' +
        ' THEN 2147483647 ELSE wt.wait_duration_ms END),' +
        ' wt.wait_type, wt.resource_description,' +
        ' vs.host_name, vs.host_process_id, vs.program_name,' +
        ' vt.transaction_id, bt.transaction_id,' +
        ' bs.status, bs.program_name,' +
        ' CASE WHEN bs.last_request_end_time IS NULL THEN 0' +
        ' WHEN DATEDIFF(day,' + LastEnded + ',@now) > 20 THEN 2147483647' +
        ' ELSE DATEDIFF(ms,' + LastEnded + ',@now) END,' +
        ' CASE WHEN bs.open_transaction_count IS NULL THEN 0' +
        ' ELSE bs.open_transaction_count END,' +
        ' CASE WHEN EXISTS (SELECT 1 FROM sys.dm_exec_requests r' +
        ' WHERE r.session_id = wt.blocking_session_id) THEN 1 ELSE 0 END,' +
        ' CASE WHEN ba.transaction_begin_time IS NULL THEN 0' +
        ' ELSE DATEDIFF(day,CONVERT(datetime,0),' + TranBegan + ') END,' +
        ' CASE WHEN ba.transaction_begin_time IS NULL THEN 0' +
        ' ELSE DATEDIFF(ms,CONVERT(date,' + TranBegan + '),' +
        ' ' + TranBegan + ') END' +
        Cols +
        // Дописано ПОСЛЕ Cols нарочно, и это не небрежность. Cols - всегда ровно четыре
        // столбца (при выключенных текстах он отдаёт четыре пустышки), поэтому хвост
        // неподвижен и ни один прежний номер колонки не поехал. Сдвиг на единицу в этом
        // же запросе уже стоил нам молчаливой поломки - заметка о ней выше по коду.
        ', bs.login_name, bs.host_name, vs.login_name' +
        ' FROM sys.dm_os_waiting_tasks wt' +
        ' JOIN sys.dm_exec_sessions vs ON vs.session_id = wt.session_id' +
        ' LEFT JOIN sys.dm_tran_session_transactions vt ON vt.session_id = wt.session_id' +
        ' LEFT JOIN sys.dm_exec_sessions bs ON bs.session_id = wt.blocking_session_id' +
        ' LEFT JOIN sys.dm_tran_session_transactions bt' +
        ' ON bt.session_id = wt.blocking_session_id' +
        ' LEFT JOIN sys.dm_tran_active_transactions ba' +
        ' ON ba.transaction_id = bt.transaction_id' +
        Joins +
        ' WHERE wt.wait_type LIKE ' + Q + 'LCK[_]%' + Q +
        ' AND wt.session_id <> @@SPID' +
        ' ORDER BY wt.wait_duration_ms DESC;';
      EXIT(Sql);
    END;

    LOCAL PROCEDURE EpisodeClassOf@11(VAR SqlReader@1000000000 : DotNet "'System.Data, Version=2.0.0.0, Culture=neutral, PublicKeyToken=b77a5c561934e089'.System.Data.SqlClient.SqlDataReader") : Integer;
    VAR
      Status@1000000001 : Text;

      OpenTrans@1000000003 : Integer;

    BEGIN
      // Спящая транзакция - это не медленный запрос, а забытое модальное окно: виновник
      // НЕ исполняет ничего, но транзакцию держит. Лечится звонком, а не правкой кода,
      // и потому подписывается отдельным классом. Эскалацию сюда не заводим: она видна
      // не по сессии, а по счётчикам, и приходит другим слоем.
      //
      // Спящий БЕЗ открытой транзакции не держит ничего и виновником быть не может -
      // такую сессию называть "уснувшей в транзакции" значит обвинять невиновного.

      // вывод из статуса был бы пересказом первого признака вторыми словами.

      OpenTrans := ReaderInt(SqlReader,13);

    END;

    LOCAL PROCEDURE SqlMoment@14(Days@1000000000 : Integer;Milliseconds@1000000001 : Integer) : DateTime;
    VAR
      Offset@1000000002 : Duration;
    BEGIN
      // Время приходит ДВУМЯ целыми, а не строкой, и это не причуда: разбор строки даты
      // зависит от языка сеанса и от языка сервера, и ошибётся он не отказом, а сдвигом.
      // Сутки от начала отсчёта SQL плюс миллисекунды внутри суток врать не умеют.
      //
      // Части обязаны приезжать по UTC. Часы серверной сессии NAV идут по UTC, и в колонку
      // момент ложится как есть - перевода при записи нет; в пояс читателя его переводит
      // клиент, при показе. Момент, собранный здесь из МЕСТНЫХ частей, лёг бы в колонку
      // местным временем, и клиент прибавил бы пояс к нему ещё раз. Проверено замером
      // 10.09.2026, и цена ошибки - три часа вранья в каждой строке журнала на сервере с
      // московским поясом.
      IF Days <= 0 THEN
        EXIT(0DT);
      Offset := Milliseconds;
      EXIT(CREATEDATETIME(SqlEpoch + Days,0T) + Offset);
    END;

    LOCAL PROCEDURE TryReadCounters@59(MaxRows@1000000000 : Integer;VAR Reading@1000000001 : TEMPORARY Record 110235;VAR Rows@1000000002 : Integer);

    LOCAL PROCEDURE CountersSql@60(MaxRows@1000000000 : Integer) : Text;
    VAR
      Sql@1000000001 : Text;
      WaitMs@1000000002 : Text;
    BEGIN
      // Ожидания строчные и страничные складываются в одно число нарочно: лечатся они
      // одинаково, а порознь вторая колонка почти всегда была бы нулём.
      //
      // Приведение к bigint стоит ДО сложения. Счётчики объявлены bigint по отдельности,
      // но сумма по разделам таблицы, сложенная как int, переполнилась бы молча - и на
      // самой нагруженной таблице первой.
      //
      // ОТБОР СТОИТ ДО СОЕДИНЕНИЯ С КАТАЛОГОМ, и это не вкусовщина, а замер: тот же ответ
      // с соединением снаружи стоил 82,7 мс, а с соединением после отбора - 0,46. Разница
      // в сто восемьдесят раз, и берётся она из того, что у табличной функции нет
      // статистики: оптимизатор готовится к огромному набору и соединяет с каталогом всё,
      // хотя после отбора остаются единицы строк.
      WaitMs := 'SUM(CONVERT(bigint,s.row_lock_wait_in_ms) + CONVERT(bigint,s.page_lock_wait_in_ms))';
      Sql :=
        'SELECT TOP (' + FORMAT(MaxRows) + ') g.object_id, g.index_id, g.waits, g.wait_ms,' +
        ' g.esc, o.name, i.name' +
        ' FROM (SELECT s.object_id AS object_id, s.index_id AS index_id,' +
        ' SUM(CONVERT(bigint,s.row_lock_wait_count) + CONVERT(bigint,s.page_lock_wait_count))' +
        ' AS waits, ' + WaitMs + ' AS wait_ms,' +
        ' SUM(CONVERT(bigint,s.index_lock_promotion_count)) AS esc' +
        ' FROM sys.dm_db_index_operational_stats(DB_ID(),NULL,NULL,NULL) s' +
        ' GROUP BY s.object_id, s.index_id' +
        ' HAVING ' + WaitMs + ' > 0) g' +
        ' JOIN sys.objects o ON o.object_id ' + Eq + ' g.object_id' +
        ' JOIN sys.indexes i ON i.object_id ' + Eq + ' g.object_id' +
        ' AND i.index_id ' + Eq + ' g.index_id' +
        ' WHERE o.is_ms_shipped ' + Eq + ' 0' +
        // Потолок отсекает ХВОСТ, а не голову: порядок по насчитанным миллисекундам.
        // Выпавшая из потолка строка счёту не вредит - вернувшись, она отдаст прирост за
        // всё время отсутствия разом.
        ' ORDER BY g.wait_ms DESC;';
      EXIT(Sql);
    END;

    LOCAL PROCEDURE SqlEpoch@22() : Date;
    BEGIN
